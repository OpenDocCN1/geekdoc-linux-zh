/*
* UpdateDevice
* 更新设备信息。如果该
* 设备目前不存在，则创建并
* 进行初始化。
*
* sDevName - 设备名称。"eth0:", "cpu1" 等。 
* nValue - 当前值。
* nType - DELTA/ACTUAL。
* ACTUAL 表示值是“原始”的。
* DELTA 表示该值是相对于
* 上一个值的，需要在知道
* 上一个值的情况下进行计算。
* nMax - 这是要显示的正常最大值。
*/
void UpdateDevice(char *sDevName, long nValue, int nType, long nMax) {
    typDevice *dev;
    int i;
    /* -- 没有名字？滚蛋！ --- */
    if (sDevName == NULL) return;
    /* -- 通过名称查找设备 --- */
    dev = LookupDevice(sDevName);
    /* -- 我们有关于该设备的信息吗？ --- */
    if (dev) {
        /* -- 只添加一个值。 --- */
        NewValue(dev, nValue, FALSE);
    } else {
        /* -- 没有该设备！ --- */
    }
    /* -- 创建一个新设备。 --- */
    dev = AddDevice();
    /* -- 初始化值。 --- */
    dev->sName = strdup(sDevName);
    dev->nType = nType;
    dev->nMax = nMax;
    /* -- 清空值。 --- */
    for (i = 0; i < MAX_VALUES; i++) {
        dev->nValues[i] = (typHistoryValue)0;
    }
    /* -- 用当前位置更新 --- */
    NewValue(dev, nValue, TRUE);
}
}
