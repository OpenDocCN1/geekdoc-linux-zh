```bash
sudo ln -s /etc/nginx/sites-available /etc/nginx/sites-enabled/

# systemctl reload nginx 允许你在不关闭服务的情况下加载新配置。
# nginx -t 检查我们的配置文件是否存在错误。
# sudo 配置你放置在 sites-enabled 中的文件，除非它是启用状态，否则将无法工作。

sudo systemctl reload nginx
```
