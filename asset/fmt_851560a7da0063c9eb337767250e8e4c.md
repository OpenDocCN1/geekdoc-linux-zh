
```c
#include <stdlib.h>
#include <stdio.h>
#include <unistd.h>

/* 向流中写入 COUNT 份 MESSAGE，每次之间暂停一秒。 */

void writer (const char* message, int count, FILE* stream)
{
    for ( count = 0; --count) {
        /* 将消息写入流中，并立即发送出去。 */
        fprintf(stream, "%s", message);
        fflush(stream);
        /* 休息一会儿。 */
        sleep(1);
    }
}

/* 尽可能从流中读取随机字符串。 */

void reader (FILE* stream)
{
    char buffer[1024];
    /* 一直读取到流的结束。fgets 直到遇到换行符或文件结束符。 */
    while (!feof(stream)
        && !ferror(stream)
        && fgets(buffer, sizeof(buffer), stream) != NULL)
        fputs(buffer, stdout);
}

int main ()
{
    int fds[2];
    pid_t pid;
    /* 创建一个管道。管道两端的文件描述符放在 fds 数组中。 */
    pipe(fds);
    /* 分生一个子进程。 */
    pid = fork ()
    if (pid == (pid_t)0) {
        FILE* stream;
        /* 这是子进程。关闭描述符写端的副本。 *//
        close(fds[1]);
        /* 将读描述符转换为 FILE 对象，并从其进行读取。 */
        stream = fdopen(fds[0], "r");
        reader(stream);
    } else {
        /* 这是父进程。 */
        FILE* stream;
        /* 关闭文件描述符读端的副本。 *//
        close(fds[0]);
        /* 将写文件描述符转换为 FILE 对象，并向其写入。 */
        stream = fdopen(fds[1], "w");
        writer("Hello, world.", 5, stream);
        close(fds[1]);
    }

    return 0;
}
```

在 `main` 函数开头，声明 `fds` 为一个大小为 2 的整数数组。`pipe` 调用创建一个管道并将读、写文件描述符填充该数组。程序派生一个子进程。在关闭管道的读端后，父进程开始向管道写入字符串。在关闭管道的写端后，子进程从管道中读取字符串。

注意，在 `writer` 函数之后，父进程通过调用 `fflush` 刷新管道。否则，字符串可能不会立即通过管道发送。当你调用命令 `ls | less` 时，会发生两次 `fork`：一次用于 `ls` 子进程，一次用于 `less` 进程。两个进程都会继承管道文件描述符，因此它们可以使用管道进行通信。要让无关的进程进行通信，可以使用第 5.4.5 节“FIFOs”中讨论的 FIFO 输入。

## 5.4.3 重定向标准输入、输出和错误流

通常，你可能需要创建一个子进程，并将管道的一端设置为它的标准输入或输出。使用 `dup2` 调用，你可以将一个文件描述符等于另一个。例如，要将进程的标准输入重定向到文件描述符 `fd`，请使用此行代码：

```c
dup2(fd, STDIN_FILENO);
```

符号常量 `STDIN_FILENO` 代表标准输入的文件描述符，其值为 0。该调用关闭标准输入并将其重新打开作为 `fd` 的复制副本，以便两者可以互换使用。等效的文件描述符共享相同的文件位置和相同的状态标志位集。因此，从 `fd` 读取的字符不会再次从标准输入中读取。

### 5.4 管道

列表 5.8 中的程序使用 `dup2` 将管道的输出发送到 `sort` 命令。在创建管道后，程序进行 `fork`。父进程向管道打印字符串。子进程使用 `dup2` 将管道的读描述符附加到它的输入上。它执行了 `sort` 程序。

### 列表 5.8 (dup2,c) 使用 dup2 重定向管道输出

```c
#include <stdio.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <unistd.h>

int main () {
    int fds[2];
    pid_t pid;

    /* 创建一个管道。管道两端的文件描述符被
       放在 fds 数组中。 */
    pipe(fds);
    /* 分生一个子进程。 */
    pid = fork();
    if (pid == (pid_t)0) {
        /* 这是子进程。关闭你对描述符写端的副本。 */
        close(fds[1]);
        /* 将管道的读端连接到标准输入。 */
        dup2(fds[0], STDIN_FILENO);
        /* 用 "sort" 程序替换子进程。 */
        execl("sort", "sort", NULL);
    } else {
        /* 这是父进程。 */
        FILE *stream;
        /* 关闭我们对读描述符的副本。 */
        close(fds[0]);
        /* 将文件描述符转换为 FILE 对象，并向其写入。 */
        stream = fdopen(fds[1], "w");
        fprintf(stream, "Hello, world.\n");
    }
}
```

# 第5章 进程间通信

FIFO 可以有多个读取者或多个写入者。来自每个写入者的字节将原子化地写入，最大长度为 `PIPE_BUF`（在 Linux 上为 4KB）。来自同时写入的数据块会交织在一起。同时读取也遵循类似的规则。

## 与 Windows 命名管道的区别

Win32 操作系统中的管道与 Linux 管道非常相似（技术细节请参阅 Win32 库文档）。主要区别在于命名管道，它们在 Win32 中更像套接字（sockets）。Win32 命名管道可以连接通过网络连接的独立计算机上的进程。在 Linux 上，此用途 socket 实现。此外，Win32 允许在命名管道上建立多个读写连接而不会产生数据交织，且管道可双向通信。

### 5.5 套接字

套接字是一种双向设备，可与同一机器上的另一个进程或运行在其他机器上的进程进行通信。在本章讨论的进程间通信中，只有套接字允许不同计算机之间的进程进行通信。如 Telnet、login、FTP、talk 和万维网（WWW）等互联网程序都使用了套接字。

例如，你可以使用 Telnet 程序从 Web 服务器获取 WWW 内容，因为它们使用套接字进行网络通信。要打开与 codesource.com WWW 服务器的连接，请使用 `telnet codesource.com 80`。魔数 80 指定连接到运行 www.codesource.com 的 Web 服务器，而不是其他进程。建立连接后，尝试输入 `GET /`。这将通过套接字向 Web 服务器发送消息，服务器通过发送主页的 HTML 源码回复并关闭连接——例如：

```
% telnet codesource.com 80
Trying 206.168.99.1...
Connected to merlin.codesource.com (206.168.99.1)
Escape character is ']'.
GET /
<html>
<head>
<meta http-equiv="Content-Type" type="text/html;charset=8859-1">
...
```

- 注意，只有 Windows 可以创建命名管道；套接字仅用于客户端连接。
- 通常，你会使用套接字连接到不同类型的服务器并直接进行登录。

### 5.5 套接字概念

创建套接字时，必须指定三个参数：通信类型、命名空间和协议。

通信类型控制套接字如何处理数据并指定通信伙伴。当数据通过套接字发送时，会被分成数据包（chunks）。通信类型决定了这些数据包如何从发送方到达接收方。

- 连接类型保证数据包的交付。如果数据包在网络中丢失或乱序，接收方会自动请求发送方重传。
- 连接类型套接字就像电话通话：发送方和接收方的地址在连接建立时就固定了。
- 数据报类型不保证交付或到达顺序。数据包在传输过程中可能由于网络错误或其他原因丢失或乱序。系统只保证“尽力而为（best effort）”，因此数据包可能会以不同的顺序到达。
- 数据报类型套接字像邮政。每个消息必须标记有接收者的地址。

命名空间指定套接字地址的写法。套接字地址标识一个连接。例如，在“本地命名空间”中，套接字地址是文件名。在“互联网命名空间”中，套接字地址由网络地址（IP 地址）和端口号组成。端口号用于区分同一主机上的多个套接字。

协议指定数据如何传输。某些协议包括 TCP/IP（互联网使用的）、AppleTalk 网络协议以及 UNIX 本地协议。并不是所有的类型、命名空间和组合都是支持的。

#### 5.5.2 系统调用

套接字比之前讨论的技术更灵活。这些是套接字系统调用：

- `socket`—创建套接字
- `close`—销毁套接字
- `connect`—创建两个套接字之间的连接
- `bind`—将地址绑定到套接字
- `listen`—配置套接字接受连接
- `accept`—接受连接并创建一个新的套接字

套接字由文件描述符表示。

## 创建和销毁套接字

`socket` 和 `close` 函数分别创建和销毁套接字。创建套接字时，指定三个选项：命名空间、通信类型和协议。对于命名空间参数，使用以 `PF_` 开头的常量。例如 `PF_LOCAL` 或 `PF_UNIX` 指定本地命名空间，`PF_INET` 指定互联网命名空间。对于通信类型参数，使用以 `SOCK_` 开头的常量。连接类型套接字使用 `SOCK_STREAM`，数据报类型套接字使用 `SOCK_DGRAM`。

第三个参数协议指定传输和接收数据的底层机制。对于特定的套接字类型组合，每个协议都是有效的。对于每一对，使用 0 通常是正确的协议。如果 `socket` 成功，将返回套接字的文件描述符。你可以使用 `read`、`write` 等像操作其他文件描述符一样读写套接字。

## 调用 connect

要创建两个套接字之间的连接，客户端必须指定服务器的地址。第一个参数是套接字文件描述符。第二个参数指向套接字结构的指针；第三个参数是结构的字节长度。调用 `connect` 发起连接。如果连接成功，将返回一个与服务器通信的文件描述符。

## 发送信息

任何写入套接字的技术必须绑定到一个地址。第一个参数是套接字文件描述符。第二个参数是要发送的数据。第三个参数是长度。

#### 5.5.3 服务器

服务器的生命周期包括创建套接字、为其绑定地址以及调用 `listen` 指明它是一个服务器。这不会限制服务器可以处理的连接总数；它限制的是尚未被接受的尝试连接的客户端数量。

服务器接受连接请求。第一个参数是套接字文件描述符。第二个参数指向一个套接字结构，该结构填充了客户端的地址。第三个参数是套接字结构的字节数。服务器可以使用客户端地址判断是否真的想与该客户端通信。该调用为客户端创建一个新的套接字并返回相应的文件描述符。原始服务器套接字继续接受新连接。

要从套接字读取数据但不从输入队列中移除，请使用 `recv`。它接受与 `read` 相同的参数，还有一个额外的 `FLAGS` 参数。`MSG_PEEK` 会导致数据被读取但不会从输入队列中移除。

### 5.5.4 本地套接字

连接同一计算机上程序的套接字可以使用由 `PF_LOCAL` 和 `PF_UNIX` 别名表示的本地命名空间。这些被称为本地套接字或 UNIX 域套接字。它们的套接字地址由文件名指定，且仅在创建连接时使用。

名称在 `struct sockaddr_un` 结构中指定。你必须将 `sun_family` 设置为 `AF_LOCAL`，表示这是一个本地命名空间。`sun_path` 指定要使用的文件名，最大长度为 108 字节。`struct sockaddr_un` 的长度应使用 `SUN_LEN` 宏进行计算。如果使用文件名连接套接字，进程必须对文件拥有读取权限。尽管不同的计算机可能共享文件系统，但只有运行在同一台计算机上的进程才能与本地套接字通信。

本地命名空间允许的协议只有 0。由于它位于文件系统中，本地套接字被列为一个文件。例如，提示如下：

```
% ls -l /tmp/socket
srw-rw-x  1 user group  @ Nov 13 19:18 /tmp/socket
```

在使用本地套接字完成后，调用 `unlink` 将其删除。

## 5.5.5 使用本地命名空间套接字的示例

我们通过两个程序来演示套接字。列表 5.10 中的服务器程序创建了一个本地命名空间套字并在上面监听连接。当它接收到连接时，它会从连接中读取文本消息并打印，直到连接关闭。如果其中某条消息是 "quit"，服务器程序将移除套字并结束。socket-server 程序将套字路径作为命令行参数。

### 列表 5.10 (socket-server.c) 本地命名空间套字服务器

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/socket.h>
#include <sys.h>
#include <unistd.h>
```

```c
/* 从套字读取文本并打印。持续到套字关闭。如果客户端发送了 "quit" 消息，返回非零，否则返回零。 */
```

```c
int server (int client_socket) {
  while (1) {
    int length;
    char *text;
    if (read(client_socket, &length, sizeof(length)) == 0)
      return 0;
    text = (char*)malloc(length);
    read(client_socket, text, length);
    printf("%s\n", text);
    free(text);
    if (!strcmp(text, "quit"))
      return 1;
  }
}

int main (int argc, char *argv[]) {
  const char *const socket_name = argv[1];
  int socket_fd;
  struct sockaddr_un name;
  int client_sent_quit_message;

  socket_fd = socket(PF_LOCAL, SOCK_STREAM, 0);
  name.sun_family = AF_LOCAL;
  strncpy(name.sun_path, socket_name);
  bind(socket_fd, &name, SUN_LEN(&name));
  listen(socket_fd, 5);

  do {
    struct sockaddr_un client_name;
    socklen_t client_name_len;
    int client_socket_fd;
    client_socket_fd = accept(socket_fd, &client_name, &client_name_len);
    client_sent_quit_message = server(client_socket_fd);
    close(client_socket_fd);
  }
  while (!client_sent_quit_message);

  close(socket_fd);
  unlink(socket_name);
  return 0;
}
```
