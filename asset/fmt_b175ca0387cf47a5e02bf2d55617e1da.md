
# `killproc()` 与 `pidofproc()` 函数实现

```bash
# killproc()
# 用法: killproc [-p pidfile] [signal]
# 目的: 向运行的进程发送控制信号
# 输入: -p pidfile 使用指定的文件
#   pathname 的程序路径
#   signal 向 pathname 发送此信号
# 返回值（由 LSB 退出代码定义）：
# 0 - 程序 (pathname) 已发送信号并停止
# 1 - 通用或未指定的错误
# 2 - 参数无效或过多
# 5 - 程序未安装
# 7 - 程序未运行且提供了信号

killproc()
{
    local pidfile
    local prefix
    local program
    local signal="-TERM"
    local fallback="-KILL"
    local nosig
    local pidlist
    local retval
    local pid
    local delay="30"
    local piddead
    local dtime

    # 处理参数
    while true; do
        case "${1}" in
            -p)
                pidfile="${2}"
                shift 2
                ;;
            *)
                program="${1}"
                if [ -n "${2}" ]; then
                    signal="${2}"
                    fallback=""
                else
                    nosig=1
                fi
                # 额外参数错误
                if [ -n "${3}" ]; then
                    return 2
                else
                    break
                fi
                ;;
        esac
    done

    # 检查程序是否存在
    if [ ! -e "${program}" ]; then return 5; fi

    # 检查信号是否有效
    check_signal "${signal}"
    if [ "${?}" -ne "0" ]; then return 2; fi

    # 获取 pid 列表
    if [ -z "${pidfile}" ]; then
        # 通过发现确定 pid
        pidlist=$(pidof "${1}")
        retval="${?}"
    else
        # PID 文件包含了所需的 PID
        # 注意根据 LSB 要求，必须向 pidof 提供路径，
        # 但当前的实现或标准未使用它。
        pidlist=$(pidof -p "${pidfile}" "${1}")
        retval="${?}"
    fi

    # 仅返回一个值
    # 记录日志是 init 脚本（或发行版函数）的责任
    case "${retval}" in
        0)
            # 程序运行正常
            # 在此处什么都不做，让 killproc 继续。
            ;;
        1)
            # 程序未运行，但存在无效的 pid 文件
            # 删除 pid 文件。
            rm -f "${pidfile}"
            # 如果没有传递信号，则成功。
            if [ -n "${nosig}" ]; then
                return 0
            else
                return 7
            fi
            ;;
        3)
            # 程序未运行且不存在 pidfile
            # 如果没有传递信号，则成功。
            if [ -n "${nosig}" ]; then
                return 0
            else
                return 7
            fi
            ;;
        *)
            # 状态值返回的其他值不应被解释
            # 并作为错误返回。
            return 1
            ;;
    esac

    # 对退出信号和控制信号执行不同的操作
    check_sig_type "${signal}"
    if [ "${?}" -eq "0" ]; then # 信号用于终止程序
        # 处理空 pidlist（pid 文件存在但给了
        # 信号）
        if [ "${pidlist}" != "" ]; then
            # 杀死 pid 列表
            for pid in $(echo "${pidlist}"); do
                kill -0 "${pid}" 2>/dev/null
                if [ "$?" -ne "0" ]; then
                    # 进程已死，继续下一个并假设一切正常
                    :
                else
                    kill "${signal}" "${pid}" 2>/dev/null
                    # 等待 $(delay)/10 秒让 "$(pid)"
                    # 以十分之一秒为单位终止
                    while [ "${delay}" -ne "0" ]; do
                        kill -0 "${pid}" 2>/dev/null && piddead="1"
                        if [ "${piddead}" = "1" ]; then break; fi
                        sleep 0.1
                        delay=$((delay - 1))
                    done

                    # 如果 fallback 程序仍在运行，
                    # 使用 fallback
                    if [ -n "${fallback}" -a "${piddead}" != "1" ]; then
                        kill "${fallback}" "${pid}" 2>/dev/null
                        sleep 1
                    fi

                    # 再次检查，如果仍在运行则失败
                    kill -0 "${pid}" 2>/dev/null && return 1
                fi
            done
        fi
    else
        # 对于不期望程序退出的信号，只需
        # 让 kill 执行工作，并评估 kill 的返回值
        for pid in $(echo "${pidlist}"); do
            kill "${signal}" "${pid}"
        done
        if [ "$?" -ne "0" ]; then return 1; fi
    fi

    # 检查并移除过期的 PID 文件。
    if [ -z "${pidfile}" ]; then
        # Find $(program) 的文件名
        prefix=$(echo "${program}" | sed 's/[^\/]*$//')
        progame=$(echo "${program}" | sed 's@${prefix}@@')
        if [ -e "/var/run/${progame}.pid" ]; then
            rm -f "/var/run/${progame}.pid" 2>/dev/null
        fi
    else
        if [ -e "${pidfile}" ]; then
            rm -f "${pidfile}" 2>/dev/null
        fi
    fi
}

# 执行启动！
nice -n "${nice}" "$@"
```

```bash
# pidofproc()
# 用法: pidofproc [-p pidfile] pathname
# 目的: 此函数返回特定守护进程的一个或多个 pid
# 输入: -p pidfile, 使用指定的 pidfile 代替 pidof
#   pathname, 指指定程序的路径
# 返回值（如 LSB 状态码定义）:
#   0 - 成功（将 PIDs 输出到标准输出）
#   1 - 程序已停止，但 PID 文件仍然存在（剩余 PIDs）
#   0 - 程序未运行（无输出）
```
