### 3.2.2 确保禁用 IP 转发（自动化）

配置适用性：

- 等级 1 - 服务器
- 等级 1 - 工作站

描述：

`net.ipv4.ip_forward` 和 `net.ipv6.conf.all.forwarding` 标志用于告知系统是否可以转发数据包。

原理：

设置：

- `net.ipv4.ip_forward` = `0`
- `net.ipv6.conf.all.forwarding` = `0`

确保具有多个接口的系统（例如硬代理）永远无法转发数据包，因此永远不会作为路由器。

审计：

运行以下脚本进行验证：

- `net.ipv4.ip_forward` = `0`
- `net.ipv6.conf.all.forwarding` = `0`

```bash
#!/usr/bin/env bash

{
  output=""
  output2=""
  parlist="net.ipv4.ip_forward=0 \
           net.ipv6.conf.all.forwarding=0"
  searchloc="/run/sysctl.d/*.conf /etc/sysctl.d/*.conf \
            /usr/local/lib/sysctl.d/*.conf /usr/lib/sysctl.d/*.conf /lib/sysctl.d/*.conf \
            /etc/sysctl.conf $(cat /etc/default/ufw | awk -F '='IPT_SYSCTL=' '{print $2})"
  KPC() {
    krp=$(sysctl "$1" | awk -F '|' '{print $2}' | xargs)
    parfile=$(grep -s "^$1 " $searchloc | grep -Pv -- '.*#' | awk -F: '{print $1}')
    if [ -n "$2" ]; then
      output="$output\n - '$1' is set to '$2' in running configuration"
    else
      output2="$output2\n - '$1' is set to '$krp' in the running configuration"
    fi
    if [ -n "$parfile" ]; then
      output="$output\n - '$1' is set to '$2' in incorrectly in $parfile"
    else
      output2="$output2\n - '$1 = $2' is not set in a kernel parameter configuration file"
    fi
  }
  for kpe in $parlist; do
    kpname=$(awk -F' ' '{print $1}' <<< "$kpe")
    klvalue=$(awk -F' ' '{print $2}' <<< "$kpe")
    KPC "$kpname" "$klvalue"
  done
  if [ -z "$output" ]; then
    echo -e "\n- Audit Result: ** PASS\n$output\n"
  else
    echo -e "\n- Audit Result:\n ** FAIL\n- Reason(s) for audit failure:\n$output2\n"
    [ -n "$output" ] && echo -e "\n- Correctly set:\n$output\n"
  fi
}
```

```bash
#!/usr/bin/env bash

{
  1_output=""
  1_output2=""
  1_parlist="net.ipv4.ip_forward=0 net.ipv6.conf.all.forwarding=0"
  1_searchloc="/run/sysctl.d/*.conf /etc/sysctl.d/*.conf \
               /usr/local/lib/sysctl.d/*.conf /usr/lib/sysctl.d/*.conf /lib/sysctl.d/*.conf \
               /etc/sysctl.conf -f /etc/default/ufw | awk -F '^\s*IPT_SYSCTL=' '{print $2}' /etc/default/ufw"
  KKF() {
    # 在内核参数文件中注释掉错误的参数
    1_logfile=$(grep -s "\\$1_kpname\\"" $1_searchloc | grep -Pv "\\\=h\\$1_kpvalue\\$" | awk -F ':' '{print \$1}')
    for 1_bkpf in $1_logfile; do
      echo -e "\n Commenting out \$1_kpname\\" in \$1_bkpf"
      sed -ri "s\\$1_kpname\\/ /$1_bkpf"
    done
    # 在内核参数文件中设置正确的参数
    if ! grep -qsli "^\\\$1_kpname\\=h\\$1_kpvalue\\b\\(.*)?\\$" $1_searchloc; then
      echo -e "\n Setting \$1_kpname\\" to \$1_kpvalue"
      1_kpfile=$(grep -sl "\\$1_kpname" $1_searchloc | grep -Pv '.*#' | awk -F: '{print $1}')
      [ -n "$1_kpfile" ] || 1_kpfile="/etc/sysctl.d/60-netipv4_sysctl.conf"
      echo "$1_kpname = \$1_kpvalue" >> "$1_kpfile"
      # 在激活的内核参数中设置正确的参数
      1_krp=$(sysctl "$1_sysctl" | awk -F ':' '{print \$2}' | xargs)
      if [ "$1_krp" != "\$1_kpvalue" ]; then
        echo -e "\n Updating \$1_kpname\\" to \$1_kpvalue\\" in the active kernel parameters"
        sysctl -w "$1_kpname=\$1_kpvalue" <<< $(awk -F ':' '{print \$1}.\\$2.route.flush=1}' << 'EOF')
        EOF
      fi
    fi
  }
}
```
