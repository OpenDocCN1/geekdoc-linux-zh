## 列表 13-11: mm/oom_kill.c: oom_evaluate_task()

```c
if (oom_unkillable_task(task))
    goto next;

/*
 * 此任务已经访问了内存保留并正在被杀死
 * 不允许任何其他任务访问保留内存，除非
 * 该任务具有 MMF_OOM_SKIP 标记，因为它释放
 * 任何内存的概率非常低。
 */
if (!is_sqrq_oom(oc) && tsk_is_oom_victim(task)) {
    if (test_bit(MMF_OOM_SKIP, &task->signal->oom_mm->flags))
        goto next;
    goto abort;
}

/*
 * 任务正在分配大量内存并标记为在触发
 * oom 时首先杀死，然后进行选择。
 */
if (oom_task_origin(task)) {
    points = LONG_MAX;
    goto select;
}

npoints = oom_badness(task, oc->totalpages);

if (points == LONG_MIN || points < oc->chosen_points)
    goto next;

select:
    if (oc->chosen)
        put_struct(oc);
    oc->chosen = task;
    oc->chosen_points = points;

next:
    return 0;

abort:
    if (oc->chosen)
        put_struct(oc);
    oc->chosen = task;
    return 1;
}
```

### 内存溢出（OOM）杀器

## 列表 13-13: oom_kill.c: oom_badness()

```c
/**
 * oom_badness - 确定杀死候选对象的启发式函数
 * @obj 需要计算分的任务 struct
 * @totalpages: 允许的总页数
 *
 * 确定杀死任务的启发式应尽可能简单
 * 可预测。目标是为消耗内存的任务返回
 * 最高值，以避免后续的 OOM 失败。
 */
long oom_badness(struct task_struct *p, unsigned long totalpages)
{
    long points;
    long adj;

    if (oom_unkillable_task(p))
        return LONG_MIN;

    p = find_lock_task_mm(p);
    if (!p)
        return LONG_MIN;

    /*
     * 不要考虑显式标记为 oom 不可杀
     * 或已经被 oom 回收或正在
     * 执行 vfork() 的任务
     */
    adj = (long)p->signal->oom_score_adj;
    if (adj == OOM_SCORE_ADJ_MIN ||
        test_bit(MMF_OOM_SKIP, &p->mm->flags)) {
        task_unlock(p);
        return LONG_MIN;
    }

    /*
     * 分数的基准是任务 RSS、页表和交换空间
     * 内存的比例。
     */
    points = get_mm_rss(p->mm) + get_mm_counter(p->mm, MM_SWAPENTS) +
        mm_pgptables_bytes(p->mm) / PAGE_SIZE;
    task_unlock(p);

    /* 归一化为 oom_score_adj 单位 */
    adj *= totalpages / 1000;
    points += adj;
}
```

我们再次断言该任务无法通过 `oom_unkillable_task()` 杀死（参见列表 ??），这里使用了 `LONG_MIN`（即可能的最小有符号长型 `long` 值）来提供一个有效的最小分值，以防止该任务被选中。

接下来，我们获取该任务的锁以防止其通过 `find_lock_task_mm()` 消失，我们将在列表 13-14 中检查该函数。

## 列表 13-14: oom_kill.c: find_lock_task_mm()

在 `find_lock_task_mm()` 中，我们遍历代表与传入任务同一进程内的每个线程，确定是否有任何线程维护着对 `struct mm_struct` 的引用，该结构体描述了它们共享的虚拟地址空间。

这样做是因为在销毁过程中，至少有一个线程可能仍然维持着虚拟地址空间的引用，这足够满足我们的需求。

回到列表 13-1 中的 `oom_badness()`，我们注意到再次如果 `find_lock_task_mm()` 返回 `NULL`，则将分数设置为最小值，以指示该进程不应被选中。

我们考虑了三个最终的边界情况——如果该进程的 `/proc/$pid/oom_score_adj` 被设置为可能的最小值（`OOM_SCORE_ADJ_MIN`），或者该进程设置了 `MMF_OOM_SKIP` 标记（表示该进程可能不会释放任何内存），或者我们正处于 `fork()` 操作中间（`CLONE_VFORK`）；在这种情况下，我们通过返回 `LONG_MIN` 来指示该任务不应被选中（当然，在此之前会释放对该任务的锁）。

这两种情况是自言自足的——它们明确表示该进程不被考虑用于 OOM 杀进程，但第三种情况需要一些解释。

一个 `fork()`（也可以通过 `clone()` 系统调用实现）使用 `CLONE_VFORK` 进行复制，它与普通的 `fork()` 类似，只是复制了页表并且父进程会被挂起直到子进程退出或执行 `execve()`。

在 `oom_badness()` 中，我们检查该任务是否是 `vfork()` 中父进程的 fork 子进程。在这种情况下，它没有内存可以释放，我们对其进行单独处理。

在考虑所有边界情况后，我们计算进程的“坏度”分数，即一个考虑了任何 `/proc/$pid/oom_score_adj` 修饰符的比例索引。

我们通过获取进程使用的驻留基页和交换页表的总和，加上 `oom_score_adj` 修饰符来计算“坏度”分数，以基页为单位表示（调整分数以系统可用总页数的千分之一为单位）。

从任务中检索相关数据后，我们在返回之前释放其锁。

## 13.4.5 杀死受害者

回到 `out_of_memory()`（如列表 13-5 所示），我们看到一旦进程被选中终止，进程的实际杀死操作将推迟到 `oom_kill_process()`，我们将在列表 13-15 中检查该函数（省略了 cgroup 处理）。

## 列表 13-15: mm/oom_kill.c: oom_kill_process()

```c
static void oom_kill_process(struct oom_control *oc, const char *message)
{
    struct task_struct *victim = oc->chosen;

    static struct DEFINE_RATELIMIT_STATE(oom_rs, DEFAULT_RATELIMIT_INTERVAL,
                              DEFAULT_RATELIMIT_BURST);

    /*
     * 如果任务已经在退出，不要惊扰管理员或杀死
     * 它的子进程或线程，只需为其访问内存储备的权限
     * 这样它可以快速死亡
     */
    task_lock(victim);
    if (task_will_free_mem(victim)) {
        mark_oom_victim(victim);
        queue_oom_reaper(victim);
        task_unlock(victim);
        put_task_struct(victim);
        return;
    }
    task_unlock(victim);
    if (__ratelimit(&oom_rs)) {
        dump_header(oc, victim);
    }
    __oom_kill_process(victim, message);
}
```

我们建立了一个速率限制变量 `oom_rs` 以限制 OOM 杀进程输出的重复。关于此细节的讨论超出本书的范围。

我们获取任务的锁，然后如列表 13-7 所示，通过 `task_will_free_mem()` 评估进程是否正在被杀死并将轻易地释放内存。

如果是，则通过 `mark_oom_victim()`（参见列表 13-18）将任务标记为 OOM 受害者（这反过来会允许进程访问内存储备以服务于释放内存），并通过 `queue_oom_reaper()`（参见列表 13-21）将其加入收割队列，最后在解锁并减少其引用计数之前。

在 `oom_evaluate_task()` 中（如列表 13-11 所示），我们通过 `get_task_struct()` 增加任务的引用计数，这样我们可以确保任务不会在此点之后消失。

然后我们通过 `dump_header()`（以速率限制的方式以避免跨内核缓冲区）详细说明被选中的进程，我们在列表 13-16 中检查该函数。

回到 `out_of_memory()` 和列表 13-15，我们看到我们将杀死进程的沉重工作交给了 `__oom_kill_process()`，我们在列表 13-17 中检查它（省略 cgroup 逻辑）。

## 列表 13-17: oom_kill.c: __oom_kill_process()

```c
static void __oom_kill_process(struct task_struct *victim, const char *message)
{
    struct task_struct *p;
    struct mm_struct *mm;
    bool can_oom_reap = true;

    p = find_lock_task_mm(victim);
    if (!p) {
        pr_info("%s: OOM victim %d (%s) is already exiting. Skip killing the task\n", message, task_pid_nr(victim), victim->comm);
        put_task_struct(victim);
        return;
    } else if (victim != p) {
        get_task_struct(p);
        put_task_struct(victim);
        victim = p;
    }
    /* 在 task_unlock(victim) 后安全获取 mm 的引用 */
    mm = victim->mm;
    mmgrab(mm);

    /*
     * 我们应该在授予内存储备之前发送 SIGKILL
     * 以防止 OOM 受害者耗尽用户空间控制的储备。
     */
    do_send_sig_info(SIGKILL, SEND_SIG_PRIV, victim, PIDTYPE_TGID);
    mark_oom_victim(victim);
    pr_err("Killed process %d total-vm=%lu, anon-rss=%lu, UID=%lu, file-rss=%lu, shmem-rss=%lu, pgtables=%lu oom_adj=%d\n",
           message, task_pid_nr(victim), victim->mm->total_vm,
           mm_counter(victim->mm, MM_FILEPAGES),
           mm_counter(victim->mm, MM_SHMEMPAGES),
           from_kuid(&user_ns, task_uid(victim)),
           mm_pgtables_bytes(victim->mm) >> 10, victim->signal->oom_adj);
    task_unlock(victim);
}
```

在 `__oom_kill_process()` 中，我们首先调用 `find_lock_task_mm()`（如列表 13-14 所示）来查找并锁定属于目标受害者同一进程且引用了描述进程地址空间的有效 `mm_struct` 对象的任务。

## 列表 13-18: mm/oom_kill.c: mark_oom_victim()

```c
/*
 * oom 已经被禁用。
 *
 * tsk->signal 不为 NULL，调用者必须保证它是稳定的（
 * 在 task_lock 锁下或对当前任务操作）。
 */
static void mark_oom_victim(struct task_struct *tsk)
{
    struct mm_struct *mm = tsk->mm;

    WARN_ON(oom_killer_disabled);
    /* oom mm 与信号生命周期绑定。 */
    if (!cmpxchg(&tsk->signal->oom_mm, NULL, mm)) {
        mmgrab(tsk->signal->oom_mm);
        set_bit(MMF_OOM_VICTIM, &mm->flags);
    }

    atomic_inc(&oom_victims);
}
```

`signal_struct->oom_mm` 字段类型为 `struct mm_struct`，用于跟踪受 OOM 回收的虚拟地址空间计数。

我们原子地将此字段与该任务的受害者进行比较并交换，如果还没有设置（这表明该进程已经处于 OOM 杀进程状态），那么我们调用 `mmgrab()` 来增加虚拟地址空间的引用计数，并使用 `MMF_OOM_VICTIM` 标志将该地址空间标记为受害者，在增加当前 OOM 受害者的原子计数 `oom_victims` 之前。

标记完成后，我们现在继续确保在执行列表 13-19 中的最终清理任务之前，我们涵盖了所有共享 `mm_struct` 但并不存在于同一线程组中的任务。
