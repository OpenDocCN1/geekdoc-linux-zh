```c
/* 每当一个已经打开设备文件的进程尝试从中读取数据时，
 * 就会调用此函数。
 */
static ssize_t device_read(struct file *file, /* 参见 include/linux/fs.h */
                           char __user *buffer, /* 要填充的缓冲区 */
                           size_t length, /* 缓冲区长度 */
                           loff_t *offset)
{
    /* 实际写入缓冲区的字节数 */
    int bytes_read = 0;
    /* 读取消息的进程读到了什么位置？如果消息
     * 比我们在 device_read 中可填充的缓冲区大小更大，这个信息很有用。
     */
    const char *message_ptr = message;
    if (!*(message_ptr + *offset)) { /* 我们已经到达消息末尾 */
        *offset = 0; /* 重置偏移量 */
        return 0; /* 表示文件结束 */
    }
    message_ptr += *offset;
    /* 实际将数据放入缓冲区 */
    while (length && *message_ptr) {
        /* 因为缓冲区位于用户数据段，而不是内核数据段，
         * 所以直接赋值行不通。我们必须改用 put_user，
         * 将数据从内核数据段复制到用户数据段。
         * 这正是 put_user 的作用。
         */
        put_user(*(message_ptr++), buffer++);
        length--;
        bytes_read++;
    }
    pr_info("Read %d bytes, %ld left\n", bytes_read, length);
    *offset += bytes_read;
    /* 读函数应返回实际插入缓冲区的字节数，
     * 也就是实际写入缓冲区的字节数。
     */
    return bytes_read;
}
```

```c
/* 当有人试图向我们的设备文件写入数据时调用。 */
static ssize_t device_write(struct file *file, const char __user *buffer,
                            size_t length, loff_t *offset)
{
    int i;
    pr_info("device_write(%p,%p,%ld)", file, buffer, length);
    for (i = 0; i < length && i < BUF_LEN; i++)
        get_user(message[i], buffer + i);
    /* 同样，返回所使用的输入字符数。 */
    return i;
}
```

```c
/* 每当进程尝试对我们的设备文件执行 ioctl 时，就会调用此函数。
 * 我们得到两个额外的参数（除了所有设备函数都有的 inode 和 file
 * 结构之外）：被调用的 ioctl
 * 的编号
 * 以及传递给 ioctl 函数的参数。
 *
 * 如果 ioctl 是写操作或读/写操作（意味着输出会返回给调用进程），
 * 则 ioctl 调用会返回此函数的输出。
 */
static long
device_ioctl(struct file *file, /* 同上 */
             unsigned int ioctl_num, /* ioctl 的编号和参数 */
             unsigned long ioctl_param)
{
    int i;
    long ret = SUCCESS;
    /* 我们不想同时与两个进程通信。 */
    if (atomic_cmpxchg(&already_open, CDEV_NOT_USED, CDEV_EXCLUSIVE_OPEN))
        return -EBUSY;
    /* 根据所调用的 ioctl 进行切换 */
    switch (ioctl_num) {
    case IOCTL_SET_MSG:
        /* 接收指向消息（位于用户空间）的指针，并将其设置为
         * 设备的消息。获取传递给 ioctl 的参数，
         * 该参数由进程提供。
         */
        char __user *tmp = (char __user *)ioctl_param;
        char ch;
        /* 查找消息的长度 */
        get_user(ch, tmp);
        for (i = 0; ch && i < BUF_LEN; i++, tmp++)
            get_user(ch, tmp);
        device_write(file, (char __user *)ioctl1_param, i, NULL);
        break;
    }
    case IOCTL_GET_MSG: {
        loff_t offset = 0;
        /* 将当前消息传递给调用进程 - 参数
         * 我们得到的是一个指针，填充它。
         */
        i = device_read(file, (char __user *)ioctl1_param, 99, &offset);
```
