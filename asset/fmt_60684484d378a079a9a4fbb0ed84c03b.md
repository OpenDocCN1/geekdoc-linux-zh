/* stksrv.c: * * 股票指数广播示例: */
#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>
#include <errno.h>
#include <string.h>
#include <time.h>
#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

#define TRUE
#define TRUE 1
#define FALSE 0

extern int maddr(
    void *addr,
    int *addrlen,
    char *str_addr,
    char *protocol);
#define MAXQ 4

static struct {
    char *index;
    int start;
    int vol;
    int current;
} quotes[] = {
    { "DJIA", 1030330, 375 },
    { "NASDAQ", 276175, 125 },
    { "S&P 500", 120331, 50 },
    { "TSE 300", 689572, 75 },
};

/*
 * 初始化:
 */
static void
initialize(void) {
    short x;
    time_t td;

    /*
     * 初始化随机数生成器:
     */
    time(&td);
    srand((int)td);

    for ( x=0; x < MAXQ; ++x )
        quotes[x].current =
            quotes[x].start;
}

/*
 * 随机更改某个指数报价:
 */
static void
gen_quote(void) {
    short x;     /* 指数 */
    short v;     /* 指数的波动性 */
    short h;     /* v 的一半 */
    short r;     /* 随机变化 */

    x = rand() % MAXQ;
    v = quotes[x].vol;
    h = (v / 2) - 2;
    r = rand() % v;

    if ( r < h )
        r = -r;
    quotes[x].current += r;
}

/*
 * 此函数报告错误并
 * 返回到 shell:
 */
static void
bail(const char *on_what) {
    fputs(strerror(errno), stderr);
    fputs(" : ", stderr);
    fputs(on_what, stderr);
    fputc('\n', stderr);
    exit(1);
}

int
main(int argc, char **argv) {
    short x;        /* 股票指数索引 */
    double i0;      /* 初始指数值 */
    double i;       /* 指数值 */
    char bcbufr[512], *bp;/* 缓冲区和指针 */
    int z;          /* 状态返回码 */
    int s;          /* 套接字 */
    struct sockaddr_in adr_srvr;/* AF_INET */
    int len_srvr;   /* 长度 */
    struct sockaddr_in adr_bc; /* AF_INET */
    int len_bc;     /* 长度 */
    static int so_broadcast = TRUE;
    static char
    *sv_addr = "127.0.0.1:9099",
    *bc_addr = "127.255.255.255:9099";

    /*
     * 形成服务器地址:
     */
    if ( argc > 2 )
    {
        /* 服务器地址: */
        sv_addr = argv[2];
    }
    if ( argc > 1 )
    {
        /* 广播地址: */
        bc_addr = argv[1];
    }

    /*
     * 形成服务器地址:
     */
    len_srvr = sizeof adr_srvr;

    z = maddr(
            &adr_srvr,  /* 返回的地址 */
            &len_srvr,  /* 返回的长度 */
            sv_addr,    /* 输入的字符串地址 */
            "udp");     /* UDP 协议 */

    if ( z == -1 )
        bail("无效的服务器地址");

    /*
     * 形成广播地址:
     */
    len_bc = sizeof adr_bc;

    z = maddr(
            &adr_bc,    /* 返回的地址 */
            &len_bc,    /* 返回的长度 */
            bc_addr,    /* 输入的字符串地址 */
            "udp");     /* UDP 协议 */

    if ( z == -1 )
        bail("无效的广播地址");

    /*
     * 创建一个要使用的 UDP 套接字:
     */
    s = socket(AF_INET, SOCK_DGRAM, 0);
    if ( s == -1 )
        bail("socket()");

    /*
     * 允许广播:
     */
    z = setsockopt(s,
                SOL_SOCKET,
                SO_BROADCAST,
                &so_broadcast,
                sizeof so_broadcast);

    if ( z == -1 )
        bail("setsockopt(SO_BROADCAST)");

    /*
     * 将一个地址绑定到我们的套接字，以便
     * 客户端程序可以侦听此
     * 服务器:
     */
    z = bind(s,
            (struct sockaddr *)&adr_srvr,
            len_srvr);

    if ( z == -1 )
        bail("bind()");

    /*
     * 现在开始提供报价:
     */
    initialize();

    for (;;) {
        /*
         * 更新列表中的一个报价:
         */
        gen_quote();

        /*
         * 形成一个要发送的数据包:
         */
        bp = bcbufr;
        for ( x=0; x<MAXQ; ++x ) {
            i0 = quotes[x].start / 100.0;
            i = quotes[x].current / 100.0;
            sprintf(bp,
                    "%-7.7s %8.2f %+.2f\n",
                    quotes[x].index,
                    i,
                    i - i0);
            bp += strlen(bp);
        }

        /*
         * 广播更新的信息:
         */
        z = sendto(s,
                    bcbufr,
                    strlen(bcbufr),
                    0,
                    (struct sockaddr *)&adr_bc,
                    len_bc);

        if ( z == -1 )
            bail("sendto()");

        sleep(4);
    }

    return 0;
}
