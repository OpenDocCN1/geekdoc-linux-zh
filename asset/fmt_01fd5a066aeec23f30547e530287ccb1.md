
尽早检测内存分配失败是一个好主意。否则，分配失败会将空指针引入程序中意料之外的地方。由于分配失败很难减少，调试这些问题可能会非常困难。分配失败通常是灾难性的，因此终止程序是一个可以接受的选择。

- `error` 函数用于报告致命错误。它向 `stderr` 打印消息并结束程序。对于由于系统调用或库调用失败引起的错误，`system_error` 根据 `errno` 的值生成部分错误消息（参见第 2 章“编写良好的 GNU/Linux 软件”中的第 2.2.3 节“来自系统的错误代码”）。

- `get_executable_directory` 返回包含当前进程正在运行的可执行文件的目录。该目录路径可以用于定位程序的其他组件，这些组件在运行时安装在相同的位置。此函数通过检查 `/proc` 文件系统中的符号链接 `/proc/self/exe` 来工作（参见第 7 章“proc 文件系统”中的第 7.2.1 节“/proc/self”）。

此外，`common.c` 定义了两个全局变量：

### 11.2.2 加载服务器模块

`module.c`（参见列表 11.3）动态加载服务器模块的实现。加载的服务器模块由 `struct server_module` 的实例表示，该结构体定义在 `server.h` 中。

#### 列表 11.3 (module.c) 服务器模块的加载与排序

```c
#include <dlfcn.h>
#include <stdlib.h>
#include <stdio.h>
#include <string.h>

#include "server.h"

char* module_dir;

struct server_module* module_open(const char* module_name)
{
    char* module_path;
    void* handle;
    void (* module_generate)(int);
    struct server_module* module;
    /* 构建我们将尝试加载的模块共享库的完整路径。 */
    module_path = (char*)xmalloc(strlen(module_dir) + strlen(module_name) + 2);
    sprintf(module_path, "%s/%s", module_dir, module_name);

    /* 尝试将 MODULE_PATH 作为共享库打开。 */
    handle = dlopen(module_path, RTLD_NOW);
    free(module_path);
    if (handle == NULL) {
        /* 失败；要么该路径不存在，要么它不是共享库。 */
        return NULL;
    }
    /* 从共享库解析 module_generate 符号。 */
    module_generate = (void (*)(int)) dlsym(handle, "module_generate");
    /* 确保找到了符号。 */
    if (module_generate == NULL) {
        /* 符号缺失。虽然这是一个共享库，但它
           可能不是服务器模块。关闭并指示失败。 */
        dlclose(handle);
        return NULL;
    }

    /* 分配并初始化一个 server_module 对象。 */
    module = (struct server_module*) zmalloc(sizeof(struct server_module));
    module->handle = handle;
    module->name = strdup(module_name);
    module->generate_function = module_generate;
    /* 返回它，表示成功。 */
    return module;
}

void module_close(struct server_module* module)
{
    /* 关闭共享库。 */
    dlclose(module->handle);
    /* 释放模块名。 */
    free ((char*)module->name);
    /* 释放模块对象。 */
    free(module);
}
```

每个模块都是一个共享库文件（参见列表 11.3）。`module.c` 包含两个函数：

- `module_open` 尝试加载共享库并必须导一个名为 `module_generate` 的函数。该函数打开共享库并返回一个 `struct_module` 对象。
- `module_close` 关闭共享库并释放模块名和对象。

`module.c` 还定义了两个全局变量：

- `program_name` 的值正在运行的程序名称，如其参数列表所示。
- 如果程序在详细模式下运行，则变量 `verbose` 为非零。

```c
/* 创建一个 TCP 套字。 */
server_socket = socket(PF_INET, SOCK_STREAM, 0);
if (server_socket == -1)
    system_error('socket');
/* 为我们想要监听连接的本地地址构建
一个套字结构体。 */
memset(&socket_address, 0, sizeof(socket_address));
socket_address.sin_family = AF_INET;
socket_address.sin_addr = local_address;
/* 将套字绑定到地址。 */
rval = bind(server_socket, &socket_address, sizeof(socket_address));
if (rval != 0)
    system_error('bind');
/* 指令套字接受连接。 */
rval = listen(server_socket, 10);
if (rval != 0)
    system_error('listen');

if (verbose) {
    /* 在详细输出模式下，显示我们正在监听的
    本地地址和端口号。 */
    socklen_t address_length;

    /* 查找套字的本地地址。 */
    address_length = sizeof(socket_address);
    rval = getockname(server_socket, &socket_address, &address_length);
    assert(rval == 0);
    /* 打印消息。端口号需要从
    网络序（大端序）转换为主机序。 */
    printf("server listening on %s:%d\n",
           inet_ntoa(socket_address.sin_addr),
           ntohs(socket_address.sin_port));
}
/* 循环处理连接。 */
while (1) {
    struct sockaddr_in remote_address;
    socklen_t address_length;
    int connection;
    pid_t child_pid;

    /* 接受连接。此调用会阻塞直到连接
    就绪。 */
    address_length = sizeof(remote_address);
    connection = accept(server_socket, (struct sockaddr *)&remote_address, &address_length);
    if (connection == -1) {
        /* accept 调用失败。 */
        if (errno == EINTR) {
            /* 调用被信号中断。重试。 */
            continue;
        } else {
            /* 出错了。 */
            system_error("accept");
        }
    }
    /* 创建子进程。 */
    child_pid = fork();
    if (child_pid == 0) {
        /* 这是子进程。它不应该使用标准输入或标准输出，
        所以将其关闭。 */
        close(STDIN_FILENO);
        close(STDOUT_FILENO);
        /* 处理连接。 */
        handle_connection(connection);
        /* 关闭连接。 */
        close(connection);
        exit(0);
    } else if (child_pid > 0) {
        /* 这是父进程。我们不需要这个 connection 的
        副本。将其关闭。 */
        close(connection);
    } else {
        /* fork 调用失败。 */
        system_error("fork");
    }
}
```

`server_run` 是运行服务器的主入口点。此函数启动服务器并开始接受连接，除非发生错误，否则不会返回。服务器使用 TCP 套字（参见第 5 章进程间通信第 5.5.3 节“服务器”）

`server_run` 的第一个参数指定接受连接的本地地址。GNU/Linux 可能有多个地址，每个地址可能绑定到不同的网络接口。要限制服务器从特定接口接受连接，请指定相应的网络地址。指定本地地址 `INADDR_ANY` 以接受任何本地地址的连接。

`server_run` 的第二个参数是接受连接的端口号。如果端口已被占用，或者对应于没有超级用户权限运行的特权端口，服务器将失败。值 `0` 指令 Linux 自动选择一个未使用的端口。有关互联网域名地址和端口的更多信息，请参阅 man 页。
