### 列表 2.11 使用 Set-VMHostLockdown 函数

```powershell
function Set-VMHostLockdown (
  # SYNOPSIS
  为 ESXI 主机启用或禁用锁定模式。
  .EXAMPLE
  为 ESXi 主机启用锁定模式
  Set-VMHostLockdown -VMHost (Get-VMHost $name) -Enable
  .EXAMPLE
  使用流水线禁用集群的锁定模式
  Get-Cluster someCluster | Set-VMHostLockdown -Enable:$false
  .PARAMETER VMHost
  要修改的主机。
  .PARAMETER Enable
  为指定的主机启用锁定模式。
  .PARAMETER Disable
  为指定的主机禁用锁定模式。
  .INPUTS
[VMware.VimAutomation.ViCore.Impl.V1.Inventory.VMHostImpl
```

### 自动化安装

```powershell
[CmdletBinding(SupportsShouldProcess=$true))
Param(
    # 要启用或禁用锁定的 VMHost
    Parameter(
        Mandatory=$true,
        ValueFromPipeline=$true,
        HelpMessage="VMHost"
    )
)
[VMware.VimAutomation.ViCore.Impl.V1.Inventory.VMHostImpl]
$VMHost
### 启用/禁用锁定
[Parameter(Mandatory=$true)]
[switch]
$Enable
Process {
    $HostView = $VMHost | Get-View -Property Name
    if ($Enable) {
        $msg = "进入锁定模式"
        if ($PsCmdlet.ShouldProcess($VMHost.Name, $msg)) {
            $HostView.EnterLockdownMode()
        } else {
            $msg = "退出锁定模式"
            if ($PsCmdlet.ShouldProcess($VMHost.Name, $msg)) {
                $HostView.ExitLockdownMode()
            }
        }
    }
}
```
