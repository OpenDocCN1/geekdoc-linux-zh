main() {

    tput clear

    echo "$PROG started from $LOGNAME@$HOST on $TERM at $TODAY."

    rm -f $LIST 2>/null

    # Trap on exit/interrupt/break to clean up

    trap "cleanup $TRAPP_EC \"Abnormal termination. $PROG\"" 0 1 25

    # Check command options

    [ "$#" -ne 1 ] && cleanup "$USAGE_EC" ""

    # Check device name

    [ 'expr "$DEVICE" : "[/"' -eq 0 ] && cleanup "$NOMNT_EC" "$PROG: Backup device or file name must start with a '/'."

    # Check tape device

    case "$DEVICE" in

        # Fix the block size if $DEVICE is a tape device

        /dev/rmt[0-9]*)

            echo "$PROG: Rewinding tape to begin.........(please wait)"

            tctl -f $DEVICE rewind 2>/null

            ;;

        *)

            ;;

    esac

    # Fix the block size if $DEVICE is a tape device

    /dev/rmt[0-9]*)

        echo "$PROG: Rewinding tape to begin.........(please wait)"

        tctl -f $DEVICE rewind 2>/null

        ;;

        *)

            ;;

    esac

    rm -f ${LIST} ${PIPE} 2>/null

    [ -n "$error" ] && echo "$error"

    trap '' 0 25

    exit "$ec"

}
