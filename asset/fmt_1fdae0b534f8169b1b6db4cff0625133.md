```bash
# Should-Start: modules
# Required-Stop: localnet
# Should-Stop: localfs
# Default-Start: S
# Default-Stop:
# Short-Description: 激活和禁用交换分区。
# Description: 激活和禁用 /etc/fstab 中定义的交换分区。
# X-LFS-Provided-By: LFS
### END INIT INFO
```

```bash
# 激活和禁用交换分区。
# 是在 /etc/fstab 中定义的。
/lib/lsbinit-functions
# /config/
 - - / /etcconfig// ] && . /etcconfig/clock
case "${1}" in
yes|true)
)
start)
  log_info_msg "正在激活交换分区"
  swapon -a
  evaluate_retval
  ;;
stop)
  log_info_msg "正在禁用交换分区"
  swapoff -a
  evaluate_retval
  ;;
restart)
  log_info_msg "正在重启交换分区"
  swapoff -a
  swapon -a
  evaluate_retval
  ;;
status)
  log_info_msg "获取交换分区状态"
  swapon -s
  evaluate_retval
  ;;
*)
  echo "Usage: $0 (start|stop|restart|status)"
  exit 1
;;
esac
exit 0
```


```bash
### BEGIN INIT INFO
# Provides:
# Required-Start: modules
# Should-Start:
# Required-Stop: syslog
# Should-Stop: S
# Default-Start:
# Default-Stop:
# Short-Description: 从硬件时钟存储并恢复时间。
# Description: 启动时，系统时间从 hwclock 获取。关机时也可以设置硬件时钟。
# X-LFS-Provided-By: LFS
### END INIT INFO
```

```bash
/lib/lsb/init-functions
[ -r /etc/sysconfig/clock ] && . /etc/sysconfig/clock
case "${UTC}" in
yes|true)
    CLOCKPARAMS="utc"
;;
no|false)
    CLOCKPARAMS="localtime"
;;
esac
case $1 in
start)
    hwclock --hset ${CLOCKPARAMS} >/dev/null
    ;;
stop)
    log_info_msg "设置硬件时钟..."
    hwclock --systoc ${CLOCKPARAMS} >/dev/null
    evaluate_retval
    ;;
*)
    echo "Usage: ${0} {start|stop}"
    exit 1
;;
esac
```
