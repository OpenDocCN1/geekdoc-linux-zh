```c
/* splitline 将行拆分为字符串数组
#define is_delim(x)((x)==' '||(x)=='\t')
char **splitline(char *line)
/**
* 目的：将按空白分隔的标记拆分为数组
* 返回：以 NULL 结尾的标记数组
* 如果行没有标记则返回 NULL
* 动作：遍历数组，复制
* 注意：可以使用 strtok()，但我们可能需要添加引号
 */
{
    char **args;
    int spots = 0;           /* 表中的槽位 */
    int bufspace = 0;        /* 表中的字节 */
    int argum = 0;          /* 已用槽位 */
    char *cp = line;         /* 字符串中的位置 */
    char *start;
    int len;

    if ( line == NULL )           /* 特殊情况 */
        return NULL;

    args = malloc(BUFSIZ);                /* 初始化数组 */
    bufspace = BUFSIZ;
    spots = BUFSIZ/sizeof(char *);

    while (*cp != '\0')               /* 跳过前导空格 */
        while ( is_delim(cp) ) 
            cp++;
    if (*cp == '\0' )             /* 字符串结束 */

    /* 确保数组有空间 (+1 用于 NULL) */
    if (argum+1 >= spots) {
        args = realloc(args,bufspace+BUFSIZ);
        bufspace += BUFSIZ;
        spots += (BUFSIZ/sizeof(char *));
    }

    /* 标记开始，然后找到单词结束 */
    start = cp;
    len = 1;
    while (*cp != '\0' && !(is_delim(*cp))) {
        len++;
        args[argum++] = newstr(start, len);
    }

    args[argum] = NULL;
    return args;
}
```
