function get_current_and_previous_months() {
    # 使用 "set -A" (ksh) 或 "declare -a" (bash)
    # 月份数组。
    set -A Month
    Month[1]="Jan"
    Month[2]="Feb"
    Month[3]="Mar"
    Month[4]="Apr"
    Month[5]="May"
    Month[6]="Jun"
    Month[7]="Jul"
    Month[8]="Aug"
    Month[9]="Sep"
    Month[10]="Oct"
    Month[11]="Nov"
    Month[12]="Dec"
    current_numeric_month=$(date +%m)
    (( previous_numeric_month = current_numeric_month - 1 ))
    current_alpha_month=$Month[$current_numeric_month]
    previous_alpha_month=$Month[$previous_numeric_month]
}

function get_list_of_available_files() {
    # 开始函数
    count=$(echo $INPUT_DIR/*SALES_DATA_$previous_alpha_month 2>&1|grep -ic 'No such file')
    # $count 是“此类文件”出现的次数。
    # (grep -c 选项为我们返回计数)。
    # 如果计数不是 1，则我们发出警告消息并退出。
    if [ $count -ne 1 ]; then
        print "\n调查：无可用输入文件！正在退出。" >>$REPT_TS5
        exit
    else
        LIST_OF_INPUT_FILES=$INPUT_DIR/*SALES_DATA_$previous_alpha_month
    fi
}
