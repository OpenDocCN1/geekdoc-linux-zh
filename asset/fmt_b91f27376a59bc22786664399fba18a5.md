# 将特殊字符替换为 URL 中的 %xx 十进制

```perl
sub escape
{
  ($_)=@_;
  s/([a-zA-Z0-9\.\-])/uc sprintf("%%02x",ord($1))/eg;
  $_;
}
```

# 将 + 替换为空格并将 %xx 替换为相应的 ASCII 字符

```perl
sub unescape {
  ($_)=@_;
  tr/+/ /;
  s/%%(..)/pack("c",hex($1))/ge;
  $_;
}
```
