import os
import time
import random
import sys
import socket
import subprocess

def MAX_START():
    while True:
        start = random.randint(1, 100)
        print(f"Connecting to {start}...")
        time.sleep(random.uniform(0.1, 0.5))
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(1)
            sock.connect(("127.0.0.1", start))
            print(f"Success!")
            sock.close()
        except:
            print(f"Failed")
