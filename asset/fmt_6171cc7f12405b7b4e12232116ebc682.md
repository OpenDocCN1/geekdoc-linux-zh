# 示例 A-35. 在文本文件中定位拆分的段落

```bash
#!/bin/bash
# find-splitpara.sh
# 查找文件中拆分的段落。
# 标记行号。

ARGCOUNT=1      # 期望参数。
OFF=0           # 状态。
ON=1
WRONGARGS=85

file="$1"      #目标文件名。
linen=1         # 行号。从1开始。
Flag=$OFF       # 空行。

if [ $# -ne "$ARGCOUNT" ]; then
  echo "用法: `basename $0`"
  exit $WRONGARGS
fi

file_read () {
  # 扫描文件模式，然后打印。
  while read line; do
    if [[ "$line" =~ ^[a-z] ]] && [ $Flag -eq $ON ]; then
      # 小写字母开头，后续行。
      echo -n "$linen:"
      echo "$line"
    fi
    if [[ "$line" =~ ^$ ]]; then
      # 如果是空行
      Flag=$ON # 设置标记。
    else
      Flag=$OFF
    fi
    ((linen++))
  done < "$file"
}
file_read
```

# 标准差表示一组数据的一致性。
# 它显示了单个数据点偏离算术平均值的程度，
# 即它们如何“波动”（或聚集）。
# 它本质上是数据点距离平均值的平均偏差距离。

# 计算标准差：
# 找到所有数据点的算术平均值（平均值）。
# 从算术平均值减去每个数据点，# 并平方。
# 将 #2 中所有的单个差平方相加。
# 将 #3 中的和除以数据点的数量。
# 这被称为“方差”。
# #4 的平方根给出标准差。

count=0 # 数据点数量；全局。

 fi

```bash
arith_mean() {

  local rt=0 # 运行和。

  local am=0 # 算术平均值。

  local ct=0 # 数据点数量。

  while read value; # 每次读取一个数据点。

  do

    rt=$(echo "scale=$SC; $rt + $value" | bc)

    ((ct++))

  done

  am=$(echo "scale=$SC; $rt / $ct" | bc)

  echo $am return $ct # 这个函数“返回”两个值！

  # 注意：如果 $ct > 255，这个小技巧将不起作用！

  # 要处理更多数量的数据点，

  # 只需注释掉上面的 "return $ct" 即可。

  echo "\$datafile"

}
```

```bash
mean1=$1 # 算术平均值（传递给函数）。

n=$2 # 数据点数量。

sum2=0 # 平差和（“方差”）。

avg2=0 # $sum2 的平均值。

dev0=0 # 标准差。
```

```bash
while read value; # 每次读取一行数据。

do

  diff=$(echo "scale=$SC; $mean1" | bc)

  # 平均值与数据点之间的差异。

  ndiff2=$(echo "scale=$SC; $diff * $diff" | bc) # 平方。

  sum2=$(echo "scale=$SC; $sum2 + $ndiff2" | bc) # 平方和。

done
```

```bash
avg2=$(echo "scale=$SC; $sum2 / $n" | bc) # 平方平均值。

dev=$(echo "scale=$SC; sqrt($avg2)" | bc) # 平方根

echo $dev # 标准差。
```
