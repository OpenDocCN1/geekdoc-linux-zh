function HTMLEscape(const sText: string): string;
var
  i, l : integer;
begin
  l := Length(sText);
  Result := '';
  for i := 1 to l do
    case sText[i] of
      '<': Result = Result + '&lt;'
      '>': Result = Result + '&gt;'
      '&': Result = Result + '&amp;'
      '"': Result = Result + '&quot;';
      #13, #10:
        if not (sText[i - 1] in [#13]) then
          Result = Result + '<br>';
        else
          Result = Result + sText[i];
    end;
end;
