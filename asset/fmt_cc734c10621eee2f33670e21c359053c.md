# 在启动时创建文件/目录
create_files() {
  # 输入到文件符 9，输出到标准输入（重定向）
  exec 9< /etc/sysconfig/createfiles
  while read name dtype min; do
    case "${name}" in
      "#") continue ;;
    esac
      # 忽略现有文件。
      if [ ! -e "${name}" ]; then
        # 根据其类型创建。
        case "${type}" in
          dir)
            mkdir "${name}"
            ;;
          char)
            mknod "${name}" c ${maj} ${min}
            ;;
          block)
            mknod "${name}" b ${maj} ${min}
            ;;
          pipe)
            mknod "${name}" p
            ;;
          *)
            log_warning "unknown type: ${type}"
            continue
            ;;
        esac
        # 同时设置权限。
        chown "${user}:${group}" "${name}"
        chmod "${perm}" "${name}"
      fi
  done <&9
  # 关闭文件符 9（结束重定向）
  exec 9>&-
  return 0
}

case "${1}" in
start)
  log_info_msg "正在清理文件系统..."
  if [ "${SKIPT_CLEAN}" = "" ]; then
    log_info "/tmp"
    find -xdev -mindep 1 ! -name last_ -delete || failed=1
  fi
  >/var/run/utmp
  if grep -q "${utmp}" /etc/group; then
    chmod 664 /var/run/utmp
    chmod utmp /var/run/utmp
  fi
  (exit ${failed})
  evaluate_retval
  if grep -qv '^#|$' /etc/sysconfig/createfiles 2>/dev/null; then
    log_info_msg "正在创建文件和目录..."
    create_files # 总是返回0
  fi
  exit $failed
  ;;
*)
  echo "Usage: $0 {start}"
  exit 1
esac
