cout << "Press CTRL+C to quit..." << endl;
pinMode(LED_GPIO, OUTPUT); // GPIO17 是一个输出引脚

while(1) {
    digitalWrite(LED_GPIO, HIGH);
    for (int i=0; i<50; i++) {
        digitalWrite(LED_GPIO, LOW);
        for (int i=0; i<49; i++) {
        }
    }
    // 循环等待 ^C 按
    // LED 开启
    // 阻塞延迟黑科技
    // LED 关闭
    // 较短的延迟以平衡
    // 空空比
    // 程序永远不会到达这里
    return 0;
}
