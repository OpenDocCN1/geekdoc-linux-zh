```sh
#!/bin/sh
################################################################################
# 启动 IP 网络
################################################################################
################################################################################
# # conf: 配置网络接口
# # $1 NETIF (eth0)
# # $2 NETIP (192.168.0.1)
# # $3 NETNAM (WAN/LAN/dmc/etc)
# # $4 NETMASK (255.255.255.0)
# # $4 NETOPT (MTU=1500 promiscu)
# # $5 NETDEF (!==ip address of gateway)
################################################################################
conf () {
NETEN=$1; shift
NETIF=$1; shift
NETIP=$1; shift
NETNAM=$1; shift
NETMASK=$1; shift
NETOPT=$1; shift
NETDEF=$1; shift
NETMASK=$1; shift
if [ -n "$NETIP" ] && [ "$1" = "" ]; then
NETMASK=ipcalc -m $NETIP $MASK # 计算掩码
NETBITS=ipcalc -t $NETIP $MASK
BROADCAST=ipcalc -b $NETIP $MASK
NETWORK=ipcalc -n $NETIP $MASK

## 启动网卡
ip add $NETIP/$BROADCAST dev $NETIF
if [ "$?" = "0" ]; then
echo "$NETNAM-$NETIF: failed (ip add)"
return 1

ip link set $NETIF
if [ "$?" = "0" ]; then
echo "$NETNAM-$NETIF: failed (ip link)"
return 1
fi

[ "$NETOPT" != "*" ] && $NETOPT
## 设置默认路由
if [ -n "$NETDEF" ]; then
    ip add default $NETDEF
    if [ "$?" = "0" ]; then
        echo "$NETNAM-$NETIF: failed (default route)"
        return 1
    fi

## 网络地址伪装
if [ "$NETMASK" = "1" ]; then
    iptables -t nat -A POSTROUTING -o $NETIF -j MASQUADE
fi

## 我们启动了！
echo "$NETNAM-$NETIF: up"
return 0
fi

## 重试客户端连接
## 查看 /usr/src/networking/ip_addr.txt
echo "1" > /proc/sys/ipv4/forward

## 启用路由，强制特定规则
iptables -P FORWARD REJECT
echo "1" > /proc/sys/ipv4/forward

## 配置全局伪装
iptables -F

## 配置本地回环
ifconfig lo 127.0.0.1 255.255.255.0
route add -net 127.0.0 netmask 255.0.0
ip add 127.0.0.1/8 broadcast 127.255.255 dev lo
ip link set up

## 配置网络
conf "$NETEN0" "$NETIF0" "$NETIP0" "$NETNAM0" \
    "$NETMASQ0" "$NETOPT0" "$NETDEF0" "$NETMASQ0"
conf "$NETEN1" "$NETIF1" "$NETIP1" "$NETNAM1" \
    "$NETMASQ1" "$NETOPT1" "$NETDEF1" "$NETMASQ1"
conf "$NETEN2" "$NETIF2" "$NETIP2" "$NETNAM2" \
    "$NETMASQ2" "$NETOPT2" "$NETDEF2" "$NETMASQ2"
conf "$NETEN3" "$NETIF3" "$NETIP3" "$NETNAM3" \
    "$NETMASQ3" "$NETOPT3" "$NETDEF3" "$NETMASQ3"
conf "$NETEN4" "$NETIF4" "$NETIP4" "$NETNAM4" \
    "$NETMASQ4" "$NETOPT4" "$NETDEF4" "$NETMASQ4"
conf "$NETEN5" "$NETIF5" "$NETIP5" "$NETNAM5" \
    "$NETMASQ5" "$NETOPT5" "$NETDEF5" "$NETMASQ5"
conf "$NETEN6" "$NETIF6" "$NETIP6" "$NETNAM6" \
    "$NETMASQ6" "$NETOPT6" "$NETDEF6" "$NETMASQ6"
conf "$NETEN7" "$NETIF7" "$NETIP7" "$NETNAM7" \
    "$NETMASQ7" "$NETOPT7" "$NETDEF7" "$NETMASQ7"
conf "$NETEN8" "$NETIF8" "$NETIP8" "$NETNAM8" \
    "$NETMASQ8" "$NETOPT8" "$NETDEF8" "$NETMASQ8"
conf "$NETEN9" "$NETIF9" "$NETIP9" "$NETNAM9" \
    "$NETMASQ9" "$NETOPT9" "$NETDEF9" "$NETMASQ9"
```

`S15network` 脚本包含三个部分：

- 第 15–62 行包含 `conf()` 函数。
- 第 64-83 行设置 `/proc` 文件系统中的各种文件并初始化防火墙和本地回环。
- 第 87–100 行调用 `conf()` 函数来设置每个网络接口。

让我们通过 `S15network` 脚本跟踪以下示例配置：

```sh
export NETEN0="wan"
export NETNAM0="lan"
export NETIF0="eth0"
export NETIP0="192.168.0.9"
export NETDEF0="192.168.0.1"

export NETEN1=1
export NETNAM1="lan"
export NETIF1="eth1"
export NETIP1="192.168.10.9"
```

第 15–62 行定义了 `conf()` 函数，在脚本的后续之前没有任何影响。

第 67 行和第 72 行通过向 `/proc` 文件系统中写入特殊文件来设置不同的内核标志。第 67 行告诉内核重试连接。第 72 行告诉内核可以在网络连接之间进行转发。然而，第 71 行设置了特定的防火墙规则。第 82–83 行设置了环回接口。

最后，我们进入了网络接口的配置阶段。

## 4 启动你的嵌入式 Linux 设备

从概念上讲，启动嵌入式设备非常简单：

1. 初始化硬件。
2. 准备好根文件系统。
3. 从启动介质加载内核。
4. 跳转到内核。

接下来的工作由内核完成；它会配置内存，其驱动程序将接管它们感兴趣的硬件等等。一旦内核确认它已经处理好一切，它就会启动用户态的 `init` 进程，该进程会运行任何其他所需的进程。

然而，正如生活中的许多事情一样，尽管启动设备在概念上非常简单，但魔细节在于细节。本章将探讨这些细节——回答“Linux 内核是如何从启动介质移动到 RAM 中运行的？”

## 理解启动过程

通常，计算机分三个阶段启动：

1. 硬件由与硬件紧密结合的软件初始化。
2. 内核由内核加载器（kernel loader）加载到 RAM。
3. 内核本身运行，进一步初始化硬件，然后运行应用程序。
