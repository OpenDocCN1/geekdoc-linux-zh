    then print "ZIPCODE must not be blank." >>list_of_errors
else 
    format_users_input "$zipcode"
    zipcode=$(revised)
    echo "$zipcode | grep "[0-9]{5} \-?[0-9]{4}" >/dev/null 2>&1

        then print "zipcode" >>$list_of_errors

```

## 关于模式的说明：`"[0-9]{5}[ -]?[0-9]{4}"`

# 以下邮政编码是被接受的：

# 12345, 12345, 12345 6789, 123456789

# 邮政编码必须以 `[0-9]{5}` 开头 (`^`)

# 后面可以可选 (`?`) 空格或连字符 (`\-`)：`[ -]`

# 邮政编码可以可选 (`?`) 以 4 个数字结尾：`[0-9]{4}$`

# END_function
