    if (sockfd < 0) {

        exit(1);
    }

        exit(1);

    while (1) { /* 等待消息和控制台 */
        FD_ZERO(&fd_set);
        FD_SET(STDIN, &fd_set);
        FD_SET(sockfd, &fd_set);
        printf("等待中...\n");
        select(-1, &fd_set, NULL, NULL, NULL);
        if (FD_ISSET(sockfd, &fd_set)) {
            printf("从服务器读取的消息:\n");
            nread = read(sockfd, buf, SIZE);
            if (nread < 0) {
                perror("read");
            } else if (nread == 0) {
                close(sockfd);
                exit(0);
            } else {
                printf("%s\n", buf);
            }
        }
        if (FD_ISSET(STDIN, &fd_set)) {
            printf("从用户读取:\n");
            nread = read(STDIN, buf, SIZE);
            if (nread < 0) {
                perror("read");
            } else {
                write(sockfd, buf, nread);
            }
        }
    }
