### 示例 32-9。运行多个进程（在 SMP 机器上）

```bash
#!/bin/bash
# parent.sh
# 在 SMP 机器上运行多个进程
# 作者: Tedman Eng

# 这是两个脚本中的第一个，
# 两个都必须存在于当前目录。

LIMIT=$1                # 要启动的进程总数
NUMPROC=4              # 并发线程数（fork）
PROCID=1               # 起始 ID
echo "My PID is $$"

function start_thread() {
  if [ $PROCID -le $LIMIT ]; then
    ./child.sh $PROCID
  else
    echo "Limit reached."
    wait
  fi
}

while [ $NUMPROC -gt 0 ]; do
  start_thread;
  let "$NUMPROC--"
done

while true
do
  trap "start_thread" SIGTERM
done
```
