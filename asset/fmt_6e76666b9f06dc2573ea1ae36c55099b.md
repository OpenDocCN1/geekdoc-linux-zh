function TMazeTown.CheckForwardMove: Boolean;
var
  X, Y: Integer;
begin
  X := FXSpot;
  Y := FYSpot;
  case FDirection of
    North: Dec(Y);
    South: Inc(Y);
    East: Inc(X);
    West: Dec(X);
  end;
  if FMap.Tiles[X, Y] <> 1 then begin
    FXSpot := X;
    FYSpot := Y;
    FChanged := True;
    Result := True
  else
    FChanged := False;
    Result := False;
  end;
end;
