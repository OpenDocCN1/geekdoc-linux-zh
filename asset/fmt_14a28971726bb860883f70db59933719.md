asmlink sys_bind_int(int fd, struct sockaddr *ummaddr, int addlen) {
    struct socket *sock;
    char address[MAX_SOCKDR_ADDR];

    int err;

    if((sock = sockfd_lookup(fd,&err))!=NULL) {
        if((err=move_addr_to_kernel(ummaddr,addrlen,addlen))==0)
