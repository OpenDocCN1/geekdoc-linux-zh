# start_daemon
# 用法：start_daemon [-f] [-n nicelevel] [-p pidfile] pathname [args...]
# 目的：作为守护进程运行程序
# 输入：
#   -f：如果已经在运行，则运行。
#   -n nicelevel：指定 nice 值。参见 man nice(1)
#   -p pidfile：使用指定的文件确定 PID。
#   pathname：指定程序的完整路径
#   args：传递给程序的额外参数
# 返回值（定义 LSB 退出码）：
#   0 - 程序运行或 OK
#   1 - 普通或未指定的错误
#   2 - 无效参数

################################################
# 处理参数
while true; do
  case "$1" in
    -f)
      force="1"
      shift 1
      ;;
    -n)
      nice="$2"
      shift 2
      ;;
    -p)
      pidfile="$2"
      shift 2
      ;;
    *)
      program="$1"
      break
      ;;
  esac
done

# 检查程序是否有效
if [ ! -e "$program" ]; then return 5; fi

if [ -z "$force" ]; then
  # 通过发现确定 pid

  retval=$?

  retval=$?
fi

    return 0

  3)

    ;;

    return 1

# killproc
# 用法：killproc [-p pidfile] pathname [signal]
# 目的：向运行进程发送控制信号
# 输入：
#   -p pidfile，指定的 PID 文件
#   pathname，指定程序的路径
#   signal，向路径发送此信号
# 返回值（由 LSB 退出码定义）：
#   0 - 程序已停止/已停止或运行程序收到信号并成功停止
#   1 - 普通或未指定的错误
#   2 - 参数无效或过多
#   5 - 程序未安装
#   7 - 程序未运行且提供了信号
################################################
killproc()
{
local pidfile
local program
local prefix
local signal="-TERM"
local fallback="-KILL"
local nosig
local pidlist
local retry
local pid
local delay="30"
local piddead
local dtime

# 处理参数
while true; do
  case "${1}" in
    -p)
      pidfile="${2}"
      shift 2
    *)
      program="${1}"
      if [ -n "${2}" ]; then
        signal="${2}"
        fallback=""
      else
        nosig=1

        # 额外参数错误
        if [ -n "${3}" ]; then
        return 2
        else
        break
        fi
      fi
  esac
done

# 检查程序是否有效
if [ ! -e "${program}" ]; then return 5; fi

# 检查信号是否有效
check_signal "${signal}"
if [ "${?}" -ne "0" ]; then return 2; fi

# 获取 pid 列表
if [ -z "${pidfile}" ]; then
  # 通过发现确定 pid
  pidlist=$(pidofproc "${1}")
  retval=$?
else
  # PID 文件包含所需的 PID
  # 注意 LSB 要求必须将路径传递给 pidproc，
  # 但当前实现或标准并不使用它。
  pidlist=$(pidofproc -p "${pidfile}" "${1}")
  retval=$?
fi

# 仅返回值
# init 脚本（或发行版函数）有责任记录消息！
case "${retval}" in
  0)
    # 程序运行正常
    # 此处什么都不做，让 killproc 处理。
    ;
    ;;
  1)
    # 程序未运行，但存在无效的 PID 文件
    # 删除 PID 文件。
    progname=${program##*/}
    if [ -e "run/${progname}.pid" ]; then
      pidfile=run/${progname}.pid
      rm -f "${pidfile}"
    fi
    ;;
  3)
    # 程序未运行且不存在 PID 文件
    # 如果没有传递信号，则视为成功。
    if [ -n "${nosig}" ]; then
      return 0
    else
      return 7
    fi
    ;;
  *)
    # 其他返回的状态值将被解释
    # 并作为未指定的错误返回。
    return 1
    ;;
esac

# 对退出信号和控制信号执行不同的操作
check_sig_type "${signal}",
}
