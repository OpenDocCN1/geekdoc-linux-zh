### `start_daemon`

```bash
###########################################################
# start_daemon
# 用法：start_daemon [-f] [-n nicelevel] [-p pidfile] pathname [args...]
# 目的：作为守护进程运行程序
# 输入：
# -f：已经在运行
# -n nicelevel：指定 nice 值。见 nice(1)
# -p pidfile：使用指定的文件确定 PID
# pathname：指定程序的完整路径
# args：传递给程序的参数
# 返回值（由 LSB 退出码定义）：
# 0 - 程序运行或服务正常
# 1 - 通用或未指定的错误
# 2 - 参数无效或过多
# 5 - 程序未安装
###########################################################
```

```bash
start_daemon()
{
local force=""
local nice=""
local pidfile=""
local pidlist=""
local retval=""
```

```bash
# 处理参数
while true;
  case "${1}" in
    -f)
      force="1"
      shift 1
      ;;
    -n)
      nice="${2}"
      shift 2
      ;;
    -p)
      pidfile="${2}"
      shift 2
      ;;
    -*)
      return 2
      ;;
*)
```

```bash
program="$(1)"
break
  ;
esac
done
# 检查程序是否有效
if [ -e "$program" ]; then return 5; fi
# 执行
if [ -z "$force" ]; then
if [ -z "$pidfile" ]; then
# 通过发现确定 pid
pidlist=pidofproc "$(1)"
retval="$?"
else
# PID 文件包含所需的 PID
# 注意：根据 LSB 要求，必须给 pidofproc 传递路径
# 然而，当前的实现或标准并不使用它。
pidlist=pidofproc -p "$pidfile" "$(1)"
retval="$?"
fi
fi
# 仅返回值
# 记录消息是初始化脚本（或发行版函数）的责任！in
case "$retval" in
0) # 程序已正确运行，这是一个
# 成功的启动。
return 0
  ;
1) # 程序未运行，但存在无效 PID 文件
# 删除 PID 文件并继续
rm -f "$pidfile"*
  ;;
3) # 程序未运行且不存在 PID 文件
# 此处什么都不做，让 start_daemon 继续。
  ;;
*) # 状态值返回的其他值将不被解释
# 并作为未指定的错误返回。
return 1
  ;
esac
fi
# 执行启动！
nice -n "$nice" "$@"
}
```
