```perl
#!/usr/bin/perl

# 使用此程序时要小心！
# 这将删除所有包含给定单词的行
# 用法: remove <word> <file>
$word=ARGV[0];
$file=ARGV[1];
if ($file) {
    # 打开文件读取
    open (FILE, "\$file") or die "Could not open file: $!"; @lines=<FILE>
    close FILE;
    # 打开文件写入
    open (FILE, ">\$file") or die "Could not open file for writing: $!";
    for (@lines) {
        print FILE unless /$word/;
    }
    # 结束 FILE
    close FILE;
} else {
    print "Usage: remove <word> <file>\n";
} # 结束 if...else
```

```perl
#!/usr/bin/perl

# 将 post 数据加载到 @post 
open (POST, "post.file*");
@post = <POST>;
close POST;
# 导入 NNTP 模块
Net::NNTP;
$NNTPost = 'news';
# 连接连接到远程主机
# 失败时打印错误消息
$nntp = Net::NNTP->new($NNTPost);
or die "Cannot connect to $NNTPost: $!";
# $nntp->debug(1);
$nntp->post();
or die "Could not post article: $!";
# 发送帖子的头部
$nntp->datasend("Newgroups: news.announce\n");
$nntp->datasend("From: ADMIN <root>\n");
$nntp->datasend("\n\n");
# 遍历 $post 数组中的每一行并发送它
for ($post) {
$nntp->datasend($);
}
# 结束 for
$nntp->quit();
```
