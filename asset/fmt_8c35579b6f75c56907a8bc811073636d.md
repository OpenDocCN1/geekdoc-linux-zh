```c
tokenize ()
{
    char *ptr = line; /* 指向输入缓冲区 */
    char token [MAX_TOKEN_LENGTH]; /* 持有当前 token */
    char *tptr = token; /* 指向当前字符 */
    tIndex = 0; /* 全局：指向当前 token */

    /* 将当前输入行拆分为多个 token */
    while (TRUE)
        {
        tptr = token;
        while (*ptr == ' ') ++ptr; /* 跳过前导空格 */
        if (*ptr == NULL) break; /* 行结束 */
        do
            {
                *tptr++ = *ptr++;
            }
        while (*ptr != ' ' && *ptr != NULL);
        *tptr = NULL;
        strcpy(tokens[tIndex++], token); /* 存储 token */
        }

    /* 在 token 列表末尾放置一个换行 token */
    strcpy(tokens[tIndex++], "\n");
    toldCount = tIndex; /* 记住 token 总计数 */
    tIndex = 0; /* 重置 token 索引到列表开头 */
}
```
