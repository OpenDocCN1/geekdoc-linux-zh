    speech = NULL;
    memset(file, 0, 1024);
memset( spbBuffer, 0, 2048 * sizeof( WCHAR ));
state = idle;
SetMessageQueue(96);
CoInitialize(NULL);
speech = new CTTSMode();
wcopy(mode.szSpeaker, L"Mary");
if ((hRes = speech->Init ( &mode, NULL )) != S_OKERROR) {
cantalk = true;

TTSI_NAMelen ));

#endif // TTS

}

{
#ifdef TTS
if(cantalk == true || speech != NULL) {
delete speech;
}
CoUnitialize();
#endif
delete midi;
}

{
#ifdef TTS

state = Speech;

wcsncpy( spksbuffer,
(WCHAR *) (msgs.unicode(), 1023);
spksbuffer[strlen(p)] = 0;
speech->Speak( (PCWSTR)spksbuffer, true, NULL );
#endif
}

}
