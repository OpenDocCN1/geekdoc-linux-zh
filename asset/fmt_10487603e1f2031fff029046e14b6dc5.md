#### 示例 14-46. 生成质数

```bash
#!/bin/bash
# primes2.sh

# 以快速且简单的方法生成质数，
# 而无需求助于复杂的算法。

CEILING=10000 # 1 到 10000
PRIME=0
E_NOTPRIME=

is_prime () {
 local factors
factors=($(factor $)) # 将 `factor` 的输出加载到数组。
 if [ -z "${factors[2]}" ]; then
  # "factors" 数组的第三个元素：
  # ${factors[2]} 是参数的第二个因数。
  # 如果为空，则说明没有第二个因数，
  # 那么该参数就是质数。
  return $PRIME # 0
else
  return $E_NOTPRIME # null
fi
}

echo
for n in $(seq $CEILING);
do
  if is_prime $n
then
    printf %5d $n
  fi    ^ # 每个数字占用五个位置。
done  # 如果需要更高的 $CEILING，根据需要向上调整。

echo
exit
```

```bash

#!/bin/bash

######################################################

# Shell: base.sh - 将数字打印到不同的进制 (Bourne Shell)

# Author : Heiner Steven (heiner.steven@od.de)

# Date : 07-03-95

# Category : Desktop

# ID: base.sh in 2000/02/06 19:55

# 上行是 RCS 信息。

######################################################

# 描述

# 修改: 修复了当 0xb 输入时发生的错误

# 经脚本作者许可在 ABS 指南中使用。

# ABS 作者添加的注释。

NOARGS=8

PN=$(basename "$0") # 程序名

VER=$1 | cut -d ' ' -f2 # 版本号：

用法: [数字]

如果未给出数字，则将从标准输入读取数字。

数字可以是：

  二进制 (base 2)        以 0 开头（例如 0b1100）

  八进制 (base 8)         以 0 开头（例如 014）

  十六进制 (base 16)     以 0 开头（例如 0xc）

  十进制 (base 10)       其他

)

Msg () {

  for i in "$@";

  do echo "$PN: $i" >&2

  done

}

Fatal () {

  Msg "$0"; exit 6;

}

PrintBases () {

  # 确定数字的进制

  for i in "$@";

do # 处理命令行参数

    # 移除前缀，将十六进制字母转换为大写（bc 需要）。

    number="$i" | sed 's/^0[xX]//g' | tr '[a-f]' '[A-F]'

    # 将数字转换为十进制

    dec=$(echo "obase=$base; $number" | bc) # bc 是计算器工具。

    case $dec in

      [0-9]*) # 数字正确

        continue; 

      *) # 错误：忽略

    esac

    # 在一行内打印所有转换。

    # = <<'document' 将命令列表传递给 bc

    echo <<<

      obase=16; "hex"; $dec

      obase=10; "dec"; $dec

      obase=8; "oct"; $dec
```
