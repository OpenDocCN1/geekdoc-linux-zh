这反映了服务器，只是没有指定 `IPC_CREAT`，以表示我们不是在创建共享内存段，而是在访问它。

注意，这是对该机制工作原理的非常简要的概述，在生产版本中，利用接口暴露的信号量和其他机制来同步对共享数据的访问是明智的。

这种方法高度地抽象了底层机制，但总体方法与任何其他共享内存方法相同，即建立一个共享文件，多个进程映射该文件以访问共享内存。

这在内核中具体实现，其中系统调用 `shmget()` 最终调用 `newseg()`，后者使用 `shm_kernel_file_setup()` 生成一个未链接的 `tmpfs` 文件（即存在于 `tmpfs` 但任何目录中都没有文件项的文件）。

## 14.7.2 POSIX 共享内存

POSIX 共享内存是 Linux 中可用的一种更新的内存共享方式，旨在作为 System V 方法的继承者。同样，除了进程间简单的共享内存外，还提供了在使用 IPC 机制进行共享内存所需的许多构建模块。

这类似于 System V 接口，只是使用文件描述符通过 `mmap()` 映射到共享进程中，并在 `/dev/shm/<name>` 处建立一个 `shm`（即由内存支持的）文件，其中 `name` 在 `shm_open()` 中指定。

```c
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <sys/shm.h>
#include <unistd.h>

#define SHM_NAME "test"
#define SIZE (4096)
#define NUM_LOOPS (10)

int main(void) {
    int i, fd;
    char *ptr;

    fd = shm_open(SHM_NAME, O_CREAT | O_RDWR, 0666);
    if (fd < 0) {
        perror("shm_open");
        return EXIT_FAILURE;
    }

    if (ftruncate(fd, SIZE)) {
        perror("ftruncate");
        return EXIT_FAILURE;
    }

    ptr = mmap(NULL, SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (ptr == MAP_FAILED) {
        perror("mmap");
        return EXIT_FAILURE;
    }

    memset(ptr, 0, SIZE);

    for (i = 0; i < NUM_LOOPS; i++) {
        ptr[i] = 'a' + i;
        printf("server: %s\n", ptr);
        sleep(1);
    }

    /* 等待完成 */
    if (shm_unlink(SHM_NAME)) {
        perror("shm_unlink");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

注意，与 System V 共享内存类似，必须在 `shm_open()` 中指定 `O_CREAT` 以表明该函数创建文件（如果它不存在）。文件的大小通过 `ftruncate()` 设置。共享文件通过 `shm_unlink()` 移除。

## 14.7.3 fork 进程间的匿名共享内存

当进程 fork 出子进程时，子进程以写时复制的方式获取内存的访问权限，这意味着默认内存是不共享的。

然而，通过 `mmap()` 使用匿名共享内存映射，内存可以在父子进程之间轻松共享：

```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/mman.h>
#include <sys/wait.h>

#define NUM_LOOPS (10)

int main(void) {
    int i;
    pid_t pid;
    const long page_size = sysconf(_SC_PAGESIZE);

    char *ptr = mmap(NULL, page_size, PROT_READ | PROT_WRITE,
                     MAP_ANON | MAP_SHARED, -1, 0);
    if (ptr == MAP_FAILED) {
        perror("mmap");
        return EXIT_FAILURE;
    }

    pid = fork();
    if (pid == -1) {
        perror("fork");
        return EXIT_FAILURE;
    }

    /* 子进程。 */
    if (pid == 0) {
        for (i = 0; i < NUM_LOOPS; i++) {
            ptr[i] = 'a' + i;
            printf("child: %s\n", ptr);
            sleep(1);
        }
        return EXIT_SUCCESS;
    }

    /* 父进程。 */
    for (i = 0; i < NUM_LOOPS; i++) {
        ptr[i] = 'a' + i;
        printf("parent: %s\n", ptr);
        sleep(1);
    }

    if (waitpid(pid, NULL, 0) == -1) {
        perror("waitpid");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

列表 14-39：跨 fork 的匿名共享映射示例

在这里我们通过组合 `MAP_ANON` 和 `MAP_SHARED` 建立了一个匿名共享映射。注意，在底层，这实际上创建了一个未链接的 tmpfs 文件（即存在于文件系统的文件，详见第 5.0.2 节）。

因此，当执行 `fork()` 时，该指针会在父进程和子进程之间共享。此示例代码遍历字母，并在父进程和子进程中均输出，以证明内存实际上是共享的。

## 14.7.4 通过 memfd 共享内存

进程间共享内存的另一种方法是生成一个 memfd，它会生成一个未链接的共享 tmpfs 文件，类似于上述共享匿名 `mmap()` 和 System V 共享内存方法，允许对文件描述符进行进一步的特殊操作。

检查此方法的简单服务器：

```c
#define _GNU_SOURCE
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <unistd.h>

#define NAME "test"
#define SIZE (4096)
#define NUM_LOOPS (10)

int main(void) {
    int i, fd;
    pid_t pid;
    char *ptr;

    fd = memfd_create(NAME, MFD_ALLOW_SEALING);
    if (fd < 0) {
        perror("memfd_create");
        return EXIT_FAILURE;
    }

    if (ftruncate(fd, SIZE)) {
        perror("ftruncate");
        return EXIT_FAILURE;
    }
}
```

列表 14-40：memfd 共享映射服务器示例

这通过 `memfd_create()` 建立了一个文件描述符，指定 `MFD_ALLOW_SEALING` 允许该文件描述符被密封，从而允许对对其执行的操作进行限制。

然后我们通过 `ftruncate()` 建立共享 RAM 内存文件的大小，在此之前使用 `F_SEAL_GROW`（一种禁止共享文件大小增长的封印）和 `F_SEAL_SEAL`（禁止客户端进一步调整封印）。

最后，我们能够对文件描述符进行 `mmap()`，然后输出该文件描述符在进程 `proc fd` 目录中的条目位置。为了便于说明，这是一个简化的示例；在实际操作中，你会使用另一种 IPC 方法（如 UNIX socket）来传输文件描述符。

最后，我们可以将此路径传递给客户端：

```c
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <unistd.h>

#define SIZE (4096)
#define NUM_LOOPS (10)

int main(int argc, char **argv) {
    int i, fd;
    char *ptr;

    if (argc < 2) {
        fprintf(stderr, "usage: %s <fd path>\n", argv[0]);
        return EXIT_FAILURE;
    }

    fd = open(argv[1], O_RDWR);
    if (fd == -1) {
        perror("open");
        return EXIT_FAILURE;
    }

    ptr = mmap(NULL, SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (ptr == MAP_FAILED) {
        perror("mmap");
        return EXIT_FAILURE;
    }

    close(fd);

    for (i = 0; i < NUM_LOOPS; i++) {
        ptr[i] = 'a' + i;
        printf("client: %s\n", ptr);
        sleep(1);
    }

    return EXIT_SUCCESS;
}
```

列表 14-41：memfd 共享映射客户端示例

这只是映射了文件描述符并像访问任何文件一样访问它，修改映射中的数据，以便我们可以观察服务器和客户端更新数据。
