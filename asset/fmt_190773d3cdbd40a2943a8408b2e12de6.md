### 77.1 人工反编译

在IDA中，程序的具体指令如下：

```
sub_401510	proc near
	;ECX = input
	mov	rdx, 5D7E0D1F2E0F1F84h
	mov	rax, rcx			;input
	imul	rax, rdx
	mov	rdx, 388D76AEE8CB1500h
	mov	ecx, eax
	and	rcx, 0Fh
	ror	rax, cl
	xor	rax, rdx
	mov	rdx, 0D2E9EE7E7B3C4285Bh
	mov	ecx, eax
	and	ecx, 0Fh
	rol	rax, cl
	lea	r8, [rax+rdx]
	mov	rdx, 8888888888888888h
	mov	rax, r8
	mul	rdx
	shr	rdx, 5
	mov rax, rdx
	lea rcx, [r8+rdx*4]
	shl rax, 6
	sub rcx, rax
	mov rax, r8
	rol rax, cl ; EAX = output
	retn
sub_401510 endp
```

EC寄存器传递函数的第一个参数，由此可以判断是GCC编译的程序。

如果您手头没有Hex-Rays类的反编译程序，或者您根本不相信这些自动化工具，那么您可以自己进行反编译。在进行反编译时，可以将CPU寄存器当作C语言的变量，然后把汇编语言直接翻译为等效的C指令。例如，上述程序可反编译为：

```
uint64_t f (uint64_t input)
{
    uint64_t rax, rbx, rcx, rdx, r8;
    cx=input;
    rdx=0x5D7E0D1F2E0F1F84;
    rax=rcx;
    rax*=rdx;
    rdx=0x388D76AEE8CB1500;
    ecx=eax;
    ecx&=0xF;
    ror(rax, ecx);
    rax^=rdx;
    rdx=0xD2E9EE7E7B3C4285B;
    ecx=eax;
    ecx&=0xF;
    rol(rax, ecx);
    r8=rax+rdx;
    rdx=0x8888888888888888;
    rax=r8;
    mul(rdx);
    rdx>>=5;
    rax=rdx;
    rcx=[r8+rdx*4];
    shl(rax, 6);
    sub rcx, rax;
    rax=r8;
    rol(rax, cl);
    return rax;
}
```
