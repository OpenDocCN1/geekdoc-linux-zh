
# 输入来自标准输入

```bash
if (( $# == 1 ));
then
    # 输入来自标准输入
    PATTERN="$1" # 要高亮的模式
    FILENAME="" # 为 FILENAME 赋值为空值
elif (( $# == 2 ));
then
    # 输入来自 $FILENAME 文件
    PATTERN="$1" # 要高亮的模式
    FILENAME="$2" # 用作输入的文件
    # 该文件是否存在一个“普通”文件？
    if [ ! -f "$FILENAME" ];
    then
        echo -e "\nERROR: $FILENAME 不存在..."
        usage
        exit 2
    fi
    # 文件是否为空？
```

# 在文件中是否存在模式

```bash
grep -q "$PATTERN" "$FILENAME"
if [ $? -eq 0 ]
then
    # 在文件中找到模式
    :
else
    # 在文件中未找到模式
    echo -e "\nERROR: $PATTERN 未在文件中找到"
    exit 2
fi
```
