
# 注意：由于可能的远程日志配置，`syslogd` 不在运行级别（`runlevel`）2 中启动

```bash
. /lib/lsb/init-functions

case "${1}" in
start)
  log_info_msg "启动系统日志守护进程..."
  parms=${SYSLOGD_PARMS+-m 0}
  start_daemon=$sbin/syslog $parms
  evaluate_retval
  log_info_msg "启动内核日志守护进程..."
  start_daemon=$sbin/klogd
  evaluate_retval
  ;;
stop)
  log_info_msg "停止内核日志守护进程..."
  killproc /sbin/klogd
  evaluate_retval
  log_info_msg "停止系统日志守护进程..."
  killproc /sbin/syslogd
  evaluate_retval
  ;;
reload)
  log_info_msg "正在重载系统日志守护进程配置文件..."
  pid=`pidofproc syslog`
  kill -HUP "$pid"
  evaluate_retval
  ;;
restart)
  $(0) stop
  sleep 1
  $(0) start
  ;;
status)
  statusproc /sbin/syslogd
  statusproc klogd
  ;;
*)
  echo "用法: $(0) {start|stop|reload|restart|status}"
  exit 1
  ;;
esac

exit 0

# End syslogd
```

```bash
case "$1" in
start)
  # 启动所有接口
  for file in /etc/sysconfig/network/ifcfg-* do
    if [ -f "$file" ]; then
      source "$file"
      ifup "$INTERFACE"
    fi
  done
  ;;
stop)
  # 停止所有接口
  for file in /etc/sysconfig/network/ifcfg-* do
    if [ -f "$file" ]; then
      ifdown "$INTERFACE"
    fi
  done
  ;;
restart)
  $(0) stop
  sleep 1
  $(0) start
  ;;
status)
  ifconfig -a
  ;;
*)
  echo "用法: $(0) {start|stop|restart|status}"
  exit 1
  ;;
esac
```

```bash
/lib/lsb/init-functions
case "${1}" in
stop)
  log_info_msg "向所有进程发送 TERM 信号..."
  for pid in /var/run/*.pid; do
    if [ -f "$pid" ]; then
      kill -15 `cat "$pid")`
    fi
  done
  sleep 2
  log_info_msg "向所有进程发送 KILL 信号..."
  for pid in /var/run/*.pid; do
    if [ -f "$pid" ]; then
      kill -9 `cat "$pid")`
    fi
  done
  ;;
*)
  echo "用法: $(0) {stop}"
  exit 1
  ;;
esac

exit 0
```
