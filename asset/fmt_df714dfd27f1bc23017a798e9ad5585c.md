# `check_sig_type()` { -signal }

# 用法: `check_signal` -{signal} | {signal}

# 目的: 检查信号是程序终止信号还是控制信号

#   LSB 草案中未定义此项，但需要检查

#   信号以确定它们是意用于结束程序

#   还是仅仅为了控制程序。

# 输入: 接受格式为 - (signal) 或 {signal} 的单个字符串值

# 返回值:

#   0 - 信号用于程序终止

#   1 - 信号用于程序控制

#################################################################

```bash
check_sig_type() {
    local valsig
    # 终止信号列表（限制为通常使用的项目）
    valsig="-ALRM -INT -KILL -TERM -STOP -ABRT -QUIT -2 -3 -6 -9 -14 -15"
    echo "${valsig}" | grep - "${1}" >/dev/null
    if [ "${?}" -eq "0" ]; then
        return 0
    else
        return 1
    fi
}
```
