```csharp
private string parseDNSName(string URL) {
Regex search = new Regex("\\\\\\\/(?<DNSNAME>[\\.\\w]+)\\\\\\\\\/\\\\\\//");
Match m = search.Match(URL);
if (m.Success) {
string DNSName = m.Result("${DNSNAME}");
if (DNSName.Length > 0) {
return DNSName;
}
}
return "*";
}
```
