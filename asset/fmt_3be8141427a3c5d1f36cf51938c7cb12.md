case "$1" in
start)
  log_info_msg "正在重试 uevents，如果有的话..."
  # 与 udev-186 一样，--run 选项不再有效
  # rundir=$(sbin/udevadm info --run)
  rundir=/run/udev
  # 来自 Debian: "复制在 / 挂载前生成的规则"
  # 读写:
  for file in ${rundir}/tmp-rules--*; do
    dest=${file##*/tmp-rules--}
    [ "$dest" = '' ] && break
    cat "$file" > /etc/udev/rules.d/$dest
    rm -f "$file"
  done
  # 重新触发可能失败的 uevents，
  # 于它们现在可以成功
  /bin/sed -e 's///' /etc/sysconfig/udev_retry | /bin/grep -v '^$' | \
  while read line ; do
    for subsystem in $line ; do
      /sbin/udevadm trigger --subsystem=$subsystem --action=add
    done
  done
  # 现在等待 udev 处理我们触发的 uevents
  if [ $is_true = "$OMIT_UDEV_RETRY_SETTLE" ]; then
    /sbin/udevadm settle
  fi
  evaluate_retval
  ;;
  echo "用法 $0 (start)"
  exit 1
  ;;
esac
exit 0

# 在启动时创建文件/目录的函数
create_files()
{
    # 输入到文件描述符 9 并输出到标准输入（重定向）
    exec 9< /etc/sysconfig/createfiles

    while read name type user maj
    do
        # 忽略注释和空行
        case "${name}" in
            ""|#*") continue;
        esac

        # 忽略已存在的文件
        if [ ! -e "${name}" ]; then
            # 根据类型创建
            case "${type}" in
                dir)
                    mkdir "${name}"
                    chmod 755 "${name}"
                    ;;
                dev)
                    case "${type}" in
                        char)
                            mkod "${name}" c "${maj}"
                            ;;
                        block)
                            mkod "${name}" b "${maj}"
                            ;;
                        pipe)
                            mkod "${name}" p "${maj}"
                            ;;
                    esac
                *)
                    log_warning_msg "未知类型: ${type}"
                    ;;
            esac
        else
            # 设置权限
            chown "${user}" "${name}"
            chmod "${perm}" "${name}"
        fi
    done
    # 关闭文件描述符 9
    exec 9<&
}

case "$1" in
start)
  log_info_msg "清理文件系统:"
  if [ "$CLEAN" = "" ]; then
    log_info_msg "/tmp"
    cd /tmp &&
    find . -xdev -mindepth 1 -name + -delete || failed=1
  fi
  if grep -q "^utmp" /run/utmp; then
    chmod 664 /run/utmp
    chgrp utmp /run/utmp
  fi
  evaluate_retval
  if grep -E '^|$' /etc/sysconfig/createfiles 2>/dev/null; then
    log_info_msg "创建文件和目录..."
    create_files
    # 总是返回 0
    evaluate_retval
  fi
  exit $failed
  ;;
  echo "用法: $0 (start)"
  exit 1
  ;;
esac
