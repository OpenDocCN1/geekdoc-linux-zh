
如你所见，这是一个相当直接的函数。它只是循环遍历字节，边循环边计数，直到遇到空字符。然后它返回计数。
我们的读取记录的程序也会非常直接。它将执行以下操作：

- 打开文件
- 尝试读取记录

## 第 6 章 读取和写入简单记录

- 如果我们在文件末尾，则退出
- 否则，计算第一个名字的字符数
- 将第一个名字写入 STDOUT
- 向 STDOUT 写入换行符
- 回到读取另一个记录

为了编写这个，我们需要另外一个简单函数——一个向 STDOUT 写入换行的函数。将以下代码放入 `write-newline.s` 中：

```
.include "linux.s"

.glob write_newline

.type write_newline, @function

.section .data

newline:

ascii "\n"

.section .text

equ ST_FILEDES, 8

write_newline:

push %ebp

mov %esp, %ebp

mov $SYS_WRITE, %eax

mov ST_FILEDES(%ebp), %ebx

mov $newline, %ecx

mov $1, %edx

int $LINUX_SYSCALL

mov %ebp, %esp

pop %ebp

ret
```

## 第 6 章 读取和写入简单记录

现在我们准备好编写主程序了。这里是 `read-records.s` 的代码：

```
.include "linux.s"

.include "record-def.s"

.section .data

file_name:

.ascii "test.dat\0"

.section .bss

lcomm record_buffer, RECORD_SIZE

.section .text

#主程序

globl _start

_start:

# 这些是栈上的位置

# 我们在这里存储输入和输出描述符

#(供参考) - 我们本可以可以在 .data

# 段中使用内存地址

.equ ST_INPUT_DESCRIPTOR, -4

.equ ST_OUTPUT_DESCRIPTOR, -8

# 将栈指针移动到 %ebp

mov $esp, %ebp

# 分配空间持有描述符

sub $8, %esp

## 打开文件

mov $SYS_OPEN, %eax

mov $file_name, %ecx

mov $666, %edx

int $LINUX_SYSCALL

# 保存文件描述符

movl %eax, ST_INPUT_DESCRIPTOR(%ebp)

read_loop:

# 读取记录

mov $SYS_READ, %eax

mov $ST_INPUT_DESCRIPTOR(%ebp), %ebx

mov $record_buffer, %ecx

mov $RECORD_SIZE, %edx

int $LINUX_SYSCALL

# 如果读取的字节数等于记录大小，说明没结束

cmpl $RECORD_SIZE, %eax

jne read_loop

# 计算第一个名字的字符数

mov ST_STRING_START_ADDRESS (%ebp), %edx

call count_chars

# 将第一个名字写入 STDOUT

mov $SYS_WRITE, %eax

mov $ST_OUTPUT_DESCRIPTOR(%ebp), %ebx

mov %edx, %ecx

mov $1, %edx

int $LINUX_SYSCALL

# 写入换行符

call write_newline

### 退出

mov $SYS_EXIT, %eax

mov $0, %ebx

int $LINUX_SYSCALL
```

