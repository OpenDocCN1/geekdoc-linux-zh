```bash
# 解析命令行参数以查找精度值（scale），如果存在。

while getopts :s:S: ARGUMENT
do
    case $ARGUMENT in
        s) SCALE=$OPTARG ;;
        \?) # 因为我们有负数，我们需要
            # 测试以连字符 (-) 开头的参数
            # 是否为数字，而不是无效选项！！
            for TST_ARG in $@
            do
                if [[ $(echo $TST_ARG | cut -c1) = '-' ]] && [ $TST_ARG != '-s' -a $TST_ARG != '-S' ]; then
                    case $TST_ARG in
                        +(-([0-9]))) : ;; # 无操作，什么都不做
                        +(-([0-9].[0-9]))) : ;; # 无操作，什么都不做
                        *) echo "\nERROR: 无效参数，请"
                           usage
                           exit 1
                           ;;
                    esac
                fi
            done
            ;;
    esac
done
```

```bash
# 解析命令行参数并收集待
# 相加的数字列表，并测试每个值。

while ((COUNT < MAX_COUNT))
do
    ((COUNT = COUNT + 1))
    TOKEN=$1 # 在每次循环迭代中获取命令行参数
    case $TOKEN in # 测试每个值并寻找精度值。
        -s|-S) shift 2
            ((COUNT = COUNT + 1))
            ;;
        -s$(SCALE)) shift ;;
        -S$(SCALE)) shift ;;
        *) # 将数字 ($TOKEN) 添加到列表
            NUM_LIST="$NUM_LIST $TOKEN"
            ((COUNT < MAX_COUNT)) && shift
            ;;
    esac
done
```

```bash
for NUM in $NUM_LIST
do
    case $NUM in
        +([0-9])) # 检查是否为整数
            : ;; # 无操作，什么都不做。
        -([0-9])) # 检查是否为负整数
            : ;; # 无操作，什么都不做。
        +([0-9].[0-9])) # 检查是否为正浮点数
            : ;; # 无操作，什么都不做。
        -([0-9].[0-9])) # 检查是否为负浮点数
            : ;; # 无操作，什么都不做。
        +([.-0-9])) # 检查是否为负浮点数
            : ;; # 无操作，什么都不做。
        +(-([.+0-9]))) # 检查是否为正浮点数
            : ;; # 无操作，什么都不做。
        *) echo "\nERROR: $NUM 不是有效的数字"
           usage
           exit 1
           ;;
    esac
done
```
