# 向警报日志添加消息。重复的消息将被丢弃。
# 参数 1 = 要添加的消息
#
________

function alarm {
    # 如果本次有任何记录，意味着系统不正常，即使是
    # 一条被抑制的重复消息

    SYSTEM_OK=

    # 忽略重复的警报
    [ "$1" = "$LASTLOG" ] && return
    LASTLOG=$1

    # 将日志消息添加到警报列表中
    LOG8="$LOG7"
    LOG7="$LOG6"
    LOG6="$LOG5"
    LOG5="$LOG4"
    LOG4="$LOG3"
    LOG3="$LOG2"
    LOG2="$LOG1"
    LOG1="$TIME - "$1"

    LOG_UPDATED=1
}
readonly -f alarm
declare -t alarm
