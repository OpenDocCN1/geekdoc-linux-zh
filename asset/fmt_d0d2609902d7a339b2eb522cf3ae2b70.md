    int iThreadCounter = 0;
    for (int count = 2; count < argc; count += 2) {
        if (strcmp(globalMsg.filename, argv[count]) == 0) {
            // 找到了消费者线程并发送了消息。
            cout << "Kill Thread: " << iThreadCounter << endl;

        }
