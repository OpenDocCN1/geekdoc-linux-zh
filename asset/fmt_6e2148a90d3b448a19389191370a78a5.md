
## PartNode 类

```cpp
// ************ PartNode *
class PartNode
{
public:
    PartNode();
    ~PartNode();
    void SetNext(PartNode* node) { itsNext = node; }
    PartNode* GetNext() const;
    Part* GetPart() const;

private:
    Part* itsPart;
    PartNode* itsNext;
};
```

### PartNode 的实现

```cpp
PartNode::PartNode()
{
    itsPart = 0;
    itsNext = 0;
}

PartNode::~PartNode()
{
    delete itsPart;
    itsNext = 0;
}

PartNode* PartNode::GetNext() const
{
    return itsNext;
}

Part* PartNode::GetPart() const
{
    if (itsPart)
        return itsPart;
    else
        return NULL;
}
```

## PartsList 类

```cpp
// ************* PartsList *
class PartsList
{
public:
    PartsList();
    ~PartsList();
    int GetCount() const;
    Part* GetFirst() const;
    static PartsList& GetGlobalList();
    void Insert(Part* part);
    void Iterate(void (Part::*f()) const);
    Part* operator[](int index) const;

private:
    PartNode* pHead;
    int itsCount;
    static PartsList globalList;
};
```

### PartsList 的实现

```cpp
PartsList::PartsList()
{
    pHead = 0;
    itsCount = 0;
}

PartsList::~PartsList()
{
    while (pHead != 0) {
        PartNode* temp = pHead;
        pHead = pHead->GetNext();
        delete temp;
    }
}

int PartsList::GetCount() const
{
    return itsCount;
}

Part* PartsList::GetFirst() const
{
    if (pHead)
        return pHead->GetPart();
    else
        return NULL;
}

static PartsList& PartsList::GetGlobalList()
{
    return globalList;
}

void PartsList::Insert(Part* part)
{
    PartNode* newNode = new PartNode();
    newNode->itsPart = part;
    newNode->itsNext = pHead;
    pHead = newNode;
    itsCount++;
}

void PartsList::Iterate(void (Part::*f()) const)
{
    PartNode* temp = pHead;
    while (temp != 0) {
        (temp->GetPart()->*f)();
        temp = temp->GetNext();
    }
}

Part* PartsList::operator[](int index) const
{
    if (index < 0 || index >= itsCount)
        return NULL;
    PartNode* temp = pHead;
    for (int i = 0; i < index; i++) {
        temp = temp->GetNext();
    }
    return temp->GetPart();
}
```

## 列表 R2.1 续

void PartsList::Iterate(void Part::*func())() const
{
    if (!pHead)
        return;
    PartNode* pNode = pHead;
    do
        (pNode)->GetPart()->(*func());
    while (pNode = pNode->GetNext());
}

void PartsList::Insert(Part* pPart)
{
    PartNode* pNode = new PartNode(pPart);
    PartNode* pCurrent = pHead;
    PartNode* pNext = NULL;

    int New = pPart->GetPartNumber();
    int Next = 0;
    itsCount++;

    if (!pHead)
    {
        pHead = pNode;
        return;
    }

    // 如果这个部分比头部小
    // 如果这个部分是新的头部
    if (pHead->GetPart()->GetPartNumber() > New)
    {
        pNode->SetNext(pHead);
        pHead = pNode;
        return;
    }

    for (;);
    {
        // 如果没有下一个节点，则追加这个新节点
        if (!pCurrent->GetNext())
        {
            pCurrent->SetNext(pNode);
            return;
        }

        // 如果这个部分在当前部分之后、下一个部分之前
        // 则将其插入此处，否则获取下一个部分节点
        pNext = pCurrent->Next();
        Next = pNext->GetPart()->GetPartNumber();
        if (Next > New)
        {
            pCurrent->SetNext(pNode);
            pNode->SetNext(pNext);
            return;
        }
        pCurrent = pNext;
    }
}
