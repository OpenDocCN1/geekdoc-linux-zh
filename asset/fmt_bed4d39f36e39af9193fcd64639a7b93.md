package main

import (
	"bufio"
	"os"
	"strings"
)

func main() {
	file, err := os.Open("dict.txt")
	if err != nil {
		return err
	}

	var words []string
	scanner := bufio.NewScanner(file)
	for scanner.Scan() {
		words = append(words, strings.ToLower(scanner.Text()))
	}

	if err = scanner.Err(); err != nil {
		return err
	}

	fmt.Println(len(words))
}
