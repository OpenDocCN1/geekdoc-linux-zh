01  #!/usr/bin/env python
02
03  from zipfile import ZipFile, ZIP_DEFLATED
04  from sys import stdout, argv    # command line arguments
05  from xml.etree import ElementTree as et
06  from re import findall
07
08  filename,outputfilename = argv[1],argv[2]
09
10  # Step 1: Read /content.xml inside zip file      # open zip file
11  inZipf = ZipFile(filename, 'r')
12  content = inZipf.open("content.xml",'r').read()  # access /content.xml
13
14  # Step 2: Build the XML namespace dictionary
15  ns = {}                                            # {} = empty dictionary
16  for namespace in findall('xmlns:.*=".*"', content):
17      namespace = namespace.split(".")[1]          # get rid of "xmlns:"
18      nsurl = namespace.split("=")[1]      # cut " in front/back
19      nsurl = nsurl[1:-1]                # add to dictionary AND
20      ns[namespace] = nsurl              # register for ElementTree module
21      et.register_namespace(namespace,nsurl)
22
23  # Step 3: Access the XML tree
24  root = et.fromstring(content)
25  for x in root.findall("//", ns):
26      if x.text:                               # any text inside?
27          x.text = x.text.replace("Linux","Windows")  # modify the content
28
29  # Step 4: Create output file
30  with ZipFile(outputfilename, "w") as outZip:  # open zip file for writing
31      outZip.comment = inZipf.comment          # copy archive comment
32      for file in inZipf.infolist():          # copy all files except/ /content.xml
33          if file.filename != "content.xml":
34              outZip.writestr(file, inZipf.read(file.filename))
35      with ZipFile(filename, mode="a", compression=ZIP_DEFLATED) as outZip:
36          outZip.writestr("content.xml", et.tostring(root))  # add new content.xml
