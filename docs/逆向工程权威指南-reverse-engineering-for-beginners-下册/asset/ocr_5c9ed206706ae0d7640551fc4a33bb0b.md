```
_f1:
  sll $v0, $a0, 3
; $v0 = $a0<3 = $a0*8
  jr $ra
  subu $v0, $a0 ; branch delay slot
; $v0 = $v0-$a0 = $a0*8-$a0 = $a0*7
```

```
_f2:
  sll $v0, $a0, 5
; $v0 = $a0<5 = $a0*32
  sll $a0, 2
; $a0 = $a0<2 = $a0*4
  jr $ra
  subu $v0, $a0 ; branch delay slot
; $v0 = $a0*32-$a0*4 = $a0*28
```

```
_f3:
  sll $v0, $a0, 4
; $v0 = $a0<4 = $a0*16
  jr $ra
  addu $v0, $a0 ; branch delay slot
; $v0 = $a0*16+$a0 = $a0*17
```
