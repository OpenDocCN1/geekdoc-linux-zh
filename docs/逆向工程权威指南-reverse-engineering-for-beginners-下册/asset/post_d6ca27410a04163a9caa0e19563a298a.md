// structure is not packed
struct tree_node {
    tree_node *Left;
    tree_node *Parent;
    tree_node *Right;
    char Color; // 0 - Red, 1 - Black
    char Isnil;
    std::pair Myval; // called Myval in std::set
    unsigned int first;
    const char *second; // not present in std::set
};

struct tree_struct {
    tree_node *Myhead;
    size_t Msize;
};

void dump_tree(struct tree_node *n, bool is_set, bool traverse) {
    printf("ptr=0x%p Left=0x%p Right=0x%p Color=%d Isnil=%d\n",
           n->Left, n->Parent, n->Right, n->Color, n->Isnil);
    if (n->Isnil == 0) {
        if (is_set)
            printf("%d\n", n->first);
        else
            printf("%d [%s]\n", n->first, n->second);
    }
    if (traverse) {
        if (n->Isnil == 1)
            dump_tree(n->Parent, is_set, true);
        else {
            if (n->Left->Isnil == 0)
                dump_tree(n->Left, is_set, true);
            if (n->Right->Isnil == 0)
                dump_tree(n->Right, is_set, true);
        }
    }
}

void dump_as_tree(int tabs, struct tree_node *n, bool is_set) {
    const char ALOT_OF_TABS = "\t\t\t\t\t\t";
    for (int i = 0; i < tabs; i++) printf("%s", ALOT_OF_TABS);
    if (is_set)
        printf("%d\n", n->first);
    else
        printf("%d [%s]\n", n->first, n->second);
    if (n->Left->Isnil == 0) {
        printf("%s-------\n", ALOT_OF_TABS);
        dump_as_tree(tabs + 1, n->Left, is_set);
    }
    if (n->Right->Isnil == 0) {
        printf("%s-------\n", ALOT_OF_TABS);
        dump_as_tree(tabs + 1, n->Right, is_set);
    }
}

void dump_map_and_set(struct tree_struct *m, bool is_set) {

    printf("As a tree:\n");
    printf("---\n");
    dump_as_tree(1, m->Myhead->Parent, is_set);
}

## 1.4 std::map() and std::set()

“二叉树”是另一种常见的数据结构。二叉树是每个节点最多有两个子节点的树，每个节点都有关键字（key）—值（value）。二叉树通常用于联合数组（associative arrays）。联合数组被称为字典（dictionary）或 hash 表，它由一系列的“关键字”-值”构成。

二叉树最少具备三个特性：

- 所有的关键字有序。
- 任何类型的关键字都能构成二叉树。二叉树的算法与关键字的数据类型无关。只要规定一个二叉树关键字的排序函数，就可以构造一个二叉树。
- 与链表和哈表相比，二叉树中搜索特定关键字的速度比较快。

现在开始举例：我们将存储序列0, 1, 2, 3, 5, 6, 9, 10, 11, 12, 20, 29, 100, 101, 1001, 1010，并形成下二叉树结构：

从下面的图中我们很容易发现这样的规律：所有左侧节点的值都比其右侧节点的小，而所有右侧节点的值都比左侧节点大。

在遵守以上规则下，我们查询算法查找就简单了。我们只需要关注当前节点的值以及我们要查询的值，将这两者进行比较，当前者比后者小时我们需要向右搜索查找；反之，当前者比后者大时，我们就会向左搜索查找。直到找到节点的值与要搜索的值相等为止。这就是为什么才说只需要一个对比即可进行节点或者字符串的对比。

![](img/a1381a53f805f55074b2e5d2ec754b6_902_0.png)

所有的都有一个唯一的值，不能重复。我们可以计算需要花费的步数，公式约等于：log2(n)(是二叉树中包含的个数），这个公式计算是我们找到一个平衡树的搜索步骤。这就意味着在一个1000个二制树中，通过大约10步我们就能找到需要的值；而10000个二制树则需要13层。看起来还是很高的。树总是需要像这样平衡的，也就是说，必须在所有水平上平衡分布。插入和移除节点可能都需要保持平衡的状态。

我们几个可用的行的平衡算法，包括AVL树和红黑树。后面的算法采用的是每个节点标记为红（红）或黑（黑），以简化节点的平衡过程。不是GC而是MSVC的模板函数std::map()和std::set()都用了红黑树(red-black)的算法。std::set()中只包含键，std::map()排除了键(std::set)还有节点的数值。
