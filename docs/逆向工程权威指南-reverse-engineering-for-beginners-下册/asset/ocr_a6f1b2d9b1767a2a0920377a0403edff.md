而在有符号的计算过程中，魔术因子则是0x55555556，乘法的中间结果要除以2的32次。虽然这里没有进行除法运算，但是根据前面的讨论可知：商的有效位来自于EDX寄存器。

请别记住一步的除法计算过程中：积的高32位位右移31位，在EAX寄存器的最低位保留有符号数的符号位（正数为0，负数为1）。将符号位加入的积高32位值，可实现负数补码的"+1”修正"

_f3_32_signed PROC
    mov     eax, 1431655766      ; 55555556H
    imul    DWORD PTR, a$[esp-4]  ; signed multiply
    ; take high part of product
    ; it is the same as if shift product by 32 bits right
    ; or to divide it by 2^32
    mov     eax, edx              ; EAX=EDX=
    (input*0x55555556)/2^32
    shr     eax, 31               ; 00000001F
    add     eax, edx              ; add 1 if sign is negative
    ret                           0
f3_32_signed ENDP
