```
; load address of jumpable, LUI and ADDIU pair are there in fact:
; LA is pseudointruction
	la	$v0, off_120
; multiply input value by 4:
	sll	$a0, 2
; sum up multiplied value and jumpable address:
	adddu	$a0, $v0, $a0
; load element from jumpable:
	lw	$v0, 0($a0)
	or	$at, $zero ; NOP
; jump to the address we got in jumpable:
	rj	$v0
	rj	$at, $zero ; branch delay slot, NOP
```
