
# 第53章 16位的Windows程序

虽然16位的Windows程序已经几乎绝迹，但是有关复古程序以及研究密集知识的研究，往往会涉及此部分知识。

微软于1993年8月发布了最后一个16位的Windows系统，即Windows 3.11（同年发行的中文操作系统Windows 3.2也是16位操作系统）。在此时之后问世的16/32位操作系统Windows 96/98，以及32位的Windows NT系统都可以运行16位应用程序。后来推出的64位Windows NT系列系统不再支持16位应用程序。

16位应用程序的代码结构与MSDOS的程序十分相似。这种类型的可执行文件采用了一种“New Executable (NE)”的可执行程序格式。

本章的所有程序均OpenWat 1.9编译。编译时的选项如下：

```
wcl.exe -i=C:/WATCOM/h/win/ -s -os -bt=windows -bcl=windows example.c
```


```pascal
int PASCAL WinMain( HINSTANCE hInstance,
               HINSTANCE hPrevInstance,
               LPSTR lpCmdLine,
               int nCmdShow )
{
    MessageBeep (MB_ICONEXCLAMATION);
    return 0;
}
```

```assembly
WinMain
    proc near
    push bp
    mov bp, sp
    mov ax, 30h ; '0'      ; MB_ICONEXCLAMATION
    push ax
    call MESSAGEBEEP
    xor ax, ax             ; return 0
    pop bp
    retn 0Ah
WinMain
    endp
```

## 53.2 例子 #2

```c
#include <windows.h>
```

```pascal
int PASCAL WinMain( HINSTANCE hInstance,
                   HINSTANCE hPrevInstance,
                   LPSTR lp1pcCmdLine,
                   int nCmdShow )
{
    MessageBox (NULL, "hello, world", "caption",
               MB_YESNOCANCEL);
    return 0;
}
```

```assembly
WinMain
    proc near
    push bp
    mov bp, sp
    xor ax, ax             ; NULL
    push ax
    push bp
    mov ax, offset aHelloWorld ; 0x18. "hello,\nworld"
    push ax
    push ds
    mov ax, offset aCaption ; 0x10. "caption"
    push ax
    mov ax, 3             ; MB_YESNOCANCEL
    push ax
    call MESSAGEBOX
    xor ax, ax ; return 0
    pop bp
    retn 0Ah
endp

WinMain
dseg:00210 aCaption db 'caption',0
dseg:00218 aHelloWorld db 'hello, world',0
```

基于Pascal语言的调用约定要求：参数从左到右入栈（withcdecl）。调用函数参数依次传递NULL,"hello, world","caption"和MB_YESNOCANCEL。这个规则还需要被调用函数恢复指针，只是RETN指令有一个0Ah参数，被调用函数退出的时候要释放10字节的空栈间。这种调用规则和stdcall（参数64.2）非常相似，只是数据传递顺序是从左到右的“自然语言”顺序。

```pascal
	HINSTANCE hInstance,
	HINSTANCE hPrevInstance,
	LPSTR lpCmdLine,

	int result=MessageBox(NULL,"hello, world","caption",
		MB_YESNOCANCEL);

		MessageBox(NULL,"you pressed cancel","caption",
		MB_OK);
	else if(result==IDYES)
		MessageBox(NULL,"you pressed yes","caption", MB_OK);
	else if(result==IDNO)
		MessageBox(NULL,"you pressed no","caption", MB_OK);

	return 0;
}
```

## 53.4 例子#4

```c
#include <windows.h>

int PASCAL func1 (int a, int b, int c)
{
    return a*b+c;
};

long PASCAL func2 (long a, long b, long c)
{
    return a*b+c;
};

long PASCAL func3 (long a, long b, long c, int d)
{
    return a*b+c-d;
};

int PASCAL WinMain( INSTANCE hInstance,
                   INSTANCE hPrevInstance,
                   LPSTR lpCmdLine,
                   int nCmdShow )
{
    func1 (123, 456, 789);
    func2 (600000, 700000, 800000);
    func3 (600000, 700000, 800000, 123);
    return 0;
}
```

```assembly
func1 proc near
c = word ptr 4
b = word ptr 6
a = word ptr 8
push bp
mov bp, sp
mov ax, [bp+a]
imul [bp+b]
add ax, [bp+c]
pop bp
retn 6

func2 proc near
arg_0 = word ptr 4
arg_2 = word ptr 6
arg_4 = word ptr 8
arg_6 = word ptr 0Ah
arg_8 = word ptr 0Ch
arg_A = word ptr 0Eh
push bp
mov bp, sp
mov ax, [bp+arg_8]
mov dx, [bp+arg_A]
mov bx, [bp+arg_4]
mov cx, [bp+arg_6]
call subB2 ; long 32-bit multiplication
add ax, [bp+arg_0]
adc, [bp+arg_2]
pop bp
retn 12

func3 proc near
arg_0 = word ptr 4
arg_2 = word ptr 6
arg_4 = word ptr 8
arg_6 = word ptr 0Ah
arg_8 = word ptr 0Ch
arg_A = word ptr 0Eh
arg_C = word ptr 10
push bp
mov bp, sp
mov ax, [bp+arg_A]
mov dx, [bp+arg_C]
mov bx, [bp+arg_6]
mov cx, [bp+arg_8]

cx, [bp+arg_2]
add ax, [bp+arg_4]
adc bx, dx ; BX=high part, CX=low
```

## 53.5 例子5

```c
#include <windows.h>
```

```pascal
int PASCAL string_compare (char *s1, char *s2)
{
    while (1)
    {
        if (*s1!=*s2)
            return 0;
        if (*s1==0 || *s2==0)
            return 1; // end of string
        s1++;
        s2++;
    }
}

int PASASCAL string_compare_far (char far *s1, char far *s2)
{
    while (1)
    {
        if (*s1!=*s2)
            return 0;
        if (*s1==0 || *s2==0)
            return 1; // end of string
        s1++;
        s2++;
    }
}

void PASASCAL remove_digits (char *s)
{
    while (*s)
    {
        if (*s>='0' && *s<=='9')
            *s='-';
        s++;
    }
}
```

```pascal
char str[]="hello 123 world";
int PASCAL WinMain(HINSTANCE hInstance,
HINSTANCE hPrevInstance,
LPSTR lpCmdLine,
nCmdShow)
{
    string_compare("asd", "def");
    string_compare_far("asd", "def");
    remove_digits(str);
    MessageBox(NULL, str, "caption", MB_YES);
    return 0;
}
```
