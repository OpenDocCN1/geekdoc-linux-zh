```
get_by_coordinates:
; sign-extend input 32-bit values "a" and "b" to 64-bit ones
movsx rsi, esi
movsx rdx, edx
lea rax, [rdi+a*4]
RAX=RDI+ESI*4-address of array+a*4
movzx eax, BYTE PTR [rdx]
; AL=load byte at address RAX+RDX=address of array+a+b
ret
```
