void decrypt_file(char *fin, char* fout, char *pw)
{
  FILE *f;
  int real_flen, flen;
  BYTE *buf;

  f=fopen(fin, "rb");

  if (f==NULL) {
    printf("Cannot open input file!\n");
    return;
  }

  fseek(f, 0, SEEK_END);
  flen=ftell(f);
  fseek(f, 0, SEEK_SET);

  buf=(BYTE*)malloc(flen);
  fread(buf, flen, 1, f);

close(f);

  if (memcmp(buf, "QR9", 3)!=0) {
    printf("File is not encrypted!\n");
    return;
  }

  memcpy(&real_flen, buf+3, 4);

  decrypt(buf+(3+4), flen-(3+4), pw);
