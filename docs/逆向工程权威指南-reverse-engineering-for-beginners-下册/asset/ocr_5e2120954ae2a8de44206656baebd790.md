```
#include <stdio.h>
struct inner_struct {
    int a;
    int b;
}

struct outer_struct {
    char a;
    int b;
    struct inner_struct c;
    char d;
    int e;
}
void f(struct outer_struct s) {
    printf("a=%d; b=%d; c.a=%d; c.b=%d; d=%d; e=%d\n", s.a, s.b, s.c.a, s.c.b, s.d, s.e);
}

int main() {
    struct outer_struct s;
    s.a=1;
    s.b=2;
    s.c.a=100;
    s.c.b=101;
    s.d=3;
    s.e=4;
    f(s);
}
```

这个程序把结构体inner_struct当作另一个结构体outer_struct的字段用来，它和outer_struct的a、b、d、一样，都是一个字段。
