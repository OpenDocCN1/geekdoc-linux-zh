#include <stdio.h>

unsigned int f(unsigned int a)
{
    int i=0;
    int j=1;
    unsigned int rt=0;
    for(;i<=28; i+=4, j*=10)
        rt+=((a>>i)&0xF) * j;
    return rt;
}

int main()
{
    // test
    printf("%d\n", f(0x012345678));
    printf("%d\n", f(0x01234567));

    printf("%d\n", f(0x012345));
    printf("%d\n", f(0x01234));
    printf("%d\n", f(0x0123));

}
