// bit=31..0
uint32_t set_bit (uint32_t input, int bit)
{
    return input=input|(1<<bit);
}

uint32_t form_netmask (uint8_t netmask_bits)
{
    uint32_t netmask=0;
    for (i=0; i<netmask_bits; i++)
        netmask=set_bit(netmask, 31-i);
    return netmask;
}
