```
; a*28
||f2|| PROC
RSB r0,r0,r0,LSL #3
; R0=R0<<3-R0=R0*8-a=8-a=7
LSL r0,r0,#2
; R0=R0<<2-R0*4=a*7*4=28
BX lr
ENDEP
```

```
; a*17
||f3|| PROC
ADD r0,r0,r0,LSL #4
; R0=R0<<4=R0+R0*16=17=a*17
BX lr
ENDEP
```
