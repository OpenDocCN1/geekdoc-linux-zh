# 指令清单43.16 采用MSVC 2010优化

```
_inbuf$ = 8      ; size = 4
outbuf$ = 12     ; size = 4
_memcopy_7 PROC
mov     ecx, DWORD PTR inbuf$[esp-4]
mov     edx, DWORD PTR [ecx]
mov     eax, DWORD PTR outbuf$[esp-4]
mov     DWORD PTR [eax+10], edx
mov     dx, WORD PTR [ecx+4]
mov     WORD PTR [eax+14], dx
mov     cl, BYTE PTR [ecx+6]
mov     BYTE PTR [eax+16], cl
mov     0
ret
_memcopy_7: 
```

# 指令清单43.17 采用GCC 4.8.1优化

```
memcopy_7:
push    
mov     eax, DWORD PTR [esp+8]
mov     ecx, DWORD PTR [esp+12]
mov     ebx, DWORD PTR [eax]
lea     edx, [ecx+10]
mov     DWORD PTR [ecx+10], ebx
movzx ecx, WORD PTR [eax+4]
mov     WORD PTR [edx+4], cx
movzx eax, BYTE PTR [eax+6]
mov     BYTE PTR [edx+6], al
pop
ebx
ret
```

上述指令的操作流程：首先复制4个字符，然后复制一个字（就是2个字节），

此外，编译器还会通过赋值指令MOV结构体（structure）类型数据。详情请参阅书籍的21.4.1。
