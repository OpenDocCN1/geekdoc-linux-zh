and eax, 32767 ;
00007fffH
ret 0
my_rand ENDP

# 初始化的TLS数据

编程人员通常会给变量rand_state设置一个固定的初始值，以防止后期忘记它进行初始化。如下述代码第9行所示，我们对它进行初始化赋值。

```

2 #include <windows.h>
3 #include <winnt.h>
4

8

10

13 rand_state=init;
14 }
15
16 int my_rand ()

18 rand_state=rand_state*RNG_a;

21 }
```
