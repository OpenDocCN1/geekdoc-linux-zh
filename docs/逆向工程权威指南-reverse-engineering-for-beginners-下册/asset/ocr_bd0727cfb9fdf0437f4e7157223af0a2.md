# 指令清单65.2 优化的MSVC 2013 x86

```
_TLS SEGMENT
_rand_state DD 01H DUP (?)
_TLS ENDS

DATA SEGMENT
$SG84851 DB    '%d', 0aH, 00H
DATA ENDS
TEXT SEGMENT
```

```
_init$ = 8                               ; size = 4
my_rand PROC
    ; FS:0=address of TIB
    mov     eax, DWORD PTR fs:_tls_array ; displayed in
    IDA as FS:2Ch
    EAX=address of TLS of process
    mov     ecx, DWORD PTR  _tls_index
    mov     ecx, DWORD PTR [eax+ecx*4]
    ; ECX=current TLS segment
    mov     eax, DWORD PTR  _init$[esp-4]
    mov     dword PTR  _rand_state[ecx], eax
    ret
_my_rand ENDP
```

```
_my_rand PROC
    ; FS:0=address of TIB
    mov     eax, DWORD PTR fs:_tls_array ; displayed in
    IDA as FS:2Ch
    EAX=address of TLS of process
    mov     ecx, DWORD PTR  _tls_index
    mov     ecx, DWORD PTR [eax+ecx*4]
    ; ECX=current TLS segment
    imul    eax, DWORD PTR  _rand_state[ecx], 1664525
    add     eax, 1013940223 ; 3c6ef35fh
    mov     dword PTR  _rand_state[ecx], eax
    and     eax, 32767
    ret     0
_my_rand ENDP
_TEXT ENDS
```

参数rand_state现在位于TLS段中。此后每个线程都会拥有自己的rand_state。这里表示的是如何寻址：从FS:2Ch调用线程信息块
