# MSVC 2013的编译方法可圈可点，三个函数的指令完全一致。

指令清单18.22 Optimizing MSVC 2013 x64

```
array$ = 8
as = 16
b$ = 24
get_by_coordinates3 PROC
; RCX=address of array
; RDX=a
; R8=b
    movsd rax, r8d
    EAX=b
    movsd r9, edx
    R9=a
    add rax, rcx
    RAX=b+address of array, BYTE PTR [rax+r9*4]
    movzx eax, BYTE PTR [rax+r9*4]
    AL=load byte at address RAX+R9*4=address of array+a*4=address of array+=4*b
    ret 0
get_by_coordinates3 END
```

```
array$ = 8
as = 16
b$ = 24
get_by_coordinates2 PROC
    movsd rax, r8d
    movsd r9, edx
    add rax, rcx
    movzx eax, BYTE PTR [rax+r9*4]
    ret 0
get_by_coordinates2 END
array$ = 8
as = 16
b$ = 24
get_by_coordinates1 PROC
    movsd rax, r8d
    movsd r9, edx
    add rax, rcx
    movzx eax, BYTE PTR [rax+r9*4]
    ret 0
get_by_coordinates1 END
```

GCC 生成等效的指令，只是各函数的指令略有区别。
