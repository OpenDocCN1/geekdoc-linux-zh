# $SG1305 DB      'read=%d, write=%d', 0aH, 00H

```
_write = -12      ; size= 4
_read = -8       ; size= 4
tv64 = -4       ; size= 4
_type = 8       ; size= 4
_f               PROC

push           ebp
mov            ebp, esp
sub            esp, 12
mov            DWORD PTR _read$[ebp], 0
mov            DWORD PTR _write$[ebp], eax
mov            eax, DWORD PTR _type$[ebp]
mov            DWORD PTR tv64[ebp], eax
cmp            DWORD tv64[ebp], 1 ; R
je             SHORT LN2@
cmp            DWORD tv64[ebp], 2 ; W
je             SHORT LN3@
cmp            DWORD tv64[ebp], 3 ; RW
je             SHORT LN4@
jmp            SHORT LN5@
LN4f: : case RW:
mov            DWORD PTR _read$[ebp], 1
$LN3f: case W:
mov            DWORD PTR _write$[ebp], 1
jmp            SHORT LN5@
$LN2f: case R:
mov            DWORD PTR _read$[ebp], 1
$LN5f: : default
mov            ecx, DWORD PTR _write$[ebp]
push           ecx
mov            edx, DWORD PTR _read$[ebp]
push           edx
mov            OFFSET $$S1305 ; 'read=%d, write=%d'
call           _printf
add            esp, 12
mov            ebp, esp
pop            ebp
ret
_f               ENDP
```

上述汇编指令与C语言代码几乎一一对应。因为在LN4@f和$LN3@之间没有转移指令，所以当程序执行了LN4@f处的“令赋值的值为1”的指令后，它就会执行后面“fall through”（滑落）这个名字的来源：当执行前面那个陈述语句的指令（read赋值）之后，继续执行下一个陈述语句的指令（write赋值）。如果type的值为W，程序将执行$LN3@f的指令，而不会执行前面那个read指令。
