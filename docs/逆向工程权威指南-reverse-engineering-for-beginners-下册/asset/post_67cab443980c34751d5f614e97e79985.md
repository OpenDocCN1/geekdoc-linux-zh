```c
#include <sys/types.h>
#include <sys/stat.h>
#include <time.h>
#include <stdio.h>
#include <stdlib.h>

int main(int argc, char *argv[])
{
    struct stat sb;

    if (argc == 2)
    {
        fprintf(stderr, "Usage: %s pathname\n", argv[0]);
        return 0;
    }
    if (stat(argv[1], &sb) == -1)
    {
        // error
        return 0;
    }

    printf("%ld\n", (long) sb.st_uid);
}
```

### G. 3.1 练习题2.1

这个函数可以根据牛尔顿法计算平方根，其算法源自Harold Abelson, Gerald Jay Sussman, and Julie Sussman:《Structure and Interpretation of Computer Programs》(1996)。

程序源代码如下：

```c
// algorithm taken from SICP
#include <math.h>
double average(double a, double b)
{
    return (a + b) / 2.0;
}
double improve(double guess, double x)
{
    return average(guess, x/guess);
}
int good_enough(double guess, double x)
{
    double d = abs(guess*guess - x);
    return (d < 0.001);
}
double square_root(double guess, double x)
{
    while (good_enough(guess, x)==0)
        guess = improve(guess, x);
    return guess;
}
double my_sqrt(double x)
{
    return square_root(1, x);
}
int main()
{
    printf("%g\n", my_sqrt(123.456));
}
```

### G. 3.2 练习题2.4

答案: 函数strstr()。其源代码如下所示：

```c
char *strstr (
    const char *str1,
    const char *str2
)
{
    char *cp = (char *) str1;
    char *s1, *s2;
    if (!*s2)
        return((char *) str1);
    while (*cp)
    {
        s1 = cp;
        s2 = (char *) str2;
        while (*s1 && *s2)
            s1++, s2++;
        if (!*s2)
            return(cp);
        cp++;
    }
    return(NULL);
}
```

### G. 3.4 练习题2.13

答案：线性反馈位寄存器。详细资料参见：https://en.wikipedia.org/wiki/Linear_feedback_shift_register 程序源代码所示：

```c
// reworked from https://en.wikipedia.org/wiki/Linear_feedback_shift_register
#include <stdint.h>

uint16_t f(uint16_t in) {
    /* taps: 16 14 13 11; feedback polynomial: x^16 + x^14 + x^13 + x^11 + 1 */
    uint16_t bit = ((in >> 0) ^ (in >> 2) ^ (in >> 3) ^ (in >> 5)) & 1;
    return (in >> 1) | (bit << 15);
}

#define C 0xBADF

int main(void)
{
    uint16_t fsr = C;
    unsigned period = 0;

    do
    {
        fsr=f(fsr);
        printf("period=0x%x, fsr=0x%x\n", period, fsr);
        ++period;
    } while (fsr != C);

    return 0;
}
```

int ack (int m, int n)
{
    if (m==0)
        return n+1;
    if (n==0)
        return ack(m-1, 1);
    return ack(m-1, ack (m, n-1));
}
