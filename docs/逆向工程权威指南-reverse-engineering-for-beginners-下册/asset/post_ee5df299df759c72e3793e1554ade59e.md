STRB        r4,[r0,r3]
; i++      ADDS       r3,r3,#1
|L0.12|
; i<size?
            CMP        r3,r2
; jump to the loop begin if its so:
            BCC        |L0.6|

## 14.2.2 编译为ARM模式的程序

在编辑ARM模式的应用时，Keil能够充分利用相应指令的条件执行指令：

指令清单14.13 Optimizing Keil 6/2013 (ARM mode)

```assembly
my_memcpy PROC
; R0 = destination address
; R1 = source address
; R2 = size of block

; initialize counter (i) at 0
            MOV        r3,#0
|L0.4|
; all bytes copied?
            CMP        r3,r2
; the following block is executed only if "less than" condition,
; i.e. if R2<R3 or isize.
; load byte at R1+:
            LDBCC      r2,[r1,r3]
; store byte at R1+:
            STRBCC     r2,[r0,r3]

; i++
            ADDCC      r3,r3,#1
; the last instruction of the "conditional block".
; jump to loop begin if i<size
; do nothing otherwise (i.e., if i>=size)
            BCC        |L0.4|
; return

            BX         lr
ENDP
```

32位的ARM程序只用了一个转移指令，比程序少了一个转移指令。

## 14.2.3 MIPS

指令清单14.14 GCC 4.4.5 optimized for size (-Os) (IDA)

```assembly
my_memcpy;
; jump to loop check part:
    b loc_14
; initialize counter (i) at 0
; it will always remain in $v0;
    move $v0, zero; branch delay slot
```

```assembly
loc_8:\                               # CODE XREF:
my_memcpy+1C
; load by unsigned at address in $t0 to $v1:
    lbu $v1, 0($t0)
; increment counter (i):
    addiu $v0, $1
; store byte at $a3
    sb $v1, 0($a3)
```

```assembly
loc_14:\                               # CODE XREF: my_memcpy
; check counter (i) in $v0 is still less then 3rd function argument
("cnt" in $a2):
    sltu $v1, $v0, $a2
; form address of byte in source block:
    addiu $t0, $a1, $v0
; $t0 = $a1+$v0 ; src+
; jump to loop body if still less then "cnt":
    bnez $v1, loc_8
; form address of byte in destination block ($a3 = \$a0+\$v0 =
    addiu $a3, $a0, $v0 ; branch delay slot
; finish if BNEZ triggered:
    jr $ra
    nop $at, zero ; branch delay slot, GOP
```

这里介绍的两个指针寄存器分别是LBU (Load Byte Unsigned)和SB (Store Byte)。在两个平台上，所有的MIPS寄存器都是32位寄存器。在两个平台上，我们无法像x86那样对寄存器的低位进行操作。因此在处理单字节字符操作时，我们都是访问整个32位寄存器。LB指令加载带符号的数据并清除寄存器的其他位。由于有符号位的问题，所以在处理有符号的数据的时候通常要使用LB (Load Byte)指令把单字节的有符号数据扩展为等值的32位数据、再存储在寄存器中。SB指令的作用是把寄存器的低8位数据、存储到内存里。

## 14.3 总结

当循环变量从2增到9时，循环语句对应的汇编代码如下：

指令清单14.15 x86

```assembly
mov [counter], 2 ; initialization
jmp check
body:
  ; loop body
  ; do something
  ; use counter in local stack
  add [counter], 1 ; increment
check:
  cmp [counter], 9
  jne body
```

如果没有开启编译器的优化功能，那么循环变量的增操作可能对应3条汇编指令。

指令清单14.16 x86

```assembly
MOV [counter], 2 ; initialization
JMP check
body:
  ; loop body
  ; do something
  ; use counter in local stack
  MOV REG, [counter] ; increment
  INC REG
  MOV [counter], REG
check:
  CMP [counter], 9
  JLE body
```

当循环体短时，编译器可能给循环变量分配用的寄存器。

指令清单14.17 x86

```assembly
MOV EBX, 2 ; initialization
JMP check
body:
  ; loop body
  ; do something
  ; use counter in EBX, but not modify it
  INC EBX ; increment
check:
  CMP EBX, 9
  JLE body
```

编译器还可能调部分的顺序。

指令清单14.18 x86

```assembly
MOV [counter], 2 ; initialization
JMP label_check
label_increment:
ADD [counter], 1 ; increment
label_check:
CMP [counter], 10
JGE exit
; loop body
  ; do something
  ; use counter in local stack
JMP label_increment
```

通常情况下，程序应当判断循环条件是否满足，但是在编译器优化的情况下，它可能会交换循环体和判断语句的顺序。下面这个就是一个典型的例子。

指令清单14.19 x86

```assembly
MOV REG, 2 ; initialization
body:
  ; loop body
  ; do something
  ; use counter in REG, but not modify it
  INC REG; increment
  CMP REG, 10
JL body
```

编译器不会使用LOOP指令。如果某代码带有LOOP指令，那么应当认为它是手写的汇编程序。

指令清单14.20 x86

```assembly
; count from 10 to 1
MOV ECX, 10
body:
  ; loop body
  ; do something
  ; use counter in ECX, but not modify it!
LOOP body
```

ARM平台的R4寄存器专门存储循环变量。

指令清单14.21 ARM

```assembly
MOV R4, 2 ; initialization
B check
body:
```

```assembly
; loop body
  ; do something
  ; use counter in R4, but not modify it!
ADD R4,R4, #1; increment
check:
CMP R4, #10
BLT body
```
