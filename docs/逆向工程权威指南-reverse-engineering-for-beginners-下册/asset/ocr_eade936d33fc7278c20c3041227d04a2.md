才的修改方法可以迁移指令流。除此之外我们还可以强制程序进行
转我们可以把jcc对应的opcode的第一个字母替换为“EB”，不去修改第二
节(offset)。这种方法把条件指令替换为了无条件跳转指令。经
过的这样的调整之后，本例中的可执行文件都会无条件显示错误处理信息
“What you entered? Huh?

# 7.3.5 MSVC:x64

本例使用的变量是int整数变量。在x64系统中，int型变量还是32位
数据。在64位平台上访问寄存器的（低）32位时，计算机就需要使用助记符
E-头的寄存器名称。而在汇编语言中x64的64位指针中，我们就需要使用R-
头的寄存器名称，处理完善的64位数据。

```
DATA SEGMENT
$SG2924 DB  'Enter X: ', 0AH, 0H
$SG2926 DB  '%d', 00H
$SG2927 DB  'You entered %d ', 0AH, 0H
$SG2929 DB  'What you entered? Huh?', 0AH, 0H
_DATA      ENDS

TEXT SEGMENT
x$ = 32
main  PROC
$LN5:
    sub     rsp, 56
    lea     rcx, OFFSET FLAT:$SG2924 ; 'Enter X: '
    call    printf
    mov     rdx, QWORD PTR x$[rsp]
    lea     rcx, OFFSET FLAT:$SG2926 ; '%d'
    call    scanff
    cmp     eax, 1
    jne     SHORT LN2@main
    mov     edx, DWORD PTR x$[rsp] ; 'You entered %d...'
    lea     rcx, OFFSET FLAT:$SG2927
    call    printf
    jmp     SHORT LN1@main
$LN2@main:
    lea     rcx, OFFSET FLAT:$SG2929 ; 'What you entered? Huh?'
    call    printf
$LN1@main:
    ; return 0
    xor     eax, eax
    add     rsp, 56
    ret
main  ENDP
```
