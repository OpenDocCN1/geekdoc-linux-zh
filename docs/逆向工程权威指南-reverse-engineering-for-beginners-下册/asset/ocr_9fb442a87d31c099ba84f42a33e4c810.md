可rotate1()的源码如下:

void rotate2(int v)
{
    bool tmp[8]; // internal array
    int i, j;

    for (i=0; i<8; i++)
        for (j=0; j<8; j++)
            tmp[i][j]=get_bit (v, i, j);

}
