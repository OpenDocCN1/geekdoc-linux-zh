对应章节: 21.7.1节

这个程序将显示文件所有人的 uscr ID。其源代码如下所示

```
#include <sys/types.h>
#include <sys/stat.h>
#include <time.h>
#include <stdio.h>
#include <stdlib.h>

int main(int argc, char *argv[])
{
    struct stat sb;

    if (argc == 2)
    {
        fprintf(stderr, "Usage: %s pathname\n", argv[0]);
        return 0;
    }
    if (stat(argv[1], &sb) == -1)
    {
        // error
        return 0;
    }

    printf("%ld\n", (long) sb.st_uid);
}
```
