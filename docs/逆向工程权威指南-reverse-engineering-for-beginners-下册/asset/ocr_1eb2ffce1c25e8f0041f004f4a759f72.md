    test ecx, ecx
    je SHORT $LN11@f
$LN5@f:
    test eax, eax
    jne SHORT $LN1@f
    mov eax, 1
    jmp SHORT $LN12@f
$LN1@f:
    lea edx, DWORD PTR [rax-1]
    mov ecx, ebx

$LN12@f:
    dec ebx
    test ebx, ebx
    jne SHORT $LN5@f
