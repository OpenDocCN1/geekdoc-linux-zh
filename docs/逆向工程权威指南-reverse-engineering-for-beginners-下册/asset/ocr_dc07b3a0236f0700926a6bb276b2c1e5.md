```
public f
proc near
arg_0    = dword ptr 8
arg_4    = dword ptr 0Ch
arg_8    = dword ptr 10h

push     ebp
mov      ebp, esp
mov      eax, [ebp+arg_0]; ; 1st argument
imul     eax, [ebp+arg_4]; ; 2nd argument
add      eax, [ebp+arg_8]; ; 3rd argument
pop      ebp
retn
endp
```

```
public main
proc near
var_10   = dword ptr -10h
var_C   = dword ptr -0Ch
var_8   = dword ptr -8

push     ebp
mov      ebp, esp
and      esp, 0xFFFFFF0h
sub      esp, 10h
mov      [ebp+var_8], 3; ; 3rd argument
mov      [ebp+var_C], 2; ; 2nd argument
mov      [ebp+var_10], 1; ; 1st argument
call     f
mov      edx, offset AD ; "%d\n"
mov      [ebp+var_C], eax
mov      [ebp+var_10], edx
call     _printf
mov      eax, 0
leave
retn
endp
```

GCC的编译结果和MSVC的编译结果十分相似。

不同的是两个调用函数(f和printf)没有还原指针SP。这是由于函数尾部的倒数第二条指令——LEAVE指令（参见附录A.6.2）能够还原指针。
