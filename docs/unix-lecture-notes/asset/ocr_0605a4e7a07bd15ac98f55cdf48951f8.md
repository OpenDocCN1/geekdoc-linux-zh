    fprintf(stderr , "upcased is not reading the pipe.\n");
    unlink(msg.raw_text_fifo);
    unlink(msg.converted_text_fifo);

void on_signal( int sig )
{
    close( publicfifo );
    if ( convertedtext_fd != -1 )
        close(convertedtext_fd);
    if ( rawtext_fd != -1 )
        close(rawtext_fd);
    unlink(msg.converted_text_fifo);
    unlink(msg.raw_text_fifo);
    exit(0);
}

void clean_up()
{
    close( publicfifo );
    close(rawtext_fd);
    unlink(msg.converted_text_fifo);
    unlink(msg.raw_text_fifo);
