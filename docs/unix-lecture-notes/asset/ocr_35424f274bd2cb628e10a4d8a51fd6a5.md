while ( (c = fgetc( fp )) != EOF ) {

     * to using it.
     */
    switch ( c ) {
        case 'q' :            /* 'q' pressed */
            return QUIT;
        case ' ' :            /* ' ' pressed */
            return SPACEBAR;
        case '\n' :           /* Enter key pressed */
            return RETURN;
        default :             /* invalid if anything else */
            return INVALID;
    }
    return INVALID;
}
