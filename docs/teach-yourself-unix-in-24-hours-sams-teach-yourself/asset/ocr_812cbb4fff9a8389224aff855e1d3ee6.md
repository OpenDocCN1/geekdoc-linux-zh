```
# Set up shell environment:
alias diff    /usr/bin/diff -c w
alias from    "frm -n"
alias ll      "ls -l"
alias ls      /bin/ls -F"
alias mail
alias mailq  /usr/lib/sendmail -bp
```

```
alias newaliases "echo you mean newelias..."
alias rd      *readmsg S| page"

# and some special stuff if we're in an interactive shell
($ ?prompt) # shell is interactive.

```
alias cd      *chdir \*; setprompt
alias env     *printenv
alias setprompt *setprompt=$system (cwd): \! : "
