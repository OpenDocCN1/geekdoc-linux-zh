# This function described in dot.array.bash
# dot_array <array_name>
dot_array() {
    if [ $# -ne 1 ] || return 1; # Single argument required.
    local -a da_input
    eval da_input=("${$1[@]}")
    local IFS=${DOT_IFS}
    local da_output=${da_input[0]}
    IFS=${WSP_IFS}
    echo ${da_output}
    return 0
}

# This function described in file_to_array.bash
# file_to_array <name> <line_name>
file_to_array() {
    if [ $# -ne 2 ] || return 1; # Two arguments required.
    local IFS=${NO_WSP}
    local -a fta_tmp
    fta_tmp=($(cat $1))
    eval $2=("${fta_tmp[@]}")
    return 0
}

# Columized print of an array of multi-field strings.
# col_print <array_name> <min_space>
# +tab_stop [tabs_stops]
col_print() {
    if [ $# -lt 2 ] || return 0;
    local -a cp_inp
    local -a cp_sp
    local -a cp_line
    local cp_min
    local cp_cnt
    local cp_pos
    local cp_tab
    local -i _cp
    local -i _cpf
    local cp_fld
    # WARNING: FOLLOWING LINE NOT BLANK - IT IS QUOTED.
    local _cp_max
    set -f
    local IFS=${NO_WSP}
    eval cp_inp=("${$1[@]}")
    if [ ${#cp_inp[@]} -eq 0 ] || return 0; # Empty is easy.

    _cp_max=$2
    cp_min=${2:1}
    shift
    shift
    cp_cnt=0
    for ((cp=0; _cp < cp_cnt; _cp++));
    do
        cp_sp[$cp]=${_cp_max:cp_min}"
        shift
    done
    cp_cnt=${#cp_inp[@]}
    for ((cp=0; _cp < cp_cnt; _cp++));
    do
        cp_pos=1
        IFS=' '
        cp_line=("${cp_inp[$cp]}")
        IFS=${WSP_IFS}
        for ((cpf=0; _cpf < ${#cp_line[@]}; _cpf++));
        do
            cp_tab=${cp_sp[$cpf]}:${cp_pos}
            if [ "${#cp_tab}" -lt "${cp_min}" ];
            then
                cp_tab="${cp_min}"
            fi
            echo -n "${cp_tab}"
            ((cp_pos += ${#cp_tab}))
            cp_fld=${cp_line[$cpf]}
            echo -n "${cp_fld} "
            ((cp_pos += ${#cp_fld}))
        done
    done
    set +f
    return 0
}
