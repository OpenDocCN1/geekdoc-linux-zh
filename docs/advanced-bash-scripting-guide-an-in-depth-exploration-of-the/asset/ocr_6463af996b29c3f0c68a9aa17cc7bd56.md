pm -i --test $1 # Query whether rpm file can be installed.
if [ "$2" -eq SUCCESS ];
then
    echo "$1 can be installed."
else
    echo "$1 cannot be installed."
fi
echo # End code block.

} > "$1.test"

echo "Results of rpm test in file $1.test"
# See rpm man page for explanation of options.
