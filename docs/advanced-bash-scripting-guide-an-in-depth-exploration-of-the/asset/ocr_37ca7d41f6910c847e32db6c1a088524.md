prohame () 
{
    echo -n "Program name? "n    read name

    echo -n "Manpage section? [Hit RETURN for default (\"1\\)] "n    read section
    if [ -z "$section" ]
    then
        section=l # Most man pages are in section 1.
    fi

    if [ -n "$name" ]
    then
        savefile="$name".$section" # Filename suffix - section.
        echo -n "$1" >>$savefile
        name=$(echo "$name" | tr a-z A-Z) # Change to uppercase,
        echo -n "$name" >>$savefile # per man page convention.
    else
        echo "Error! No input." # Mandatory input.
        exit $E_NOINPUT # Critical!
        # Exercise: The script-abort if no filename input is a bit clumsy.
        # Rewrite this section so a default filename is used
        if no input.
    fi

    echo -n "\$name\">>$savefile # Append, always append.

    echo -n "Version? " 
    read ver
    echo -n "\Version \$ver>>$savefile

    echo -n "Short description [0 5 words]" 
    read dsdesc
    echo "$NameHeader">>$savefile
    echo "$BOLD "$name">>$savefile
    echo "-\$dsdesc>>$savefile
}
