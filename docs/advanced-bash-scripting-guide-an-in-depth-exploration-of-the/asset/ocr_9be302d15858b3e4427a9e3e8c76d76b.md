# Advanced Bash-Scripting Guide

```
# Adds some text in the terminal frame (if applicable)

function xtitle()
{
    case "$TERM" in
        *term | rxvt)
            echo -en \0;$*\a"
        *)
            ;;
    esac
}
```

```
# Aliases use xtitle
alias top='xtitle
alias make='xtitle Making $(basename PWD); make

# and functions
function man()
{
    for i; do
        xtitle The $(basename $i|tr -d .[:digit:]]) manual
        command -a "$i"
    done
}
```

```
# Make the following commands run in background automatically:

function te()  # wrapper around emacs/gnuserv
{
    if [ "$(gnucle -batch -el 2>&-)" == "t" ]; then
        gnucle -q "$@"
    else
        emacs "$@"
    fi
}
```

```
function soff() { command soff "$@" }
function firefox() { command firefox "$@" }
function xpdf() { command xpdf "$@" }
```

```
# File & strings related functions:
# Find a file with a pattern in name:
function ff() { find . -type f -iname '*"$1*\' -ls; }
```

```
# Find a file with pattern $1 in name and Execute $2 on it:
function fe() { find . -type f -iname '*"$1*\' -maxdepth 1 -exec ${2:(2-file)} {} \; }
```
