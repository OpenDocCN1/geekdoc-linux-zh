# Example 12-1. Stupid tricks

```
#!/bin/bash
# stupid-tricks.sh: Don't try this at home, folks.
# From "Stupid Script Tricks," Volume I.

exit 99 ### Comment out this line if you do
dangerous_variable=cat /boot/mlinuz # The compressed Linux kernel itself.
echo "string-length of \$dangerous_variable = ${$dangerous_variable}"
# string-length of $dangerous_variable = 794151
# (Never kernels are bigger).
# Does not give same count as 'wc -c /boot/vmlinuz'.

echo "$dangerous_variable"
# Don't try this! It would hang the script.
```
