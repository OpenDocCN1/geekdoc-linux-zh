```
# A valid reply is at least 4 digits plus 3 dots.
if [ ${$ip_adr} -gt 6 ]
then
  echo
  declare query

  # Parse by assignment at the dots.
  declare -a dns
  IFS=$ADR_IFS
  dns=(${$ip_adr})
  IFS=$WSP_IFS

  # Reorder octets into dns query order.
  rev_dns='$${dns[3]}'.'${dns[2]}'.'${dns[1]}'.'${dns[0]}'
```
