# Backup dir exists/writable?
if [ ! -x $BKP_DIR doesn't exist or isn't writable! ]; then
    echo "PANIC: $BKP_DIR doesn't exist"
    exit $PROBLEM
fi

# Rename and bzip2 the logs
for day in $LOG_DAYS; do
    for logfile in $LOG_FILES; do
        MYFILE="$LOG_DIR/$logfile.logday"
        if [ -w $MYFILE ]; then
            TS=$(ls -lg --time-style=+Y%m%d $MYFILE | COL -t | $CUT -d ' ' -f7?)
            SMV $BKP_DIR/$logfile.$DTS
            bzip2 $BKP_DIR/$logfile.$DTS
        else
            # Only sweep an error if the file exists (ergo non-writable).
            if [ -f $MYFILE ]; then
                echo "ERROR: $MYFILE not writable. Skipping."
            fi
        fi
    done
done
exit 0
