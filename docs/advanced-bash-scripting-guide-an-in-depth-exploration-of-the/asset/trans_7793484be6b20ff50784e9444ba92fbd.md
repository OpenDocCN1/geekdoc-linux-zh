```bash
#!/bin/bash

# show-all-colors.sh
# 使用 ANSI 转义序列显示 256 种可能的背景颜色。
# 作者：Chetankumar Phulgare
# 经许可在 ABS Guide 中。

T1=8
T2=6
T3=36
offset=0

for num1 in {0..7}
do
  for num2 in {0..1}
  do
    shownum=$(echo "$offset + $T1 * $num2" | bc)
    echo -e "\E[48;5;${shownum}m color \E[0m"
  done
done
echo
offset=16
for num1 in {0..5}
do
  for num2 in {0..5}
  do
    for num3 in {0..5}
    do
      shownum=$(echo "$offset + $T2 * $num3" | bc)
      echo -e "\E[48;5;${shownum}m color \E[0m"
    done
  done
done
echo
offset=23
for num1 in {0..23}
do
  shownum=$(echo "$offset + $num1" | bc)
  echo -e "\E[48;5;${shownum}m color \E[0m"
done
echo
```

```bash
extract_letters () {
    local pos=0 # 按字母切分字符串。
    local len=1 # 从字符串左端开始。
    strim=$1 # 一次一个字母。
    while [ $pos -lt ${#strim} ]; do
        letter=${strim:pos:len}
        # 参见第 10.1 节。
        play_letter $letter
        echo -n " " # 标记已播放的字母。
        ((pos++))
    done
}
```
