```
#!/bin/bash

uparrow='\x1[A'
downarrow='\x1[B'
leftarrow='\x1[D'
rightarrow='\x1[C'

read -s -n3 p "Hit an arrow key: " x

case $x" in
$uparrow)
    echo "You pressed up-arrow"
    ;;
$downarrow)
    echo "You pressed down-arrow"
    ;;
$leftarrow)
    echo "You pressed left-arrow"
    ;;
$rightarrow)
    echo "You pressed right-arrow"
    ;;
esac
```
