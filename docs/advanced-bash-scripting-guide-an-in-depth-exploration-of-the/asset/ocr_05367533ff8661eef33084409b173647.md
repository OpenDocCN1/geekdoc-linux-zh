# Advanced Bash-Scripting Guide

```
# isalpha.sh: Using a "case" structure to filter a string.

SUCCESS=0
FAILURE=-1
  # Was FAILURE=-,
  # but Bash no longer allows negative return value.

isalpha () {
  # Tests whether *first character* of input string is alphabetic.
  if [ -z "$1" ]; then
    # No argument passed?
    return $FAILURE
  fi
  case "$1" in
    [a-zA-Z]*) return $SUCCESS;; # Begins with a letter?
    *) return $FAILURE;
  esac
}
# Compare with "isalpha()" function in C.
isalpha2 () {
  # Tests whether "entire string" is alphabetic.
  [ $ -eq 1 ] || return $FAILURE

  case $1 in
    *[!a-zA-Z]*\*) return $FAILURE;
    *) return $SUCCESS;
  esac
}
disdigit () {
  # Tests whether "entire string" is numerical.
  # In other order, tests for integer variable.
  [ $ -eq 1 ] || return $FAILURE

  case $1 in
    *[!0-9]*\*) return $FAILURE;
    *) return $SUCCESS;
  esac
}
```

```
check_var () {
  if isalpha "$0"; then
    echo "\$0\*" begins with an alpha character.
    if isalpha2 "$0"; then
      # No point in testing if first char is non-alpha.
      echo "\$0\*" contains only alpha characters.
    else
      echo "\$0\*" contains at least one non-alpha character.
    fi
  else
    echo "\$0\*" begins with a non-alpha character.
    # Also "non-alpha" if no argument passed.
  fi
}

echo
```
