#!/bin/bash

while true
do
    read -snl a
    test "$a" == 'echo -en \\e' || continue
    read -snl a
    test "$a" == "[" || continue
    read -snl a
    case "$a" in
        A) echo "up";
        B) echo "down";
        C) echo "right";
        D) echo "left";
    esac

# Example 15-7. Using read with file redirection

```
#!/bin/bash

read var1 <data-file
echo var1 = $var1
# var1 set to the entire first line of the input file "data-file"

read var2 <data-file
echo var2 = $var2 var3 = $var3
# Note non-intuitive behavior of "read"
# 1) Rewinds back to the beginning of input file.
# 2) Each variable is now set to a corresponding string,
   separated by whitespace, rather than an entire line of text.
```
