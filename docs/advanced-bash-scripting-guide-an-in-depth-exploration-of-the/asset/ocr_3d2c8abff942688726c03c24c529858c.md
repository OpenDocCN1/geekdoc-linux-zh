local i
local rowcheck

for ((i=0; i<$element_count; i++))
do

# Insert newline at end of each row.
let "rowcheck = $i COLS"
if [ "$rowcheck" -eq 0 ];
then
    echo # Newline.
    echo -n "    # Indent.
fi

cell=$$(arr[i])

if [ "$cell" = . ];
then
    let "alive += 1"
fi

echo -n "$cell" | sed -e 's/_/ /g'
# Print out array, changing underscores to spaces.
done
return

IsValid () # Test cell coordinate valid.
{
    if [ -z "$1" -o -z "$2" ];
then
        return FALSE
    fi
    local row
    local lower_limit=0 # Disallow negative coordinate.
    local upper_limit
    local left
    local right
    let "upper_limit = $COLS-1" # Total number of cells.
    if [ "$1" -lt "$lower_limit" -o "$1" -gt "$upper_limit" ];
then
        return FALSE # Out of array bounds.
    fi
    row=$2
    let left = $row * $COLS # Left limit.
    let right = $left + $COLS - 1 # Right limit.

    if [ "$1" -lt "$left" -o "$1" -gt "$right" ];
then
        return FALSE # Beyond row boundary.
    fi
    return $TRUE # Valid coordinate.
}
