```
process_text () {
    while read line;            # Read one line at a time
    do
        if [ ! "$line" ]         # Blank line?
        then
            echo                    # Then new paragraph must follow.
            echo "$SLF"\           # Insert two <br> tags.
            echo "$SLF"
            echo
            continue                # Skip the underscore test.
        else                        # Otherwise .
        fi
        if [ "$line" = "*\.jpg" ]; # Is a graphic?
        then\                        # Strip away brackets.
            temp=$(echo "$line" | sed 's/\\/\\\\/g' -e 's/\//g')
            line="<CENTER><img src="$temp">$END_CENTER>" # Add image tag.
        else                        # And, center.
        fi
    }
```
