include=
exclude=
  for p in "${INCLUDE[@]}"; do
    include=\$include --$p\"
  done
  for p in "${CLUDE[@]}"; do
    exclude=\$exclude --$p\"
  done
}

echo -n "Retrieving and refining update list ..."
# Retrieve list -- 'eval' is needed to run rsync as a single command.
# $3 and $4 is the date and time of file creation.
# $5 is the full package name.
previous=
pre_file=
pre_date=0
eval /bin/nice /usr/bin/rsync \
  -r rsync exclude $SURL | \
  grep 'dr.*-r' | \
  awk '{print $3, $4, $5}' | \
  sort -k3 | \
  while read line; do
    # Get seconds since epoch, to filter out obsolete pkgs.
    cur_date=$(date -d "$line | awk '{print $1, $2}' +%s)
    echo $cur_date

# Get file name.
cur_file=$(echo $line | awk '{print $3}')
echo $cur_file
# Get rpm pkg name from file name, if possible.
if [ "$cur_file" == *.rpm ]; then
  pkg_name=$(echo $cur_file | sed -r -e '\/s/([a-z0-9_-]+):([0-9]+\).*/\1/')
else
  pkg_name=
fi
echo $pkg_name
if [ -z "$pkg_name" ]; then # If not a rpm file,
  echo $cur_file >> $TMP # then append to download list.
elif [ "$pkg_name" != "$previous" ]; then # A new pkg found.
  echo $S_pkg_file >> $TMP # Output latest file.
  previous=$pkg_name
  pre_date=$cur_date
  pre_file=$cur_file
