function strcmp ()
{
  if [ -z "$3" -o "$3" -le "0" ]; then
    return 0
  fi

  if [ $3 -ge $($1) -a $3 -ge $($2) ]; then
    strcmp "$1" "$2"
    return $?
  else
    s1=${1:0:$3}
    s2=${2:0:$3}
    strcmp $s1 $s2
    return $?
  fi
}

function strstr ()
{
    # 如果 s2 指向零长度的字符串，strstr 输出 s1
    if [ "$s2" -eq 0 ]; then echo "$s1"; return 0; fi

    # 如果 s2 不在 s1 中，strstr 不输出任何内容
    case "$s1" in
        *$s2)
            return 0;;
    esac

    # 使用模式匹配代码剥离匹配项及其
    # 其后的所有内容
    first="${s1//$2*/}"

    # 然后剥离字符串中第一个不匹配的部分
    echo "${s1#$first}"
}
