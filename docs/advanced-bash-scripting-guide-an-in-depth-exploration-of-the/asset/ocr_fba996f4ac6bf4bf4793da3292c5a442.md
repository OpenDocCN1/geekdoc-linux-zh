decide_move () {
  # Which move will we make?
  for mov in {1..8};
  do
    # But, fails on startpos=37 !!!
    let square=$scurpos + ${moves[mov]}
    is_on_board $square
    if [[ $7 -eq $SUCCESS && ${board[square]} -eq $UNVISITED ]];
    then # Find accessible square with least possible moves.
      # What happens if the knight wanders towards the outer edge
      # +of the board, then perhaps miserly inward.
      # Given two or more moves with same value of
      # +least-future-moves, this implementation chooses
      # +the_first_ of moves.
      # This means there is not necessarily a unique solution
      # +for any given starting position.
    fi
  }
