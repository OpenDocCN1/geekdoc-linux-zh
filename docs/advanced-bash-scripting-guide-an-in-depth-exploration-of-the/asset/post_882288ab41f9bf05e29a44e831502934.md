## A-56. The Gronsfeld Cipher

```bash
#!/bin/bash
# gronsfeld.bash

# License: GPL
# This is an implementation of a stripped-down variant of the Cipher.
# It's essentially polyalphabetic Vigenère Tableau, but with only digits.
# The classic Gronsfeld has a numeric sequence as key, but here we substitute a letter for ease of use.
# Although, this cipher was invented by Count Gronsfeld in the 17th Century, it was one time considered to be unbreakable.
# Note that this is a secure cipher by modern standards.

# Global Variables
Enc-suffix="###"
Default_key=gronsfeld # The script uses this if key not entered below.
# Change the two values frequently for security.

GROUPPLEN=5          # Output in groups of 5 letters, per tradition.
alpha=(abcdefghijklmnopqrstuvwxyz) # Output in all caps, per tradition.
alpha2=(A..Z)          # Use alpha2=(a..z) for password generator.
wraplen=26              # crypt if past end of alphabet.
dflag=                   # flag (if set $Enc-suffix present).
E_NOARGS             # Missing command-line args?
DEBUG=0                # Debugging flag.
declare -a offsets       # This array holds the numeric shift values for encryption/decryption.

# Function: (Encrypt or decrypt, depending on whether dflag is set. Why : as a function name? Just to prove that it can be done.)
local idx keyxd mlen shft
local plaintext="$1"
local mlen=${#plaintext}

for (( idx=0; idx<mlen; idx++ ))
do
    let keyxd=$idx % $keylen
    shft=${offsets[keyxd]}

    if [ -z "$dflag" ]; then
        let off=$(expr "[[:al]*]" ${plaintext:idx:1}) - $shft
        # Shift backward to decrypt.
    else
        # Encrypt!
        let off=$(expr "[[:al]*]" ${plaintext:idx:1}) + $shft
        # Shift forward to encrypt.
        test (( idx % $GROPLEN )) == 0 && echo -n " " # Groups of 5 letters.
        # Comment out above line for output as a string without whitespace, for example, if using the script as a password generator.
    fi

    ((off--)) # Normalize. Why is this necessary?

    if [ $off -lt 0 ]; then
        # Catch negative indices.
        let off +=$wraplen
    fi
    let off=$((off % wraplen)) # Wrap around if end of alphabet.

    echo -n "${alpha2[$off]}"
done

    if [ -z "$dflag" ]; then
        echo "$Enc_suffix"
        echo "$Enc_suffix" # For password generator.
    else
        echo
    fi
) # End encrypt/decrypt function.

# Check for command-line args.
if [ -z "$1" ]; then
    echo "Usage: $0 TEXT TO ENCODE/DECODE"
    exit $E_NOARGS
fi

if [ "$1" = "$Enc_suffix" ]; # Final command-line arg.
  dflag=ON
  echo -n "+" # Flag decrypted text with a "+" for easy ID.
fi

if [ -z "$key" ]; then
  key="$Default_key" # "gronsfeldk" per above.
fi
keylen=$#key
```
