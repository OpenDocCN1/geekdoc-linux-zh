num_b=0
r0=${Rows[row_b]}

if [ "$r0" -eq 1 ]
then
    num_b=1
else
    let "num_b = $r0 - 1"
    # Leave only a single peg in the row.
    # Not a very strong strategy,
    # -- but perhaps a bit better than totally random.

let "Rows[row_b]" -= $num_b
echo -n "Bot: "
echo "Removing from row $row_b ..."

if [ "$num_b" -eq 1 ]
then
    peg_msg=$peg
else
    peg_msg=$peg
fi

echo "$num_b $peg_msg."

then
    echo " Bot wins!"
    tput sgr0 # Restore display.
