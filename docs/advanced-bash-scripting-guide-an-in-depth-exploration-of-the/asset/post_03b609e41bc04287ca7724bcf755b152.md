# Example 11-8. A grep replacement for binary files

```bash
#!/bin/bash
# bin-grep.sh: Locates matching strings in a binary file.

# A "grep" replacement for binary files.
# Similar effect to "grep -a"

E_BADARGS=65
E_NOFILE=66

if [ $# -ne 2 ];
then
    echo "Usage: $0 basename search_string filename"
    exit $E_BADARGS
fi

if [ ! -f "$2" ];
then
    echo "File \$2" does not exist.\n    exit $E_NOFILE
fi

IFS=$'\01'
# Per suggestion of Anton Filippov.
# was: IFS=$'\' | grep "$1"
for word in $strings; do
    echo "$2" | grep "$1"
    do
    echo $word
done

# The "strings" command lists in binary files.
# Output piped to "grep", which tests for desired result.

# As S.C. points out, lines 23 - 30 could be replaced with the simpler
strings "$2" | grep "$1" | tr -s "$IFS" '[\n*]'
```

# Example 11-10. Checking all the binaries in a directory for authorship

```bash
#!/bin/bash
# findstring.sh
# find a particular string in the binaries of a specified directory
directory=/usr/bin/
fstring="Free Software Foundation" # See which files come from the FSF.

for file in $(find $directory -type f -name '*' | sort)
do
  strings = $file | grep "$fstring" | sed -e "s/$directory%%"
  # In the "sed" expression,
  #+ it is necessary to substitute for the normal "/ delimiter
  # because / happens to be one of the characters filtered out.
  # Failure to do gives an error message. (Try it.)
done
```
