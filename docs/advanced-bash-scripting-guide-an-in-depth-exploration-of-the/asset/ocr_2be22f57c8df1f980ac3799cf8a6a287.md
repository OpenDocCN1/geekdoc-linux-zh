split_input() {
  [$#uc_name[@]}] -gt 0 ] || return 0
  local -i _si_cnt
  local -i _si_len
  local _si_str
  unique_lines uc_name uc_name
  _si_cnt=${#uc_name[@]}
  for ((i = 0; $i < _si_cnt; ++i++))
  do
    _si_str=${uc_name[$i]}
    if is_address "${uc_address[$i]}"
    then
      uc_address[${uc_address[$i]}]=${_si_str}
      unset uc_name[$i]
    else
      if uc_name[$i] == "${name_fix ${_si_str}}"
      then
        unset uc_name[$i]
      fi
    fi
dnd

# limit_chk() <next_level>
limit_chk {
  local -i _lc_lmt
  # Check direction limit.
  if [($indirect) -eq 0 ] || [ $# -eq 0 ];
  then
    # The 'do-foreover' choice
    echo 1 # Any value will do.
    return 0 # OK to continue.
  else
    # Limiting is in effect.
    if [($indirect) -lt 1 ]
    then
      echo $1 # Whatever.
      return 1 # Stop here.
    else
      _lc_lmt=$($1)+1 # Bump the given limit.
      echo ${_lc_lmt # Echo it.
      return 0 # OK to continue.
    fi
}
