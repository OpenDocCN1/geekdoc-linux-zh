```sh
# /bin/sh
# letter-count2.sh: Counting letter occurrences in a text file.
# Script by nyal [nyal@voila.fr].
# Used in ABS Guide with permission.
# Recommended and reformatted by ABS Guide author.
# Version 1.1: Modified to work with gawk 3.1.3.

INIT_TAB_AWK=""
# Parameter to initialize awk script.
count_case=0
FILE_PARSE=$1

E_PARAMETRR=85

usage() {
  echo "Usage: letter-count.sh file letters" 2>&1
  # For example: ./letter-count2.sh filename.txt a b c
  exit $E_PARAMETRR Too many arguments passed to script.
}

if [ ! -f "$1" ]; then
  echo "$1: No such file." 2>&1
  usage # Print usage message and exit.
fi

if [ -z "$2" ]; then
  echo "$2: No letters specified." 2>&1
  usage
fi

shift
for letter in "$@";
do
  INIT_TAB_AWK="$INIT_TAB_AWK tabsearch([${count_case}]) = \\letter\; final_tab([${count_case}]) = 0"
  # Pass as parameter to awk script below.
  count_case=expr $count_case + 1
done

DEBUG:
  echo $INIT_TAB_AWK;

cat $FILE_PARSE |
```

# Appendix D. Parsing and Managing Pathnames

Emmanuel Rouat contributed the following example of parsing and transforming filenames, and in particular, pathnames. It draws heavily on the functionality of sed.

```bash
#!/usr/bin/env bash

# Management of PATH, LD_LIBRARY_PATH, MANPATH variables...
# By Emmanuel Rouat <noemail>
# Management of PATH, LD_LIBRARY_PATH, MANPATH variables...
# By Emmanuel Rouat <noemail>

# The following functions handle spaces.
# These functions belong in .bash_profile rather than .bashrc
# I guess the modular structure of these functions makes it easy
# to expand them to handle substitutions instead of path removal...
# See http://www.catman.net/blog/awk-liners-explained/part-two/ (item 4) for explanation of 'duplicate-entries' removal

# Show (usually PATH) as list.
function p_show() {
  local p="$1"; for p in $p; do echo "${p//}"; done
}

# Filter empty lines, multiple/trailing slashes, and duplicate entries.
function p_filter() {
  awk '/^[[:space:]]*$/{next} sub(/\/+$/, "/")
}

# Clean (typically PATH) and rebuild it
function p_build() {
  local p="$1"; echo "$p" | p_filter
}
}

# Remove $ from $2 (found on stackoverflow, with modifications).
function p_rm() {
  local d=$(echo "$1" | p_filter); p=$($2) && eval "$(p_show $p | p_filter | grep -v "${d}" | p_build)"
}

# Same as previous, but filters on pattern (dangerous...).
# don't use 'bin' or */ as pattern!.
function p_rmpat() {
  local d=$(echo "$1" | p_filter); p=$($2) && eval "$(p_show $p | p_filter | grep -v "${d}" | p_build)"
}

# Delete $1 from two and append cleanly.
function p_append() {
  local d=$(echo "$1" | p_filter); p=$($2) p_rm="${d}" && eval "$(p_show $p | d | p_build)"
}

# Delete $1 and prepend cleanly.
function p_prepend() {
  local d=$(echo "$1" | p_filter); p=$($2) && p_rm "$d" && p=$($p_show $d) | p_build
}
```

So, how can you process filenames in shell? Here's a quick summary about how to do it correctly, for the impatient "just want the answer". In short: Double-quote to use "Variable" instead of $variable, set IFS to join name and tab, prefix all globs/filenames so they cannot begin with "*" when expanded, and use one of a few templates that work correctly. Here are some of those templates that work correctly:

```bash
IFS=$(printf '\n')
# Remove SPACE, so filenames with spaces work well.

# Correct glob use:
## always use "for" loop, prefix glob, check for existence:
for file in ./*; do
  # Use "*" ... NEVER bare "*"
  if [ -f "$file" ]; then
    # Make sure it isn't an empty match.
    COMMAND ... "$file"
  fi
done
```

```bash
# Correct glob use, requires nonstandard bash extension.
shopt -s nullglob # Bash extension.
# that empty glob matches will work.
for file in ./*; do
  # Use "*/*", NEVER bare "*"
  COMMAND "$file"
done
```
