    then
        if $(TEST) -z "$cd_npwd"

        fi
    fi
    tm=$(echo "$(cd_npwd)" | cut -b 1)
    if $(TEST) "$(tm)" = "_"
    then
        pm=$(echo "$(cd_npwd)" | cut -b 2)
        nm=$(echo "$(cd_npwd)" | cut -d $ -f2)
        case "$pm" in
            a) cd_npwd=$(cd_sugg[$nm]);
            s) cd_npwd=$CDS[$nm]);
            S) cd_npwd=$CDS[$nm]); $CDS[pwd]=;
            r) cd_npwd="$2"; $cd_SpecDir=$nm "$1" "$2";
            R) cd_npwd="$2"; $CDS[$nm] = pwd; cd_doselection "$1" "$2";
        esac
    fi
