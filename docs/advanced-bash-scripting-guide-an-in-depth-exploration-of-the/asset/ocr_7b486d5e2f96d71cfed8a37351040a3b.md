Show_Slots () {
    echo; echo
    echo -n " $( seq $NUMSLOTS # Pretty-print array elements.
    for i in $(seq $NUMSLOTS); do
        printf "%3d" "${Slots[$i]}" # Allot three spaces per result.
    done
    echo # Row of slots:
    echo "__________________________________________"
    echo "__________________________________________"
    echo # Note if the count within any particular slot exceeds 99,
    # it misses up the display.
    # Running only (!) 500 passes usually avoids this.
}

Move () {
    Move=$RANDOM # Move right / left, or stay put.
    Move=$%RANDOM # How random is $RANDOM? Well, let's see
    let Move=$RANGE # Normalize range of 0 - 2.
    case "$Move" in
        0) # Do nothing, i.e., stay in place.
        1) ((--POS)); # Left.
        2) ((POS)); # Right.
        *) echo -n "Error"; # Anomaly! (Should never occur.)
    esac
}

Play () {\ # Single pass (inner loop).
    i=0
    while [ "$i" -lt "$ROWS" ]; do # One event per row.
        Move
        ((i++));
    done

    SHIFT=11
let POS += $SHIFT # Why 11, and not 10?
    # Shift "zero position" to center.
    (( Slots[$SPOS])); # DEBUG: echo $SPOS
    echo -n "$SPOS"
}

Run () {\ # Outer loop.
    p=0
    while [ "$p" -lt "$PASSES" ]; do
        Play
        ((p++));
        POS=0 # Reset to zero. Why?
    done
}
