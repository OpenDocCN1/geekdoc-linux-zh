play_letter () {
    eval $morse[$1] # Play dots, dashes from appropriate sound letters.
    # Why is 'eval' necessary here?
    usleep $LINESPACE # Pause between letters.
}
```

```
extract_letters () {
    local pos=0 # Slice string argument, letter by letter.
    local len=1 # Starting at left end of string.
    strim=$1 # One letter at a time.
    while [ $pos -lt ${#strim} ]; do
        letter=${strim:pos:len}
        # See Chapter 10.1.
        play_letter $letter
        echo -n " " # Mark letter just played.
        ((pos++))
    done
}
```

```
no_args () {
    declare -a usage
    usage=( "Usage: word1 word2 ..." )

    echo "Usage:"; echo
    echo "${usage[*]}"
    for index in 0 1 2; do
        extract_letters ${usage[$index]}
        usleep $WORDSPACE
        echo -n " " # Print space between words.
    done
    echo "Usage: $0 word1 word2 ..."
    echo; echo
}
