# list_array <array_name>
list_array() {
  [ -eq 1 ] || return 1 # One argument required.

  local -a _la_lines
  set -f $NO_WSP
  eval _la_lines=( ${$1[@]} )
  echo
  echo 'Element count ${#_la_lines[@]} array "${$1}\'\n  ln_cnt=${#_la_lines[@]}

  for (( _i = 0; _i < ${_ln_cnt}; _i++ ));
    do
      echo 'Element \'_$_i\>'${_la_lines[$_i]}'
    done
  set +f
  return 0
}
