```
x=24
y=25
b=`expr $x = $y`      # Test equality
echo "b = $b"      # 0 ( $x -ne $y )

a=3
b=`expr $a \ 10`
echo "b = expr $a > 10, therefore...""
echo "If a > 10, b = 0 (false)"
echo "b = $b"      # 0 ( 3 -lt 10 )
echo
```

```
b=`expr $a < 10`
echo "If a < 10, b = 1 (true)"
echo "b = $b"      # 1 ( 3 -lt 10 )
echo
# Note escaping of operators.
```

```
b=`expr $a <= 3`
echo "If a <= 3, b = 1 (true)"
echo "b = $b"      # 1 ( 3 -le 3 )
# There is also a "\\=" operator (greater than or equal to).
```
