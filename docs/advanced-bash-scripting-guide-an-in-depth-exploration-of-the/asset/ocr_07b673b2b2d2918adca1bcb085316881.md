```
# Use IFS (Internal Field Separator variable) to split a line of input to
# "read", if you do not want the default to be whitespace.

echo "List of all users"
IFS=$IFS; IFS=":" # /etc/passwd ":" for field separator.
while read name passwd uid fullname ignore

  echo "$name ($fullname)"
done </etc/passwd  # I/O redirection.
IFS=$IFS  # Restore original $IFS.
# This code snippet added by Heiner Steven.
```

```
# Setting the $IFS variable within the loop itself
## eliminates the need for storing the original $IFS
## + on a temporary variable.
# Thanks, Dim Segebart, for pointing this out.

echo "----------------------------------------------------------------------"
echo "List of all users"
```

```
while IFS= read name passwd uid fullname ignore
done
  echo "$name ($fullname)"
done </etc/passwd  # I/O redirection.
```
