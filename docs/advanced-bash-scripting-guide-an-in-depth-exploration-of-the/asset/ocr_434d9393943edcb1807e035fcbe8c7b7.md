```
/usr/bin/chroot
/usr/bin/fakefile
/usr/bin/badblocks
/usr/bin/ypbind

# List of files you are curious about.
# Threw in a dummy file, /usr/bin/fakefile.

echo

for file in $FILES
do

if [ ! -e "$file ]    # Check if file exists.
then
    echo "$file does not exist"; echo
    continue
fi

ls -l $file | awk '{ print $8 " " $ $5 }'    # Print 2 fields.
whatis basename $file    # File info.
# Note that the whatis database needs to be set up for this to work.
# To do this, as root run /usr/bin/makewhis
echo
done

exit 0
```

The **[list]** in for loop may be parameterized.

**Example 11-4.** Operating on a parameterized file list

```
#!/bin/bash

filename="*.txt"

for file in $filename
do
    echo "Contents of $file"
    echo "--"
    cat "$file"
    echo
done
```

If the **[list]** in for loop contains wildcards (* and ?) used in filename expansion, *globbing* takes.
