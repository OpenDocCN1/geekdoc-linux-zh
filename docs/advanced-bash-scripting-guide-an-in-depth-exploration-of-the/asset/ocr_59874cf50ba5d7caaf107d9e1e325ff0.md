_lr_dns=${1}'.in.addr.arpa.
echo -n ','

_lr_reply={
  dig +noafail +answer +authority +additional \
  ${_lr_dns} -t so ${_lr_dns} -t any 2>/dev/null
}
_lr_rc?=
if [ ${_lr_rc} -ne 0 ];
then
  trace_log(${_trace_log[0]})='# Dleg lek ekpr '${_lr_rc}' '#
  # ${_lr_rc} -ne 9 } &&
    return ${_lr_rc}
else
  # Some versions of 'dig' return warnings on stdout.
  _lr_cnt=${0; _lr ${_lr_cnt};
  for ((i=0; i < ${_lr_cnt}; i++));
do
    ['x'${_lr_reply[${i}]:0}] == 'x'] &&
      unset _lr_reply[${i}]
  done
  eval 2=\${_lr_reply[\0]\\}
fi
return 0
```

# ## Application specific functions

```
# Mung a possible name; suppresses root and TLDs.
# name_fixup<string>
name_fixup() {
  local -a _nf_tmp
  local -i _nf_end
  local _nf_str
  local IFS
  _nf_str=${to_lower ${1}}
  _nf_str=${to_dot ${_nf_str}}
  _nf_end=${_nf_str}-1
  [ ${_nf_str:${_nf_end}} != '.' ] &&
    _nf_str='_nf_str}'.'
  IFS=${IFS_IFS}
  _nf_tmp=(${nf_str})
  IFS=${WSP_IFS}
  _nf_end=${_nf_tmp[0]}
  case ${_nf_end} in
    0) # Dots, only dots.
      echo
      return 1
    1) # Only a TLD.
      echo
      return 1
    2) # Maybe okay.
      echo ${_nf_str}
      return 0
    # Needs a lookup table?
    if [ ${_nf_tmp[1]} -eq 2 ];
    then
      # Country coded TLD.
      echo
      return 1
    else
  esac
}
