```
case "$#" in
  3) case "$1" in
      -if)
        if=1; shift; 
        return 1;;
      esac
  2) :; # Poor man's "continue"
      return 1;;
esac
```

```
case $if in
  0) eval T1=(\\\\${1}[][\]\\\" \)
      T2=$(echo ${T1[0]} | md5sum -)
      ;;
  1) T2=$(md5sum $1)
      ;;
esac
```

```
case $${T2[@]} in
  0) return 1;;
  1) return 1;;
  2) case ${T2[1]}:0-1 in  # SanScript-2.0.5
      \\*${T2[@2]}-=${T2[1:1]}
      T2[1]=\*\
      ;;
      T2[${T2[@2]}]=${T2[1]}
      T2[1]"
      ;;
    esac
  3) :; # Assume it worked
      return 1;;
esac
```
