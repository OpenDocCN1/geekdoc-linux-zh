#!/bin/bash
# tohtml.sh [v. 0.2.01, reldate: 04/13/12, a teeny bit less buggy]

# Convert a text file to HTML format.
# Author: Mendel Cooper
# License: GPL3
# Usage: sh tohtml.sh <textfile> target htmlfile
# Script can easily be modified to accept source and filenames.

# Assumptions
# 1) Paragraphs in (target) text file are separated by a blank line.
# 2) Jpeg (.jpg) are located in "images".
# In target file, image names are enclosed in square brackets,
# for example, [image01.jpg].
# 3) Emphasized (italic) phrases begin with a space or underscore
# the first character on the line is an underscore and end with space or underscore-of-line.

# Settings
# FNTSIZE=2           # Small-medium font size
# IMGDIR="images"    # Image directory
# Headers
# HDR01="<!DOCTYPE HTML PUBLIC "-//W3//DTD HTML 4.0 Transitional//EN">
# HDR02="<!-- Converted to HTML by ***tohtml.sh*** -->
# HDR03="<!-- author: Leo Cooper <leo@gmail.com> -->"
# HDR11="<head>"
# HDR1a="</head>"
# HDR12="<title>"
# HDR20="</title>"
# HDR21="<meta NAME="GENERATOR" CONTENT="tohtml.sh">
# HDR13="<body bgcolor="#ddd"> # Change background color to size.
# HDR14a="<font size="
# HDR14b="</font>"
# Footers
# FTR10="</body>"
# FTR11="</html>"

# Tags
# BOLD="<b>"
# CENTER="<center>"
# END_CENTER="</center>"
# LF="br>"
