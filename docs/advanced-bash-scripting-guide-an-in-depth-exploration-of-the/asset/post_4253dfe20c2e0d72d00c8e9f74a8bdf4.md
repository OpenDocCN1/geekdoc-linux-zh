Bash permits operations on variables, even if the variables are not explicitly declared as.

```bash
string=ABC123ABCabc
echo ${string[#]} # ABC123ABCabc
echo ${string[*]} # ABC123ABCabc
echo ${string[0]} # ABC123ABCabc
echo ${string[1]} # No output!
echo ${string[@]} # Why?
# 1
# One element in the array.
# The string itself.
```

```bash
echo "----------------------------------"
# starting at position # 1 (2nd character)

# Length of first element of array.
echo ${#array}         # 4
# Length of first element of array.
# (Alternate notation)

# Length of second element of array.
# Arrays in Bash have zero-based indexing.

# Number of elements in array.
echo ${#array[@]}      # 6
# Number of elements in array.

echo "----------------------------------"
array2=( [0]="first element" [1]="second element" [3]="fourth element" )
# Quoting permits embedding whitespace within individual array
echo ${array2[0]}      # first element
echo ${array2[1]}      # second element
echo ${array2[2]}      # 
echo ${array2[3]}      # fourth element
# Skipped in initialization, and therefore null.
echo ${#array2[0]}      # 13 (length of first element)
echo ${#array2[*]}      # 3 (number of elements in array)
exit
```

# Substring Removal
echo ${array[@]:1:2}      # two
# All elements following element[0]

# Removes shortest match from front(s)
echo ${array[@]}%f*}      # two three five
# Applied to all elements of the array.
# Matches "Four" and removes it.

# Longest match from front(s)
echo ${array[@]}##t*}     # two four five
# Applied to all elements of the array.
# Matches "three" and removes it.

# Shortest match from back(s)
echo ${array[@]}%%*h*}     # two four five
# Applied to all elements of the array.
# Matches "three" and removes it.

# Longest match from back(s)
echo ${array[@]}%%*t*}     # two four five
# Applied to all elements of the array.
# Matches "three" and removes it.

echo "-----------------"

# Substring Replacement

# Replace first occurrence of substring with replacement.
echo ${array[@]}//fv/     # one two three four
# Applied to all elements of the array.

# Replace all occurrences of substring.
echo ${array[@]}//ve/ZZ     # one two four five
# Applied to all elements of the array.

```bash
# Delete all occurrences of substring.
# Not specifying a replacement defaults to 'delete'...
echo ${array[@]}//fi/          # one two four five
# Applied to all elements of the array.

# Replace front-end occurrences of substring.
echo ${array[@]}#fi/         # one two four five
# Applied to all elements of the array.

# Replace back-end occurrences of substring.
echo ${array[@]}%%/ve/   # one two three four
# Applied to all elements of the array.

echo ${array[@]}%/          # one two three four five
# Why?
```
