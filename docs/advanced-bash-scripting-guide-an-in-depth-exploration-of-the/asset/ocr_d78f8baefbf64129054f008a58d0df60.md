```
# pend_func function_name $(printf '%q\n' arguments)
pend_func() {
    local IFS=$(NO_WSP)
    set -f
    _pending_[$${#_pending[@]}]=$0
    set +f
}
```

```
# The function which stops the release:
pend_stop_mark() {
    _p_ctrl_=0
}

pend_mark() {
    pend_func pend_stop_mark
}
```

```
# Execute functions until 'pend_stop_mark' . .
pend_release() {
    local -i _top_ # Declare _top_ as integer.
    _p_ctrl=$${#_pending[@]}
    while [ $_p_ctrl -gt 0 ];
    do
        _top_${#_pending[@]}-1
        _pend_current=${_pending[$_top_]}
        unset _pending[$_top_] # Debug only.
        eval $_pend_current_$
    done
}
```

```
# Drop functions until 'pend_stop_mark' . ..
pend_drop() {
    local -i _top_
    local -i _pd_ctrl=$${#_pending[@]}
    while [ $_pd_ctrl -gt 0 ];
    do
        _top_=$${#_pending[@]}-1
        if [ "${_pending[$_top_]}" == 'pend_stop_mark' ];
        then
            unset _pending[$_top_]
            break
        else
            unset _pending[$_top_]
            _pd_ctrl=$_top_
        fi
    done
    fi
    if [ ${#_pending[@]} -eq 0 ];
    then
        pend_func pend_stop_mark
    fi
}
```
