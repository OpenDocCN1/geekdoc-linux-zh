check_database ()
{
  local n=0
  local filename
  local checksum

  # This file check should be unnecessary
  # but better safer than sorry.

  if [ ! -r "$dbfile" ];
  then
    echo "Unable to read checksum database file!"
    exit $E_BAD_DBFILE
  fi

  while read record[n]
  do
    directory_checked="${record[0]}"
    if [ "$directory_checked" != "$directory" ];
    then
      echo "Directories do not match up!"
      # Tried to use file for a different directory.
      exit $E_DIR_MATCH
    fi

    if [ "$Sn" -gt 0 ]; # Not directory name.
    then
      filename[n]=$(echo "${record[n]}" | awk '{ print $2}')
      # mdsum writes records backwards,
      # checksum checksum first, then.
      checksum[n]=$(mdsum "${filename[n]}")
    fi

    if [ "${record[n]}" = "${checksum[n]}" ];
    then
      echo "${filename[n]} unchanged."
    elif [ "$(basename ${filename[n]})" != "$dbfile" ]; ];
      # Skip over checksum database file,
      # as it will change with each invocation of script.
      # This unfortunately means that when running
      # this script on $PWD, tampering with the
      # checksum database file will not be detected.
      # Exercise: Fix this.
      then
        echo "${filename[n]} : CHECKSUM ERROR!"
        # File has been changed since last checked.
      fi
    fi
  let "n+1" # Read from checksum database file.
done "$dbfile";
