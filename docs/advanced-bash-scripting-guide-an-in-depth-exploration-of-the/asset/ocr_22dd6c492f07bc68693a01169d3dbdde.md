Exercise: Analyze the following script.

```
#! /usr/bin/env bash

mkfifo /tmp/fifol /tmp/fifol & exec /tmp/fifol
while read a; do echo "FD1: $a"; done < /tmp/fifol & 7 > /tmp/fifol
exec 8 >(while read a; do echo "FD8: $a", to fd$7")

ec 3>&1
(
    while read a; do echo "FD2: $a"; done < /tmp/fifol | tee /dev/stderr \\
    tee /dev/fd/4 | tee /dev/fd/5 | tee /dev/fd/6 &&7 & exec 3 > /tmp/fifol

    echo lst1, to stdout
    sleep 1
    echo 2nd, to stderr &2
    sleep 1
    echo 3rd, to fd 3 &3
    sleep 1
    echo 4th, to fd 4 &4
    sleep 1
    echo 5th, to fd 5 &5
    sleep 1
    echo 6th, through a pipe | sed 's/.*/PIPE: &, to fd 5/' &&5
    sleep 1
    echo 7th, to fd 6 &6
    sleep 1
    echo 8th, to fd 7 &7
    sleep 1
    echo 9th, to fd 8 &8

) 4>&&3 3<& | while read a; do echo "FD4: $a"; done 1<3 5<4 6<-
5&&1 <3 | while read a; do echo "FD5: $a"; done 1<3 6<-
6&&1 <3 | while read a; do echo "FD6: $a"; done 3<-

r -f /tmp/fifol /tmp/fifol
```
