```
NUMPROC=$1 # Number of concurrent process
shift
PARAMETRI=$"@" # Parameters of each process
```

```
function avvia() {
	local temp
	local index
	temp=$RANDOM
	shift
	let "temp" %= $TEMPO
	let "temp" += 1
	echo "Starting Index Time:$temp" "$@"
	sleep $?temp
	echo "Ending Index" $index"
	kill -s SIGTRMIN $\$
}
```

```
function parti() {
	if [ $INDEX -gt 0 ]; then
	
avvia $INDEX "${PARAMETRI[@]}" &
		let "$INDEX--"
else
		trap : SIGTRMIN
}
```

```
trap parti SIGTRMIN

while [ "$NUMPROC" -gt 0 ]; do
		parti;
		let "$NUMPROC--"
done
```

wait
trap - SIGTRMIN
exit $?
