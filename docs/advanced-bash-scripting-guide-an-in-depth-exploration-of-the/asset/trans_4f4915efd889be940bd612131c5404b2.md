# 状态 Y=是, N=否
#                                   ：
# 未设置         Y      $( ... ) == 0
# 空         N Y      $( ... ) == 0
# 内容      N N      $( ... ) > 0

# 测试的第一部分或第二部分

t=$(VarEmpty)
$(VarEmpty--decT)             # 未执行 _decT 函数。
echo $t

# 内容，设置：t = 非空字符数量
VarSomething='simple'           # 设置有效函数名。
t=$(VarSomething)               # 非零长度
$(VarSomething--decT)         # 执行函数。
