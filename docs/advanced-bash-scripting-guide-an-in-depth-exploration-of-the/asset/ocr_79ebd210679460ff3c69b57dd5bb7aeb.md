# Advanced Bash-Scripting Guide

```
# "N" is another file descriptor.

# Redirecting stdout, one line at a time.
LOGFILE=script.log
echo "This statement is sent to the log file, \$LOGFILE\*" 1>>$LOGFILE
echo "This statement is appended to \$LOGFILE\*" 1>>$LOGFILE
echo "This statement is also appended to \$LOGFILE\*" 1>>$LOGFILE
echo "This statement is echoed to stdout, and will not appear in \$LOGFILE\*"
# These redirection commands automatically \"reset" after each line.

# Redirecting stderr, one line at a time.
ERRORFILE=script.errors
bad_command1 >>$ERRORFILE                # Error message sent to $ERRORFILE.
bad_command2 >>$ERRORFILE                # Error message appended to $ERRORFILE.
bad_command3                             # Error message echoed to stderr.
                                         # and does not appear in $ERRORFILE.
# These redirection commands automatically \"reset" after each line.
```

```
2&1
# Redirects stderr to stdout.
# Error messages sent to same place as standard output.
filename 2&1
bad_command filename >>2&1
# Appends both stdout and stderr to the file "filename" ...
2&1 | [command(s)]
bad_command 2&1 | awk '{print $5}' # found
# Sends stderr through a pipe.
# & is added to Bash 4 as an abbreviation for 2&1 |
```

```
i>j
# Redirects descriptor i to j.
# All output of file pointed to by i gets to file pointed by j.
```

```
>3
# Redirects, by default, file descriptor l (stdout) to j.
# All stdout gets sent to file pointed by j.
```

```
[j]>filename
# Open file "filename" for reading and writing,
#+ add file descriptor "j" to it.
# If "filename" does not exist, create it.
# If file descriptor "j" is not specified, default to fd 0, stdin.
# An application of this is writing at a specified place in a file.
echo 1234567890 > File # Write string to "File".
exec 3> File                                 # Open "File" and assign fd 3 to it.
```
