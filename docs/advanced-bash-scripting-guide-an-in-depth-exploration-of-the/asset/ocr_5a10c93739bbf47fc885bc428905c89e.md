  # Recursive function.

      ((Moves++))

    esac
  *)
}
case $8 in
  1) $((1>0)) in # Must have at least one disk.
    1) # Nested case statement.
      dohanoi 1 1 3 2;
      echo "Total moves $Moves" # 2^n - 1, where n = # of disks.
      exit 0;
    *)
      echo "$0: illegal value for number of disks";
      exit $E_BADPARAM;
    esac
  ;;;
*)
echo "usage: $0 N"

esac
