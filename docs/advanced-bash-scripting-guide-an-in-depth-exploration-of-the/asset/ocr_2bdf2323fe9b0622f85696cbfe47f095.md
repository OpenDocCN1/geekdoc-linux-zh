Example 36-10. A (useless) script that recursively calls itself

```
#!/bin/bash
# recurse.sh

# Can a script recursively call itself?
# Yes, but this is of any practical use?
# (See the following.)

RANGE=10
MAXVAL=9

i=$RANDOM
let i="%s" $RANGE # Generate a random number between 0 and $RANGE - 1

if [ "$i" -lt "$MAXVAL" ];
then
    echo "i = $i"
    # Script recursively spawns a new instance of itself.
    ./$0 # Each child does the same, until
    #+ a generated $i equals $MAXVAL.
fi

# Using a "while" loop instead of an "if/then" test causes problems.
# Explain why.

exit 0
```

Example 36-11. A (useful) script that recursively calls itself

```
#!/bin/bash
# pb.sh: phone book

# Written by Rick Boovie, and used with permission.
# Modifications by ABS Guide author.

MINARS=1
DATAFILE=./phonebook
# A data file in current working directory
#+ named "phonebook" must exist.

PROGNAME=$0
# No arguments error.

if [ $# -lt $MINARS ]; then
    echo "Usage: $PROGNAME data-to-look-up"
    exit $_NOARGS
fi

if [ $# -eq $MINARS ]; then
    grep $1 "$DATAFILE"
    # 'grep' prints an error message if $DATAFILE is not present.
else
    shift "$PROGNAME" $* | grep $1
    # Script recursively calls itself.
fi
```
