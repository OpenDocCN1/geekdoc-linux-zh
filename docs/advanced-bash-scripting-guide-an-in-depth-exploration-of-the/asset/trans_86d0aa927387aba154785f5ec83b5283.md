# 高级 Bash 脚本指南

bash
# 当前格式：[时间 用户主机 路径]
# 时间：
# 绿色 == 机器负载低
# 橙色 == 机器负载中等
# 红色 == 机器负载高
# ALERT == 机器负载极高

# 用户：
# 青色 == 普通用户
# 橙色 == 用户
# 红色 == root
# 主机：
# 青色 == 本地会话
# 绿色 == 安全连接 (通过 ssh)
# 红色 == 不安全连接

# 路径：
# 绿色 == 剩余空间大于 10%
# 橙色 == 剩余空间小于 10%
# ALERT == 剩余空间小于 5%
# 红色 == 当前用户没有写权限
# 青色 == 当前文件系统大小为 0 (如 /proc)
# ：
# 白色 == shell 中没有后台或挂起任务
# 青色 == shell 中至少有一个后台任务
# 橙色 == shell 中至少有一个挂起任务

# 测试连接类型：
if [ -n "$SSH_CONNECTION" ]; then
CNX=$(Green) # 连接到远程机器，通过 ssh (良好)。
elif [ "$DISPLAY" != "*" ]; then
CNX=$(ALERT) # 连接到远程机器，非通过 ssh (糟糕)。
else
CNX=$(Cyan) # 连接到本地机器。
fi

# 测试用户类型：
if [ "$USER" == "root" ]; then
SU=$(Red) # 用户是 root。
elif [ "$USER" != "$(hostname)" ]; then
SU=$(BRed) # 用户不是登录用户。
else
SU=$(Cyan) # 用户是普通用户。
fi

bash
NCPU=$(grep -c processor /proc/cpuinfo)
SLO=$((100*$NCPU))    # 小负载
MLOAD=$((200*$NCPU))   # 中负载
XLoad=$((400*$NCPU))   # 高负载

# 以百分比返回系统负载，例如 '40' 而不是 '0.40'
function load_color() {
    local SYLOAD=$(cut -d " " -f1 /proc/loadavg | tr -d '.')
    # 当前主机的系统负载。
    echo $(1*$SYLOAD))
}

# 返回表示系统负载的颜色。
function load_color() {
    local SYLOAD
    if [ ${SYLOAD} -gt ${XLoad} ]; then
        echo -en ${ALERT}
    elif [ ${SYLOAD} -gt ${MLOAD} ]; then
        echo -en ${Red}
    elif [ ${SYLOAD} -gt ${SLO} ]; then
        echo -en ${BRed}
    else
        echo -en ${Green}
    fi
}

# 返回 $PWD 磁盘剩余空间的颜色。
function disk_color() {
    if [ ! -w "$PWD" ]; then
        echo -en ${Red} # 当前目录没有权限。
    elif [ -s "$PWD" ]; then
        local used=$(df -P "$PWD" | awk 'NR==2 {print $5}' | sed 's/%//')
        if [ ${used} -gt 95 ]; then
            echo -en ${ALERT} # 磁盘满 (>95%)
        elif [ ${used} -gt 90 ]; then
            echo -en ${BRed} # 剩余空间将尽。
        else
            echo -en ${Green} # 剩余空间正常。
        fi
    else
        echo -en ${Cyan} # 当前目录大小为 '0' (如 /proc, /sys 等)。
    fi
}

# 根据运行/挂起的任务返回颜色。
function job_color() {
    if [ ${jobs -s | wc -l} -gt "0" ]; then
        echo -en ${BRed}
    elif [ ${jobs -r | wc -l} -gt "0" ]; then
        echo -en ${BCyan}
    fi
}

# 在终端框中添加一些文本（如果适用）。

# 现在我们构建提示符。
PROMPT_COMMAND="history -a"
case [$TERM] in
  term*|linux|load_color)
    PS1="$(load_color)\[A${NC}\]"
    # 时间（带有日期信息）：
    PS1="$(load_color)\[A${NC}\]"
    # 带有连接类型的主机：
    PS1="$(hostname)\[U${CNX}\]-h\[NC}\]"
    # 带有“磁盘空间”信息的路径：
    PS1="$(disk_color)\[W\]${NC}\]"
    # 带有“任务”信息的提示符：
    PS1="$(job_color)\[NC}\]"
    # 设置当前终端的标题：
    PS1="$(PS1)\e]0;%l"
  )
esac

PS1="\[A u@h \w\] " # 显示完整路径。

export TIMEFORMAT="\nreal 3Ruser %3ys %3Ut %P\n"
export HISTNORE="&bg;fg:li"
export TIMEFORMAT="$(echo -e ${BCY}) [%d/%m %H:%M:%S] $(echo -e ${NC})"
export HISTCONTROL=ignoreps
export HOSTFILE="$HOME/.hosts"
