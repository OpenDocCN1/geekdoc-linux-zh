```
let n1 = $n - 1      # n-1
let n2 = $n - 2      # n-2

t0=expr $n - ${Q[n1]}  # n Q[n-1]
t1=expr $n - ${Q[n2]}  # n Q[n-2]

T0=${Q[t0]}           # Q[n - Q[n-1]]
T1=${Q[t1]}           # Q[n - Q[n-2]]
Q[n]=expr $T0 + $T1      # Q[n-1] + Q[n-2]
echo -n "${Q[n]}"
```
