# 示例 A-35. 在文本文件中定位拆分的段落

```bash
#!/bin/bash
# find-splitpara.sh
# 查找文件中拆分的段落。
# 标记行号。

ARGCOUNT=1      # 期望参数。
OFF=0           # 状态。
ON=1
WRONGARGS=85

file="$1"      #目标文件名。
linen=1         # 行号。从1开始。
Flag=$OFF       # 空行。

if [ $# -ne "$ARGCOUNT" ]; then
  echo "用法: `basename $0`"
  exit $WRONGARGS
fi

file_read () {
  # 扫描文件模式，然后打印。
  while read line; do
    if [[ "$line" =~ ^[a-z] ]] && [ $Flag -eq $ON ]; then
      # 小写字母开头，后续行。
      echo -n "$linen:"
      echo "$line"
    fi
    if [[ "$line" =~ ^$ ]]; then
      # 如果是空行
      Flag=$ON # 设置标记。
    else
      Flag=$OFF
    fi
    ((linen++))
  done < "$file"
}
file_read
```

# # 高级 Bash 脚本指南

```bash
# 从列表第二个到最后一个循环*pivot*。
for ((i=1;i<n; i++)); do
    ((DEBUG))show list "[$i]"
    # 当前 pivot 回到第一个元素。
    for ((j=i; j>=0; j--)); do
        # 查找第一个小于 "pivot" 的元素
        while [[ ${list[$j-1]} -le ${list[$i]} ]]; do
            ((j--))
        done
        ((j)) && break # 此元素不需要插入。
        # 将 list(pivot) 移动到 list[j] 的左侧：
        list=(${list[@]:0:j} ${list[$i]} ${list[@]:j+1})
        ((j+1))
        # ${list[$i]:j+1}:-(j+1)} ${list[@]:i+1} (i+1,最后一个)
    done
    ((DEBUG))show j "<="
done
```
