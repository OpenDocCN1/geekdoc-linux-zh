initialize_board() {
  local idx

  for idx in {0..63}
  do
    board[$idx]=$UNVISITED
  done
}

print_board() {
  local idx

  echo "   # 行的倒序 ..."
  for row in {7..0}
  do # 所以符合棋盘顺序。
    let rownum=$row + 1 # 从1开始行号。
    echo -n "$rownum |" # 用边界标记棋盘边缘
    for column in {0..7}
    do
      let idx=$row + $column
      if [ ${board[$idx]} -eq $UNVISITED ];
      then
        echo -n "SYM " # 用移动次数标记格子。
      else
        printf "%02d"; "${board[$idx]}"; echo -n ""
      fi
      echo -e "\b\b\b" # \b 是退格。
    done
    echo # -e 允许转义字符。
  done

  echo ""
  echo "   a b c d e f g h"
}

failure() {
  # 抱怨一下，然后退出。
  echo
  print_board
  echo
  echo "   Waah!! 没有格子可以移动了！"
  echo -n "   骑士巡游原子结束"
  echo
  echo "   在 $(to_algebraic $currpos) [格子 $currpos]"
  echo
  echo "   在仅经过 $moves 次移动后!"
  echo
  exit FAIL
}
