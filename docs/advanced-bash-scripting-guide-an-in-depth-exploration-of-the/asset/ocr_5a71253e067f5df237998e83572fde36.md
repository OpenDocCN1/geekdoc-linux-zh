# Advanced Bash-Scripting Guide

```
stack[$SP]=$1
return

pop() # Pop item off stack.
Data=\ # Empty out data item.
if [ "$SP" -eq "$SP" ]; then # Stack empty?
then
    return # This also keeps SP from getting past 100,
    # i.e., prevents a runaway stack.
fi

Data=${stack[$SP]}
let SP += 1 # Bump stack pointer.
return
}

status_report() {
echo "--------------------------------" # Find what's happening.
echo "--------------------------------"
echo "REPORT"
echo "Stack Pointer = $SP"
echo "Just popped \"\$Data\" off the stack."
echo "--------------------------------"
echo
echo
}

#=================================================
# Now, for some fun.
echo

# See if you can pop anything off empty stack.
pop
status_report
echo

push garbage # Garbage in, garbage.
pop
status_report

value1=23; push $value1
value2=skidoo; push $value2
value3=LAST; push $value3

pop # LAST
status_report
pop # skoo
status_report
pop # 23 # Last-in, first-out!

# Notice the stack pointer decrements with each push,
# and increments with each pop.
echo
```
