```
seed_random() {
	if [ -n "$randomseed" ] # Can specify random seed.
	then # for play in competitive mode.
	RANDOM="$randomseed"
	echo "RANDOM set to \$randomseed"
	else
	randomseed="$" # Or get random seed from process ID.
	echo "RANDOM seed not specified, set to Process ID of script ($$)"
	fi
}

RANDOM="$randomseed"
echo
}
```

```
get_letset() {
	element=0
	echo -n "Letterset"
	for letset in $(seq $NVLET)
	do # Pick letters to fill out letterset.
	LS[element]=\${letters[\$((RANDOM%numelements))]}
		((element++))
	done
	echo
	echo "${LS[0]}"
}
```
