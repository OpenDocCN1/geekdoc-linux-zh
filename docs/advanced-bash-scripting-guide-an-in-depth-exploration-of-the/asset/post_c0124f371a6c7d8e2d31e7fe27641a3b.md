## Return Codes

- 0 - All OK
- 1 - Script failure
- 2 - Something is Blacklisted

## Optional environment variables

### SPAMMER_TRACE
If set to a writable file, script will log an execution trace.

### SPAMMER_DATA
If set to a writable file, script will dump its discovered data in the form of GraphViz See: http://www.research.att.com/sw/tools/graphviz

### SPAMMER_LIMIT
Limits the depth of resource tracing.

Default is 2 levels.

A set of 0 (zero) means 'unlimited'... Caution: script might recurse the whole Internet!

A limit of 1 or 2 is most useful when processing a file of domain names and addresses. A higher limit can be useful when hunting spam gangs.

# Drop functions until 'pend_stop_mark' . ..
pend_drop() {
    local -i _top_
    local -i _pd_ctrl=$${#_pending[@]}
    while [ $_pd_ctrl -gt 0 ];
    do
        _top_=$${#_pending[@]}-1
        if [ "${_pending[$_top_]}" == 'pend_stop_mark' ];
        then
            unset _pending[$_top_]
            break
        else
            unset _pending[$_top_]
            _pd_ctrl=$_top_
        fi
    done
    fi
    if [ ${#_pending[@]} -eq 0 ];
    then
        pend_func pend_stop_mark
    fi
}

# This function described in unique_lines.bash.
# unique_lines <in_name> <out_name>
unique_lines() {
    [ $# -eq 2 ] || return 1
    local -aul_in
    local -aul_out
    local -aul_cnt
    local -aul_pos
    local IFS=$(NO_WSP)
    set -f
    _aul_in=($1[@])
    _aul_cnt=${#_aul_in[@]}
    _aul_pos=0
    for (( i=0; i<_aul_cnt; i++ )); do
        if [ -n "${_aul_in[$i]}" ]; then # If defined and not empty
            _aul_tmp=${_aul_in[$i]}
            _aul_out[_aul_pos]="${_aul_tmp}"
            for (( zap=0; zap<_aul_pos; zap++ )); do
                if [ "${_aul_in[$zap]}" == "${_aul_tmp}" ]; then
                    unset _aul_in[$zap]
                    break
                fi
            done
            ((_aul_pos++))
        fi
    done
    eval $2=("${aul_out[@]}")
    set +f
    return 0
}

# This function described in is_number.bash.
# is_number <input>
is_number() {
  [ "$# -eq 1 ] || return 1 # is blank?
  [ "$1" -eq '0' ] && return 0 # is zero?
  local -i ts
  let ts=$1>/dev/null  # else is numeric!
  return $7
}
