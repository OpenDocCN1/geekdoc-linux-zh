```
# Color definitions (taken from Color Bash Prompt HowTo)
# Some colors look different of some terminals.
# For example, I see 'Bold Red' as 'orange' on my screen,
# hence the 'Green' 'Blue' 'Red' sequence I often use in prompt.
```

```
# Normal Colors
Black='e[0;30m'      # Black
Red='e[0;31m'       # Red
Green='e[0;32m'     # Green
Yellow='e[0;33m'    # Yellow
Blue='e[0;34m'      # Blue
Purple='e[0;35m'    # Purple
Cyan='e[0;36m'      # Cyan
White='e[0;37m'     # White
```

```
# Bold
Black='e[1;30m'     # Black
BRed='e[1;31m'       # Red
BGreen='e[1;32m'     # Green
Byellow='e[1;33m'    # Yellow
BBlue='e[1;34m'      # Blue
BPurple='e[1;35m'    # Purple
BCyan='e[1;36m'      # Cyan
BWhite='e[1;37m'     # White
```

```
# Background
OnBlack='e[4;40m'    # Black
OnRed='e[4;41m'      # Red
OnGreen='e[4;42m'    # Green
OnYellow='e[4;43m'   # Yellow
OnBlue='e[4;44m'      # Blue
OnPurple='e[4;45m'    # Purple
OnCyan='e[4;46m'      # Cyan
OnWhite='e[4;47m'     # White
```

```
NC='e[0m'
# Color Reset
```

```
ALERT-$(BWhite)${On_Red} # Bold White on red background
```

```
echo - "$(BCyan)This is Bash $(BRed)${BASH_VERSION.*}$(BCyan)
DISPLAY is $(BRed)$DISPLAY$(NC)"
date
```
