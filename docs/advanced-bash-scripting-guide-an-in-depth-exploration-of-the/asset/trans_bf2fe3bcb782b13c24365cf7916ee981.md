ARRAY1=(

### 示例 27-13. 埃拉托尼筛法

```bash
#!/bin/bash
# sieve.sh (ex68.sh)
# 埃拉托尼筛法，寻找质数的古代算法。
# 运行速度比对应的 C 语言程序慢几个数量级。

LOWER_LIMIT=1 # 从 1 开始。
UPPER_LIMIT=1000 # 到 1000（如果你有时间，可以设置更高）。
PRIME=1
NON_PRIME=0

let SPLIT=UPPER_LIMIT/2 # 优化：仅测试到上限一半的数字。为什么？

declare -a Primes # Primes[] 是一个数组。

initialize() {
# 初始化数组。
i=$LOWER_LIMIT
until [ $i -gt $UPPER_LIMIT ]; do
  Primes[$i]=$PRIME
  let i += 1
done
# 假设所有数组成员都是有罪的（质数），直到证明无辜。
}

print() {
# 打印 Primes[] 数组中标记为质数的成员。
i=$LOWER_LIMIT
until [ $i -gt $UPPER_LIMIT ]; do
  if [Primes[$i]} -eq $PRIME ]; then
    printf "%8d" "$i" # 每个数字 8 个空格对齐
  fi
  let i += 1
done
}

sift () {
# 筛掉非质数。
i=$LOWER_LIMIT+1 # 从 2 开始。
until [ $i -gt $UPPER_LIMIT ]; do
  if [Primes[$i]} -eq $PRIME ]; then # 已经筛过的（标记为非质数）。
    t=$i
    while [ $t -le $UPPER_LIMIT ]; do
      let t += 1
      Primes[$t]=$NON_PRIME # 标记为非质数的倍数。
    done
  fi
  let i += 1
done
}

#===========================
# main ()
# 顺序调用函数。
initialize
sift
print # 这就是所谓的结构化编程。
#============================

echo
# 以下代码因为 'exit' 不执行。
# Stephane Chazelas 改进的 Sieve 版本运行快一些。
# 必须带命令行参数调用（质数限制）。

UPPER_LIMIT=$1 # 来自命令行。
let SPLIT=UPPER_LIMIT/2 # 到上限一半。

Primes=$(seq $UPPER_LIMIT)

i=1
until (( i += 1 )); do # 只需要一半。
  if [ -nPrimes[$i]} ]; then
    t=$i
    until (( t += i ) > UPPER_LIMIT); do
      Primes[$t]=;
    done
  fi
done
echo ${Primes[*]}
exit $?

##### 示例 27-14. 优化的埃拉托尼筛法

```bash
#!/bin/bash
# 优化的埃拉托尼筛法
# Jared Mattin 编写脚本，ABS 作者微调。
# 在 ABS Guide 中使用（感谢他！）。

# 基于高级 Bash 脚本指南。
UPPER_LIMIT=$1 # 质数的上限。

Primes=$(seq $UPPER_LIMIT)

typeset -i t
Primes[$i-1]=$i # 1 不是质数。
until (( i += 1 )) > $((UPPER_LIMIT)/i); do # 只需要 i 方式。
  if [Primes[t*(i-1)] -eq $i ]; then # 为什么？晦涩但有启发性，下标中使用算术扩展。
    until (( t += i) > UPPER_LIMIT )); do
      Primes[t]=;
    done
  fi
done

## 27-15. 模拟栈

```bash
#!/bin/bash
# stack.sh: 栈模拟

# 类似于 CPU 栈，栈按顺序存储数据项，
# 但按相反顺序释放它们，后进先出。

BP=100 # 栈基指针。
# 从 100 开始。

SP=$BP

declare -a stack # 栈数组。

push() {
  if [ -z "$1" ]; then
    return
  fi

  let SP -- "$1" # 移动栈指针。
  stack[$SP]=$1
}

pop() { # 从栈弹出元素。
  if [ $SP -eq $BP ]; then # 栈为空？
    return
  fi

  Data=${stack[$SP]}
  let SP += 1 # 移动栈指针。
  return
}

status_report() { # 查看发生了。
echo "--------------------------------"
echo "--------------------------------"
echo "REPORT"
echo "Stack Pointer = $SP"
echo "Just popped \"\$Data\" off the stack."
echo "--------------------------------"
echo
}

#=================================================
# 现在玩点。
echo

# 看看能否从空栈中弹出东西。
pop
status_report
echo

push garbage # 垃圾入，垃圾出。
pop
status_report

value1=23; push $value1
value2=skidoo; push $value2
value3=LAST; push $value3

pop # LAST
status_report
pop # skidoo
status_report
pop # 23 # 后进先出！

# 注意栈指针每次 push 递减，
# 每次 pop 递增。
echo
```
