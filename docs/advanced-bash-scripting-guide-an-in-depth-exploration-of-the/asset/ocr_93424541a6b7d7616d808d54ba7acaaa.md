# Advanced Bash-Scripting Guide

```
# Nothing out of the ordinary here.

# Octal: numbers preceded by '0' (zero)
let oct = "032"
echo "octal number = $oct"      # 26
# Expresses result in decimal.
--------------------------------
```

```
# Hexadecimal: numbers preceded by '0x' or '0X'
let hex = "032"
echo "hexadecimal number = $hex"      # 50

echo $((0x9abc))               # 39162
# ^^           double-parentheses arithmetic expansion/evaluation
# Expresses result in decimal.
```

```
# Other bases: BASE#NUMBER
# BASE between 2 and 64.
# NUMBER must use symbols within the BASE range, see below.
```

```
let bin = "2111001100101"
echo "binary number = $bin"      # 31181

let b32 = "32=7"
echo "base-32 number = $b32"      # 231

let b64 = "64=$_"
echo "base-64 number = $b64"      # 4031
# This notation only works for a limited range (2 - 64) ASCII characters.
# 10 digits + 26 lowercase characters + 26 uppercase characters + @ +
```

```
necho $((36#zz)) $((2#10101010)) $((16#AF6)) $((53#1aa))
                               # 1295 17 44822 6395
```

```
# Important note:
----
# Using a digit out of range of the specified base notation
## gives an error message.
```

```
let bad_oct = "081"
# (Partial) error message output:
# bad_oct = 081: value too great for base (error token "081")
# Octal numbers use only digits in the range 0 - 7.
```

```
exit $ # Exit value = 1 (error)
```
