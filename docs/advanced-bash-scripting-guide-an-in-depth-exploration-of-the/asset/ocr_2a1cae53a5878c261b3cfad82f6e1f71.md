```
dohanoi() {
  # Recursive function.
  case $1 in
    0)
      ;
    *)
      dohanoi "$($1-1))" $2 $4 $3
      if [ "$Moves" -ne 0 ]
        then
          tput cup 0 0
          echo "+ Position after move $Moves"
        fi
      ((Moves++))
      echo -n " Next move will be: "
      echo $2 "--" $3
      display 2 $3
      dohanoi "$($1-1))" $4 $3 $2
      ;
  esac
}
```
