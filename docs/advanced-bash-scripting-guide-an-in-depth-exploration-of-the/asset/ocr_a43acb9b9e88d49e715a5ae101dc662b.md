```
echo
----------------------------------------------------------------------
# Must match at beginning / end of string,
#+ otherwise replacement results.
# # Matches, but not at beginning.
v3=$(v0/123/000)  # abc1234zip1234abc
echo "v3 = $v3"      # NO REPLACEMENT.

v4=$(v0/123/000) # Matches, but not at end.
echo "v4 = $v4"      # abc1234zip1234abc
                               # NO REPLACEMENT.

exit 0
```

${!varprefix*}, ${!(varprefix@}

Matches names of all previously declared variables with varprefix

+   # This is a variation of indirect reference, but with a " * or @.
# Bash, version 2.04, adds this feature.

xyz=whatever
xyz24=

```
a=${!xyz*}      # Expands to "name*" of declared variables
# ^ ^               + beginning with "xyz".
echo "a = $a"      # a = xyz23 xyz24
a=${!xyz0}       # Same as above.
echo "a = $a"      # a = xyz23 xyz24
```

---

```
echo
abc23=something_else
b=${!abc*}
echo "b = $b"      # b = abc23
c=$(!b)             # Now, the more familiar type of indirect reference.
echo $c              # something_else
```
