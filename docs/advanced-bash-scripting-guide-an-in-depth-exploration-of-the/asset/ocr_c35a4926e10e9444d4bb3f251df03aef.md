```
# This function moves the cursor to line $1 column $2 and prints $3.
# E.g.: "move_and_echo 5 10 linux" is equivalent to
# "tput cup 4 9; echo linux", but with one command instead of two.
# Note: "tput cup defines 0 0 the upper left angle of the terminal.\n# echo defines 1 1 the upper left angle of the terminal.
move_and_echo() {
echo -ne "\E[$1];${2}H"\$3"
}
```

```
# Function to generate a pseudo-random number between 1 and 9.
random_1_9() {
head -c10 /dev/random | md5sum | tr -d [a-z] | tr -d 0 | cut -c1
}
```
