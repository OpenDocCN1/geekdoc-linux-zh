# Example A-12. behold: Removing mail and news message headers

```
#! /bin/sh
# Strips the header from a mail/News message i.e. till the first
# empty line.
# Author: Mark Moraes, University of Toronto

# == These comments added by author of this document.

if [ $# -eq 0 ]; then
    # If no command-line arguments present, then work on redirected to stddin.
    sed -e '1,\$/d' -e '*$/d' /
    # -- Delete empty lines and all lines until
    # -- first beginning with white space
else
    # If command-line arguments present, then work on files named.
    for i; do
        sed -e '1,\$/d' -e '*$/d' $i
        # -- Ditto, as above.
    done

exit

# == Exercise: Error checking and other options.
# == Note that the small script repeats, except for the arg passed.
# == Does it make sense to embed it in a function? Why or why not
```
