A more elaborate example:

Example 5-3. Detecting key-presses

```
#!/bin/bash
# Author: Sigurd Solaas, 20 Apr 2011
# Used in ABS5 Guide with permission.
# Requires version 4.2+ of Bash.

key="no value yet"
while true; do
    clear
    echo " Extra Keys Demo. Keys to try:"
    echo " * Insert, Delete, Home, End, Page_Up and Page_Down"
    echo " * The four arrow keys"
    echo " * Tab, enter, escape, and space key"
    echo " * The letter and number keys, etc."
    echo ""
    d = show date/time
    echo " q="quit"
    echo "------------------------------------------------"

# Convert the separate home-key to home-key_num_7:
if [ "$key" = "$\x11\xb5\x31\x48" ]; then
    key=$x11\xb5\x34\x37'
    # Quoted string-expansion construct.
    fi

# Convert the separate end-key to end-key-num_1.
if [ "$key" = "$\x11\xb5\x34\x46" ]; then
    key=$x11\xb5\x34\x37'
    fi

case "$key" in
    "$\x11\xb5\x32\x3e") # Insert
        echo Insert Key
        ;;
    "$\x11\xb5\x33\x3e") # Delete
        echo Delete Key
        ;;
    "$\x11\xb5\x31\x3e") # Home_key_num_7
        echo Home Key
        ;;
    "$\x11\xb5\x34\x3e") # End_key_num_1
        echo End Key
        ;;
    "$\x11\xb5\x35\x3e") # Page_Up
        echo Page_Up
        ;;
    "$\x11\xb5\x36\x3e") # Page_Down
        echo Page_Down
        ;;
    "$\x11\xb5\x41") # Up_arrow
        echo Up arrow
        ;;
    "$\x11\xb5\x42") # Down_arrow
        echo Down arrow
        ;;
esac
```
