export LANG=C

limit -c 0 # SIGQUIT 不产生转储。
trap '' TSTOP # 忽略 Ctrl-Z。
stty -g; trap "stty save_tty" EXIT # 退出时恢复 tty。
stty quit "?" 空格代替 Ctrl-.
stty eof '?' # 代替 Ctrl-D。
stty -echo # 不回显输入。

cache_progs() {

date > /dev/null

(echo "import time" | python) 2> /dev/null
