# 通过将环境变量 SPAMMER_TRACE 设置为
# 可写文件的名来启用执行跟踪。
declare -a _trace_log
declare _log_file

# 填充跟踪日志的函数
trace_logger() {
    _trace_log[$#_trace_log[@]]=${pend_current}
}

# 将跟踪日志导出到文件变量。
declare -f _log_dump
dump_dump=pend_dummy # 初始为无操作。
# 将跟踪日志导出到文件。
dump_log() {
    local -l dl_cnt
    _dl_cnt=${#_trace_log[@]}
    for (( _dl = 0; _dl < _dl_cnt; _dl++ ));
    do
        echo ${_trace_log[$_dl]} >> ${_log_file}
    done
    _dl_cnt=${#_pending[@]}
    if [ ${_dl_cnt} -gt 0 ];
    then
        _dl_cnt=$((_dl_cnt-1))
        echo '# # 操作栈不为空 # #' >> ${_log_file}
        for (( _dl = ${_dl_cnt}; _dl >= 0; --dl-- ));
        do
            echo ${_pending[$_dl]} >> ${_log_file}
        done
    fi
}

# 工具程序 'dig' 包装器
# 这些包装器派自
# dig_wrappers.bash 中所示示例。
# 主要区别在于它们将结果以数组列表的形式返回。
# 详情请参见 dig_wrappers.bash，
# 并使用该脚本开发任何更改。

# 短格式回答：'dig' 解析答案。
# 正向查询：名称 -> 地址
# short_fwd :<domain_name> <array_name>
short_fwd() {
    local -a _sf_reply
    local -i _sf_rc
    local -i _sf_cnt
    IFS=$-
    echo -n ' '
    echo 'sfwfd: '${1}'
    _sf_reply=$(dig +short ${1} -c n -t 2>>/dev/null)
    _sf_rc=$?
    if [ ${_sf_rc} -ne 0 ];
    then
        _trace_log(${#_trace_log[@]})===='# 查询错误 '${_sf_rc}' 在 '${1}' ''
        # [ ${_sf_rc} -ne 9 ] && pend_drop
        return ${_sf_rc}
    else
        # 某些版本的 'dig' 在标准输出上返回警告。
        sf_cnt=${#_sf_reply[@]}
        for (( _sr = 0; _sr < ${_sf_cnt}; _sr++ ))
        do
            [ 'x${_sf_reply[$_sr]}:0:2' == 'x;' ] &&
                unset _sf_reply[$_sr]
        done
        eval 2="${_sf_reply[0]}"
    fi
    return 0
}

# 反向查询：地址 -> 名称
# short_rev <ip_address> <array_name>
short_rev() {
    local -a _sr_reply
    local -i _st_rc
    local -i _st_cnt
    IFS=$NO_WSP
    echo -n ' '
    echo "${1}'
    _sr_reply=$(dig +x ${1} 2>/dev/null)
    _sr_rc=$?
    if [ ${_sr_rc} -ne 0 ]
    then
        _trace_log(${#_trace_log[@]})===='# 查询错误 '${_sr_rc}' 在 '${1}'''
        # [ ${_sr_rc} -ne 9 ] && pend_drop
        return ${_sr_rc}
    else
        # 某些版本的 'dig' 在标准输出上返回警告。
        _sr_cnt=${#_sr_reply[@]}
        for (( _sr = 0; _sr < ${_sr_cnt}; _sr++ ))
        do
            [ 'x${_sr_reply[$_sr]}:0:2' == 'x;' ] &&
                unset _sr_reply[$_sr]
        done
        eval 2="${_sr_reply[0]}"
    fi
    return 0
}

# 用于查询黑名单服务器的特殊格式查询。
# short_text <ip_address> <array_name>
short_text() {
    local -a _st_reply
    local -i _st_rc
    local -i _st_cnt
    IFS=$NO_WSP
    echo -n '${1}'
    _st_reply=$(dig +short ${1} -in -t txt 2>/dev/null)
    _st_rc=$?
    if [ ${_st_rc} -ne 0 ]
    then
        _trace_log(${#_trace_log[@]})===='#文本查询错误 '${_st_rc}' 在 '${1}'''
        # [ ${_st_rc} -ne 9 ] && pend_drop
        return ${_st_rc}
    else
        # 某些版本的 'dig' 在标准输出上返回警告。
        _st_cnt=${#_st_reply[@]}
        for (( _st = 0; _st < ${_st_cnt}; _st++ ))
        do
            [ 'x${_st_reply[$_st]}:0:2' == 'x;' ] &&
                unset _st_reply[$_st]
        done
        eval 2="${_st_reply[0]}"
    fi
    return 0
}
