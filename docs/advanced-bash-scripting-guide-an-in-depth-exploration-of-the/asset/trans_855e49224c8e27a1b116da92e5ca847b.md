```bash
#!/bin/bash

# 警告：运行此脚本可能会导致你的系统锁死！
# 如果你足够幸运，它会在用尽可用内存之前发生段错误。
recursive_function() {
  echo "$1" # 让函数执行某些操作，并加速默认过程。
  (( $1 < $2 )) && recursive_function $(( $1 + 1 )) $2;
  # 只要列表参数小于第二个参数，
  # 递增并递归。
}
recursive_function 1 50000 # 递归 50000 层！
# 最可能发生段错误（取决于栈大小，受 ulimit -m 限制）。
# 这种深度的递归可能会导致 C 程序也发生段错误，
# 因为耗尽分配给栈的所有内存。
```
