```
Usage ()               # "Usage: message.
{
if [ -z "$UID" ]      # No arg passed.
then
    msg=$filename
else
    msg=$@
fi
echo "Usage: $basename $0 \$msg"
}
```

```
Check_if_root ()       # Check if root running script.
{
if [ -ne "$ROOT_UID" ]  # From "ex39.sh" example.
then
    echo "Must be root to run this script."
    exit $E_NOTROUTE
}
```

```
CreateTemplateName ()  # Creates a "unique" temp filename.
{
    prefix=$temp          # From "ex51.sh" example.
    suffix=$(date +ts)
    filename=$prefix.$suffix
}
```

```
isalpha2(2)           # Tests whether *entire string* is alphabetic.
{
    if [[ $# -eq 1 ]] && [[ "$1" =~ ^[a-zA-Z]*$ ]]; then
        return $FAILURE
fi}
```
