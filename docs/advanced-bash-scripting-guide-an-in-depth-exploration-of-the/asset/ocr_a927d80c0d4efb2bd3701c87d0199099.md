```
# And, it gets better.
# Consider the following snippet get variable named getSparc,
# but no such variable getIa64:
chkMirrorArches () {
  arch="$S1"
  if [ "$arch" = "$$(get$(echo -ne $arch |) ]; then
    sed 's/\\/(.*)\\\/1/g' | tr 'a-z A-Z'; echo $arch |
    sed 's/\\\/(.*)\\\/1/g')\:false"
  then
    return 0
  else
    return 1
  fi
}
```

```
getSparc="true"
unsetIa64
chkMirrorArches sparc
echo $?    # 0
               # True
chkMirrorArches Ia64
echo $?    # 1
               # False
```
