# Preparing an SD card with a bootable image for the Raspberry Pi.

```
# $1 = imagefile name
# $2 = sdcard (device file)
# Otherwise defaults to the defaults, see below.

DEFAULTsize=4M
DEFAULTcard="2013-07-26-weezy-raspberry.img"
DEFAULTsdcard="/dev/mmcblk0"
ROOTUSER_name=root
E_NOTROOT=1
E_NOIMAGE=82

username=$($id -nu)                # Who is running this script?
if [ "$username" != "$ROOTUSER_name" ];
then
    echo "This script must run as root or with root privileges."
    exit $E_NOTROOT
fi

if [ -n "$1" ];
then
    imagefile="$1"
else
    imagefile="$DEFAULTsize"
fi

if [ -n "$2" ];
then
    sdcard="$2"
else
    sdcard="$DEFAULTsdcard"
fi

if [ ! -e $imagefile ];
then
    echo "Image \$imagefile\" not found!"
    exit $E_NOIMAGE
fi
```
