
# Advanced Bash-Scripting Guide

The default pathname for the -f and -u commands is $CDPath
The default filename for the -f and -u commands is $CDFile

Set the following environment variables:
CDL_PROMPTLEN - Set to the length of prompt you require.
Prompt string is set to the right characters of the
current directory.
If not then prompt is left unchanged
CDL_PROMPT_PRE - Set to the string to prefix the prompt.
Default is:
non-root: "\\[001;34m\] (sets colour to blue)
root: "\\[001;31m\] (sets colour to red).
CDL_PROMPT_POST - Set to the string to suffix the prompt.
Default is:
non-root: "\\[00m\] (resets colour and displays $)
root: "\\[00m\] (resets colour and displays #)
CDPath - Set the default path for the -f & -u options.
Default is home directory
CDFile - Set the default filename for the -f & -u options.
Default is cdfile

# Truncate right.
#
# params:
# p1 string
# p2 length to truncate to
#
# returns string in tcd
cd_right_trunc () {
    local tlen=$2
    local plen=$1
    local str="${1}"
    local diff
    local filler="<--"
    if [ ${plen} -le ${tlen} ]; then
        tcd="${str}"
    else
        let diff=$(${plen}-${tlen})
        let len=3
        if [ ${diff} -le 2 ]; then
            let len=$(${diff})
        fi
        tlen=$(${tlen})
        let tlen=$(${tlen}+${len})
        tcd="${str:0:len}${filler}${str:tlen}"
    fi}

cd_distory ()
{
    cd_getrc
    printf "History:\n"
    local -i count=${cd_histcount}
    while [${TEST} ${count} -ge 0 ];
    do
        cd_right_trunc "${CD[count]}" ${cd_lchar}
        printf "%2d -${cd_lchar}s " ${count} "${tcd}"
        
        cd_right_trunc "${CD[count]}" ${cd_rchar}
        printf "%s%d -${cd_rchar}.${cd_rchar}\n" ${count} "${tcd}"
        count=$((count-1))
    done
}

cd_getrc ()
{
    cd_flchar=$(stty -a | awk -F \ ')
    '/rows/ { print $2 $3 }' | awk '{ print $4 }'
    if [${TEST} ${cd_flchar} -ne 0 ];
    then
        cd_lchar=${{cd_flchar}/2-5}
        cd_rchar=${{cd_flchar}/2-5}
        cd_flchar=${{cd_flchar}-5}
    else
        cd_flchar=${FLCHAR:=75}
        # cd_flchar is used for @s @ history
        cd_lchar=${LCHAR:=35}
        cd_rchar=${RCHAR:=35}
    fi
}

cd_doselection {
    local -i nm=0
    cd_dlag="TRUE"
    if [${TEST} "$(CD_MODE)" = "PREV" ];
    then
        if [${TEST} -z "$cd_npwd" ];
        then
            cd_npwd=0
        fi
    fi
    tm=$(echo "${cd_npwd}" | cut -b 1)
    if [${TEST} "$(tm)" = "_" ];
    then
        pm=$(echo "${cd_npwd}" | cut -b 2)
        nm=$(echo "${cd_npwd}" | cut -d -2)
        case "$pm" in
            a) cd_npwd=$(CDS[$nm]);
            s) cd_npwd=$CDS[$nm];
            S) cd_npwd=$CDS[$nm]; $CDS[0]=pwd;
            r) cd_npwd="$2"; $cd_SpecDir=$nm "$1" "$2";
            R) cd_npwd="$2"; $CDS[0]=pwd;
        esac
    fi
    if [${TEST} "$(cd_npwd)" = "*" ] || [${TEST} "$(cd_npwd)" -le "$(cd_history)" ] >>/dev/null 2>&1
    then
        cd_npwd=$(CDS[$cd_npwd])
    else
        case "$cd_npwd" in
            @) cd_history; cd_dlag="FALSE";
            h) cd_historyH; cd_dlag="FALSE";
            s) cd_historyS; cd_dlag="FALSE";
            h) hm; cd_dlag="FALSE";
            H) Hm "$2"; cd_dlag="FALSE";
            f) fsave "SHOW "$2"; cd_dlag="FALSE";
            u) upload "SHOW "$2"; cd_dlag="FALSE";
            F) fsave "NOSHOW "$2"; cd_dlag="FALSE";
            U) upload "NOSHOW "$2"; cd_dlag="FALSE";
            g) cd_npwd="$2";
            d) cd_defm 1; cd_dlag="FALSE";
            D) cd_defm 0; cd_dlag="FALSE";
            r) cd_npwd="$2"; $cd_SpecDir=0; cd_doselection "$1" "$2";
            R) cd_npwd="$2"; $CDS[0]=pwd; cd_doselection "$1" "$2";
            s) cd_npwd=$CDS[0];
            S) cd_npwd=$CDS[0]; $CDS[0]=pwd;
            V) cd_version; cd_dlag="FALSE";
        esac
    fi
}
