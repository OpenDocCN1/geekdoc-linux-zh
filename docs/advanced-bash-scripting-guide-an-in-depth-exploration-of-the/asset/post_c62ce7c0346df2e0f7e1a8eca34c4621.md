```bash
# == Similar effect for dir in *
# == but in dir `echo *` will not handle filenames with blanks.
if [ -d"$dir" ]; then # If it is a directory (-d)...
  # == Temp variable, keeping track of
  zz=0

  while [ $zz -eq 1 ]; do
    # Keep track of inner nested loop.
    echo -n ""
    # == Display vertical connector symbol,
    # == with 2 spaces & no line feed
    # in order to indent.
    zz=$(expr $zz + 1)
    if [ -L"$dir" ]; then # If it is a symbolic link...
      echo "$dir" -ls -1 | sed 's/.*$dir //'
      # == Display horiz connector and list directory name, but...
      # == delete date/time part of long listing.
    else
      echo "$dir" -- # Display horizontal symbol,
      # == and print directory name.
      numdirs=$(expr $numdirs + 1) # == Increment count.
      if cd "$dir"; then # If we can move to subdirectory...
        search expr $1 + 1 # == with recursion calls itself.
        cd ..
      fi
    fi
  done
fi

if [ $# -eq 0 ]; then
  cd "$1" # Move to indicated directory.
```

```bash
# Usage: tree <recursion> <indent prefix> <min size> <directory>
function tree {
  curs=$1 # How deep nested are we?
  prefix=$2 # What do we display before file/dirname?
  minsize=$3 # What is the minimum file/dirsize?
  dirname=$4 # Which directory are we creating?
}
```

```bash
# Get (TOP) biggest subdirectories/subfiles from TMP file.
LIST=$(grep '^[[:space:]]*$(dirname)/.*' "$TMP" | awk '{if($1)}print $0}' | sort -nr | head -$TOP)
[ -z "$LIST" ] && return 4 # Empty list, then go back.
```

# Call recursively too if it's a directory
