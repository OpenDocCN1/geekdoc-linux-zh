# This is a 'universal' completion function - it works when commands
# a so-called 'long options' mode, ie: 'ls --all instead of 'ls -a'
# Needs the '-o' option of grep
# (try the commented-out version if not available)

# First, remove 'm' from completion word separators
# (this will allow completions like '--color=auto' to work correctly)

COMP_WORDBREAKS=$(COMP_WORDBREAKS=/)

get_longopts() {
    $1 --help | sed -e '//|d' -e 's/.*\([^[:space:],].*\).*/--\1|'
    grep "$2" | sort -u
    $1 --help | grep -o "--[[:space:],].*" | grep -e "$2" sort -u
}

opts() {
    local cur
    cur="$(COMP_WORDS[COMP_CWORD])"
    case "${cur#*-}" in
        -*)
            return 2;
    esac

    case "$1" in
        \*)
            eval cmd="$1"
            cmd="$1";
    esac
    COMPRELY=($ $(get_longopts $(cur)) ))
}
complete -o default -F _longopts configure bash
complete -o default -F _longopts wget id info a2ps is recode

tar() {
    local cur ext regex tar untar

COMPRELY=())
    cur="$(COMP_WORDS[COMP_CWORD])"

    # If we want an option, return the possible long options.
    case "$cur" in
        -*))
            COMPRELY=$( $(get_longopts $cur )); return 0;
    esac

    if [ $COMP_CWORD -eq 1 ]; then
        COMPRELY=$( $( compgen -W 'c t x r d A' -- $cur ))
        return 0
    fi

    case "${COMP_WORDS[1]}" in
        ?(-)c*f)
            COMPRELY=( $( compgen -f $cur ))
            return 0;
    esac
}
