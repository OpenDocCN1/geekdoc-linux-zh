# Current Format: [TIME USERHOST PWD]
# TIME:
# Green == machine load is low
# Orange == machine load medium
# Red == machine load is high
# ALERT == machine load is very high

NCPU=$(grep -c processor /proc/cpuinfo)
SLO=$((100*$NCPU))    # Small load
MLOAD=$((200*$NCPU))   # Medium load
XLoad=$((400*$NCPU))   # High load

# Returns system load as percentage, i.e., '40' rather than '0.40'
function load_color() {
    local SYLOAD=$(cut -d " " -f1 /proc/loadavg | tr -d '.')
    # System load of the current host.
    echo $(1*$SYLOAD))
}

# Returns a color indicating system load.
function load_color() {
    local SYLOAD
    if [ ${SYLOAD} -gt ${XLoad} ]; then
        echo -en ${ALERT}
    elif [ ${SYLOAD} -gt ${MLOAD} ]; then
        echo -en ${Red}
    elif [ ${SYLOAD} -gt ${SLO} ]; then
        echo -en ${BRed}
    else
        echo -en ${Green}
    fi
}

# Returns a color free disk space in $PWD.
function disk_color() {
    if [ ! -w "$PWD" ]; then
        echo -en ${Red} # No privilege in the current directory.
    elif [ -s "$PWD" ]; then
        local used=$(df -P "$PWD" | awk 'NR==2 {print $5}' | sed 's/%//')
        if [ ${used} -gt 95 ]; then
            echo -en ${ALERT} # Disk full (>95%)
        elif [ ${used} -gt 90 ]; then
            echo -en ${BRed} # Free space almost goes.
        else
            echo -en ${Green} # Free space is ok.
        fi
    else
        echo -en ${Cyan} # Current directory is size '0' (like /proc, /sys etc).
    fi
}

# Returns a color according to running/suspended jobs.
function job_color() {
    if [ ${jobs -s | wc -l} -gt "0" ]; then
        echo -en ${BRed}
    elif [ ${jobs -r | wc -l} -gt "0" ]; then
        echo -en ${BCyan}
    fi
}

  term*|linux|load_color)

    # Time of day (with day info):

    # Host with connection type info:

    # PWD with "disk space" info:
    PS1="$(disk_color)\[W\]${NC}\]"
    # Prompt with "job" info:

    # Set title of current xterm:
