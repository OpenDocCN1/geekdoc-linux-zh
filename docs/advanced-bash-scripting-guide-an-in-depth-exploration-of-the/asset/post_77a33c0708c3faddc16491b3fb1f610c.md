Example 14-46. Generating prime numbers

```bash
#!/bin/bash
# primes2.sh

# Generating prime numbers the quick-and-easy,
# without resorting to fancy algorithms.

CEILING=10000 # 1 to 10000
PRIME=0
E_NOTPRIME=

is_prime () {
 local factors
factors=($(factor $)) # Load output of `factor` into array.
 if [ -z "${factors[2]}" ]; then
  # Third element of "factors" array:
  # ${factors[2]} is 2nd factor of argument.
  # if it is blank, then there is no 2nd factor,
  # and the argument is therefore prime.
  return $PRIME # 0
else
  return $E_NOTPRIME # null
fi
}

echo
for n in $(seq $CEILING);
do
  if is_prime $n
then
    printf %5d $n
  fi    ^ # Five positions per number suffixes.
done  # For a higher $CEILING, adjust upward, as necessary.

echo
exit
```

bc can't handle floating point calculations, and it lacks operators for certain important mathematical functions. Fortunately, bc gallops to the rescue.

Not just a versatile, arithmetic precision calculation utility, bc offers many of the facilities of a programming language. It has a syntax vaguely resembling C.

Since it is a fairly well-behaved UNIX utility, and may therefore be used in a pipe, bc comes in handy scripts.

Here is a simple example for using bc to calculate a script variable. This uses command substitution.

```bash
variable=$(echo "OPTIONS; OPERATIONS" | bc)
```

# Example 16-47. Monthly Payment on a Mortgage

```bash
#!/bin/bash
# monthlypymnt.sh: Calculates monthly payment on a mortgage.

# This is a modification of code in the
# + "mcalc" (mortgage calculator) package,
# + by Jeff Schmidt
# + and
# + Mendel Cooper (ours truly, the ABS Guide author).
# http://www.ibiblio.org/pub/Linux/apps/financial/mcalc-1.6.tar.gz

echo
echo "Given the principal, interest rate, and term of a mortgage,
echo "calculate the monthly payment."

bottom=1.0

echo
echo -n "Enter principal (no commas) "
read principal
echo -n "Enter interest rate (percent) " # If 12%, enter "12", not ".12"
read interest_r
echo -n "Enter term (months) "
read term

interest_=$(echo "scale=9; $interest_r/100.0" | bc) # Convert to decimal.
-----------------
Divide by 100.
# "scale" determines how many decimal places.

interest_rate=$(echo "scale=9; $interest_r/12 + 1.0" | bc)
top=$(echo "scale=9; $principal$interest_rate$term" | bc)
#---------------------------------------
# Standard formula for figuring interest.

echo; echo "Please be patient. This may take a while."

let months=$term - 1
#---------------------------------------
for ((x=$months; x > 0; x--)); do
  bot=$(echo "scale=9; $interest_rate$x" | bc)
  bottom=$(echo "scale=9; $bot" | bc)
  bottom=$((bottom + bot))
done
#---------------------------------------
# Bob pointed out a more efficient implementation
# of the above loop, decreases computation time by 2/3.
for ((x=1; x <= months; x++)); do
  bottom=$(echo "scale=9; $interest_rate + 1" | bc)
done

And then we can use an even more efficient alternative,
bottom={
  echo "scale=9; $interest_rate"
  for ((x=1; x <= months; x++)); do
    echo 'bottom = $bottom * interest_rate + 1'
  done
  echo 'bottom'
} | bc
# Because the loop is actually a sum of geometric progression.
# The formula is a(1-q^n)/(1-q),
# where a is the first element and n is the number of elements.
payment=$(echo "scale=2; $top" | bc)
# Use two decimal places for dollars and cents.
echo "monthly payment = \$payment" # Echo a dollar sign in front of amount.
exit 0

If no number is given, the numbers are read from standard input.

  binary (base 2)        starting with 0 (i.e. 0b1100)
  octal (base 8)         starting with 0 (i.e. 014)
  hexadecimal (base 16)  starting with 0x (i.e. 0xc)
  decimal               otherwise (i.e. 12)
>&2 $NOARGS
) # Prints usage message.

Msg () {
  for i in "$@"; listing. Why?
  do echo "$PN: $1" >&2
  done
}

Fatal () {
  Msg "$0"; exit 6;
}

PrintBases () {
  # Determine the base of the number
  for i in "$@";
do # so operates on command-line args(s).
    case "$i" in
      ibase=2)          # binary
      0x['a-f][A-F]*)  # hexadecimal
      ibase=16);
      0*) ibase=8; # octal
      [1-9]*) ibase=10; # decimal
      *)
        Msg "illegal $i - ignored"
        continue;
    esac

    # Remove prefix, convert hex digits to uppercase (bc needs this).
    number="$i" | sed 's/^0[xX]//g' | tr '[a-f]' '[A-F]'
    # Uses \ as sed separator, rather than ""

    # Convert number to decimal
    dec=$(echo "obase=$base; $number" | bc) # 'bc' is calculator utility.
    case $dec in
      [0-9]*) # number ok
        continue; 
      *) # error: ignore
    esac

    # Print all conversions in one line.
    # = <<'document' feeds command list to 'bc'
    echo <<<
      obase=16; "hex"; $dec
      obase=10; "dec"; $dec
      obase=8; "oct"; $dec
      obase=2; "bin"; $dec
    BC
  | sed -e 's/ : /g'
  done
}
