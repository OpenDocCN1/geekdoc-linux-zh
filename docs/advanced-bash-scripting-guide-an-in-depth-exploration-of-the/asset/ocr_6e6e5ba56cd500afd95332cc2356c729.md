```
# Before we scan for targets, see if a Makefile was named
# specified with -f.
for ((i=0; i<${COMP_WORDS[@]}; i++)); do
  if [ "${COMP_WORDS[$i]}" == -f ]; then
    # eval for tilde expansion
    eval make=${${COMP_WORDS[$i+1]}}
    break
  fi
done
[ -f $make ] && return 0
```

```
# Deal with included Makefiles.
make_inc=$(grep -E '^?include$' $make | 
  sed -e "s,.*,"$make_dir/",")
for file in $make_inc; do
  [ -f $file ] && make="$file"
done
```
