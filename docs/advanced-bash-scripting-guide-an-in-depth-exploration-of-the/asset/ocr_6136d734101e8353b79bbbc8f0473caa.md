```
sort -k 9 <(ls -1 /bin) <(ls /usr/bin) <(ls /usr/X11/bin)
# Lists all the files in the 3 'bin' directories, and sorts by filename.
# Note that three (count 'em') distinct commands are to 'sort'.
```

```
tar cf <(bzip2 -c file.tar.bz) $directory_name
# Calls "tar /dev/fd/$ $directory_name", and bzip2 -c file.tar.bz2.
```

```
bzip2 -c < pipe > file.tar.bz2
tar cf $directory_name
```
