move_stupid() # Digba algorithm,
(
  # courtesy of script author, *not* Warnsdorf.
  local valid_moves=0
  local movloc
  local square
  aapos
  local sclocpos="$1"

  for movloc in [1..8]
  do # Move to first-found unvisited square.
    let square=$sclocpos+${moves[movloc]}
    is_on_board=$square &&
    if [ $? -eq $SUCCESS ]; then [board[square]] -eq $UNVISITED ];
    then # Add conditions to if to improve algorithm.
      ((++movenum))
      board[square]=$movenum
      squareloc=$square # Update current position.
      ((valid_moves++)); # moves[0]=$valid_moves
      aapos=$to_algebraic $squareloc
      echo -n "$aapos "
      test $((moves % $LINELEN)) -eq 4 && echo # Print 4 moves/line.
      return $valid_moves # Found a square to move!
    fi
  done
  return $FAIL # If no square found in 8 iterations, then Knight's Tour ends in failure.
)

```bash
decide_move () {
  # Which move will we make?
  for mov in {1..8};
  do
    # But fails on startpos=37
    let square=$currpos+${moves[mov]}
    is_on_board $square
    if [[ $? -eq $SUCCESS && ${board[square]} -eq $UNVISITED ]];
    then # Find accessible square with least possible moves.
      # What happens if the knight wanders towards the outer edge, then perhaps inward.
      # Given two or more moves with same value of future-moves, this implementation chooses the first of moves.
      # This means there is not necessarily a unique solution for any starting position.
    fi
  done
}
```

```bash
possible_moves () # Calculate number of possible moves
{
  # given the current position.
  if [ -z "$1" ];
  then
    return $FAIL
  fi

  local curr_pos=$1
  local valid_mov=0
  local icx=0
  local mov
  local sq
  declare -a movesloc

  movesloc=$(generate_curr_pos)
  for mov in {1..8}
  do
    let sq=$curr_pos+${movesloc[mov]}
    is_on_board $sq
    if [ $? -eq $SUCCESS ] && [ $(board[sq]) -eq $UNVISITED ];
    then
      ((valid_mov++))
    fi
  done
  return $valid_mov # Found a square to move!
}
```
