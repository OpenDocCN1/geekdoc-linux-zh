There are other, more complex, ways of implementing timed input in a script. One alternative is to set a timing loop to signal the script when it times out. This also requires a signal handling routine to trap (Example 32-5) the interrupt generated by the timing loop (whel!).

# Example 9-2. Timed Input

```
#!/bin/bash
# timed-input.sh

# TIMEOUT=3      Also works, as of newer versions of Bash.

TIMER_INTERRUPT=14
# Three seconds in this instance.
TIMELIMIT=3      # May be set to different values.

PrintAnswer() {
  if [ "$answer" = TIMEOUT ]
  then
    echo $answer
  else
    # Don't want to mix up the two instances
    echo "Your favorite veggie is $answer"
    kill $!      # Kills no-needed TimerOn function
    # + running in background.
    $! is PID of last job running in background.
  fi
}

TimerOn() {
  sleep TIMELIMIT && kill -s 14 $ &
  # Waits 3 seconds, then sends alarm to script.
}

Int14Vector() {
  answer="TIMEOUT"
  PrintAnswer
  exit $TIMER_INTERRUPT
}

trap Int14Vector $TIMER_INTERRUPT
# Timer interrupt (14) subverted for our purposes.
```

echo "What is your favorite vegetable "
TimerOn
read answer
PrintAnswer
