```
foundone=true
# -----------------
echo "Subshell level = $BASH_SUBSHELL"
# Subshell level = 1
# Yes, we're inside a subshell.
# -----------------
done

# foundone will always be false since it is
+# set to true inside a subshell
if [ $foundone = false ];
then
    echo "No files need archiving."
fi

# ===================Now, here is the correct way:=================
foundone=false
for f in $($HOME -type f -atime +30 -size 100k) # No pipe here.
do
    echo "$f is over 100KB and has not been accessed in over 30 days"
    echo "Consider moving the file to the archives."
done

if [ $foundone = false ];
then
    echo "No files need archiving."
fi

# ===================And there is another alternative=================

# Places parts of the script that reads the variables
+# within a code block, so they share the same subshell.
# Thank you, W.B.

find $HOME -type f -atime +30 -size 100k | {
  foundone=false
  while read f;ndo
    echo "$f is over 100KB and has not been accessed in over 30 days"
    echo "Consider moving the file to the archives."
    foundone=true
  done

  if [ $foundone ];
then
    echo "No files need archiving."
  fi
}
```

A lookalike problem occurs when trying to write the stdout of a tail -f piped to grep.

```
tail -f /var/log/messages | grep "$ERROR_MSG" >> error.log
# The .log file will not have anything written to it.
# As Samuli Kaiainen points out, this results from grep
# buffering its output.
# The fix is to add a "--line-buffered" parameter to grep.
```
