The -u option gives the UTC (Universal Coordinated Time).

```
bash$ date
Fri Mar 29 21:07:39 MST 2002

bash$ date -u
Sat Mar 30 04:07:42 UTC 2002
```

This option facilitates calculating the time between different dates.

# Example 16-11. Date calculations

```
#!/bin/bash
# date-calc.sh
# Author: Nathan Coulter
# Used in ABS Guide with permission (thanks!)

MPHR=60
# Minutes per hour.
HPD=24
# Hours per day.

diff () ( 
    printf '%s' $(( $(date -u -d"$TARGET" +%s)) \
      -$(date -u"$CURRENT" +%s)) \
    -%d = day of month. 
)

CURRENT=$(date -u -d '2007-09-01 17:30:24' '+%F %T.%N %Z')
TARGET=$(date -u -d '2007-12-25 12:30:00' '+%F %T.%N %Z')
# %F = full date, %T=H:%M:%S, %N = nanoseconds, %Z = time zone.

printf '\n2007, %s, \
    \$(date -d"$CURRENT' \
    \(( ($diff) / MPHR / $HPD / 2 )) days\'+%d %B')\n    # %B = name of month
printf ' was halfway between %s' '\$(date -d"$CURRENT" +%d %B)\' \
    and '%s' '\$(date -d"$TARGET" +%d %B)\'

printf '\n%s after there were \n\n    %s' '\$(date -u -d"$CURRENT" +%T)'
DAYS=$(( ($diff) / MPHR / $HPD ))
CURRENT=$(date -d "$CURRENT" +$DAYS days') '+%F %T.%N %Z')
HOURS=$(( ($diff) % MPHR ))
CURRENT=$(date -d "$CURRENT" +$HOURS hours') '+%F %T.%N %Z')
MINUTES=$(( ($diff % 60 ))
CURRENT=$(date -d "$CURRENT" +$MINUTES minutes') '+%F %T.%N %Z')
printf '%s days, %s, and %s "$DAYS" "$HOURS"
printf '%s minutes, and %s seconds "$MINUTES" "($diff)"
printf 'until Christmas Dinner!\n\n'
```
