expand_input_name() {
    [$#uc_name[@]} -gt 0 ] || return 0
    local -a _ein_addr
    local -a _ein_new
    local -i _uc_cnt
    local -i _ein_cnt
    local _ein_tst
    uc_cnt=$(${uc_name[@]})

    if ! _ein_cnt=$((limit_chk $1))\n    then
        return 0
    fi

    for (( _ein=0; _ein<_uc_cnt; _ein++ ))
    do
        if short_fwd ${uc_name[$_ein]} ${_ein_new
        then
            for (( _ein_cnt=0; _ein_cnt<${_ein_new[0]}; _ein_cnt++ ))
            do
                _ein_tst=${_ein_new[$_ein_cnt]}
                if is_address ${_ein_tst}
                then
                    _ein_addr[$${_ein_addr[0]}]=${_ein_tst}
                fi
            done
        fi
    do

done
fi
