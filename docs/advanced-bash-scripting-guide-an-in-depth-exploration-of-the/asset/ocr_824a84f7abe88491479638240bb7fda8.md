# Usage: strcat s1 s2 $n
# Line scrat, but strcat appends a maximum of n characters from the value
# of variable s2. It copies fewer if the value of variabl s2 is shorter
# than n characters. Echoes result on stdout.

# Example:
# a=foo
# b=bar
# strcat a b 3
# echo $a
# => foobar

:end docstring

```
###;;;autoload
function strcat() {
    local s1="$1"
    local s2="$2"
    local -i n="$3"
    local s1_val=$s2_val

    s1_val=$!1  # => indirect variable expansion
    s2_val=$!2

    if [($s1_val) -gt $n ]; then
        s2_val=$($s1_val:0:$n)
    fi

    eval "\${${s1_val}}{\${s2_val}}""
    # => eval "\${${s1_val}}{\${s2_val}}"" avoids problems,
    # => if one of the variables contains a single quote.
}
```
