```
# Output width when encoding
# (64 characters, just like openssl output).
text_width=64
```

```
function display_base64_char {
# Convert a 6-bit number (between 0 and 63) to its corresponding values
# In base64, then display the result with the specified text width.
printf "${base64_charset[$1]}" $((width++))\n(( width == text_width == 0 )) && printf "\n"
}
```

```
function encode_base64 {
# Encode 8-bit hexadecimal codes into 4-bit numbers.
# We need two local array variables:
c8[0]: to store the codes of the 8-bit characters to encode
c6[0]: to store the corresponding values on 6-bit
declare -a c8 c6
```

```
# Hexadecimal decimal.
c8=$(printf "base=16:%s:1:2\n%s:1:2\n" $1 $2 $3 | bc )
```

```
# Let's play with bitwise operators
# (38-bit into 6-bit conversion).
(( c6[0] = ((c8[0] >> 2)) ))
(( c6[1] = ((c8[0] >> 3) << 4) | ((c8[1] >> 4)) ))
```

```
# The following operations depend on the c8 element number.
case $c8[$3] in
3) ((c6[0] = ((c8[1] << 5) | (c8[2] >> 6)) ))
(( c6[1] = (c8[2] & 63 )) ))
2) ((c6[2] = (c8[1] << 2)) ))
(( c6[3] = 64 )) ))
1) ((c6[2] = c6[3] = 64 )) ))
esac
```

```
for char in ${c6[$6]}; do

done
```
