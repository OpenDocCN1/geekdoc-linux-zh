# 示例 9-16. 为 RANDOM 重新种子

```bash
#!/bin/bash
# seeding-random.sh: 为 RANDOM 变量种子。
# v 1, 日期 13年2月09日

MAXCOUNT=25      # 生成多少个数字。
SEED=

random_numbers () {
  local count=0
  local number

  while [ "$count" -lt "$MAXCOUNT" ]; do
    number=$RANDOM
    echo -n "$number "
    let count++
  done
  echo
}

echo
=1
RANDOM=$SEED      # 设置随机数生成器。
echo "Random seed = $SEED"
random_numbers

RANDOM=$SEED      # 为 RANDOM 使用相同的种子 . . 
echo "Again, with same seed..."
echo "Random seed = $SEED"
random_numbers    # . . 重现了完全相同的序列。
```
