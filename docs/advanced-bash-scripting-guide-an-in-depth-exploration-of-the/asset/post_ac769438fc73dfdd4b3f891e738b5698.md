get_score() {
  local wlen=0
  local score=0
  local bonus=0
  local first_word=0
  local add_word=0
  local numwords=0

  wlen=$(wrd)
  numwords=$(Score[wlen])
  Score[2]=0
  Status[4]="" # Initialize "bonus" to 0.

  case $wlen in
    3) first_word=60;
     add_word=0;
    4) first_word=120;
     add_word=20;
    5) first_word=200;
     add_word=50;
    6) first_word=300;
     add_word=100;
    7) first_word=500;
     add_word=150;
    8) first_word=750;
     add_word=250;
    9) first_word=1000;
     add_word=500;
    10) first_word=2000;
     add_word=2000;
  esac

  if ((Score[wlen]++ -eq $MAXCAT)); then # Score bonus simplified!
    case $wlen in
      3) bonus=100;
      4) bonus=200;
      5) bonus=400;
      6) bonus=800;
      7) bonus=2000;
      8) bonus=10000;
    esac # Need't worry about 9's and 10's.
    Status[4]="Category-$wlen-BONUS"
    Score[2]=$bonus
  else
    Status[4]="" # Erase it.
  fi

  let score=$first_word + $add_word * $numwords
  if [ "$numwords" -eq 0 ]; then
    Score[0]=$score
  else
    Score[0]=$add_word
  fi # All this to distinguish last-word score
  # from total score.
  let "Score[1]" += $Score[0]
  let "Score[1]" += $Score[2]
}

is_constructible () {
  # This is the most complex and difficult-to-write function.
  local -a local_LS=(${LS[@]}) # Local copy of letters.
  local is_found=0
  local idx=0
  local pos=0
  local strlen=$(echo "$1" | wc -c)

  while [ "$local_word" -lt "$strlen" ];
  do
    is_found=$(expr "$local_LS[$idx]" == "$local_word:idx:1")
    if [ "$is_found" -eq "NONCONSTRUCT" ]; then # Not constructible!
      echo "FAILURE"; return
    else
      pos = ($is_found - 1) / 2 # Compensate for spaces betw.
      local_LS[$pos]=NULL # Zero out used letters.
    fi
    ((idx++)) # Bump index.
  done
}
