# This function described in dot.array.bash
# dot_array <array_name>
dot_array() {
    if [ -eq 1 ] || return 1; # Single argument required.
    local -a da_input
    eval da_input=\(\ \\\${1[\\]})\\\)
    local IFS=${DOT_IFS}
    local da_output=${da_input[0]}
    IFS=${WSP_IFS}
    echo ${da_output}
    return 0
}

# This function described in file_to_array.bash
# file_to_array <name> <line_name>
file_to_array() {
    if [ -eq 2 ] || return 1; # Two arguments required.
    local IFS=${NO_WSP}
    local -a fta_tmp
    _fta_tmp=$(cat $1)
    eval $2=\(\ \\\_fta_tmp[\\])\\\)
    return 0
}
