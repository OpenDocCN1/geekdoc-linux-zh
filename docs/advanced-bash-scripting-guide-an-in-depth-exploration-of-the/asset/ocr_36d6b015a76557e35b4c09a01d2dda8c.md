# LocateFile [-1] FileName Location-Array-Name
# LocateFile [-1] -of FileName Location-Array-Name

LocateFile()
{
    local -a LOC1 LOC2
    local lm="of=0

    case "$#" in
        0) return 1 ;;
        1) return 1 ;;
        2) while ( ("$#" > 2 ))
        do
            case "$1" in
                -1) lk=-1 ;;
                -of) of=1 ;;
            esac
            shift
            done
            return 1 ;;
    esac
