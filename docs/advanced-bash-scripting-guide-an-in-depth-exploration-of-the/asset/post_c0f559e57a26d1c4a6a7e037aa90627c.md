# Example A-52. Cycling through all the possible color backgrounds

```bash
#!/bin/bash

# show-all-colors.sh
# Displays 256 possible background colors, using ANSI escape sequences.
# Author: Chetankumar Phulgare
# Used in ABS Guide with permission.

T1=8
T2=6
T3=36
offset=0

for num1 in {0..7}
do
  for num2 in {0..1}
  do
    shownum=$(echo "$offset + $T1 * $num2" | bc)
    echo -e "\E[48;5;${shownum}m color \E[0m"
  done
done
echo
offset=16
for num1 in {0..5}
do
  for num2 in {0..5}
  do
    for num3 in {0..5}
    do
      shownum=$(echo "$offset + $T2 * $num3" | bc)
      echo -e "\E[48;5;${shownum}m color \E[0m"
    done
  done
done
echo
offset=23
for num1 in {0..23}
do
  shownum=$(echo "$offset + $num1" | bc)
  echo -e "\E[48;5;${shownum}m color \E[0m"
done
echo
```

```bash
extract_letters () {
    local pos=0 # Slice string argument, letter by letter.
    local len=1 # Starting at left end of string.
    strim=$1 # One letter at a time.
    while [ $pos -lt ${#strim} ]; do
        letter=${strim:pos:len}
        # See Chapter 10.1.
        play_letter $letter
        echo -n " " # Mark letter just played.
        ((pos++))
    done
}
```
