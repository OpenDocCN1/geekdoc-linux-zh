void cursor(int y, int x)
{
  rg.x.ax = 0x0200;
  rg.x.bx = 0;
  rg.x.dx = ((y << 8) & 0xff00) + x;
  int86(16,&rg,&rg);
}

void curr_cursor(int *y, int *x)
{
  rg.x.ax = 0x0300;
  rg.x.bx = 0;
  int86(16,&rg,&rg);
  *x = rg.h.dl;
  *y = rg.h.dh;
}
