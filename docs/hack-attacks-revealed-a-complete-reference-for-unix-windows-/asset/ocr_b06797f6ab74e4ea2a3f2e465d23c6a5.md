int GETOPT()
{
int result;
int x;
int y;

do
{
do
{
result = MOUSE_LOCATION(&x,&y);
if (result & 1)
{
if (x >= 52 && x <= 53 && y >= 7 && y <= 15)
return y - 7;
if (x >= 4 && x <= 40 && y >= 7 && y <= 14)
return y + 10;

if (x >= 4 && x <= 40 && y == 15)
return y + 10;
}
} while(!bioskey(1));

result = bioskey(0);

x = result & 0xff;
if (x == 0)
{
result = result >> 8;
result -= 60;
}
}
while(result < 0 || result > 8);
return result;
}

void setvideo(unsigned char mode)
{
/* Sets the video display mode and clears the screen */

inreg.h.al = mode;
inreg.h.ah = 0x00;
int86(0x10, &inreg, &outreg);
}
