Sub vbWSACleanup()
' Subroutine to perform WSACleanup
iReturn = WSACleanup()
If iReturn <> 0 Then ' If WSock32 error, then tell me abou
it.
sMsg = "WSock32 Error -
" & Trim$(Str$(iReturn)) & " occurred in
Cleanup"

End
End If
End Sub
