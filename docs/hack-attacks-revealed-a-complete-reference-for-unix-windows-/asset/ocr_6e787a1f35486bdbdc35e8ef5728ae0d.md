long emmavail()
{
	/*
	Returns the number of available (free) 16K pages of expanded memo
	ry or -1 if an error occurs.

	*/
	union REGS regs;

	regs.h.ah = 0x42;
	int86(EMM,&regs,&regs);
	if (!regs.h.ah)
		return regs.x.bx;
	return -1;
}
```

```
long emmalloc(int n)
{
	/*
	Requests 'n' pages of expanded memory and returns the file handle
	assigned to the pages or -1 if there is an error
	*/
	union REGS regs;

	regs.h.ah = 0x43;
	regs.x.bx = n;
	int86(EMM,&regs,&regs);
	if (regs.h.ah)
		return -1;
	return regs.x.dx;
}
