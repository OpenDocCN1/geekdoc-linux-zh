void book_background(int color)
{
int x,
y,
xmax = vga_getxdim(),
ymax = vga_getydim();

vga_setcolor(color);

for (y = 0; y < ymax; y++)
{
for (x = 0; x < xmax; x++)
{
vga_drawline(0, y, xmax, y);
}
