int x, y, key, quit = 0, height = -1;

while (!quit) {
    mouse_update();
    x = mouse_getx();
    y = mouse_gety();

    angle += (float)(x - (RENDERWIDTH / 2)) / 1024.0f;
    height += ((RENDERHEIGHT / 2) - (int)(y))*16;

    if (height > 1000*256)
        height = 1000*256;

    if (height < ((int)heightmap[iINDEX(mainview.cx,
        mainview.cy)] << 8))
        height = ((int)heightmap[iINDEX(mainview.cx,
            mainview.cy)] << 8);

    mainview.z = height;

    if ((key = vga_getkey()) > 0)
    {
        switch (key)
        {
        case '+': speed += 1.0 / 4.0;
            if (speed > 16) speed = 16;
            break;
        case '-': speed -= 1.0 / 4.0;
            if (speed < 0) speed = 0;
            break;
