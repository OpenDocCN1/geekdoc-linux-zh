void threed_circle(int xc, int yc, int radius)
{
    int a,
        b,
        i,
        j = 0,
        color = 1;

    if (radius < 56)
    {
        gl_setpalettecolor(color, 63, j, j);
        book_fillcircle(xc, yc, radius, color);
        xc += (radius / 4);
        yc -= (radius / 4);
        radius *= .60;

        a = 64 / radius;
        color++;
        if (radius < 16)
        {
            j = 20;
        } else if (radius < 37) {
            j = 15;
        }

        for (i = 0; i <= 63; i += a)
        {
            gl_setpalettecolor(color, 63, j, j);
            book_fillcircle(xc, yc, (radius - i), color);
            color = (color + 1 < 64) ? color + 1 : color;
            j = (j + 3 <= 63) ? j + 3 : 63;
        }
    } else if (radius > 110) {
