void move_cursor(int x, int y, int ox, int oy)
{
	vga_waitretreace();
	gl_putbox(ox, oy, 11, 19, mouse_box);
	gl_getbox(x, y, 11, 19, mouse_box);
	draw_cursor(x, y);
}
