```
#include <vgagl.h>
#include <lug.h>
#include <lugfnts.h>

int main(void)
{
  int        i,
             r,
             g,
             b;
  byte       *ptr;

  bitmap_hdr in;

  vga_init();
  vga_setmode(G1024x768x256);
  gl_setcontextvga(G1024x768x256);
  gl_setrgbpalette();

  read_gif_file("sample.gif", &in);

  for (i = 0, ptr = in.cmap; i < in.colors; i++)
  {
    r = (double)*ptr++;
    g = (double)*ptr++;
    b = (double)*ptr++;
    gl_setpalettecolor(i, r>>2, g>>2, b>>2);
  }

  gl_putbox(0, 0, in.xsize, in.ysize, in.r);

  vga_getch();
  vga_setmode(TEXT);

  return 0;
}
```

> Note. Because liblug is a static library, you don’t need to worry if people on other systems will be able to use your compiled code. Any liblug routines you include will be hard-compiled into your executable. Just use `-llug` in exactly the same manner as you’d use `-lvga`.
