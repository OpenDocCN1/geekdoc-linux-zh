        l3d_rasterizer_3d_imp *i3) :

      imp3d = i3;
    }
    virtual ~l3d_rasterizer_3d(void) {}

      imp3d->draw_point(x,y,col);
    }

      imp3d->draw_line(x0,y0,x1,y1,col);
    }

      imp3d->clear_buffer();
    }

    {imp3d->draw_polygon_textured_lightmapped(p_poly); }

    {imp3d->draw_text(x, y, string); }
};
