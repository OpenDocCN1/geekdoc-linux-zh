sub get_form_name {
  my $referrer = shift;

  # If the HTTP referrer has a query string in it
  # return only the nonquery string part of it
  #
  if ($referrer =~ /^/()) {
     ($referrer, undef) = split(//?, $referrer);
  }

  return basename($referrer);
}
