sub specialClient{
#
# This subroutine outputs a proxy server configuration
#
    print <<FUNC;

        function FindProxyForURL(url, host)
        {
            if (isPlainHostName(host) ||
               dnsDomainIs(host, "*.nietec.com"))
               return "DIRECT";
            else if (shExpMatch(host, "*.com"))
               return "PROXY com-proxy.nietec.com:8080; "
            else if (shExpMatch(host, "*.edu"))
               return "PROXY edu-proxy.nietec.com:8080; "
            else
               return "DIRECT";
        }
    FUNC
}

#
# This subroutine outputs a proxy server configuration

        function FindProxyForURL(url, host)
        {
            return "DIRECT";
        }
    FUNC
}
