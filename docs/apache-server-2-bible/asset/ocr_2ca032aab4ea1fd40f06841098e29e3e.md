#!/usr/bin/perl  -w

# Variables
my $MAX = 10;
my $i;

# Content Header
print "Content-type: text/html\n\n";

# Main loop
for ($i=0; $i < $MAX; $i++){

    print "<FONT SIZE=$i>Font Size = $i</FONT><BR>";
}

exit 0;
