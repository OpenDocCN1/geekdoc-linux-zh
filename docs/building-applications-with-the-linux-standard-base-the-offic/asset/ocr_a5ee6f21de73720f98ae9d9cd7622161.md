# 3.12  LSB Don'ts

Example 3.7: Command Options

```
#include <unistd.h>
#include <stdio.h>

int
main (int argc, char **argv)
{
    int aflag = 0;
    int bflag = 0;
    char *cvalue = NULL;
    int errflg = 0;
    int index, c;

    /* the plus sign is required to permanently set
     * POSIXLY_CORRECT
     */

    while ((c = getopt (argc, argv, "+abc:")) != -1)
    switch (c) {
        case 'a':
            aflag = 1;
            break;
        case 'b':
            bflag = 1;
            break;
        case 'c':
            cvalue = optarg;
            break;
        case ':':
            fprintf (stderr,
                "Option -%c requires an operand\n",
                optopt);
            errflg++;
            return 1;
        case '?':
            fprintf (stderr, "Unknown option -%c\n", optopt);
            errflg++;
            return 1;
        default:
            fprintf (stderr, "Hopelessly lost\n");
            errflg++;
            break;
    }
```
