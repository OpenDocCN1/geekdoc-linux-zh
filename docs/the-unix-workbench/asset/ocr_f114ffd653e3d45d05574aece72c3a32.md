```
#!/usr/bin/env bash
# File: addseq2.sh

function addseq2 {
    local sum=0

    for element in $@
    do
        let sum=sum+$element
    done

    echo $sum
}
```

Now let’s source both files so we demonstrate how local helps us avoid overwriting variables.

```
source addseq.sh
source addseq2.sh
sum=4444
addseq 5 10 15 20
echo $sum
```
