3. The easiest way to implement this command is just to fork() a child process which will use execvp() to execute the specified command, and then use the times() call in the parent process after wait()ing for the child to finish, to obtain access to the child's system and user time information:

```
#include <sys/times.h>

main(int argc, char **argv)
{
    struct tms tm;

    if (fork())
    {
        wait(0);
        times(&tm);
        printf("User CPU time = %d.%d secs\n"
            , tm.tms_cutime/100, tm.tms_cutime%100);
        printf("System CPU time = %d.%d secs\n"
            , tm.tms_cstime/100, tm.tms_cstime%100);
    }
    else
    {
        execvp(argv[1], &argv[1]);
        exit(1);
    }
}
```
