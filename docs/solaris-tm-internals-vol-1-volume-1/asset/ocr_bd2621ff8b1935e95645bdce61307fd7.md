The MADV_DONTNEED flag allows us to give back memory for a given range. The following example shows how this can be done.

```
```c
#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>
#include <sys/sysmacros.h>
#include <sys/types.h>
#include <fcntl.h>
#include <sys/mman.h>

void
main(int argc, char ** argv)
{
        int fd;
        caddr_t addr;
        off_t size;
        off_t length;
        off_t offset;
        off_t chunk = 1<<30;

        if (argc != 2) {
                (void) fprintf(stderr, "Usage: mmap_dontneed filename\n");
                exit(1);
        }

        if ((fd = open(argv[1], O_RDONLY)) < 0) {
                perror("Open failed");
                exit(1);
        }

        size = lseek64(fd, (off_t)0, SEEK_END);

        /*
         * mmap a chunksize segment and issue madvise(DONTNEED)
         * against it.
         */
        for (offset = 0; offset < size; offset += chunk) {

                length = MIN(size - offset, chunk);

                if ((addr = mmap64((caddr_t)0, length, PROT_READ,
                    MAP_SHARED, fd, offset)) == NULL) {
                        perror("mmap failed");
                        exit(1);
                }

                (void) madvise(addr, length, MADV_DONTNEED);
                (void) munmap(addr, length);
        }
}
