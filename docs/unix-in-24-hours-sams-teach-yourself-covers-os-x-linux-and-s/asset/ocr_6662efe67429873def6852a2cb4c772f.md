case $filename in
    *.gif|*.jpg )
        echo "<li><img src=$directory/$filename /><br />"
        showname $filename
        ;;
    *.htm|*.html )
        echo "<li><a href=$filename>$filename</a></li>" ;;
    *)
        if [ -d $filename ] ; then
            echo \
            "<li><tt><a href=$iamcalled?$directory/$filename>"
            echo "[$filename]</a></tt></li>"
        else
            echo "<li>"
            showname $filename
        fi
        ;;
esac
done

echo "</ul>"

exit 0
