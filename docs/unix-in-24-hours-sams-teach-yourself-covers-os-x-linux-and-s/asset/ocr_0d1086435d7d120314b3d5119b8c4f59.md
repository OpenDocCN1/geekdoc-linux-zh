sub read_exchange_rate {
    open(EXCHRATES, "exchange.txt") ||
        die "Can't find current exchange rates.\n";

    while ( <EXCHRATES> ) {
        chomp; my @fields = split;  # remove control chars, break up into words
        $curr = $fields[0];
        $val  = $fields[1];
        $rateof{$curr} = $val;
    }
    close(EXCHRATE);
}
