sub breakdown {
    @line = split("", $_);      # split at space

    $amnt = $line[0];
    if ($#line == 1) {
        $curr = $line[1];
        $curr =~ tr/a-z/A-Z/;      # normalize to uppercase
        $curr = substr($curr, 0, 1); # extract first char only
    } else { $curr = "U"; }
    return ($amnt, $curr);
}
