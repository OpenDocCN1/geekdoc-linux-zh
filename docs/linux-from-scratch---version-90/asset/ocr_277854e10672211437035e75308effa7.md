need_gateway=1

("default")
need_gateway=1
args="${args} default"
desc="default"
;

("host")
need_ip=1
;

("unreachable")
need_ip=1
args="${args} unreachable"
desc="unreachable"
;

(*)
log_failure_msg "Unknown route type (${TYPE}) in ${IFCONFIG}, cannot continue.
exit 1
esac

if [ -n "${GATEWAY}" ]; then
MSG=${GATEWAY} variable cannot be set in ${IFCONFIG} for static routes."
log_failure_msg "MSG Use STATIC_GATEWAY only, cannot continue"
exit 1
fi

if [ -n "${need_ip}" ]; then
if [ -z "${IP}" ]; then
log_failure_msg "IP variable missing from ${IFCONFIG}, cannot continue.
exit 1
fi

if [ -z "${PREFIX}" ]; then
args="${args} ${IP}/${PREFIX}
desc="${desc} ${IP}/${PREFIX}"
fi

if [ -n "${need_gateway}" ]; then
if [ -z "${STATIC_GATEWAY}" ]; then
log_failure_msg "STATIC_GATEWAY variable missing from ${IFCONFIG}, cannot continue.
exit 1
fi
args="${args} via ${STATIC_GATEWAY}"
fi

if [ -n "${SOURCE}" ]; then
args="${args} src ${SOURCE}"
fi
