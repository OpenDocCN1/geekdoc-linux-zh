```
/ /lib/lsb/init-functions
[ -r /etc/sysconfig/network ] && . /etc/sysconfig/hostname =cat /etc/hostname
[ -f $1 ] && HOSTNAME=$(cat /etc/hostname)
```

```
case "$1" in
start)
  log_info_msg "Brings up the loopback interface..."
  ip addr add 127.0.0.1/8 label 0 dev lo
  ip link set lo up
  evaluate_retval
```

```
  log_info_msg "Setting hostname to ${HOSTNAME}..."
  hostname ${HOSTNAME}
  evaluate_retval
  ;;
```

```
stop)
  log_info_msg "Brings down the loopback interface..."
  ip link set lo down
  evaluate_retval
  ;;
```

```
restart)
  $($0) stop
  sleep 1
  $($0) start
  ;;
```

```
status)
  echo "Hostname is: $(hostname)"
  ip link show lo
```
