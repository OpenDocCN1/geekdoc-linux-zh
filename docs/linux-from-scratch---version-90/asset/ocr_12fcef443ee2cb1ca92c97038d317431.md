    # The PID file contains the needed PIDs
    # Note that LSB requirement, the path must be given to pidofproc

    pidlist="pidofroc -p "$pidfile" "${1}"
    retval="${?}"
fi

# Return a value ONLY

    else

    # and returned as an unspecified error.
    return 1
    ;;
