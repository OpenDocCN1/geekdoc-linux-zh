```bash
# Function to create files/descriptor on bottom.
create_files()
{
    # Input to file descriptor 9 and output to stdin (redirection)
    exec 9< /etc/sysconfig/createfiles

    while read -name type perm usr grd dtype maj min junk
    do
        # Ignore comments and blank lines.
        case "${name}" in
            \#*") continue; 
        esac

        # Ignore existing files.
        if [ -e "${name}" ]; then
            # Create stuff based on its type.
            case "${type}" in
                dir)
                    mkdir "${name}"
                    ;
                file)
                    :> "${name}"
                    ;
                dev)
                    case "${type}" in
                        char)
                            mkod "${name}" c ${maj} ${min}
                            ;
                        block)
                            mkod "${name}" b ${maj} ${min}
                            ;
                        pipe)
                            mkod "${name}" p
                            ;
                        *)
                            log_warning_msg "Unknown device type: ${type}"
                            ;
                    esac
                esac
            fi
        fi
        log_warning_msg "Unknown type: ${type}"
        continue
    done
}
```

```bash
case "$1" in
start)
  log_info_msg "Brings up the loopback interface..."
  ip addr add 127.0.0.1/8 label 0 dev lo
  ip link set up
  evaluate_retval

  log_info_msg "Setting hostname to ${HOSTNAME}..."
  hostname ${HOSTNAME}
  evaluate_retval
  ;;

stop)
  log_info_msg "Brings down the loopback interface..."
  ip link set lo down
  evaluate_retval
  ;;

restart)
  $($0) stop
  sleep 1
  $($0) start
  ;;

status)
  echo "Hostname is: $(hostname)"
  ip link show lo
  ;;

*)
echo "Usage: ${0} {start|stop|restart|status}"
  exit 1
  ;;
esac

exit 0
```
