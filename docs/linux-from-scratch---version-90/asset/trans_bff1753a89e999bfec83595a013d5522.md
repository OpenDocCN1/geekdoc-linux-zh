# D.24. /sbin/ifup

```bash
#!/bin/sh
###############################################################################
# 开始 /sbin/ifup
# 描述 ：接口启动

# 版本 ： LFS 7.7
# 注意 ：IFONFIG 变量被传递给 /lib/services 目录下的 SERVICE 脚本，
# 用于服务应该加载哪个文件来获取接口规范。
##############################################################################################
```

```bash
up()
{
log_info_msg "正在启动 ${1} 接口..."

if ip link show $1 > /dev/null 2>&1; then
  link_status=`ip link show $1`

  if [ -n ${link_status} ]; then
    if echo "${link_status}" | grep -q UP; then
      ip link set $1 up
    fi
  fi
else
  log_failure_msg "接口 ${INTERFACE} 不存在。"
  exit 1
fi

evaluate_retrieval
}

RELEASE="7.7"

USAGE="Usage: $0 [-h -V] [--help] [--version] interface"
STR="IFS ifup, version ${RELEASE}"

while [ \$# -gt 0 ]; do
  case "$1" in
    --help | -h)
      help="y"; break; 
    --version | -V)
      echo "ifup: ${1}: invalid option" >&2
      echo "${USAGE}" >&2
      exit 2; 
    -*)
      break; 
  esac
done

if [ -n "$help" ]; then
  echo "${VERSTR}"
  echo "${USAGE}"
  echo
  cat << HERE_EOF
ifup 用于启动网络接口。接口参数（例如 eth0 或 eth0:2）必须与接口规范文件的末尾部分匹配，例如 /etc/sysconfig/ifconfig.eth0:2。
HERE_EOF
  exit 0
fi

file=/etc/sysconfig/ifconfig.${1}
# 跳过备份文件
[ "${file}" = "${file}"-"" ] || exit 0

./lib/lsb/init-functions

if [ -r "${file}" ]; then
  log_failure_msg "无法启动 ${file} 接口！${file} 缺失或无法访问。"
  exit 1

$file

if [ "${INTERFACE}" = "" ]; then
  log_failure_msg "无法启动 ${file} 接口！${file} 未定义接口 ${INTERFACE}"
  exit 1
fi

# 如果是由启动启动的且 ONBOOT 未被设置为 yes，则不处理此服务
if [ "${IN_BOOT}" = "1" -a "${ONBOOT}" != "yes" ]; then
  exit 0
fi

# 启动接口
if [ "${VIRTINT}" != "yes" ]; then
  up ${INTERFACE}
fi

for S in ${SERVICE}; do
  if [ ! -x "/lib/services/${S}" ]; then
    MSG="无法处理 \${file}。"
    MSG="${MSG}服务 '${S}' 不存在"
    MSG="${MSG}或无法执行。"
    log_failure_msg "$MSG"
    exit 1
  fi
done

if [ "${SERVICE}" = "wpa" ]; then log_success_msg; fi

# 创建/配置接口
for S in ${SERVICE}; do
  IFCONFIG=${file} /lib/services/${S} ${INTERFACE}

  # 设置虚拟接口
  if [ "${VIRTINT}" = "yes" ]; then
    up ${INTERFACE}
  fi

  # 启动任何额外的接口组件
  for I in ${INTERFACE_COMPONENTS}; do up ${I}; done

  # 如果请求则设置 MTU。检查 MTU 是否有“良好”值。
  if test -n "\${MTU}"; then
    if [[ \${MTU} =~ ^[0-9]+$ ]] && [[ \${MTU} -ge 68 ]]; then
      for I in ${INTERFACE_COMPONENTS}; do
        ip link set dev ${I} mtu ${MTU}
      done
    fi
  else
    log_info_msg "无效的 MTU \${MTU}"
  fi
done

# 如果请求则设置默认网关
if -n "${GATEWAY}"; then
  if ip route | grep -q already setup; then
    log_warning_msg "网关已设置，跳过。"
  else
    log_info_msg "向 ${IFACE} 接口添加默认网关 \${GATEWAY}"
    ip route add default via \${GATEWAY} dev \${IFACE}"
    evaluate_retrieval
  fi
fi

# 结束 /sbin/ifup
