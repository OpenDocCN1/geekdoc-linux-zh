```bash
case "${TYPE}" in
  ("" | "network")
    need_ip=1
    need_gateway=1

  ("default")
    need_gateway=1
    args="${args} default"
    desc="default"
    ;

    ("host")
      need_ip=1
      ;

    ("unreachable")
      need_ip=1
      args="${args} unreachable"
      desc="unreachable"
      ;

    *)
      log_failure_msg "在 ${IFCONFIG} 中遇到未知路由类型 (${TYPE})无法继续。
      exit 1
esac

if [ -n "${GATEWAY}" ]; then
  MSG="${GATEWAY} 变量无法在 ${IFCONFIG} 中设置静态路由。"
  log_failure_msg "MSG: ${MSG} 请仅使用 STATIC_GATEWAY，无法继续"
  exit 1
fi

if [ -n "${need_ip}" ]; then
  if [ -z "${IP}" ]; then
    log_failure_msg " ${IFCONFIG} 中缺少 IP 变量，无法继续。"
    exit 1
  fi

  if [ -z "${PREFIX}" ]; then
    args="${args} ${IP}/${PREFIX}"
    desc="${desc} ${IP}/${PREFIX}"
  fi

  if [ -n "${need_gateway}" ]; then
    if [ -z "${STATIC_GATEWAY}" ]; then
      log_failure_msg " ${IFCONFIG} 中缺少 STATIC_GATEWAY 变量，无法继续。"
      exit 1
    fi
    args="${args} via ${STATIC_GATEWAY}"
  fi

  if [ -n "${SOURCE}" ]; then
    args="${args} src ${SOURCE}"
  fi
fi
```
