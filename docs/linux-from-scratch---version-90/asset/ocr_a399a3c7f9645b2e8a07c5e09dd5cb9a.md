```
#!/bin/sh
######################################################################
# Begin setclock
# Description : Setting Linux Clock
# Authors : Gerard Beekmans - gerard@linuxfromscratch.com
# DJ Lucas - dj@linuxfromscratch.com
# Update : Bruce Dubbs - bdbubbs@linuxfromscratch.com
# Version : LFS 7.0
######################################################################

### BEGIN INIT INFO
# Provides:
# Required-Start:
# Should-Start: modules
# Required-Stop:
# Should-Stop: $syslog
# Default-Start: S
# Default-Stop:
# Description: Stores and restores time from the hardware clock
# Description: On boot, system time is obtained from hwclock. The hardware clock can also be set on shutdown.
# X-LFS-Provided-By: LFS BLFS
#### END INIT INFO

/lib/lsb/init-functions

[ -r /etc/sysconfig/clock ] && . /etc/sysconfig/clock

case ${UTC} in
yes|1)
  CLOCKPARAMS="${CLOCKPARAMS} --utc"
  ;;

false|0)
  CLOCKPARAMS="${CLOCKPARAMS} --localtime"
  ;;
esac

case $1 in
start)
  hwclock --hctctys $(CLOCKPARAMS) >/dev/null
  ;;

stop)
  log_info_msg "Setting hardware clock..."
  hwclock --systohc $(CLOCKPARAMS) >/dev/null
  evaluate_retval
  ;;
```
