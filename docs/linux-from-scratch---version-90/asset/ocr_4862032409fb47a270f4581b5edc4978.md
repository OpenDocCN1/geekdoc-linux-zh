# Inputs: -p pidfile, use the specified pidfile instead of pidof

# Return values (as defined by LSB status codes):
#   0 - Success (PIDs sent to stdout)
#   1 - Program is dead, PID file still exists (remaining PIDs exist)

        pidfile="${2}"
        shift 2
        ;;

      *)
        program="${1}"
        if [ -n "${2}" ]; then
          # Too many arguments
          # Since this is this status, return unknown
          return 4
        else
          break
        fi
        ;;
    esac
  done
