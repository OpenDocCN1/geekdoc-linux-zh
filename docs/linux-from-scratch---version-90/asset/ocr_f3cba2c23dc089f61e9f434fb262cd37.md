```
# Screen Dimensions
# Find current screen size
if [ -z "${COLUMNS}" ]; then
COLUMNS=$(stty size)
COLUMNS=${COLUMNS## *}
fi
```

```
# When using remote connections, such as a serial port, stty size returns 0
if [ "${COLUMNS}" = "0" ]; then
COLUMNS=80
fi
```

```
# Measurements for positioning return messages
COL=$((${COLUMNS} - 8))
WCOL=$((${COL} - 2))
```

```
# Set Cursor Position,, via echo
```
