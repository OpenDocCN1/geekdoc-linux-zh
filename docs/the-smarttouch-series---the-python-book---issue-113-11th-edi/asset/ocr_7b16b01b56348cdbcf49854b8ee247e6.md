class GameState:
    pass


class Game(object):

    def __init__(self):
        pygame.init()
        self.clock = pygame.time.Clock()
        self.game_font = pygame.font.Font(
            'data/OrbitrOne.ttf', 25)
        self.info_font = pygame.font.Font(
            'data/OrbitrOne.ttf', 72)
        self.screen = pygame.display.set_mode([RES[0], RES[1]])
        self.time = pygame.time.get_ticks()
        self.refresh_rate = 25
        self.rounds_won = 0
        self.level_up = 50
        self.score = 0
        self.lives = 2
        self.player_group = pygame.sprite.Group()
        self.alien_group = pygame.sprite.Group()
        self.bullet_group = pygame.sprite.Group()
        self.missile_group = pygame.sprite.Group()
        self.barrier_group = pygame.sprite.Group()
        self.all_sprite_list = pygame.sprite.Group()
        self.intro_screen = pygame.image.load(
            'data/start_screen.png').convert()
        self.background = pygame.image.load(
            'data/Space-background.png').convert()
        pygame.display.set_caption("Invaders - ESC to exit")
        pygame.mouse.set_visible(False)
        Player.image = pygame.image.load(
            'data/ship.png').convert()
        Player.image = pygame.image.load(
            'data/Space/aliens1.png').convert()
        Alien.image.set_colorkey(WHITE)
        GameState.end_game = False
        GameState.start_screen = True
        GameState.shoot_bullet = True

    def control(self):
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                GameState.start_screen = False
                GameState.end_game = True
            if event.type == pygame.KEYDOWN \
                    and event.key == pygame.K_ESCAPE:
                GameState.start_screen = True
                GameState.end_game = False
                self.kill_all()
            else:
                GameState.start_screen = True
                self.keys = pygame.key.get_pressed()
                if self.keys[pygame.K_LEFT]:
                    GameState.vector = -1
                elif self.keys[pygame.K_RIGHT]:
                    GameState.vector = 1
                else:
                    GameState.vector = 0
                if self.keys[pygame.K_SPACE]:
                    if GameState.start_screen:
                        GameState.start_screen = False
                    self.lives = 2
                    self.score = 0
                    self.make_player()
                    self.make_defenses()
                    self.alien_wave()
                    GameState.shoot_bullet = True

    def splash_screen(self):
        while GameState.start_screen:
            self.kill_all()
            self.screen.blit(self.intro_screen, [0, 0])
            self.screen.blit(self.info_font.render(
                "PIVARCERS", 1, WHITE), (255, 120))
            self.screen.blit(self.game_font.render(
                "PRESS SPACE TO PLAY", 1, WHITE), (270, 150))

Groups
This long list of groups we're creating are essentially sets. Each one has the same type of items, so when it's added to the set it can be tested for or accessed easily.

Control
Taking care of keyboard input is the control method. It checks for key events and acts accordingly depending whether we're on the start screen or playing the game.
