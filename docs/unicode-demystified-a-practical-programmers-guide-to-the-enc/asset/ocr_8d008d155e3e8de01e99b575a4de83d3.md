# Chapter 15 The Z Shell and Advanced Shell Programming

```
# Strategy: while there is still input left in both files:
# Output the lesser line.
# Read a new line from the file that line came from.
while [[ $Status1 -eq 0 && $Status2 -eq 0 ]]
do
  if [[ "$Line1" > "$Line2" ]];
    then
      print "1.\t$Line1"
      read -u3 Line1
      Status1=$?
  else
    print "2.\t$Line2"
    read -u4 Line2
    Status2=$?
  fi
done
# Now one of the files is at end-of-file.
# Read from each file until the end.
# First file:
while [[ $Status1 -eq 0 ]]
do
  print "1.\t$Line1"
  read -u3 Line1
  Status1=$?
done
# Next file2:
while [[ $Status2 -eq 0 ]]
do
  print "2.\t$Line2"
  read -u4 Line2
  Status2=$?
done
# Close and remove both input files
rm -f $file1 $file2
exit 0
```
