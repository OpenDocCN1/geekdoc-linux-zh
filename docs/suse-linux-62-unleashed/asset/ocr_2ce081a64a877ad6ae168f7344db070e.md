# LISTING 22.4 hold-A "Keep-Alive" Written in Expect

```
#!/usr/bin/expect
```

```
# Usage: "hold HOST USER PASS"
# Action: login to node HOST as USER. Offer a shell prompt to the screen word HELD
# normal usage, and also print every five seconds, to exercise the connection periodically.
# This is useful for testing and WANs with short time-outs.
# You can walk away from the keyboard, and never lose your
# connection through a time-out.
# WARNING: the security hazard of passing a password through the
# command line makes this example only illustrative. Modify to
# a particular security situation as appropriate.
set hostname [lindex $arg0]
set username [lindex $arg1]
set password [lindex $arg2]
```

```
set prompt_sequence " * ""

spawn telnet $hostname

expect "login:"
send "$username\r"
expect "Password:"
send "$password\r"
```

```
# Some hosts don't inquire about TERM. That's another
# complexification to consider before widespread use
# this application is practical.
# Note use of global [gl] pattern matching to parse ***
# as a wildcard.
```

```
expect -gl "TERM = (*)"
send "\r"
expect $prompt_sequence
send "sh -c 'while; do echo HELD; sleep 5; done\r"
interact
```
