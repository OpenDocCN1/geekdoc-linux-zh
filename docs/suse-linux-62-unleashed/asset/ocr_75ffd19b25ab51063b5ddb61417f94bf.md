```
#! /bin/sh

 /sbin/init.d/apache
#
DBROOT=/dev/null
 /etc/rc.config

# Determine the base and follow a runlevel link name.
base=$$(@##*/)
link=${base#*[SK][0-9]}[0-9]}

# Force execution if not called by a runlevel directory.
test $link=$base && START_HTTPD="yes"
test '$START_HTTPD' = yes || exit 0
#
# set DBROOT for ADABAS D driver in PHP
#
export DBROOT

# The echo return value for success (defined in /etc/rc.config).
return=$rc_done

case "$1" in
start)
if test -x /usr/sbin/httpd ; then
echo -n "Starting service httpd"
startproc /usr/sbin/httpd -f /etc/httpd/httpd.conf -D SSL !! return=$rc_failed
if test -e /var/log/ssl_scache.dir ; then
chown wwwrun /var/log/ssl_scache.dir ;
fi
if test -e /var/log/ssl_scache.pag ; then
chown wwwrun /var/log/ssl_scache.pag
fi
```
