# LISTING 1.1 Sample script for an automated installation of vCenter Server

```
#Install vCenter Server unattended
$VCMedia = "D:\vCenter-Server"
$SVC_USER = "WIN_VCENTER6\vCenter"
$SVC_PASS = "VMw@re123"
$FQDN = "MGMT-VC6.centos.com"
$vIP = "10.144.99.16"
```

```
#Database Info
$TYPE = "external"
$DSN = "vCenter"
$USER = "vCenter"
$PASS = "VMw@re123"
```

```
$SSO_DOMAIN = "vphere.local"
$SSO_PASS = "VMw@re123!"
$SSO_SITE = "MY_SITE"
```

```
# Install vCenter
Write-Host "/i \`$VCMedia\VMware-vCenter-Server.msi`"
$vars = "/1e \`c:\temp\vCenterInstallation.txt`" /gr"
$vars += "/LAUNCHED_EXE=0 FQDN=\`$FQDN`"
$vars += "/INSTALL_TYPE=$Type DB_DSN=\`$DSN`"
$vars += "/DB_USER=\`$USER`"
$vars += "/DB_PASSWORD=\`$PASS`"
$vars += "/INFRA_NODE_ADDRESS=\`$vIP`"
$vars += "/VC_SVC_USER=\`$SVC_USER`"
$vars += "/VC_SVC_PASSWORD=\`$SVC_PASS`"
$vars += "/SSO_DOMAIN=\`$SSO_DOMAIN`"
$vars += "/SSO_PASSWORD=\`$SSO_PASS`"
$vars += "/SSO_SITENAME=\`$SSO_SITE`"
Start-Process msiexec -ArgumentList $vars -Wait
```
