A switch that defines if the ExtensionData property shall be handled or not

EXAMPLE

```
Get-VM -Name VM1 | ConvertTo-Text -Depth 2
```

EXAMPLE

```
ConvertTo-Text -InputObject $xs -FullPath
```

#

#Requires -Version 4.0

```
[CmdletBinding()]
Param(
    [Parameter(ValueFromPipeline=$True)]
    object[]]$InputObject,
    [int]$Depth = 2,
    [switch]$FullPath = $false,
    [switch]$ExtensionData = $false,
    [Parameter(DontShow)]
    [string]$Indent = '',
    [Parameter(DontShow)]
    [string]$Path = ''
)
```

```
Process{
    if($Indent.Length -lt $Depth){
        foreach($object in $InputObject){
            $object.PSObject.Properties | ForEach-Object -Process {
                if ($FullPath){
                    "\$($Path + '\' + $_.Name)" - \$($_.TypeNameValue) = \$($_.Value)"
                }
                else{
                    "\$($Indent)$($_.Name)" - \$($_.TypeNameValue) = \$($_.Value)"
                }
            }
            if($_.Name -ne 'ExtensionData' -or $ExtensionData) -
                $_.PSObject.Properties{
                    $ctSplat = @{
                        InputObject = $object."\$($_.Name)"
                    }
                }
```
