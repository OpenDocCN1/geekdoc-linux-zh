```
if ($PsCmdlet.ShouldProcess "VMHost '\$($ThisHostSystem.Name)\', `\
  $strShouldProcessOperationMsg) {
  try {
    ## update the Lockdown exceptions
    $ThisHostAccessMgr.UpdateLockdownExceptions($arrLockdownExceptToSet)
    $arrNewLockdownExceptions = \
      $ThisHostAccessMgr.QueryLockdownExceptions()
    ## return info object
    New-Object -Type PSObject -Property ([ordered]@{
      Name = $ThisHostSystem.Name
      LockdownExceptionUser = \
        if ($arrNewLockdownExceptions.Count -ne 0) {
          $arrNewLockdownExceptions = $ThisHostSystem.MoRef
        }
    })
  } catch { Throw "encountered issue setting Lockdown exceptions for ` \
    VMHost '\$($ThisHostSystem.Name)'. The error: $_*"
  }
}
```

```
Set-VMHostLockdownException -UserName dom\SvcAcct0
```
