```
# For hardware v4 match disks based on controller's SCSIPPort
# disk's SCSTargetID.
# Not supported with mixed scsi adapter types.
$DiskMatch = $WinWindows | Where-Object {$_.SCSIPPort -eq $PortOffset

eq $VirtualSCSISController.BusNumber -and
$_.SCSTargetID -eq $VirtualDiskDevice.UnitNumber
}\else
{
# For hardware v7+ match disks based on controller's pciSlotNumber
# and disk's SCSTargetID
$DiskMatch = $WinWindows | Where-Object {$_.CtrlPciSlotNumber -eq
($VM.Extensiondata.Config.Extraconfig | Where-Object {$_.key -match
"scsi"($VirtualSCSISController.BusNumber).pcislotnumber")}).value
-and $_.SCSTargetID -eq $VirtualDiskDevice.UnitNumber
}
{
$VirtualDisk.WindowsDisk = "Disk $($DiskMatch.Index)"
$VirtualDisk.WindowsDiskSizeGB = $DiskMatch.Size / 1 GB
}
```
