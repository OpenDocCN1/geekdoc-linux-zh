process{
$eventMgr.Description.EventInfo |
Where-Object {$Category -contains $_.Category} | Foreach-
Object {
  New-Object PSObject -Property @{
    Name = $_.Key
    Category = $_.Category
    Description = $_.Description
    Hierarchy = &={
      $obj = New-Object -TypeName "VMware.Vim." + $+.Key
      if ($obj) {
        $obj = $obj.GetType()
        $path = ""
        do{
          $path = ($obj.Name + "/") + $path
          $obj = $obj.BaseType
        } until ($path -like "Event*")
        $path.TrimEnd("/")
      }
      else{
        "--undocumented--"
      }
    }
  }
}
