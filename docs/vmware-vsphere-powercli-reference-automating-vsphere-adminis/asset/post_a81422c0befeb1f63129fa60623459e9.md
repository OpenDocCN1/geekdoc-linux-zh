  <parameters>
    <parameter type="$(ParameterType)" name="$(ParameterName)" scope="local">

  $Body = $ParameterXML

  $Body = @"

  <parameters>
    <parameter type="string" name="$(ParameterName)" scope="local">
      <string>$(ParameterValue)</string>
    </parameter>
  </parameters>
</execution-context>
"@
}

if ($PSBoundParameters.ContainsKey('ItemHref')) {
  $URL = $ItemHref
} else {
  $URL = "https://$Server/api/workflows/$(ID)"
}

$InvokeRequest = Invoke-WebRequest -Method Post -Uri $URL -Headers $Headers -ContentType "application/xml" -Body $Body

$Object = @{
    StatusCode = $InvokeRequest.StatusCode
    StatusDescription = $InvokeRequest.StatusDescription
    Execution = $InvokeRequest.Headers.Location
}

# Querying a Workflow State

Once a workflow has been invoked, it may take some time to complete. The Get-vROState function in Listing 21.20 retrieves the state of a workflow by submitting the Execution state (the output from Listing 21.19) plus the word state to make the following URL. URL could be used as the value for the ExecutionState parameter in the Get-vROState function.

```
https://vro01.snydale.local:8281/vco/api/workflows/c0278910-9ae2-46c5-bb45-2292fe88e3ab/executions/402880244ae8e2a6014b0961089a234/state
```

- **21.20** Get-vROState

```powershell
function Get-vROState {
#SYNOPSIS
 Get vRO Workflow State
.DESCRIPTION
 Get vRO Workflow State
.PARAMETER ExecutionStateRef
 vRO Workflow Execution Reference
.PARAMETER Credential
 PS Credential Object
.EXAMPLE
 $Credential = Get-Credential
 Get-vROState -ExecutionStateRef \
https://vRO001:8281/vco/api/workflows/\
656b2c35-3607-4ab9-7-9102c139108/executions/\
402880244ae8e2a6014b045ea920213 -Credential $Credential
}
```
