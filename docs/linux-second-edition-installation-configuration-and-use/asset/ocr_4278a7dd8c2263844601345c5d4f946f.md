```
source=\$\$\$ *
# first part of parameter (before space)
target=\$\##
# second part (after the space)
source1=$source@
# first part of source pattern
source2=$source#
# second part (following *)
target1=$target%
# first part of target pattern
target2=$target#
# second part (following *)
# Check whether source or target are empty
if -z "$source" ]; then
    echo regcp: Source pattern is missing
    exit
fi
if [ "$source" = "$target" ]; then
    echo regcp: Target pattern is missing or same as source
    exit
fi
# Check whether source and target contain exactly one * each
if [ $#source1 + $#source2 +1 ] -ne $#source ]; ]; then
    echo regcp: Source pattern must contain exactly one *
    exit
fi
# Check whether any source files are found at all
files=$(echo $source) # source files to be processed
if [ "$files" = "$source" ]; then
    echo "regcp: cannot find any $source files"
    exit
fi
# copy
for i in $files; do
    tmp1=$i%%$source1 # eliminate character preceding *
    tmp2=$i##$source2 # eliminate character following *
    tmp=$target$tmp1$tmp2 # assemble target file name
    echo "cp -ip $i $tmp"
    cp -ip $i $tmp
done
else
    echo "regcp: copies files using patterns"
    echo "The whole whole parameter must be enclosed in '"'"
    echo "Source and target pattern must contain exactly one *"
    echo
    echo "Example: regcp *.bak *.old"
fi
```
