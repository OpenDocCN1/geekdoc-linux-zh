```
# Bind the background to change color when the cursor
# leaves. Cancel any pending display of an info label.
bind $toolButton <Leave> "$toolButtonName \
configure--background $background; \
[namespace current]:::toolButton::cancelInfo \
$toolButtonName";
```

```
# Whenever the button moves, reset the Motion task.
bind $toolName <Motion> \
  "[list [namespace current]::toolButton::scheduleInfo \
  $toolButtonName info"]
```

```
# And return the name
return $toolButtonName
}
```
