#
$mixMenu add separator

-menu $mixMenu.cascade
menu $mixMenu.cascade
$mixMenu.cascade add command -label "Cascaded 1" \
-command "doStuff 3"
$mixMenu.cascade add command -label "Cascaded 2" \
-command "doStuff 3"

# Define a dummy proc for the command buttons to invoke.

}
