####################################################
# proc layout {w}--
# sets the location of the subwidgets
# Arguments
#   w        The window name for this megawidget.
# Results
#   packs the prompt and entry subwidgets in the locations
#   defined by the -side option

proc layout {w} {
    upvar \#0 [namespace current]::$w data
    if {[string match $data(-side)]} {
        pack $data(prompt) -side right
        pack $data(entry) -side left
    } else {
        pack $data(prompt) -side left
        pack $data(entry) -side right
    }
}

####################################################
# proc configure {w args}--
# Process configuration commands that are specific to the
# Arguments
#   w        The window name for this megawidget.
#   args            A list of key/value pairs.
# Results
#   Will either set the -side option to the requested value and
#   rearrange the window, or generate an error.

proc configure {w args} {
    upvar \#0 [namespace current]::$w data
    foreach (key val) $args {
        switch -- $key {
            "-side"    set data(-side) $val
    }
}
