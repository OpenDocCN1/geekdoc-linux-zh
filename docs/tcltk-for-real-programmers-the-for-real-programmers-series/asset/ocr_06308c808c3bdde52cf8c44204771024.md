Check the width of the contents of each cell.
use the font measure command to get width in pixels.
This provides a font-independent value.

```
for {set i 0} {$i <= $HTable(row)} {incr i {
  for {set j 0} {$j < $HTable(maxCol)} {incr j {
    A cell may be blank.
    Check for a tag before proceeding
    if ([search $alltags tbl.$sj] != -1) {
      set range [$table ranges $sj]
    }
    Get the size of the text associated with this
    cell, check to see if it's longer than
    previous longest text in this column
    set line [$table get [lindex range 0] \
[lindex range 1]]
    set len [font measure $var(xfont) $line]
    if {$len > $colSize($sj)} {
      set colSize($sj) $len
    n    Delete the trailing \n that was added by
    # html_lib.tcl.
    $table delete "[lindex range 0] lineend"
  }}
}
```
