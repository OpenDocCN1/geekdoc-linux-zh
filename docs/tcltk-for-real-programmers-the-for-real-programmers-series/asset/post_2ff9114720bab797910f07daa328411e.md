```tk
proc infoIdentifier {toolButtonName} {
    regsub -all {\.} $toolButtonName "-" infoID
    return $infoID
}
```

```tk
# proc cancelInfo {toolButtonName)--
# Cancels the scheduled display of an Info label
# Arguments
# toolButtonName The name of the toolbutton
# Results
# If this label was scheduled to be displayed,
#    it is disabled.
```

```tk
proc cancelInfo {toolButtonName} {
    variable after
    if {[info exists after({$toolButtonName})]} {
        after cancel $afters($toolButtonName);
    }
}
```

```tk
# name The name of this toolbutton
# info The text to display in the info label
# Results
# Places an item in the "after" queue to be evaluated
# 2 seconds from now

proc scheduleInfo {toolButtonName info} {
  variable affairs
  cancelInfo $toolButtonName
  set after ($toolButtonName) \
    [after 2000 [list \
      [namespaces current]::createInfo \
      $toolButtonName info]]
}
```
