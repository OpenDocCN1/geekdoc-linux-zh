+   # Fixup the printcap file in case it has continuation characters
# at the end of each entry, like old versions of printtool did.

TMP1=`mtmp /etc/printcap.XXXXXX`

```
gawk '
  BEGIN { first = 1; cont = 0; last = "" }
  [/[:space:]*#/ { if(cont) sub("\\", "", last);
  if(first == 0) print last }
  { first = 0 }
  { last = $0 }
  { cont = 0 }
  \\$/ { cont = 1 }
  END { sub("\\", "", last); print last }
```

```
/etc/printcap > ${TMP1} && cat ${TMP1} > /etc/printcap
  && rm -f ${TMP1}
```

```
fi
```
