stop () {
# stop daemon
echo -n "Stopping lpd: "
killproc /usr/sbin/lpd
RETVAL=$?
echo
[ $RETVAL = 0 ] && rm -f /var/lock/subsys/lpd
return $RETVAL
}

restart () {
stop
start

return $RETVAL
}
