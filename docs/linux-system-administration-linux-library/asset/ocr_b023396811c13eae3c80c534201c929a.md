```
status /usr/sbin/lpd
RETVAL=$$
;;
restart)
restart
;;
condrestart)
# only restart it if it is already running
[ -f /var/lock/subsys/lpd ] && restart ||
;;
reload)
echo -n "Reloading lpd: "
killproc /usr/sbin/lpd -HUP
RETVAL=$$
echo
;;
fixup)
fixup
;;
*)
echo "Usage: lpd (start|stop|restart|condrestart|reload)"
    status|fixup"
RETVAL=1
esac
```

The format of initialization scripts is fairly uniform. The standard SysV initialization
scripts have some features that we really like. At the beginning of the script is a comment
detailing what the script is intended to do.
