stop() {
    if [ -f /var/run/news/inn.d.pid ]
    then
        echo -n "Stopping INND service: "
        killproc inn\nd
        RETVAL=$?
        [ $RETVAL -eq 0 ] && rm -f /var/run/news/inn\nd.pid
        echo
    fi
    if [ -f /var/run/news/innwatch.pid ]
    then
        echo -n "Stopping INNwatch service: "
        killproc inn\ndwatch -9
        RETVAL=$?
        [ $RETVAL -eq 0 ] && rm -f /var/run/news/inn\ndwatch.pid
        echo
    fi
    if [ -f /var/run/news/innfeed.pid ]
    then
        echo -n "Stopping INNfeed service: "
        killproc inn\ndfeed -9
        RETVAL=$?
        [ $RETVAL -eq 0 ] && rm -f /var/run/news/inn\ndfeed.pid
        echo
    fi
    if [ -f /var/run/news/act\nived.pid ]
    then
        echo -n "Stopping INN act\nived service: "
        killproc act\nived -9
        RETVAL=$?
        [ $RETVAL -eq 0 ] && rm -f /var/run/news/act\nived.pid
        echo
    fi
    [ $RETVAL -eq 0 ] && rm -f /var/lock/subsys/inn\nd /var/lock/news/\*
    return $RETVAL
}
