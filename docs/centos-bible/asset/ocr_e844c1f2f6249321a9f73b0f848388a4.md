# Networking

The edited version of the /var/named/chroot/etc/named.conf file is as follows:

```
options {
    directory "/var/named";
    dump-file "/var/named/data/cache_dump.db";
    statistics-file "/var/named/data/named_stats.txt";
};
acl "mylan" {
    127/8; 10.0.0.0/24;
};
controls {
    inet 127.0.0.1 allow { localhost; } keys { rndckey; };
};
view "inside" {
    match-clients { "mylan"; };
    recursion yes;

    zone "." IN {
        type hint;
        file "named.ca";
    };

    zone "0.0.10.in-addr.arpa" IN {
        type master;
        file "yourlan.db";
    };

    zone "example.com" {
        type master;
        file "db.example.com.inside";
        allow-transfer { 10.0.0.2; };
    };
};

view "outside" {
    match-clients { any; };
    recursion no;

    zone "." IN {
        type hint;
        file "named.ca";
    };

    zone "example.com" {
        type master;
        file "db.example.com.outside";
        allow-transfer { 123.45.67.2; };
    };
};
```
