```
# lastly, we want to masquerade the IPs from MY_NET (eth0)
# the ppp0 interface (but we may need to stop at the s0
# the interface if dialed needs to start the ppp connection!)
ipfwam -F -a masquerade -W ppp0 -S $MY_NET/24 -D 0.0.0.0/0
ipfwam -F -a masquerade -W s0 -S $MY_NET/24 -D 0.0.0.0/0
```

```
# Incoming rules
# first, flush all rules from memory
ipfwam -I -f
#then set a default policy to deny everything
ipfwam -I -p deny
```

```
# packets coming in from my local network (can only be eth0) to-
# where else are allowed
ipfwam -I - accept W eth0 -S $MY_NET/24 -D 0.0.0.0/0
```

```
# packets preending to be my local network that come from
# the outside is denied
ipfwam -I -a deny Wpp0 -S $MY_NET/24 -D 0.0.0.0/0 -o
```

```
# packets going to the firewall, from a remote network is valid
# (i.e. if we are using the firewall machine for mail, or someone
# # VALIDLY telnetting into the firewall)
ipfwam -I -a accept Wpp0 -S 0.0.0.0/0 -D $MY_STATIC/32
```

```
# We may also use localhost for testing, so make it valid
ipfwam -I -a accept -V 127.0.0.1 -S 0.0.0.0/0 -D 0.0.0.0/0
```

```
# Outgoing rules (basically the reverse of the Incoming rules)
# as usual, flush all old rules
ipfwam -O -f
# and setup a default policy of deny
ipfwam -O -p deny
```

```
ipfwam -O -a accept W eth0 -S 0.0.0.0/0 -D $MY_NET/24
```

```
# deny someone trying to fake the masquerade
ipfwam -O -a deny Wpp0 -S 0.0.0.0/0 -D $MY_NET/24
ipfwam -O -a deny Wpp0 -S $MY_NET/24 -D 0.0.0.0/0
ipfwam -O -a deny Wpp0 -S 0.0.0.0/0 -D $MY_NET/24
```

```
# all outgoing data from the firewall is valid, and
# remember, we need to make it valid for sl0, just once
# dialed hasn't enabled ppp0 yet.
ipfwam -O -a accept -W ppp0 -S $MY_STATIC/32 -D 0.0.0.0/0
ipfwam -O -a accept -W s0 -S $MY_STATIC/32 -D 0.0.0.0/0
```

```
# and.... yeah... localhost is still valid
ipfwam -O -a accept -V 127.0.0.1 -S 0.0.0.0/0 -D 0.0.0.0/0
```
