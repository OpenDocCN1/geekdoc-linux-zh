sub col_prompt
{

loop:
my $prompt = $col_name;
if (defined ($entry_ref))
{
my $cur_val = $entry_ref->{$col_name};
$cur_val = "NULL" if !defined ($cur_val);
$prompt .= " [$cur_val]";
}
$prompt .= ": ";
print STDERR $prompt;
my $str = <STDIN>;
chomp ($str);
# perform rudimentary check on the expiration date
if ($str && $col_name eq 'expiration') # check expiration date format
{
if ($str !~ /^\d+\D\d+\D\d+\d$/)
{

goto loop;
}
}
return ($str);
}
