This example creates a three-button menu and installs it into a GtkOptionMenu widget to be displayed and controlled. Figure 21-35 shows the appearance of the widget in both collapsed and expanded states.

```c
static GtkWidget *makeWidget()
{
  GtkWidget *optionmenu;
  GtkWidget *menu;
  GtkWidget *item;

  menu = gtk_menu_new();
  item = gtk_menu_item_new_with_label("Open");
  gtk_menu_append(GTK_MENU(menu), item);
  item = gtk_menu_item_new_with_label("Close");
  gtk_menu_append(GTK_MENU(menu), item);
  item = gtk_menu_item_new_with_label("Save");
  gtk_menu_append(GTK_MENU(menu), item);

  optionmenu = gtk_option_menu_new();
  gtk_option_menu_set_menu(GTK_OPTION_MENU(optionmenu), menu);

  return(optionmenu);
}
```
