/*
* op == 4 (op */
static int module_permission(struct inode *inode, int op)
{
 /* We allow everybody to read from our module, but
 * only root (uid 0) may write to it */
 if (op == 2 || (op == 2 && current->euid == 0))
    return 0;

/* If it's anything else, access is denied */
 return -EACCES;
}
