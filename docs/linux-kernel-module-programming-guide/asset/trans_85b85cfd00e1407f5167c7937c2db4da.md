```c
/* 我们不想同时与两个进程
* 交流 */
if (Device_Open)
    return -EBUSY;

/* 如果这是一个进程，我们在这里必须
* 更加小心。

* 在进程的情况下，危险在于
* 一个进程可能检查了 Device_Open，
* 然后被调度器由另一个运行
* 此函数的进程所替换了。那么，当
* 第一个进程回到 CPU 上时，它会认为
* 设备仍然未打开。

* 然而，Linux 保证进程在内核态下
* 运行期间不会被替换。

* 在 SMP（对称多处理）的情况下，当一个 CPU
* 增加 Device_Open 时，另一个 CPU 可能正在这里，
* 就在检查之后。然而，在内核 2.0 版本
* 中这不是问题，那里有一个锁
* 保证同一时间只有一个 CPU 处于
* 内核模块中。这在性能方面很糟糕，
* 所以 2.2 版本进行了更改。
* 遗憾的是，我无法访问 SMP 机器
* 来检查它在 SMP 上是如何工作的。 */

Device_Open++;

/* 初始化消息。 */
sprintf(Message,
  "If I tell you once, I told you %d times - %s",
  counter++,
  "Hello,world\n");

/* 我们被允许执行此 sprintf
* 是因为消息的最大长度
*（假设是 32 位整数 - 带有负号最多 10 位）
* 小于 BUF_LEN，而 BUF_LEN 是 80。千万不要溢出缓冲区，
* 特别是在内核中！！！ */

Device_Open--;

/* 减少使用计数，否则一旦你
* 打开了文件，你就永远无法移除该模块。 */
MOD_DEC_USE_COUNT;

#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
return 0;
#endif
}

/* 每当已经打开设备文件的进程
* 尝试从中读取数据时，就会调用此函数。 */

```c
#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
static ssize_t device_read(struct file *file,
 char *buffer, /* 用于填充数据的缓冲区 */
 size_t length, /* 缓冲区长度 */,loff_t *offset) /* 我们在文件中的偏移量 */
#else
static int device_read(struct inode *inode,
 struct file *file,
 char *buffer, /* 用于填充
* 数据的缓冲区 */
 int length) /* 缓冲区长度 */
 *（不能写入超出长度范围！） */
#endif
{
    /* 实际写入缓冲区的字节数 */
    int bytes_read = 0;

    /* 如果我们在消息的末尾，返回 0
    *（这表示文件结束） */
    if (*Message_Ptr == 0)
        return 0;

    /* 实际上将数据放入缓冲区 */
    while (length && *Message_Ptr) {
        /* 因为缓冲区在用户数据段，
        * 而不是内核数据段，赋值将无法
        * 工作。相反，我们必须使用 put_user
        * 它将数据从内核数据段拷贝到
        * 用户数据段 */
        put_user(*Message_Ptr++, buffer++);
        length--;
        bytes_read++;
    }

#ifdef DEBUG
    printk("Read %d bytes, %d left\n", bytes_read, length);
#endif

    /* 读取函数应该返回实际插入
    * 缓冲区的字节数 */
    return bytes_read;
}
```

```c
/* 此结构体包含进程与设备
* 交互时调用的函数 */
struct file_operations fops = {
    NULL,      /* seek */
    NULL,/* release */
    NULL,      /* open */
    device_read, /* read */
    device_write, /* write */
    NULL,      /* mmap */
    NULL,      /* select */
    NULL,      /* llseek */
    NULL,/*/* loff */
    NULL,      /* release */
    NULL,/*/* ig_shared */
    NULL,      /* special */
    NULL,      /* ioctl */
    NULL,      /* aasync */
    device_open,  /* open */
    device_release, /* release */
};

/* 初始化模块 - 注册字符设备 */
int init_module()
{
    /* 注册设备 */
    Major = register_chrdev(0, DEVICE_NAME, &fops);
    if (Major < 0) {
        printk("Registration failed with %d\n", Major);
        return Major;
    }

    printk("Registration success. The major number is %d\n", Major);
    return 0;
}
```

/* 清理 - 从 /proc 取消相应的文件 */
void cleanup_module()
{
    int ret;

    /* 注销设备 */
    ret = unregister_chrdev(Major, DEVICE_NAME);

    /* 如果有错误，报告它 */
    if (ret < 0)
        printk("Error in unregister_chrdev: %d\n", ret);
}
