procedure TfrmMain.UMConnect(var Msg: TMessage);
begin
  Inc(FConnections, Msg.WParam);
  if FConnections = 1 then
    pnlConnections.Caption := SOneConnection
  else
    pnlConnections.Caption := Format(SConnections, [FConnections]);
end;
