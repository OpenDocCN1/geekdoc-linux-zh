# EXAMPLE 14.20

```
# Module Employee.pm
# The Base Class
package Employee;
use strict;
use warnings;
# Constructor method
sub init {
  my $Class = shift;
  my $info = @_;
  my $self = {
    _Name=>$info["_Name"],
    _StartDate=>$info["_StartDate"],
    _BasePay=>$info["_BasePay"],
  };
  return bless($self, $class);
}
# Some instance methods
sub get_name{
  my $self=shift;
  return $self->{_Name};
}
sub get_startdate{
  my $self=shift;
  return $self->{_StartDate};
}
sub get_basepay{
  my $self=shift;
  return $self->{_BasePay};
}
1;
```
