# **Example**

In this program, we create an empty file with one set of permissions, and then use `fchmod` to change the permissions.

```
```c
/*
 * chgmod.c - Create a file and change its mode
 */
#include <stdlib.h>
#include <stdio.h>
#include <sys/stat.h>
#include <sys/types.h>
#include <fcntl.h>

int main(void)
{
    mode_t mode = 0755;
    int fd;
    /* create the file */
    if((fd = open("empty.file", O_CREAT, 0644)) < 0) {
        perror("open");
        exit (EXIT_FAILURE);
    }
    /* ls the file */
    system("ls -l empty.file");
    /* change its permissions */
    if((fchmod(fd, mode)) < 0) {
        perror("fchmod");
        exit(EXIT_FAILURE);
    }
    /* ls it again */
    system("ls -l empty.file");
    exit(EXIT_SUCCESS);
}
```
