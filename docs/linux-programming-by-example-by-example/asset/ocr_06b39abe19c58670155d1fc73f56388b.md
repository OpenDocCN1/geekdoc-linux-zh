int main(void)
{
	DB *db;
	DBC *dbc;
	DBT key, value;
	int ret;


	ret = db_open("test.db", DB_BTREE,
			DB_CREATE, 0600, NULL, NULL, &db);
	if(ret) {
		perror("dbopen");
		exit(EXIT_FAILURE);
	}

	/* Create a cursor */
	ret = db->cursor(db, NULL, &dbc, 0);
	if(ret) {
		perror("cursor");
		exit(EXIT_FAILURE);
	}
	memset(&key, 0, sizeof(key));
	memset(&value, 0, sizeof(value));

	/* Move sequentially through the database */
	while((ret = dbc->c_get(dbc, &key, &value,
		DB_NEXT)) != DB_NOTFOUND) {
		printf("key/value is %s/%s\n", (char *)key.data,
			(char *)value.data);
	}

	/* Sync the database */
	db->sync(db, 0);

	/* Close the cursor */
	dbc->c_close(dbc);

	db->close(db, 0);
	exit(EXIT_SUCCESS);
}
