    /* Child is reader, close the write descriptor */
    close(fd[1]);
    while((len = read(fd[0], buf, BUFSZ)) > 0)

        close(fdin);

    /* Close the write descriptor */
    close(fd[1]);

waitpid(pid, NULL, 0);
