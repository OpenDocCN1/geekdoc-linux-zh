/*
 * seek.c - Using lseek
 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>

int main(void)
{
  char ftmp[] = "tmpXXXXXX";
  char buf[10];
  int i, infd, outfd;

  /* open the input file */
  if((infd = open("devices.txt", O_RDONLY)) < 0) {
    perror("open devices.txt");
    exit(EXIT_FAILURE);
  }
  /* create a temporary file for output */
  if((outfd = mkstemp(ftmp)) <0) {
    perror("mkstemp");
    exit(EXIT_FAILURE);
  }
  fprintf(stdout, "Output file is %s\n", ftmp);

  /* set the initial location in the file */
  lseek(infd, 100, SEEK_SET);
  /*
  * copy the first ten out of every 100 bytes
  * to the output file
  */
  for(i = 0; i < 10; ++i) {
    read(infd, buf, 10);
    write(outfd, buf, 10);
    lseek(infd, 90, SEEK_CUR); /* jump forward 90 bytes */
  }
}
