```
int fd;
void *map;
struct stat statbuf;

/* Open the file to map */
if (argc != 2) {
    puts ("USAGE: mkmap [file]");
    exit(EXIT_FAILURE);
}
if ((fd = open (argv[1], O_RDONLY)) < 0)
    err_quit ("open");

/* Get the file's length for mmap */
fstat (fd, &statbuf);

/* Map the input file */
if ((map = mmap (0, statbuf.st_size, PROT_READ,
        MAP_SHARED, fd, 0)) == MAP_FAILED)
    err_quit ("mmap");
printf ("map created at %p\n", &map);

/* Close and unmap the file */
close (fd);
munmap (map, statbuf.st_size);
exit (EXIT_SUCCESS);
}

void err_quit (char *msg)
{
    perror (msg);
    exit (EXIT_FAILURE);
}
```

The only output from mkmap is the mapped file's starting address. Although this program has no real functionality, it should unambiguously show how to map and unmap a file. The fstat call makes it easy to determine the file's size, which is fed to the length parameter of the mmap call.
