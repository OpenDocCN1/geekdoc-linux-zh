int count_recs(void) {
  int ret, cnt = 0;
  DBT key, value;
  DBC *dbc = NULL;

  /* Create the cursor */
  ret = db->cursor(db, NULL, &dbc);
  if(ret != 0)
    return ret;

  /* Initialize the DBTs */
  memset(&key, 0, sizeof(DBT));
  memset(&value, 0, sizeof(DBT));

  while((dbc->c_get(dbc, &key, &value, DB_NEXT)) != DB_NOTFOUND) {
    ++cnt;
  }
  return cnt;
}
