    perror("socket call failed");

}

/* Bind address to socket */
if (bind(s, (struct sockaddr *)&saddr, saddrlen) == -1) {
    perror("bind failed");
    exit(1);
}

/* If it is a TCP socket, put it in passive mode, */
/* i.e., ready to listen for incoming connection  */
if (type == SOCK_STREAM) {
    if (listen(s, qlen) == -1) {
        perror("listen failed");
        exit(1);
    }
}

/* name bound to it.                            */
return s;
