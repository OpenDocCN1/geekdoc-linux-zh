calculate_difference ()
# Difference between day indices.
{
let "diff = $1 - $2"
# Global variable.

abs ()
# Absolute value
# Uses global "value" variable.
# If negative
  if [ "$1" -lt 0 ]
  then
    let value = 0 $1
    # change sign,
  else
    let value = "$1"
    # leave it alone.
  fi
}

if [ $# -ne "$ARGS" ]
# Require two command line params.
then
  Param_Error
fi

Parse_Date $1
check_date $day $month $year
# See if valid date.

strip_leading_zero $day
# Remove any leading zeroes
day=$n# on any day and/or month.
strip_leading_zero $month
month=$?

day_index $day $month $year
datel=$?

abs $datel
# Make sure it's positive
datel=$value
# by getting absolute value.

Parse_Date $2
check_date $day $month $year

strip_leading_zero $day
day=$?
strip_leading_zero $month
month=$?

day_index $day $month $year
date2=$

abs $date2
# Make sure it's positive.
date2=$value

calculate_difference $datel $date2

abs $diff
# Make sure it's positive.
diff=$value

echo $diff

exit 0
# Compare this script with the implementation of Gauss' Formula in C at
# http://buschencrew.hypermart.net/software/datedif
