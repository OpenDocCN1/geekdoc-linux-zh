# 示例 10–23. 使用 case

```bash
#!/bin/bash

echo "按下一个键，然后回车。"
read Keypress

case "$Keypress" in
  [a-z]) echo "小写字母";
  [A-Z]) echo "大写字母";
  [0-9]) echo "数字";
  *) echo "标点、空格或其他";
# 允许在 [方括号] 中使用字符范围
esac

# 练习：

# 目前脚本目前是，它接受单个按键，然后退出。
# 修改脚本使其接受连续输入，
# 报告每次按键的输入，并仅在按下 "X" 时才退出。
# 提示：将所有内容封装在 "while" 循环中。
exit 0
```

# 示例 10-26. 字符串匹配

```bash
#!/bin/bash
# 简单的字符串匹配

match_string ()
{
  MATCH=0
  NOMATCH=1
  # 匹配逻辑
  case "$1" in
    apple)
      MATCH=1
      ;;
    orange)
      MATCH=1
      ;;
  esac

  if [ $MATCH -eq 1 ]; then
    echo "匹配 $1"
  else
    echo "不匹配 $1"
  fi
}

match_string "apple"
match_string "orange"
```

 示例 10-28. 使用 select 创建菜单

```bash
#!/bin/bash
# 使用 select 创建菜单

PS3="请选择一个: "
select fruit in "apple" "banana" "cherry" "quit"; do
  case $fruit in
    apple) echo "你选择了苹果" ;;
    banana) echo "你选择了香蕉" ;;
    cherry) echo "你选择了樱桃" ;;
    quit) break ;;
    *) echo "无效选项 $REPLY" ;;
  esac
done
```

# 示例 10-30. 使用 select 处理文件数组

```bash
#!/bin/bash
# 使用 select 处理文件数组

files=("config.txt" "data.csv" "log.log")
PS3="请选择一个文件: "
select file in "${files[@]}" "quit"; do
  echo "正在处理 $file"
  [[ "$file" == "quit" ]] && break
done
```
