nIsAlive () # Test whether cell is alive.
{
    # Takes array, cell number, state of cell as arguments.
    GetCount "$1" $2 # Get alive cell count in neighborhood.
    local nhbd=$

    if [ "$nhbd" -eq "$BIRTH" ] # Alive in any case.
then
        return $ALIVE
    fi

    if [ "$3" = "." -a "$nhbd" -eq "$SURVIVE" ]
then
        return $ALIVE # Alive only if previously alive.
    fi

    return $DEAD # Default.
}

## Example A-1. fptget. Downloading files via ftp

```bash
# /bin/sh
# $Id: fptget,v1.2 91/05/07 21:15:43 morae
# Script to perform batch fptget. Essentially converts a list of
# line arguments into input to fptget
# Simple and quick: acts as comparison to ftplist
# -h specifies the remote host (default prep.ai.mit.edu)
# -d specifies the remote directory to cd - you can produce a sequence
# -o options - they will be cded in. If the paths are relative,
# make sure you get the sequence right. Be careful with relative paths -
# around too many symbols.
# (default is the ftp login directory)
# -v turns on the verbose option of ftp, and shows all responses from the
# ftp server
# -f remotefile[:filename] gets the remote file into localfile
# -m pattern does an mget with the specified pattern. Remember to quote
# characters.
# shell characters
# -c does a cd to the specified directory
# For example,
fptget -h expo.cls.mit -cd contrib -f xplaces:xplaces.sh \
# --../pub/R3/fixes -c ~ -m 'fix*'
# will xplaces.sh from -ftp on expo.cls.mit.edu, and put it in
# xplaces.sh in the working directory, and get all fixes from
# -ftp/pub/R3/fixes and put them in ~/fixes directory.
# Obviously, the sequence of the options is important, since the equivalent
# commands are executed by ftp in corresponding order
```
