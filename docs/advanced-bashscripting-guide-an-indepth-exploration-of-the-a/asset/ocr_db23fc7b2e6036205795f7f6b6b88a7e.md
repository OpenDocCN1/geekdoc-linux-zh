abs ()
{
  E_ARGERR=-999999                             # Absolute value.
                              # Caution: Max return value = 256.

  if [ -z "$1" ]                               # Need arg passed.
  then
    return $E_ARGERR                        # Obvious error value returned.
  fi

  if [ "$1" -ge 0 ]                            # If non-negative,
  then
    absval=$1                                # stays as-is.
  else
    let absval=$(( 0 - $1 ))               # Otherwise,
                                              # change sign.
  fi

  return $absval
}
