abs ()
{
  E_ARGERR=-999999                             # Absolute value.
                              # Caution: Max return value = 256.

  if [ -z "$1" ]                               # Need arg passed.
  then
    return $E_ARGERR                        # Obvious error value returned.
  fi

  if [ "$1" -ge 0 ]                            # If non-negative,
  then
    absval=$1                                # stays as-is.
  else
    let absval=$(( 0 - $1 ))               # Otherwise,
                               # change sign.
  fi

  return $absval
}

Using the $? status variable, a script may test if a parameter contains only digits, so it can be treated as an integer.

```bash
#!/bin/bash

SUCCESS=0
E_BADINPUT=65

test "$1" -ne -o "$1" -eq 0 2>/null
# An integer is either equal to 0 or not equal to 0.
# 2>/null suppresses error message.

if [ "$1" -ne "$SUCCESS" ]; then
  echo "Usage: $0 <input>"
  exit $E_BADINPUT
fi
```

## 34.6. Return value

```bash
#!/bin/bash
# multiplication.sh

multiply () { # Multiplies params passed.
  local product=1
  until [ -n "$1" ]; do  # Until arguments passed.
    let product *= $1
    shift
  done
  echo "$product"  # Will echo to stdout,
  # since this will be assigned to a variable.
}

val1=$(multiply 5 3)
echo "val1 = $val1"

val2=$(multiply 2 5)
echo "val2 = $val2"

val3=$(multiply 8 7 5)
echo "val3 = $val3"
```

The same technique also works for alphanumeric strings. This means that a function can "return" a non-numeric value.

```bash
capitalize_ichar () # Capitalizes initial character
{
  string="$@"                   # Accepts multiple arguments.

  firstchar=${string:0}       # First character.
  string1=${string:1}           # Rest of string(s).

  FirstChar=$(echo "$firstchar" | tr a-z A-Z) # Capitalize first character.

  echo "$FirstChar$string1"    # Output to stdout.
}

newstring=$(capitalize_ichar "each sentence should start with a letter.")
echo "$newstring" # Each sentence should start with a capital letter.
```

It is possible for a function to "return" multiple values with this method.

## 34.7. Even more value trickery

```bash
#!/bin/bash
# sum-product.sh
# A function may "return" more than one value.

sum_and_product () { # Calculates both sum and product of passed args.
  echo "$((1 + $2)) $((1 * $2))"  # Echos to stdout each calculated value, separated by spaces.
}

echo "Enter first number"
read first

echo "Enter second number"
read second

retval=$(sum_and_product $first $second) # Assigns output of function.
sum=$(echo "$retval" | awk '{print $1}')      # Assigns first field.
product=$(echo "$retval" | awk '{print $2}')  # Assigns second field.

echo "$first + $second = $sum"
echo "$first * $second = $product"
```

- *Next in our bag of trick are techniques for passing an array to a function, then "returning" an array back to the main body of the script.*

## 34.8. /bin/bash and returning arrays

Passing an array involves loading the space-separated elements of the array into a variable with command substitution. Getting an array as the "return value" from a function uses the previously mentioned strategy of echoing the array in the function, invoking command substitution and the ( ... ) operator to assign it to an array.

```bash
#!/bin/bash
# array-function.sh: Passing an array to a function and
#   "returning" an array from a function

Pass_Array () {
    local passed_array=() # Local variable.
    passed_array=$(echo "$1")
    echo "${passed_array[@]}"
    # List all the elements of the new array
    # declared and set within the function.
}
```

```bash
original_array=(element1 element2 element3 element4 element5)
echo "original_array = ${original_array[@]}"
# List all elements of original array.

# This is the trick that permits passing an array to a function.
argument=$(echo "${original_array[@]}")
# Pack a variable with all the space-separated elements of the original array.
# Note that attempting to pass the array itself will not work.

# This is the trick that allows an array as a "return value".
returned_array=$(Pass_Array "$argument")
# Assign 'echoed' output of function to array variable.
echo "returned_array = ${returned_array[@]}"
```

```bash
# Try it again,
# attempting to access (list) the array from outside the function.
Pass_Array "$argument"
# The function itself lists the array, but...
# accessing the array from outside the function is forbidden.
echo "Passed array (within function) = ${passed_array[@]}"
# NULL VALUE since this is a variable local to the function.
```
