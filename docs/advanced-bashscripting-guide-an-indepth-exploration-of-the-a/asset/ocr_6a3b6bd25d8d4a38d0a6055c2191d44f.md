```
:docstring strcspn
# Usage: strcspn s1 $s2
#
# Strcsp returns the length of the maximum initial segment of string s1,
# which consists entirely of characters from string s2.
#:enddocstring

###;autoload
function strcspn ()
{
    # Unset IFS allows whitespace to be handled as normal chars.
    local IFS=
    local result="${1%%[!${2}]*"

    echo "$result"
}
```
