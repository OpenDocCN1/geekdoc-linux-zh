```bash
function fastprompt() {
  unset PROMPT_COMMAND
  case $TERM in
    *term | rxvt)
      PS1="[#h] \W > \u033[0;\u007[h] \w\u007";
    )
    *)
      PS1="[#h] \W > "
  esac
}
```

```bash
function powerprompt() {
  _powerprompt() {
    LOAD=$($uptime | sed -e "s/.*([^^*]*).\1/" -e "s //g")
    TIME=$(date +%H:%M)
  }
  PROMPT_COMMAND=_powerprompt
  case $TERM in
    *term | rxvt)
      PS1="$(cyan)[\TIME]NC\u033[0;\u007[h] \w\u007";
    linux)
      PS1="$(cyan)[\TIME]NC\u033[0;\u007[h] \w\u007";
    *)
      PS1="$(cyan)[\TIME]NC\u033[0;\u007[h] \w\u007";
  esac
}
powerprompt # this is the default - might be slow
# If slow, use fastprompt instead.
```

```bash
function man () {
  man $(basename $PWD) |tr -d .[:digit:]
  man -a "*"

function ll() { ls -1 "$*" | grep "^d"; ls -l "$@" 2>&1 | grep -v "^d|total"; }

function emacs() { command emacs -private 2>&1 && disown }
function te() { # wrapper around emacs/gnuuser
{
  if [ "$(gnuclient -eval t 2>&1)" == "t" ]; then
    gnuclient -q "$@"
  else
    emacs "$@" &
  fi
```

## File & strings related functions:

```bash
function ff() { find -name '*$1*' ; } # find a file
function fe() { find -name '*$1*' -exec $2 \; \; } # find a file and run $2 on it
function fstr() { find a string in a set of files
if [ "@" -gt 2 ]; then
  echo "Usage: fsstr \"pattern\" [files]"
  return;
fi
SMS=$(tput rmso)
RMS=$(tput smso)
find . -type f -name "${2:-*}" -print | xargs grep -sin "$1" | \
sed "s/$1/$SMS$RMS/g"
}

function cutall() # cut last n lines in files, 10 by default
{
lines=${2:-10}
sed -n -e "1,\${lines}p"
}

function lowercase() { # move filenames to lowercase
{
  for file; do
    filename=${file##*/}
    case "$filename" in
      *)
        dirname="$filename"
        if [ "$file" != "$dirname" ]; then
          echo "lowercase: $file"
        fi
    esac
  done
}

function swap() { # swap around
{
  local TMPFILE=$1
  mv $1 $2
  mv $2 $TMPFILE
}

## Process/system related functions:
function my_ps() { ps -u -o cpu,$mem,command; }
function pp() { ps f; }
function kills() { # function is roughly the same as 'kill' but has no equivalent
local pid=$1
if [ "$#" -lt 1 ] || [#" -gt 2 ]; then
return;
fi
kill $pid
}

function my_ip() { # get IP addresses

## Misc utilities:
function repeat() { # repeat n times command
local i max
max=$1; shift;
for ((i=1; i<=max; i++)); do
echo "$2"
done
}

function ask() {
echo -n "$@" [y/n]
read ans
case "$ans" in
  y*|Y*) return 0 ;;
*) return 1 ;;
esac
```
