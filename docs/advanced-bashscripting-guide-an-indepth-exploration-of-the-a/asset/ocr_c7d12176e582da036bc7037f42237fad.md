count_lines_in_etc_passwd() {
  [[ -r /etc/passwd ]] && REPLY=$(echo $(wc -l < /etc/passwd))
  # If /etc/passwd is readable, set REPLY to line count.
  # Returns a parameter value and status information.
}

count_lines_in_etc_passwd
