# Example 12–32. Another way to invoke bc

```
#!/bin/bash
# Invoking 'bc' using command substitution
# in combination with a 'here document'

var1=$(bc << EOF
18.33 * 19.78
EOF)
echo $var1      # 362.56

# $(...) notation also works.
v1=23.53
v2=17.881
v3=83.501
v4=171.63

var2=$(bc << EOF
scale = 4
a = ($v1 + $v2)
b = ($v3 + $v4)
a * b + 15.35
EOF)
echo $var2      # 593847.8452

var3=$(bc -1 << EOF
scale = 9
s ( 1.7 )
EOF)
# Returns the sine of the 1.7 radians.
# The "-1" option calls the 'bc' math library.
echo $var3      # .991664810
```
