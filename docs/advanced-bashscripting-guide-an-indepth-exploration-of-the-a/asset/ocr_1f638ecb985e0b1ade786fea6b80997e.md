ParseDate () {
  # Parse date from command line params.
  month=${1%%/*}
  # Day and month.
  day=${1#*/}
  let year = `basename $1`  # Not a filename, but works just the same.
}

check_date () {
  # Checks for invalid date($s) passed.
  [ "$day" -gt "$DIM" ] || [ "$month" -gt "$MIY" ] || [ "$year" -lt "$REFURY" ] && Param_Error "Exit script on bad value($s)"
  # Uses "or -and-list"

}

strip_leading_zero () {
  # Better to strip possible leading zero(s)
  # from day/and or month
  val=${1#0}
  # since otherwise Bash will interpret them
  return $val  # as octal values (POSIX.2, sect.2.9.2.1).
}

```
day_index () {
  # Gauss' Formula:
  # Days from Jan. 3, 1600 to date passed as param.
  day=$1
  month=$2
  year=$3

  let "month = $month - 2"
  if [ "$month" -le 0 ]; then
    let "month += 12"
    let "year -= 1"
  fi

  let "year -= $REFURY"
  let "index = $year + $year / CENTURY"
}
```
