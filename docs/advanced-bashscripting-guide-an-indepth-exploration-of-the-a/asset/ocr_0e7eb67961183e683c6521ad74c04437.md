# Advanced Bash–Scripting Guide

```
#!/bin/bash
# timed-input.sh

# TMEOUT=3 useless in a script
TIMELIMIT=3 # Three seconds in this instance, may be set to different value.

PrintAnswer()
{
    if [ "$answer" = $TIMEOUT ]
    then
        echo $answer
    else
        echo # Don't want to mix up the instances
        echo "Your favorite veggie is $answer"
        kill $! # Kills longer needed TimerOn function running in background. $! is PID of last job running in background.
}

TimerOn()
{
    sleep $TIMELIMIT && kill -s 14 $$ &
    # Waits 3 seconds, then sends sigalarm to script.
}

Intl4Vector()
{
    answer="TIMEOUT"
    PrintAnswer
    exit 14
}

trap Intl4Vector 14 # Timer interrupt (14) subverted for our purposes.

echo "What is your favorite vegetable "
TimerOn
read answer
PrintAnswer
```
