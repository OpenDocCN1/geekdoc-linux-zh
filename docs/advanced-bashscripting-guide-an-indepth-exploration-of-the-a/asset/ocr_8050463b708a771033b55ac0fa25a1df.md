```
function my_ip() { # get IP addresses
{
MY_IP=$(/sbin/ifconfig p0 | awk '/inet/ (print $2)' | sed -e s/addr://;)
MY_IP=$(/sbin/ifconfig p0 | awk '/^T-P/ (print $3)' | sed -e s/P-P-T-//;)
}
```

function repeat() {
{
# repeat n times command
local i max
max=$1; shift;
for ((i=1; i=max; i++)); do # --> C-like syntax
echo "$8"
done
}
