# Advanced Bash–Scripting Guide

# Example 28–1. Finding the process associated with a PID

```
#!/bin/bash
# pid-identifier.sh: Gives complete path name to process associated with pid

ARGNO=$# Number of arguments the script expects.
E_WRONGARGS=65
E_BADPID=66
E_NOSUCHPROCESS=67
E_NOPERMISSION=68
PROFILE=exit

if [ $ -ne $ARGNO ];
then
    echo "Usage: basename $0 'PID-number" >&2 # Error message to stderr.
    exit $E_WRONGARGS
fi

pidno=$(ps ax | awk '{ print $1 }' | grep $1)
# Checks for pid in "ps" listing, field #1.
# Then makes it the actual process, not the process invoked by this script.
# The last "grep $1" filters out this process.
if [ -z "$pidno" ];
then
    # If, after all the filtering, the result is a zero-length string,
    echo "No such process running."
    # # no running process corresponds to the pid given.
    exit $E_NOSUCHPROCESS
fi

Alternatively:
if ! $1 > /dev/null 2>&1;
then
    # no running process corresponds to the pid given.
    echo "No such process running."
    exit $E_NOSUCHPROCESS
fi

# To simplify the entire process, use "pidof"

if [ -r "/proc/$1/$PROFILE" ];\ # Check for read permission.
then
    echo "Process $1 running, ..."
    echo "Can't get read permission on /proc/$1/$PROFILE."
    exit $E_NOPERMISSION # Ordinary user can't access some files in /proc.
# The last two lines may be replaced by:
if ! kill -0 $1 > /dev/null 2>&1; # '0' is not a signal, but
    # this will test whether it is possible
    # # to send a signal to the process.
then
    echo "PID does not exist or you're not its owner" >&2
    exit $E_BADPID
fi

exe_file=$(ls -1 /proc/$1 | grep "exe" | awk '{ print $11 }')
# Or 
_file=$(ls -l /proc/$1/exe | awk '{print $11 }')

# /proc/pid-number/exe is a symbolic link
# to the complete path name of invoking the process.
```
