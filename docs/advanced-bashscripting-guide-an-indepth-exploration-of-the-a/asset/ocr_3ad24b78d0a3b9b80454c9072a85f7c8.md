# Example 10–10. Linking the symbolic links in a directory

```
#!/bin/bash
# symlinks.sh: Lists symbolic links in a directory.

ARGS=1 # Expect one command-line argument.
if [ $ -ne "$ARGS" ] # If not 1 arg...
then
  directory='pwd' # current working directory
else
  directory=$1
fi

echo "symbolic links in directory \$directory\"

for file in $( find $directory -type 1 ) # -type 1 = symbolic links
do
  echo "$file"
  sort # Otherwise file list is unsorted.
done

# As Dominik 'Aene' Schnitzer points out,
#+ failing to quote $ ( find $directory -type l1 )
#+ will choke on filenames with embedded whitespace.
```

```
exit 0
```
