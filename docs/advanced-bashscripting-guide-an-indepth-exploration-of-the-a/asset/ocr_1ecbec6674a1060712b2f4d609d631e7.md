```
# Shell prompt

function fastprompt() {
  unset PROMPT_COMMAND
  case $TERM in
    *term | rxvt)
      PS1="[#h] \W > \u033[0;\u007[h] \w\u007";"
    )
    PS1="[#h] \W > "
  esac
}
```

```
function powerprompt() {
  _powerprompt() {
    LOAD=$($uptime | sed -e "s/.*:\([^^*]*).*/\1/" -e "s/ //g")n    TIME=$(date +%H:%M)
  }
  PROMPT_COMMAND=_powerprompt
  case $TERM in
    *term | rxvt)
      PS1="$(cyan)[\$TIME $LOAD]NC\n[# #] \W > \u033[0;\h] \w\u007";"
    linux)
      PS1="$(cyan)[\$TIME $LOAD]NC\n[# #] \w > ";"
    *
      PS1="[\$TIME - $LOAD]\n[# #] \w > ";"
  esac
}
```
