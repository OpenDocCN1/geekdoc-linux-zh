```bash
nIsAlive () # 测试单元格是否存活。
{
    # 接收数组、单元格编号、单元状态作为参数。
    GetCount "$1" $2 # 获取邻域内的存活单元数量。
    local nhbd=$

    if [ "$nhbd" -eq "$BIRTH" ] # 在任何情况下存活。
then
        return $ALIVE
    fi

    if [ "$3" = "." -a "$nhbd" -eq "$SURVIVE" ]
then
        return $ALIVE # 仅在之前存活时存活。
    fi

    return $DEAD # 默认。
}

Count () # 统计通过单元格邻域内的存活单元。
{
    # 两个参数
    # $1 存放数组的变量
    # $2) 单元格编号
    local cell_number=$2
    local array
    local top
    local center
    local bottom
    local r
    local row
    local i
    local t_top
    local t_cen
    local t_bot
    local count=0
    local ROW_NHBD=3

    array=( `echo "$1"` )

    let $top = $cell_number - $COLS - 1 # 设置单元编号。
    let $center = $cell_number - 1
    let $bottom = $cell_number + $COLS - 1
    let $r = $cell_number / $ROWS
```

for ((i=0; i<$ROW_NHBD; i++)) {

  let t_top = $top + $i

  IsValid $st_row # 有效单元格位置？
  if [ $? -eq "TRUE" ]
then
    if [ ${array[$st_cen]} = "$ALIVE1" ] # 存活吗？
    then
      let count += 1 # 是？
      增加计数器。
    fi
  fi
  fi

  let row = $r - 1 # 统计顶行。
  IsValid $st_top $row
  if [ $? -eq "TRUE" ]
then
    if [ ${array[$st_top]} = "$ALIVE1" ]
    then
      let count += 1
    fi
  fi

  let row = $r + 1 # 统计底行。
  IsValid $st_bot $row
  if [ $? -eq "TRUE" ]
then
    if [ ${array[$st_bot]} = "$ALIVE1" ]
    then
      let count += 1
    fi
  fi

if [ ${array[$cell_number]} = "$ALIVE1" ]
then
  let count -- 1 # 确保测试对象本身
  # 不被计数。
fi

return $count
}

next_gen () # 更新代数数组。
local array
local i=0
array=( "$1" ) # 将传递参数转换为数组。
while [ $i -lt ${#array[@]} ]
do
  IsAlive "$1" ${array[$i]} # 存活吗？
  if [ $? -eq "TRUE" ]
then
    # 如果存活，则
    array[$i]=. # 用点表示单元格。
  else
    array[$i]="" # 否则用下划线
    #（稍后会被空格）
  fi
  let i += 1
done

# generation += 1 # 增加代数计数

avar="${array[@]}" # 将数组转换回字符串。

echo "Generation --$generation"
