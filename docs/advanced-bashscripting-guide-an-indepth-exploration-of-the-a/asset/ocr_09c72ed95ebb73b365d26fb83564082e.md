display ()
{
alive=0    # How many cells "alive".
             # Initially zero.

  declare -a arr
  arr=(echo "$1")    # Convert passed arg to array.

  element_count=${arr[@]}

  local i
  local rowcheck

  for (i=0; i<${element_count}; i++))
  do
    # Insert newline at end of each row
    let rowcheck = $i % ROWS
    if [ "$rowcheck" -eq 0 ]
    then
      echo
      echo -n " "    # Indent.
    fi


  cell=${arr[i]}

  if [ "$cell" = . ]
  then
    let alive = 1
  fi

  echo -n "$cell" | sed -e 's/_/ /g'
  # Print array and change underscores to spaces.
  done

  return

n

{
  if [ -z "$1" -o -z "$2" ]    # Mandatory arguments missing?
  then
    return FALSE
  fi

  local row

  local upper_limit
  local left

  if [ "$1" -lt "$lower_limit" -o "$1" -gt "$upper_limit" ]
  then
    return FALSE    # Out of array bounds.
  fi
n}
