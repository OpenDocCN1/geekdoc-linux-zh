```bash
for i in [list] missing...
  do ==> so operates on command line arg(s)
  case "$i" in
    0*) ibase=2; # binary
    0x*[a-f]*[A-F]*) ibase=16; # hexadecimal
    ibase=8; # octal
    [1-9]*) ibase=10; # decimal
    *)
      echo "illegal number $i - ignored"
      continue;
  esac

  # Remove prefix, convert hex digits to uppercase (bc needs this)
  number=$(echo "$i" | sed -e 's:0x:[0-9a-fA-F]//' | tr '[a-f]' '[A-F]')
  # Uses \ as separator, rather than ""
  # Convert number to decimal
  dec=$(echo "ibase=$ibase;$number" | bc) # calculator utility
  case "$dec" in
    [0-9]*) # number ok
      continue;
    *) ) # error: ignore
  esac

  # Print all conversions in one line.
  # == 'here document' command list feeds to 'bc'
  bc <<EOF
    obase=16; "hex": $dec
    obase=10; "dec": $dec
    obase=8; "oct": $dec
    obase=2; "bin": $dec
EOF | sed -e 's/"//g'
done

while [ $# -gt 0 ];
  do
    case "$1" in
      -\) shift; break;
      -h) Usage; ==> Help message.
         Usage;
         break;
      *) ) # first number
    esac
    ==> More error checking for illegal input would be useful.
  shift
done

if [ $# -gt 0 ];
  then
    PrintBases "$@" # read from stdin
  else
    while read line
      do
        PrintBases $line
      done
  fi
```

## Example 12–32. Another way to invoke bc

```bash
#!/bin/bash
# Invoking 'bc' using command substitution
# in combination with a 'here document'

var1=$(bc << EOF
18.33 * 19.78
EOF)
echo $var1      # 362.56

# $(...) notation also works.
v1=23.53
v2=17.881
v3=83.501
v4=171.63

var2=$(bc << EOF
scale = 4
a = ($v1 + $v2)
b = ($v3 + $v4)
a * b + 15.35
EOF)
echo $var2      # 593847.8452

var3=$(bc -1 << EOF
scale = 9
s ( 1.7 )
EOF)
# Returns the sine of the 1.7 radians.
# The "-1" option calls the 'bc' math library.
echo $var3      # .991664810
```

## Now, try it in a function...
```bash
hyp # Declare global variable.
hypotenuse () # Calculate hypotenuse of a right triangle.
{
hyp=$(bc -1 << EOF
scale = 9
sqrt ( $1 * $1 + $2 * $2 )
EOF
# Unfortunately, can't return floating point values from a Bash function.
)
hypotenuse 3.68 7.31
echo "hypotenuse = $hyp"      # 8.18403934

exit 0
```

Another way of doing floating point math in a script is awk's built-in math functions in a shell wrapper.

## Example 12–33. Calculating the hypotenuse of a triangle

```bash
#!/bin/bash
# hypotenuse.sh: Returns the "hypotenuse" of a right triangle.
# square root of squares of the "legs"

ARGS=2      # Script needs sides of triangle passed.
E_BADARGS=65  # Wrong number of arguments.

if [ $# -ne "$ARGS" ]; then  # Test number of arguments to script.
then
    echo "Usage: $0 side_1 side_2"
    exit $E_BADARGS
fi

AWKSCRIPT='printf("%3.3f\n", sqrt($1 + $2*$2))' 
  # parameters passed to awk
  command(s)

echo -n "Hypotenuse of $1 and $2 = "
echo $1 $2 | awk "$AWKSCRIPT"

exit 0
```
