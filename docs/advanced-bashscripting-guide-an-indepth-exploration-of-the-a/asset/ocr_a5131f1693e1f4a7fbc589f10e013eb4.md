rotate () # Rotate the array "balance" on its lower lefthand corner.
local row
local column
for (( row = Rows; row > -Rows; row-- ))  # Step through the array backwards
do
for (( column = 0; column < Columns; column++ ))
do
if [ "$row" -ge 0 ]
then
let t1 = $column $row
let t2 = $column
else
let t1 = $column
let t2 = $column + $row
fi

filter $t1 $t2  # Filter out negative array indices.
done
echo; echo
done
