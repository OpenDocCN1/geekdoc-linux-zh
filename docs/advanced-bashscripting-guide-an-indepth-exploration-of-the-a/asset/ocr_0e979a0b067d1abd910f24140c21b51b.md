```
if [ -z "$1" ]
then
  Filename=names.data      # Default, if no filename specified.
else
  Filename=$1
fi
```

```
exec 3<&0      # Save stdin to file descriptor 3.
exec 0<"$Filename" # Redirect standard input.
```
