#:doc strcat:
# Usage: strcat s1 s2

# Strcat appends the value of variable s2 to variable s1.

# Example:
# a="foo"
# b="bar"
# strcat a b
# echo $a
# > foobar

#end docstring
###;;autoload --> Autoloading of function commented out.
function strcat () {
    local s1_val s2_val

s1_val=${1} # indirect variable expansion
s2_val=${2}
eval "$s1='${s1_val}\${s2_val}'*"
    # => eval s1='${s1_val}\${s2_val}' avoids problems,
    # => if one of the variables contains a single quote.
}

#:doc strcat
# Usage: strcat s1 s2 n

# Line struct, strcat appends a maximum of n characters from the value
# of variable s2. It copies if the value of variable s2 is shorter
# than n characters. Echoes result on stdout.

# Example:
# a=foo
# b=barbaz
# strcat a b 3
# echo $a
# > foobar

#end docstring:
###;;autoload
function strcat {
    local s1="$1"
    local s2="$2"
    local -i n="$3"
    local s1_val s2_val

s1_val=\${!s1} --> indirect variable expansion
}
