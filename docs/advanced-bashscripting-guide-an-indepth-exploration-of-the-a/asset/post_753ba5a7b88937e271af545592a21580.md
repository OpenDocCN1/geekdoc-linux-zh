```bash
#!/bin/bash

ENDLESS_LOOP=1
INTERVAL=1

echo "Hit Control-C to exit this script."
echo

while [ "$ENDLESS_LOOP" ]
do
    if [ "$SECONDS" -eq 1 ];
    then
        units=second
    else
        units=seconds
    fi
    echo "This script has been running $SECONDS $units"
    sleep $INTERVAL
done

exit 0
```

$TMOUT

The $TMOUT environmental variable is set to a non–zero value time, then the shell prompt will time out after time seconds. This will cause a logout.

While it would be nice to speculate on the uses of this internal variable for timed input, for in combination with read. $TMOUT does not work in that context and is virtually useless for shell scripting. (Reportedly the ksh version of a timed read does work.)

```bash
#!/bin/bash
# timed-input.sh

# TMEOUT=3 useless in a script
TIMELIMIT=3 # Three seconds in this instance, may be set to different value.

PrintAnswer()
{
    if [ "$answer" = $TIMEOUT ]
    then
        echo $answer
    else
        echo # Don't want to mix up the instances
        echo "Your favorite veggie is $answer"
        kill $! # Kills longer needed TimerOn function running in background. $! is PID of last job running in background.
}

TimerOn()
{
    sleep $TIMELIMIT && kill -s 14 $$ &
    # Waits 3 seconds, then sends sigalarm to script.
}

Intl4Vector()
{
    answer="TIMEOUT"
    PrintAnswer
    exit 14
}

trap Intl4Vector 14 # Timer interrupt (14) subverted for our purposes.

echo "What is your favorite vegetable "
TimerOn
read answer
PrintAnswer
```
