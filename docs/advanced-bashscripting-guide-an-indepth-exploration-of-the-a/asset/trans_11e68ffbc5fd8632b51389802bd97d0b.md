```bash
abs ()
{
  E_ARGERR=-999999                             # 绝对值。
                              # 注意：最大返回值 = 256。

  if [ -z "$1" ]                               # 需要传递参数。
  then
    return $E_ARGERR                        # 返回明显的错误值。
  fi

  if [ "$1" -ge 0 ]                            # 如果是非负数
  then
    absval=$1                                # 保持不变。
  else
    let absval=$(( 0 - $1 ))               # 否则，
                               # 改变符号。
  fi

  return $absval
}
```

使用 $? 状态变量，脚本可以测试参数是否仅包含数字，从而可以将其视为整数。

```bash
#!/bin/bash

SUCCESS=0
E_BADINPUT=65

test "$1" -ne -o "$1" -eq 0 2>/null
# 整数要么等于 0，要么不等于 0。
# 2>/null 抑制错误消息。

if [ "$1" -ne "$SUCCESS" ]; then
  echo "Usage: $0 <input>"
  exit $E_BADINPUT
fi
```

任何变量（而不只是命令行）都可以用任何 0 范围的返回值进行测试，这是一个严重的限制。全局变量的变通方法通常有问题。函数向主体返回值的另一种方法是让函数写入标准输出 (stdout)，并将其给变量。

## 34.6. 返回值

```bash
#!/bin/bash
# 乘法

multiply () { # 乘以参数。
  local product=1
  until [ -n "$1" ]; do # 直到参数为空。
    let product *= $1
    shift # 移动位。
  done
  echo "$product" # 输出结果。
}

result=$(multiply 2 3 4) # 将函数输出赋值给变量。
echo "Result: $result"
```

## 34.7. 多个返回值

```bash
#!/bin/bash
# 多个返回值

get_stats () { # 获取统计。
  local sum=$1
  local count=$2
  local avg=$(( sum / count ))
  echo "$sum"   # 输出总和。
  echo "$count" # 输出计数。
  echo "$avg"   # 输出平均值。
}

# 使用 read 接收多个返回值。
read s c a < <(get_stats 100 5)
echo "Sum: $s, Count: $c, Avg: $a"
```

## 34.8. 数组返回

```bash
#!/bin/bash
# 数组返回

get_array () { # 获取数组。
  local arr=("$@")
  echo "${arr[@]}" # 输出数组元素。
}

my_array=($(get_array "apple" "banana" "cherry"))
echo "Array: ${my_array[@]}"
```
