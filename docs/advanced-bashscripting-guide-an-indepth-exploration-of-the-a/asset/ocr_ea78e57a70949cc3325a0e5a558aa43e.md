```
usage() {
echo "Usage: $0 -e 'basename $0' C-program-file" >&2
exit 1
}
WEIRD=$(echo -e '\x377\') # or WEIRD='\x377\')
[[ $# -eq 1 ]] && usage
case "file $1" in inn*C program text*" sed -e "%s/\*\*/${WEIRD}%g;s/\*/${WEIRD}%g" "$1" \
t tr '\x377\n' '\x377' \
t sed -ne '\n' \
t tr -d '\n' tr '\x377' '\n';\n*) usage);
esac
```
