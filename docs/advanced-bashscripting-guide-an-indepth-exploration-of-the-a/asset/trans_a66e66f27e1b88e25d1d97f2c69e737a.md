#!/bin/bash
# collatz.sh
----------------------------------------------------------------------
# 臭名的“hallstone”或 Collatz 序列。
----------------------------------------------------------------------
# 1) 从命令行获取整数“种子”。
# 2) NUMBER <-- 种子
# 3) 打印 NUMBER：
# 4) 如果 NUMBER 是偶数，除以 2；或者
# 5) 如果是奇数，乘以 3 并加 1。
# 6) NUMBER <-- 结果
# 7) 回到 3（循环指定的迭代次数）。

# 该理论是：每一个序列，
# 无论初始值有多大，
# 最终都会稳定在循环“4,2,1...”周期中，

# 这是“迭代”的一个实例，
# 这种操作将输出反馈回输入中。

display ()

alive=0    # 活跃单元格数量。
# 初始为零。

  declare -a arr
  arr=(echo "$1")    # 将传递的参数转换为数组。

  element_count=${arr[@]}

  local i
  local rowcheck

  for (i=0; i<element_count; i++))
  do
    # 每行插入换行
    let rowcheck = $i % ROWS
    if [ "$rowcheck" -eq 0 ];
    then
      echo
      echo -n " "    # 缩进。
    fi


  cell=${arr[i]}

  if [ "$cell" = . ]
  then
    let alive = 1
  fi

  echo -n "$cell" | sed -e 's/_/ /g'
  # 数组并将下划线替换为空格。
  done

  return

}
