```
echo

if [ TRUE ]
then
  alias rr="ls -l"
  echo "Trying aliased \"rr\" if/then statement"
  rr /usr/X1LR6/bin/mk* # Error message results!
  # Aliases not expanded compound statements
  echo "However, previously expanded alias still recognized"
  ll /usr/X1LR6/bin/mk*
fi
```

```
echo

count=0
while [ $count -lt 3 ]
do
  alias rr="ls -l"
  echo "Trying aliased \"rr\" within \"while\" loop"
  rr /usr/X1LR6/bin/mk* # Aliases will not expand here either.
  ((count+=1))
done
```

```
echo;
nalias xyz="$1" # Try a positional parameter in an alias.
xyz # If you invoke the script with a filename as a parameter.
```

The unalias command removes a previously set alias.

# Example 24–2. unalias: setting and unsettling an alias

```
#!/bin/bash

opt -s expand_aliases # Enables alias expansion

alias ll='ls -al' | more'
llm

echo

unalias ll # Unset alias.
llm
# Error message results, since 'llm' no longer recognized.

exit 0
./unalias.sh
```
