```
# a few fun ones

function xtitle {
  case $term in
    *rxvt)
      echo -ne "\033];0$*\007"
    *)
  esac
}
```

```
# aliases...
alias top='xtitle Processes on $HOST & top'
alias make='xtitle Making $(basename $PWD) ; make'
alias nctftp='xtitle nCFTP ; nctftp
```

```
# and functions
function man () {
  xtitle $ $(basename $PWD) |tr -d .[:digit:]] manual
  man -a "*"
}
```
