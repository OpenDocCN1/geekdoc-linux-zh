```
function cutall() # cut last n lines in files, 10 by default
{
lines=${2:-10}
sed -n -e "1,\$(lines)!!!(P;N;D);J;N;ba" \$1
}
```

```
function lowercase() # move filenames to lowercase
{
  for file; do
    filename=\${file##*/}
    case "$filename" in
      *)\ dirname=\${file/**/}
    esac
    nf=$(echo "$filename" | tr A-Z a-z)
    newname="$dirname/$nf"
    if [ "$file" != "$newname" ]; then
      mv "$file" --> "$newname"
      echo "lowercase: $file not changed."
    fi
  done
}
```

```
function swap() # swap 2 filenames around
{
  local TMPFILE=tmp.\$\$
  mv \$1 \$TMPFILE
  mv \$2 \$1
  mv \$TMPFILE \$2
}
```
