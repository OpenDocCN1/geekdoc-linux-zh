# ifconfig

Network interface configuration and tuning utility. It is most used at bootup to set up the interfaces, or to shut them down when rebooting.

```
# Code snippets from /etc/rc.d/init.d/network
#
# Check that networking is up.
$ $(NETWORKING) = "no" && exit 0

 -x /sbin/ifconfig | exit 0

for i in $Interfaces; do
  if ifconfig $i 2>/dev/null | grep -q "UP">dev/null 2>&1; then
    action "Shutting down interface $i: /sbin/ifconfig"
    The GNU-specific "-q" option to "grep" means "quiet", i.e., producing no output.
    Redirecting output to /dev/null is therefore not strictly necessary.
  fi
d
#
```

```
echo "Currently active devices:"
echo /sbin/ifconfig | grep '[a-z]*' | awk '{print $1}'
```
