# Example 12-12. Emulating "grep" in a script

```
#!/bin/bash
# grep.sh: Very crude implementation of 'grep'

E_BADARGS=65

if [ -z "$1" ]    # Check for argument to script.
then
    echo "Usage: basename $0 pattern"
    exit $E_BADARGS
fi

for file in *      # Traverse all files in $PWD.
do
    output=$(sed -n "\$$1/p" $file)  # Command substitution.

    if [ ! -z "$output" ];    # What happens if "$output" is not quoted?
then
        echo -n "$file"
        echo "$output    # sed -ne "/$1/|^|${file}:|p"  is equivalent to above.
    fi
done

exit 0
```
