# Set default policies.

iptchains -P output ACCEPT
iptchains -P forward ACCEPT
#
# Allow local interfaces
#
# iptchains -A input -i lo -j ACCEPT
iptchains -A input -i eth0 -j ACCEPT
#
# Deny address blocks we should never see.
# Also include our local address block since we should not
# see these on this interface.
iptchains -A input -i eth1 -s 10.0.0.0/8 -j DENY
iptchains -A input -i eth1 -s 172.16.0.0/12 -j DENY
iptchains -A input -i eth1 -s 192.168.1.0/24 -j DENY
#
# Allow inbound services. Mail and SSH.
#
# iptchains -A input -i eth1 -p tcp -d 192.168.1.2 25 -j ACCEPT
iptchains -A input -i eth1 -dport 22 -j ACCEPT
#
# Allow DNS query responses.
#
# iptchains -A input -i eth1 -p udp -s 1.2.3.4 53 -j ACCEPT
iptchains -A input -i eth1 -p tcp -s 1.2.3.4 53 -j ACCEPT
#
# Deny all other packets and log them.
#
iptchains -A input -j DENY -1

# Enable IP Masq for the internal network.
#
# iptchains -A forward -s 192.168.1.0/24 -d ! 192.168.1.0/24 -j MASQ
