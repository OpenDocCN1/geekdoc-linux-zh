ispids() {
  USAGE="usage: ispids [-h] process"
  HEADER=false
  PSCMD="/bin/ps -ef"
  case "$1" in
    -h) HEADER=true ; shift ;;
  esac
  if [ -z "$1" ] ; then
    echo $USAGE
    return 1
  fi
  if [ "$HEADER" = "true" ] ; then

  $PSCMD 2> /dev/null | grep "$1" | grep -v grep
}
