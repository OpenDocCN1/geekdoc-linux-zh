# The Complete Timer Script

The complete timer script follows:

```
```sh
#!/bin/sh

AlarmHandler() {
  echo "Got SIGALARM, cmd took too long."
  KillSubProcs
  exit 14
}

KillSubProcs() {
  kill $CHPROCSID:$!
  if [ $? -eq 0 ] ; then
    echo 'Sub-processes killed.' ;
  fi
}

SetTimer() {
  DEF_TOUT=$1:10;
  if [ $DEF_TOUT -ne 0 ] ; then
    sleep $DEF_TOUT & $!
    CHPROCSID=$CHPROCSID $!
    TIMERPROC=$!
  fi
}

UnsetTimer() {
  kill $TIMERPROC
}

# main()
trap AlarmHandler 14

SetTimer 15
$PROG &
CHPROCSID=$CHPROCSID $!
wait $!
UnsetTimer
echo "All Done."
exit 0
