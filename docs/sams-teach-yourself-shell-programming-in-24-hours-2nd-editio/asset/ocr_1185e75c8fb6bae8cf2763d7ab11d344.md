#!/bin/sh

USAGE='Usage: basename $0 [-v] [-x] [-f] [filename] [-o] [filename]'

VERBOSE=false
EXTRACT=false

while getopts f:o:xV OPTION ; do
  case "$OPTION" in
    f) INFILE="$OPTARG" ;;
    o) OUTFILE="$OPTARG" ;;
    v) VERBOSE=true ;;
    x) EXTRACT=true ;;
    ?) echo 'USAGE' ;
       exit 1
       ;;
  esac
done

shift `echo "$OPTIND - 1" | bc`
