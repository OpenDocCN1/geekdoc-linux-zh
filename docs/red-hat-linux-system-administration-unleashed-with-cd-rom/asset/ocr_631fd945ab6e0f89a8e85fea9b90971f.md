sub notify_email {
    my $host = shift;
    my $fs = shift;
    my $level = shift;
    my $used = shift;
    my $recipients = shift;

    my @subjstr = ("[WARNING]", "[CRITICAL]", "[CRISIS]");

    my $msg = 'df $fs';

    if ($options{Verbose}) {
        print "Sending e-mail notification to $recipients.\n";
    }
    if (! $options{Dryrun}) {
        open(SENDMAIL, "|/usr/sbin/sendmail $recipients");
        print SENDMAIL <<EOF;
Subject: $subjstr[$level] $host:$fs $used% full

Notification from checkspace.pl:

$msg

EOF
        close(SENDMAIL);
    } else {
        print "$msg\n";
    }
}
