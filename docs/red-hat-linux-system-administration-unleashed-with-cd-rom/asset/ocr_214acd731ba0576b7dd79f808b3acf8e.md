function disable_user {

# Prompt for the user account to disable
printf "\nEnter the user account to disable: "
read unlucky

EXISTS=`grep "$unlucky:" /etc/passwd`
if [ "$EXISTS" = "" ]; then
printf "\nUser account not found...aborting.\n"
return
else
/usr/sbin/usermod -p "*" $unlucky
fi

}
