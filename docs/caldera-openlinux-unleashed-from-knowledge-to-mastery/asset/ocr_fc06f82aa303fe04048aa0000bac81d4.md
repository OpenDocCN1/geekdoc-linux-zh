S21

#
#  header sender and masquerading header recipient rewriting
#
S31
RS+             S: >$1 $1             sender/recipient common
R:; <@>         $0                     list; special case

# do special header rewriting
RS* <@> S*      $0 $1 <@> $2          pass null host through
R< @ S* > S*    $0 < @ $1 > $2        pass route-addr through
RS+             S: >$1 $1             qualify unqual'ed names
RS+             S: >$3 $1             do masquerading

#
#  convert pseudo-domain addresses to real domain addresses
#
S51

# pass <route-addr>s through
R< @ S* > S*    $0 < @ $1 > $2        resolve <route-addr>

# output fake domains as user@fake@relay

# do UUCP heuristics; note that these are shared with UUCP mailers
