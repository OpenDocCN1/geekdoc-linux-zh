: ${DOCNFSSTAT:=false}
DOCNFSSTAT="echo "${DOCNFSSTAT}"" ; tr ${AZ} '${az}'"
if [ -z "${DOCNFSSTAT}" ] \
   -o "${DOCNFSSTAT}" = "on" \
   -o "${DOCNFSSTAT}" = "true" \
   -o "${DOCNFSSTAT}" = "yes" ] ; then
    DOCNFSSTAT=true
else
    DOCNFSSTAT=false
fi

${DOCNFSSTAT} && {
    echo "Scheduled start of cfnfsstat."
    ( sleep 60 ; ${PATHBIN}/cfnfsstat -s -l ) &
}

RMFILE=${MOST_LOGS}/expire.rw
for F in ${RMFILE} ${RMFILE}.* ; do
    if [ -r ${F} -a -s ${F} ] ; then
        echo " INN: Removing articles from pre-downtime expire run (${F})."
        (
            echo 'System shut down during expire.' \
            'Unlinking articles listed in'
            echo ${F}
        ) ; eval ${MAIL}
        ${PATHBIN}/expreimr ${F}
    fi
done &
: ;;
esac
exit 0

This script does numerous housekeeping chores, which includes checking to see that the news.daily script has been run recently (news.daily takes care of things such as article expiration) and actually starting INN.

After the INN package is installed and ready to go, you still need to check the configuration information to make sure that everything will run smoothly when innnd connects to the newssfeed.
