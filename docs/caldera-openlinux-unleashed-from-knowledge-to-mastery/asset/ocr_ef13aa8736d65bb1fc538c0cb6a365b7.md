R$*  $: $>98 $1   handle local hacks
R$*  $: $>Parse1 $1   final parsing

# Parse0 -- do initial syntax checking and eliminate local addresses.
# This should either return with the (possibly modified) input
# or return with a #error mailer. It should not return with a
# #mailer other than the #error mailer.

SParse0
R<0>  $0 <0>   special case error msgs
R$* : $* ; <>$*  $Error $0 5.1.3 $: 'List; syntax illegal for recipient
@=addresses'
R@<0>$>  <0 $1>  catch '@host' bogosity
R<0>=$>  $: <>$>  $Error $0 5.1.3 $: 'User address required'
R$*  $: <$ $1>
R<> $* <0 [$*] > $*  $1 <0 [$2] > $3
R<> $* <$* : $* > $*  $Error $0 5.1.3 $: 'Colon illegal in host name part'
R<> $*  $1
R$* <0 . $*> $*  $Error $0 5.1.2 $: 'Invalid host name'
R$* <0 $* . . $*> $*  $Error $0 5.1.2 $: 'Invalid host name'

# now delete the local info -- note *0 to find characters that cause forwarding
R$* <0 > $*  $0 $>Parse0 $>3 $1  user0 > user
R<> $*=W . > : $*  $0 $-Parse0 $>3 $2  @here:..., ... $2...
RS: <0 $*=W . >  S: $(dequote $1) > : <0 $2: >  dequote 'foo'@here
R<> $* >  $Error $0 5.1.3 $: 'User address required'
R$* <0 $* <0 @ $*=W . >  $0 $-Parse0 $>3 $1 $2 $3  ...@here -> ... 
RS:  S: $(dequote $1) $> <0 *LOCAL* >  dequote 'foo'
R<> *LOCAL* >  $Error $0 5.1.3 $: 'User address required'
R$* $=O $* <0 *LOCAL* >
R$* <0 *LOCAL* >  $0 $>Parse0 $>3 $1 $2 $3  ...@*LOCAL* > : ... 
R$* <0 *LOCAL* >  S: $1

# Parse1 -- the bottom half of rule:et 0.

SParse1
# handle numeric address spec
R$* <0 [$*] > $*  $: $>98 $1 <0 [$2] > $3  numeric internet spec
R$* <0 [$*] > $*  $#emptP $0 [$2] $: $1 <0 [$2] > $3  still numeric:
send

# short circuit local delivery so forwarded email works
R$=L <0 $*=W . >  $#local S: $1   special local names
R$* <0 $*=W . >  $#local S: $1   regular local name
