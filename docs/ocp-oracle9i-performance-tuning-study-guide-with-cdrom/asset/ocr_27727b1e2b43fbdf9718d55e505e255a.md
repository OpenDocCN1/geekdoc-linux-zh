# FIGURE 8.1 Performance query on V$FILESTAT, VS$DATAFILE, and VS$TEMPFILE

```
SQL> SELECT s.file#||'D' "FILE#TYPE", d.name "DATAFILE",
2 s.phyrdts, s.phyrwts, 
s.avgiotm, s.minitotm,
s.maxiotm, s.maxiortm
4 s.avgiotm, s.maxiortm
5 WHERE v$filetype = 'Datafile' d
6 s.file# = f.file# 
7 FROM v$filestat s, vsdatafile d
8 s.file#||'T', t.name "DATAFILE", 
9 s.phyrdts, s.phyrwts, 
10 s.avgiotm, s.minitotm, 
s.maxiotm, s.maxiortm
12 FROM v$filestat s, vs$tempfile t
13 WHERE s.file# = t.file# 
14 ORDER BY 1;
```
