    setLayout(new GridBagLayout());

    GridBagConstraints c = new GridBagConstraints();
    c.gridx = 0; c.gridy = 0;
    add(new Button("topleft"), c);
    c.gridx = 1;
    add(new Button("topmiddle"), c);
    c.gridx = 2;
    add(new Button("topright"), c);

    c.gridx = 0; c.gridy = 1;
    add(new Button("lefthandsiddle"), c);
    c.gridx = 1;
    add(tallPanel, c);

    c.gridy = 2; // note, sets _y_
    add(new Button("bottomcenter"), c);
    c.gridx = 2;
    add(tallPanel2, c);
}

public static void main(String args[]) {
    JFrame f = new JFrame("GridBag 1 example");
    f.getContentPane().add(new GB1());
    f.pack();
    f.setVisible(true);
