```c
// 打印一个月天数的程序
#include <stdio.h>

int main (void)
{
    enum month { January = 1, February, March, April, May, June,
                 July, August, September, October, November, December };
    enum month aMonth;
    int days;

    printf ("输入月份编号: ");
    scanf ("%i", &aMonth);

    switch (aMonth) {
    case January:
    case March:
    case May:
    case July:
    case August:
    case October:
    case December:
        days = 31;
        break;
    case April:
    case June:
    case September:
    case November:
        days = 30;
        break;
    case February:
        days = 28;
        break;
    default:
        printf ("无效的月份编号\n");
        days = 0;
        break;
    }

    if ( days != 0 )
        printf ("天数为 %i\n", days);

    if ( aMonth == February )
        printf ("...如果是闰年则为 29 天\n");

    return 0;
}
```

```text
输入月份编号: 5
天数为 31
```

```text

...如果是闰年则为 29 天
```

```c
#include "date.h"

// 计算明天日期的函数
Date dateUpdate (Date today)
{
    Date tomorrow;

    if ( today.day != numberOfDays(today) )
        setDate (tomorrow, today.month, today.day + 1, today.year);
    else if ( today.month == December )    // 年底
        setDate (tomorrow, January, 1, today.year + 1);
    else                    // 月底
        setDate (tomorrow, today.month + 1, 1, today.year);
    
    return tomorrow;
}
```

```c
// 演示各种 printf() 格式的程序
#include <stdio.h>

int main (void)
{
    char            c = 'X';
    char            s[] = "abcdefghijklmnopqrstuvwxzy";
    int             i = 425;
    short int       j = 17;
    unsigned int    u = 0xf179U;
    long int        l = 75000L;
    long long int   L = 0x1234567812345678LL;
    float           f = 12.978F;
    double          d = -97.4583;
    char            *cp = &c;
    int             *ip = &i;
    int             c1, c2;

    printf ("整数:\n");
    printf ("%i  %o  %x  %u\n", i, i, i, i);
    printf ("%x  %X  %#x %#X\n", i, i, i, i);
    printf ("%+i % i %07i %.7i\n", i, i, i, i);
    printf ("%i  %o  %x  %u\n", j, j, j, j);
    printf ("%i  %o  %x  %u\n", u, u, u, u);
    printf ("%ld  %lo  %lx  %lu\n", l, l, l, l);
    printf ("%lli %llo %llx %llu\n", L, L, L, L);

    printf ("\n浮点数和双精度数:\n");
    printf ("%f  %e  %g\n", f, e, f);
    printf ("%.2f  %.2e\n", f, f);
    printf ("%.0f  %.0e\n", f, f);
    printf ("%7.2f  %7.2e\n", f, f);
    printf ("%f  %e  %g\n", d, d, d);
    printf ("%.*f\n", 3, d);
    printf ("%*. *f\n", 8, 2, d);

    printf ("\n字符:\n");
    printf ("%c\n", c);
    printf ("%3c%3c\n", c, c);
    printf ("%x\n", c);

    printf ("\n字符串:\n");
    printf ("%s\n", s);
    printf (".5s\n", s);
    printf ("%30s\n", s);
    printf ("%20.5s\n", s);
    printf ("%-20.5s\n", s);

    printf ("\n指针:\n");
    printf ("%p  %p\n\n", ip, cp);

    printf ("This% is fun.%n\n", &c1, &c2);
    printf ("c1 = %1, c2 = %1\n", c1, c2);

    return 0;
}
```
