`SIGCONT` signal arrives.<sup>6</sup> After that, the main program sleeps. If it returns from `sleep` because a signal arrived, it goes back to sleeping again.

```
```
int main(void)
{
   struct sigaction act;

   memset(&act, 0, sizeof(act));
   act.sa_handler = catchsig;
   ec_negl_sigaction(SIGCONT, &act, NULL) ;
   showpinfo("initial call");
   while (true)
      sleep(10000);

   EC_CLEANUP_BGN
       exit(EXIT_FAILURE);
   EC_CLEANUP_END
}

static void catchsig(int signo)
{
   if (signo == SIGCONT)
      showpinfo("got SIGCONT");
}
