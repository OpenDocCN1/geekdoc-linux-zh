```
bool smi_send_getaddr_skt(SMIQ *sqp, struct client_id *client,
void **addr)
{
SMIQ_SKT *p = (SMIQ_SKT *)sqp;

if (p->sq_entity == SMI_SERVER)
p->sq_client = *client;
*addr = p->sq_msg;
return true;
}
```
