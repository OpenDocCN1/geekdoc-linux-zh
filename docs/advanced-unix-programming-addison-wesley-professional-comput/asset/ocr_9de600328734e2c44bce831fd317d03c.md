BUFIO *Bopen(const char *path, const char *dir)
{
BUFIO *b = NULL;
int flags;

switch (dir[0]) {
case 'r':
flags = O_RDONLY;
break;
case 'w':
flags = O_WRONLY | O_CREAT | O_TRUNC;
break;
default:
errno = EINVAL;
EC_FAIL
}
ec_null( b = calloc(1, sizeof(BUFIO)) )
ec_neg!( b->fd = open(path, flags, PERM_FILE) )
b->dir = dir[0];
return b;

EC_CLEANUP_BGN
free(b);
return NULL;
EC_CLEANUP_END
}

static bool readbuf(BUFIO *b)
{
ec_neg!( b->total = read(b->fd, b->buf, sizeof(b->buf)) )
if (b->total == 0) {
errno = 0;
return false;
}
b->next = 0;
return true ;

EC_CLEANUP_BGN
return false;
EC_CLEANUP_END
}
```

```
static bool writebuf(BUFIO *b)
{
size_t n, total;

total = 0;
while (total < b->next) {
ec_neg!( n = write(b->fd, &b->buf[total], b->next - total) )
total += n;
}
