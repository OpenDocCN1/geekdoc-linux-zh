static void set_fd_hwm(SSI *ssip, int fd)
{

		ssip->ssi_fd_hwm = fd;
}
static void reset_fd_hwm(SSI *ssip, int fd)
{
	if (fd == ssip->ssi_fd_hwm)
		ssip->ssi_fd_hwm--;
}
