bool cready(CR_STATUS *statusp)
{
if (cbuf != EMPTY) {
*statusp = CR_READY;
return true;
}
setblock(STDIN_FILENO, false);
switch (read(STDIN_FILENO, &cbuf, 1)) {
case -1:
if (errno == EAGAIN) {
*statusp = CR_NOTREADY;
return true;
}
EC_FAIL
case 0:
*statusp = CR_EOF;
return true;
case 1:
return true;
default: /* "impossible" case */
errno = 0;
EC_FAIL
}

EC_CLEANUP_BGN
return false;
EC_CLEANUP_END
}
