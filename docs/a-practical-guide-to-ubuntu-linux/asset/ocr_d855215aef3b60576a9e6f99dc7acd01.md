}

function scramble ()
{
  typeset -i index count
  questions=${questions[*]}
  quescount=${#questions[@]} # Number of elements
  ((index=quescount-1));
  while [[ $index > 0 ]]; do
    (target=$RANDOM % index)
    exchange $target $index
    ((index -- 1));
}
