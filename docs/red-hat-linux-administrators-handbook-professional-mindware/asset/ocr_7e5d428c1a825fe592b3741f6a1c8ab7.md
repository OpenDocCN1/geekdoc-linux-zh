sub createNamedFile {  
my \$file = shift;  
my \$domain = shift;  
my \$database = shift;  
my \$line;  
open (OUT, ">>\$file") || die "Can't write \$file\n";  
open (FP, S NAMED _TEMPLATE) || die "Can't open  
SNAMED _TEMPLATE\n";  
while (\$line=<FP>){  
\$line = - s/<DOMAIN>/\$domain/g;  
\$line = - s/<SERIAL>/\$serial/g;  
print OUT \$line;  
}  
close (FP);  
close (OUT);  
open (FP, ">>SNAMED_CONF") || die "Can't open  
SNAMED _CONF\n";  
print FP <<ZONE;  
// \$domain was created on \$date  
zone "\$domain" {  
type master;  
file "\$database";  
};
