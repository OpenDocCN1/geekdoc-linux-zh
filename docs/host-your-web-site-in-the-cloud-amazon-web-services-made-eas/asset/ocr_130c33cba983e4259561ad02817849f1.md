function WriteBatch($ssdb, &$items)
{
    $res = $ssdb->batch_put_attributes(BOOK_FILE_DOMAIN, $items, true);
    if ($res->isOK())
    {
        print("Inserted " . count($items) . " items\n");
        return true;
    }
    else
    {
        $error = $res->body->Errors->Error->Message;
        print("Could not insert items: {$error}\n");
        return false;
    }
}
