#include <Dde.h>

static QList<DdeTopic> topics;

static HHRESULT CALLBACK
Callb back( UINT wType, UINT wFmt, HCONV hConv, HDSZ hsz1,
            HDSZ hsz2, HDEDATA hDDData, DWORD dwData1,
            DWORD dwData2 )
{
    DdeTopic *p;
    DWORD len;
    bool (*fun)(const char *);
    const char *s;

switch (wType)
{
    case XTVP_REGISTER:
    case XTVP_UNREGISTER:
        return( (HHRESULT)false );

    case XTVP_ADVDATA:
        return( (HHRESULT)DDE_FACK );

    case XTVP_XACT_COMPLETE:
        return( (HHRESULT)false );

    case XTVP_DISCONNECT:
        return( (HHRESULT)false );

    case XTVP_EXECUTE:
        s = (const char *)DdeAccessData( hDDData,
            &len );
        for(p=topics.first();p; p=0;p=topics.next()) {
            if(p->topic() == hsz1) {
                fun = p->function();
                if( fun( s ) == true )
                    return( (HHRESULT)DDE_FACK );
            } else
                return( (HHRESULT)DDE_FNOTPROCESSED);
        }
        return( (HHRESULT)DDE_FNOTPROCESSED );

    case XTVP_CONNECT:
        for(p=topics.first();p; p=0;p=topics.next()) {
            if(p->topic() == hsz1) {
                p->setConversation( hConv );
                return( (HHRESULT)true );
            }
        }
        return( (HHRESULT)false );
