bool  
Socket::connect( const char *host, unsigned short int _port )  
{  
    sock = ::socket(IP_INET, SOCK_STREAM,0);  
    if (sock == INVALID_SOCKET) {  
        return false;  
    }  
    if ( !init_sockaddr( _host, _port )) {  
#ifdef WIN32  
        ::closesocket( sock );  
#else  
        ::close( sock );  
#endif  
        sock = INVALID_SOCKET;  
        return false;  
    }  
    if ( ::connect(sock, (struct sockaddr*)(&name), sizeof( name )) < 0 )  
    {  
#ifdef WIN32  
        ::closesocket( sock );  
#else  
        ::close( sock );  
#endif  
        sock = INVALID_SOCKET;  
        return false;  
    }  
    return true;  
}
