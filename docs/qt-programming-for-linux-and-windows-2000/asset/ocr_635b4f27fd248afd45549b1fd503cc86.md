// LilMidi:::(Stopped, Playing, or Paused)
int
LilMidi::status()
{
if( isopen == true ) {
MCI_STATUS_PARMS parms;
parms.dwItem = MCI_STATUS_MODE;
if( mcSendCommand(
device,
MCI_STATUS,
MCI_WAIT | MCI_STATUS_ITEM,
(DWORD)(LPOVOID)&parms ) ) {
return( Stopped );
}
switch( parms.dwReturn ) {
case MCI_MODE_PLAY:
state = Playing;
break;
case MCI_MODE_PAUSE:
state = Paused;
break;
default:
state = Stopped;
break;
} } else {
state = Stopped;
}
return( state );
