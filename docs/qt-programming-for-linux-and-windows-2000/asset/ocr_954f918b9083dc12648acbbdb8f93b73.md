// void close()
// Stop and close the mci stream
void
LiiMidi::close()
{
if (isopen == true ) {
// Stop playing and close
mciSendCommand( device, MCI_STOP, NULL, NULL );
state = Stopped;

mciSendCommand( device, MCI_CLOSE, NULL, NULL );

device = 0;
isopen = false;
}
}
