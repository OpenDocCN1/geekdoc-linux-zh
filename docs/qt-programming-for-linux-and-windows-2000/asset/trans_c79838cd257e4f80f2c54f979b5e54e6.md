// int status()
// update and return current state
// LilMidi:::(Stopped, Playing, or Paused)
int
LilMidi::status()
{
    if( isopen == true ) {
        MCI_STATUS_PARMS parms;
        parms.dwItem = MCI_STATUS_MODE;
        if( mciSendCommand(
            device,
            MCI_STATUS,
            MCI_WAIT | MCI_STATUS_ITEM,
            (DWORD)(LPVOID)&parms ) ) {
            return( Stopped );
        }
        switch( parms.dwReturn ) {
            case MCI_MODE_PLAY:
                state = Playing;
                break;
            case MCI_MODE_PAUSE:
                state = Paused;
                break;
            default:
                state = Stopped;
                break;
        }
    } else {
        state = Stopped;
    }
    return( state );
}

// bool seek( unsigned long milliseconds )
// seek to the stream position
bool
LilMidi::seek( unsigned long msec )
{
    if( isopen == true ) {
        MCI_SEEK_PARMS params;
        params.dwTo = msec;
        switch( status() ) {
            case Playing:
                if( mciSendCommand( device,
                    MCI_PAUSE, 0, NULL ) )
                    return( false );
                if( mciSendCommand( device,
                    MCI_SEEK,
                    MCI_TO | MCI_WAIT,
                    (DWORD)(LPVOID)&params) ) {
                    return( false );
                }
                state = Paused;
                play();
                break;
            default:
                if( mciSendCommand( device,
                    MCI_SEEK,
                    MCI_TO | MCI_WAIT,
                    (DWORD)(LPVOID)&params) ) {
                    return( false );
                }
                break;
        }
        return( true );
    }
    return( false );
}

LilMidi::fastPwd( int sec )

    if( isopen == true ) {
        unsigned long p = pos();
        p += (unsigned long) ( sec * 1000 );

    }
}

// void rewind( int seconds )
// rewind the stream
void
LilMidi::rewind( int sec )
{
    if( isopen == true ) {
        unsigned long p = pos();
        if( p < (unsigned long) ( sec * 1000 ) )
            p = 0;
        else
            p -= (unsigned long) ( sec * 1000 );
        seek( p );
    }
}

## 总结

虽然在我心目中，Linux 仍然是我首选的操作系统（作为开发平台），但 Windows 2000 提供了许多功能，使得选择最喜欢的系统变得越来越困难。Qt 发行版中附带的大多数示例主要面向 Linux 用户。我这样说的依据是，其中一些示例在 Windows 平台上无法构建（且相关性不大）。我预计在未来版本的 Qt 中会看到一些仅适用于 Windows 的示例。让我们拭目以待。
