void
Audio::pause()
{
  switch(state) {
    case Speech:
      #ifdef TTS
        speech->AudioPause();
      #endif
      break;
    case WAV:
      PlaySound( NULL, NULL, SND_PURGE );
      break;
    case Midi:
      midi->pause();
      break;
    default:
      break;
  }
}
void
Audio::resume()
{
  switch(state) {
    case Speech:
      #ifdef TTS
        speech->AudioResume();
      #endif
      break;
    case WAV:
      PlaySound( file, NULL,
                  SND_ASYNC|SND_FILENAME );
      break;
    case Midi:
      midi->play();
      break;
    default:
      break;
  }
}
