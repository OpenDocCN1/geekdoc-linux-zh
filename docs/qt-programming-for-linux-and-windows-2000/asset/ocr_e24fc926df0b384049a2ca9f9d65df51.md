void
Audio::stop()
{
    switch(state) {
    case Speech:
#ifdef TTS
        speech->AudioReset();
#endif
        break;
    case Wave:
        PlaySound( NULL, NULL, SND_PURGE );
        break;
    case Midi:
        midi->stop();
        break;
    default:
        break;
    }
}
```

```
void
Audio::repeat()
{
    switch(state) {
    case Speech:
#ifdef TTS
        speech->Speak( spkbuffer, true, NULL );
#endif
        break;
    case Wave:
        PlaySound( file, NULL,
             SND_ASYNC|SND_FILENAME );
        break;
    case Midi:
        midi->seek( 0L );
        midi->play();
        break;
    default:
        break;
    }
}
