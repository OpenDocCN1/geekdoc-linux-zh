
# 1.9 Runtime Profiling with Performance Counters

```c
int group_fd,
unsigned long flags) {
    int ret;
    ret = syscall(__NR_perf_event_open, hw_event, pid, cpu,
    group_fd, flags);
    return ret;
}
```

The pid argument specifies the Linux process ID for which the measurements are made. There are two special values: if pid is 0 the measurement is made on the current process, and if pid is -1 the measurement is made on all processes. This code will set the argument to 0 to measure the current process.

The cpu argument allows the user to restrict measurements on a single CPU. If cpu is -1 the events are measured on all CPUs.

The group_fd argument allows event groups to be created, allowing more than one event to be counted with a single call to perf_event_open. The code in this chapter uses separate calls to perf_event_open to count each individual event, so set this argument to -1 to measure a single event.

Also needed is a function that will scale the results given by perf_event according to the ratio time enabled and time running. Reading a counter with perf_event returns an array of 64-bit integers, the value at index 0 represents the counter value, the value at index 1 represents the time enabled, and index 2 represents the time running.

```c
static inline unsigned long long perf_count(unsigned long long *values) {
    return (unsigned long long)((float)values[0]*
    (float)values[1]/(float)values[2]);
}
```

In perf_event, individual performance counters can be accessed through a file descriptor, so begin by declaring an array of file descriptors:

```c
int fd[5];
```

Declare an array to store the information read from each counter:

```c
unsigned long long cnts[4];
```

The following function will open the four counters. This function is designed to work for the ARM11, ARM Cortex A9, and ARM Cortex A15 processors. Opening the cache miss and reference counters is slightly different for the ARM11, so the code assumes that the ARM11 variable is set when compiling for the ARM11 (compile with the -DARM11 option).

```c
int cnts_open(int *event_attr, struct *attr) {
    memset(&attr, 0, sizeof(attr));
    attr.type = PERF_TYPE_HARDWARE;
    attr.size = sizeof(struct perf_event_attr);
    attr.read_format =
        PER_FORMAT_TOTAL_TIME_ENABLED|
        PER_FORMAT_TOTAL_TIME_RUNNING;
    attr.config = PERF_COUNT_EVENT_HW_CPU_CYCLES;
    fd[CYCLES] = perf_event_open(&attr, 0, -1, -1, 0);
    if (fd[CYCLES] == -1) 
        perror("cannot open perf_counter for cycles"); exit(0);
    attr.config = PERF_COUNT_HW_INSTRUCTIONS;
    fd[INSTRUCTIONS] = perf_event_open(&attr, 0, -1, -1, 0);
    if (fd[INSTRUCTIONS] == -1) 
        perror("cannot open perf_counter for instructions"); exit(0);
#ifdef ARM11
    attr.type = PERF_TYPE_RAW;
    attr.config = 0x9;
    fd[CACHFS] = perf_event_open(&attr, 0, -1, -1, 0);
    if (fd[CACHFS] == -1) 
        perror("cannot open perf_counter for cache refs"); exit(0);
    attr.config = 0x8;
    fd[CACHES] = perf_event_open(&attr, 0, -1, -1, 0);
    if (fd[CACHES] == -1)
        perror("cannot open perf_counter for cache misses"); exit(0);
#else
    attr.config = PERF_COUNT_HW_REFERENCES;
    fd[CACHFS] = perf_event_open(&attr, 0, -1, -1, 0);
    if (fd[CACHFS] == -1) 
        perror("cannot open perf_counter for cache refs"); exit(0);
    attr.config = PERF_COUNT_HW_MISSES;
    fd[CACHES] = perf_event_open(&attr, 0, -1, -1, 0);
    if (fd[CACHES] == -1)
        perror("cannot open perf_counter for cache misses"); exit(0);
#endif
}
```

Add the "tick" and "clock" functions, which can be called before and after the code:

```c
void cnts_tick(int *fd) {
    int i, ret;
    cnts_open(fd);
    for (i=0; i<4; i++) {
        ret = ioctl(fd[i], PERF_EVENT_IOC_RESET);
        if (ret==-1) perror("ioctl in cnts_tick() failed");
    }
}

void cnts_clock(int *fd, unsigned long long cnts[]) {
    int i, ret;
    for (i=0; i<4; i++) {
        ret = read(fd[i], cnts[i], sizeof(cnts[i]));
        if (ret!=8) perror("read in cnts_clock() failed");
    }
}
```

void cnts_close(int *fd) {
  int i;
  for (i=0; i<4; i++) close(fd[i]);
}

# 1.10 MEASURING MEMORY BANDWIDTH

To begin using perf_event, let us use it to measure the time required for the processor to write a contiguous block of data and read a contiguous block of data. This test will also allow us to measure the processor's memory bandwidth, which is often different for reads versus writes. The programmer can use these values to calculate the performance upper bound.

Write a print routine that prints our performance results and calculates bandwidth:

```c
void cnts_dump(unsigned long long cnts[][3]) {
    float time, memb;

    time = ((float)perf_count(cnts[CYCLES])/CLOCK_RATE;
    membw = (float)SIZE/time/(1024.0*1024.0);

    printf("[%perf_event] cycles = %llu (%0.0f us)\n",
           perf_count(cnts[CYCLES]),
           (float)perf_count(cnts[CYCLES])/(CLOCK_RATE/1.0E-6));

    printf("[%perf_event] instructions = %llu (CPI=%0.2f)\n",
           perf_count(cnts[INSTRUCTIONS]),
           (float)perf_count(cnts[INSTRUCTIONS])/(float)perf_count(cnts[CYCLES]));

    printf("[%perf_event] misses = %llu, references = %llu\n"
           "(miss rate=%0.4f)\n",
           perf_count(cnts[CACHES]),
           perf_count(cnts[CACHFS]),
           (float)perf_count(cnts[CACHES])/(float)perf_count(cnts[CACHFS]));
}
```
