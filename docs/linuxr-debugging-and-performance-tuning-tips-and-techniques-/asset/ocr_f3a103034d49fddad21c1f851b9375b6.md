        exit(1);
    }
    if (S_ISCHR(sbuf.st_mode))
        devtype = "char";
    else if (S_ISBLK(sbuf.st_mode))
        devtype = "block";
    else {
        fprintf(stderr, "%s is not a block or character device\n", argv[1]);
        exit(1);
    }
    printf("%s: major: %d, minor: %d\n", devtype,
        major(sbuf.st_rdev), minor(sbuf.st_rdev));
