# Delete all hosts from batchparms
sub delbatch
{
    local( $sysnam, $backup, $dummy );
    local( $uid, $gid, $mode );

    # Backup up older versions
    #backupfile( $swsbat, );
    $backup = $swsbat;
    open ORIGINAL, "$backup" || warn "Cannot open $swsbat, $!\n" && $lnk $backup,
    $swsbat && return;
    open NEW, ">$swsbat" || warn "Cannot creat $swsbat, $!\n" && $lnk $backup,
    $swsbat && return;

    while( <ORIGINAL> ) {
        if( /^$sysnam/ ) {
            print NEW;
            next;
        }
        ($sysnam, $dummy) = split /:/s;
        print NEW if (!exists $delete{$sysnam});
    }
    close ORIGINAL;
    close NEW;
    # Set ownership and permissions
    &setperm( $backup, $swsbat );
} # delbatch

# Add all hosts to batchparms
sub addbatch
{
    local( $sysnam, $backup, $dummy );
    local( $uid, $gid, $mode );

    # Backup up older versions
    #backupfile( $swsbat, );
    $backup = $swsbat;
    open ORIGINAL, "$backup" || warn "Cannot open $swsbat, $!\n" && $lnk $backup,
    $swsbat && return;
    open NEW, ">$swsbat" || warn "Cannot creat $swsbat, $!\n" && $lnk $backup,
    $swsbat && return;

    while( <ORIGINAL> ) {print NEW;}
    close ORIGINAL;

    foreach $sysnam (keys %add) {
        print NEW <CEOF
$sysnam u 100000 20 batcher
| compun | v1uxx
EOF
    }
    close NEW;
    # Set ownership and permissions
    &setperm( $backup, $swsbat );
} # addbatch
