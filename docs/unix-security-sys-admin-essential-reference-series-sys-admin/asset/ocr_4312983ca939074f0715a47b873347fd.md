As said before, I use Taylor UUCP on this system. One of the nice features of Taylor UUCP is that it can support a number of different UUCP configuration formats. The native Taylor UUCP format is the most flexible. It also has a descriptive configuration that offers the most features. The well-known HDB format. For true die-hards, it also supports the V2 style. When building Taylor UUCP, you can specify which formats you would like to have supported. You can have more than one format supported, but that definitely complicates things.

# The UUCP main entry point
sub dooucp {
local($trigger ) = @;
local($busy, $sysnam);
$busy = $trigger . ".busy";
link $trigger, $busy || warn "Cannot rename $trigger, $!\n" & return;
unlink $trigger;
open TRIGGER, "$busy" || warn "Cannot open $trigger, $!\n" && unlink $busy && return;
while (<TRIGGER>) {
chomp;
if (/\^\n\.,\*/) ||
$sysnam = substr $_, 1;
$add{$sysnam} = 1;
}
if (/\^\n\.,\*/) ||
$sysnam = substr $_, 1;
delete($sysnam) = 1;
}
close TRIGGER;
unlink $busy;
# We got a list of hosts to be added and hosts to be deleted
# First delete all hosts
&deluucp@mainnews[ ];
foreach $sysnam (keys %delete) |
if ($uucp@host($sysnam) ) |
&deluucp($sysnam) );
}
else {
if (-e $pending) { open PENDING, ">>$pendingfile"; }
else | open PENDING, ">>$pendingfile"; |
print PENDING "*$sysnam$24"";
}
close PENDING;
}
@ = keys %delete;
if ($#_ > 0) |
&notifuuucp("deleted", keys %delete);
