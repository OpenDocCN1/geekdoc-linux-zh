# Delete a user from the passwd and group file
sub deluser
{
    local( $usrnam ) = @_;
    local $nwfile;

    $usrnam = "wu" . $usrnam;

    # from the group file
    $nwfile = $group . ".new";
    open OLD, "$group" || warn "Cannot open $group, $!\n" && return;
    open NEW, "$nwfile" || warn "Cannot creat $nwfile, $!\n" && close OLD && return;
    while (<OLD>) {
        if (/^$usrnam/) {
            s/,$usrnam//;
        }
        print NEW;
    }
    close OLD;
    close NEW;
    rename $group, $group . ".old";
    rename $nwfile, $group;
    &setperm( $group, ".old", $nwfile );

    # from the passwd file
    $nwfile = $passwd . ".new";
    open OLD, "$passwd" || warn "Cannot open $passwd, $!\n" && return;
    open NEW, "$nwfile" || warn "Cannot creat $nwfile, $!\n" && close OLD && return;
    while (<OLD>) {
        if (/^$usrnam/) {
            next;
        }
        print NEW;
    }
    close OLD;
    close NEW;
    rename $passwd, $passwd . ".old";
    rename $nwfile, $passwd;
    &setperm( $passwd, ".old", $nwfile );
}
