# Notify root of successful action for UUCP
sub notifyuucp
{
    local( $action, @hostlist ) = @_;

    open MAILTO, "|mail root" || warn "Cannot start mail program, $!\n" & return;
    print MAILTO <<EOF
Subject: UUCP hosts $action.

The following UUCP hosts have been $action :

@hostlist
EOF
    ; # notifyuucp
