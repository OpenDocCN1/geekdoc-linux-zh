# Listing 13.5 s1callback.c—a login shell for the SLIP user ID that drops the current connection, resets the modem, and executes the sldial program.

```
#include <stdio.h>
#include <signal.h>
#include <strings.h>
#include <sys/types.h>
#include <pwd.h>

#define _PATH_SLDIAL  "/usr/bin/sldial"
#define CB_DELAY      30

main()
{
    struct passwd *pw ent;
    char *pgrm, *name, *slhost;

    pw ent=getpwid(getuid());
    printf("You seem to be %s $(d)\n", pw ent->pw_name, getuid());
    printf("Calling back\n\n\n");
    fflush(stdout);
    signal(SIGHUP, SIG_IGN);
    slhost=pwent->pw_name;

    /* some code to hang up the line */
    if (fork())
        exit(0);
    sleep(CB_DELAY);  /* time for IBM modems to hangup */

    /* just call sldial. It has everything set up */
    pgrm = _PATH_SLDIAL;
    name = ((name = strchr(pgrm, '/')) == NULL) ? pgrm : name + 1;

    execl(pgrm, name, "-f", "-d", "-S", slhost, slhost, (char *) 0);

    exit(0);
}
```
