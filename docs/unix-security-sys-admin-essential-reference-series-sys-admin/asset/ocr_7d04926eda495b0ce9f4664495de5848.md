weeknum ()
{
    echo $1 |
    awk '
    #
    for (i = 1; i <= length; i++) |
        i = length;
        do |
            sum += 64;
            sum += index(digits, substr($0, i, 1)) - 1;
        | while (--i);
    }
    END |
        print sum;
    |
    ' digits=$DIGITS
}
