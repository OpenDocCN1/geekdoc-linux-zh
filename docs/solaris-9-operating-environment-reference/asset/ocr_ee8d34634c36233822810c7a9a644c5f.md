```
hrtime_t
gethrime()
(
    static mutex_t lock;
    static hrtime_t (*real_gethrime) = null;
    static hrtime_t last_time = 0;

hrtime_t this_time;
    if (real_gethrime == null) {
        real_gethrime =
            (hrtime_t (*)(void)) dlsym(RTLD_NEXT, "gethrime");
    }
    this_time = real_gethrime();
    mutex_lock(&lock);
    if (this_time <= last_time) {
        this_time = ++last_time;
    } else
        last_time = this_time;
    mutex_unlock(&lock);
return (this_time);

```

Of course, this workaround does not increase the resolution of the timer, so timestamps for individual events are still relatively inaccurate. However, this technique does maintaining ordering that if event A causes event B, B never seems to happen before or at the same time as A.
