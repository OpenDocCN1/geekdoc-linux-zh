chkln: cmp [ebx],0 ; Does help msg pointer to a null?
jne .show ; If not, show the help lines
mov eax,l1 ; Load eax with number of newlines to output
call newline ; Output the newlines

.show: push ebx ; Push address of help line on the stack
call printf ; Display the line
add esp,4 ; Increment address by length of help line
add ebx,HELPSIZE
jmp .chk ; Loop back and check to see if we have done

showerr:
push eax ; On entry, eax contains address of error message
call printf ; Show the error message
add esp,4 ; Clean the stack
ret ; Go home; no returned values

newline:
mov eax, BUFSIZE ; BUFSIZE tells how many chars to pull
mov byte [buff+BUFSIZE+1],0 ; Put a null at the end of the buffer
loop: dec ecx ; BUFSIZE is 1-based, decrement
call pull6 ; Get a random number from 0-63
mov cl, (chartbl+eax) ; Use random # as offset into table
; and copy character from cl to table cl
mov [buff+ebx],cl ; Copy char from character buffer
mov ebx,0 ; Are we done having fun yet?
jne .loop ; If not, go back and pull another
mov esi, dword buff ; Copy address of the buffer into esi
ret ; and go home
