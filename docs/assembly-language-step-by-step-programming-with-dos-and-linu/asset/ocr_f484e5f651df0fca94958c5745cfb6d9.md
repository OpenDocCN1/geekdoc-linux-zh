;; Everything before this is boilerplate; use it for all ordinary apps!
;; First, an example of safely limited string using fgets. Unlike
;; gets, which does not allow limiting the number of chars entered;
;; lets you specify a maximum number. However, you must also specify a
;; file (the 'f' in 'fgets') so we must push the stdin handle.

```
push dwarf promptstr  ; Push address of the string input prompt string
call printf             ; Display it
add esp,4               ; Clean up stack for 1 arg
```

```
push dwarf [stdin]    ; Push address predefined handle for standard input
push dwarf 72           ; Accept no more than 72 characters from keybd.
push dwarf instring    ; Push address of buffer for entered characters
call fgets              ; Call fgets
add esp,12              ; 3 args X 4 bytes = 12 for stack cleanup
```

```
push dwarf instring;  ; Push address of entered string data buffer
push dwarf sshow        ; Push address of the string display prompt
call printf             ; Call printf
add esp,8                ; Clean up the stack
```

;; Next, we'll use scanf to enter numeric data. This is easier, whereas
;; unlike strings, integers can only be so big and hence self-
;; limiting.

```
push dwarf iprompt  ; Push address of the integer input prompt
call printf             ; Display it
add esp,4                ; Clean up the stack
```

```
push dwarf intval    ; Push the address of the integer buffer
push dwarf format      ; Push the address of the integer format string
call scanf              ; Call scanf to enter numeric data
add esp,8                ; Clean up the stack
```

```
push dwarf [intval]; ; Push integer value to display
push dwarf ishow       ; Push base string
call printf             ; Call printf to convert & display the integer
add esp,8                ; Clean up the stack
```
