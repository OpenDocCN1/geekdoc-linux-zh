mov [DispType],AL ; Store display adapter code in DispType

Next we determine the font size currently in force:
mov AL, 0AH ; See if board is an MCGA
jmp TryOld ; If less than code 0AH, it's not an MCGA
mov [FontSize], BYTE, 16 ; MCGA supports 'only' 16 pixel text font
jmp GetName ; Jump to look up adapter name string

TryOld: [DispType], BYTE 01 ; Is the display adapter code 1, for MDA?
jne TryGA ; If not, go for CGA code 2
mov [FontSize], BYTE, 14 ; MDA uses 'only' 14-pixel text font
jmp GetName ; Jump to look up adapter name string
TryGA: [DispType], BYTE 02 ; Is the display adapter code 2, for CGA?
jne TryVGA ; If not, go for EGA/VGA font size
mov [FontSize], BYTE, 08 ; CGA uses 'only' 8-pixel text font
jmp GetName ; Jump to look up adapter name string
TryVGA: mov AL, 11H ; Select Video Get Information
m AL 30H ; requires AL = 11H and AL = 30H
mov BH, 0 ; Get info about current font
int 10H ; Call Video
mov [FontSize],CL ; Font size in pixels returns in CL

Next we get the name string for the board into the table:
GetName: mov AL,[DispType] ; Load adapter code into AL
xor AX, AH ; Zero AH so we don't copy trash into DI
mov DI, AX ; Copy AX (with code) into DI
mov CL, 5 ; We must shift the code 5 bits to mult. by 32
sh DI, CL ; Multiply code by 32 to act as table index
mov BX,DeviceInfotbl ; Load address of origin table into BX
mov [BoardName],BX ; Save pointer to video table in BoardName
add [BoardName],DI ; Add offset into right element

Next we calculate the refresh segment from the table:
mov AX, [BX+27] ; Index into table past name string to segment
mov [VidSegment],AX ; Store segment from table to VidSegment

Here we calculate the number of lines on-screen for font:
xor AX, AH ; Make sure AH has no trash in
mov AL, [FontSize] ; Load the font size in pixels into AL
cmp AL, 8 ; Is it the 8-pixel font?
jne Try14 ; If not, the 14-pixel font
mov AL, 1 ; The 8-pixel font is table offset 1
jmp ReadLines ; Jump to read screen lines from table
cmp AL, 14 ; Is it the 14-pixel font?
Try14: jne Dol6 ; If not, it is the 16-pixel font
mov AL, 2 ; The 14-pixel font is table offset 2
jmp ReadLines ; Jump to read screen lines from table
cmp AL, 3 ; The 16-pixel font is table offset 3

Dol6: mov AL, 3 ; Add font size offset to element offset
ReadLines: add DI, AX
