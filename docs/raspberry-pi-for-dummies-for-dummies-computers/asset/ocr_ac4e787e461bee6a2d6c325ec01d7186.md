def readCard():
    global needsRedraw
    status = -1
    while checkEvent() == -1: allow pygame a look in while waiting
        status = rfidReader.RC522_ReadCard()
        # This is the (default) key for authentication
        key = [0xFF,0xFF,0xFF,0xFF,0xFF]
        block1 = rfidReader.RC522_GetSector(key,8)
        block2 = rfidReader.RC522_GetSector(key,9)
        if block1 != -1 and block2 != -1: if data OK
            x = block2[0] << 8 | block2[1]
            y = block2[2] << 8 | block2[3]
            lookAtNewRedraw(block1,x,y)
    needsRedraw = True
    return -1
