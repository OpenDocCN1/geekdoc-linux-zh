#!/usr/bin/env python3
#demo1 Apa102bang class driver

import time
from apa102bang import Apa102bang

length = 8 ; brightness = 4
dataPin = 23 ; clockPin = 24 # Blinkti! wiring
led = Apa102bang(length,dataPin,clockPin,wiring)

def main():
    print("APA102 demo - Cnt C to stop")
    while True:
        for i in range(0,8):
            led.setled(i,(1,120,0)) = red
            led.show()
            time.sleep(0.06)
        time.sleep(0.6)
        for i in range(0,4):
            led.setled(i,(0,120,0)) = green
            led.show()
            time.sleep(0.06)
        time.sleep(0.6)
        for i in range(4,8):
            led.setled(i,(0,0,120)) = blue
            led.show()
            time.sleep(0.06)
        time.sleep(0.6)
        for i in range(0,8):
            led.setled(i,(0,0,0)) = black
            led.show()
            time.sleep(0.06)
        time.sleep(0.6)

# Main program logic:
if __name__ == '__main__':
    main()
