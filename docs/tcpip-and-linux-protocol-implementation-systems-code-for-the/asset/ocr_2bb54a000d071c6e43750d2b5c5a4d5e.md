static u32 to_hash(u32 id) {
    u32 h = id&0xFF;
    if (id&0x8000)
        h += 256;
    return h;
}
static u32 from_hash(u32 id) {
    id &= 0xFFFF;
    if (id == 0xFFFF)
        return 32;
    if (!(id & 0x8000)) {
        if (id > 256)
            return 256;
        return id&0xF;
    }
