static void ipgre_tunnel_init_gen(struct net_device *dev)
{

  dev->uninit = ipgre_tunnel_uninit;
  dev->destructor = ipgre_tunnel_destructor;
  dev->hard_start_xmit = ipgre_tunnel_xmit;
  dev->get_stats = ipgre_tunnel_get_stats;
  dev->do_ioctl = ipgre_tunnel_ioctl;
  dev->change_mtu = ipgre_tunnel_change_mtu;

  dev_init_buffers(dev);
  dev->type = ARPHID_IPORE;
  dev->hard_header_len = LL_HEADER + sizeof(struct iphdr) + 4;
  dev->mtu = 1500 - sizeof(struct iphdr) - 4;
  dev->flags = IFF_NOARP;
  dev->ifflink = 0;
  dev->addr_len = 4;
  memcpy(dev->dev_addr, &parms.iph.saddr, 4);
  memcpy(dev->broadcast, &parms.iph.daddr, 4);
}
