read_lock(kmrt_num_lock);
if (reg_vif_num >= 0)
    reg_dev = vif_table[reg_vif_num].dev;
if (reg_dev)
    dev_hold(reg_dev);
unlock_lock(kmrt_num_lock);

if (reg_dev == NULL) {
    kfree_skb(skb);
    return -EINVAL;
}

skb->mac_raw = skb->nh_raw;
skb_pull(skb, (unsigned char*)encap - skb->data);
skb->nh_ip = (struct iphdr*)skb->data;
skb->dev = reg_dev;
memset(&skb->options, 0, sizeof(struct ip_options));
skb->protocol = __constant_htons(ETH_P_IP);
skb->ip_summed = 0;
skb->pkt_type = PACKET_HOST;
dst_release(skb->dst);
skb->dst = NULL;
((struct net_device_stats*)reg_dev->priv)->rx_bytes += skb->len;
((struct net_device_stats*)reg_dev->priv)->rx_packets+=1;
#ifdef CONFIG_NFFILTER
nf_conrack_put(skb->nfct);
skb->nfct = NULL;
#endif

netif_r(skb);
dev_put(reg_dev);
return 0;
