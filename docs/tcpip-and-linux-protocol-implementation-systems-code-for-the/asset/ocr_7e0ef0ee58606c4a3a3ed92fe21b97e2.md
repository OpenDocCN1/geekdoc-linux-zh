```
#define IPv6_DEFAULT_HOPPLIMIT 64
#define IPv6_DEFAULT_MCASTHDOP 1
/*
 * Addr type
 * type - unicast | multicast | anycast
 * scope - local | site | global
 *        compat
 *        v4
 *        vmapped
 *        any
 *        loopback
 */
#define IPv6_ADDR_ANY 0x0000
#define IPv6_ADDR_UNICAST 0x0001
#define IPv6_ADDR_MULTICAST 0x0002
#define IPv6_ADDR_ANYCAST 0x0004
#define IPv6_ADDR_LOOPBACK 0x0010
#define IPv6_ADDR_LINKLOCAL 0x0200
#define IPv6_ADDR_SITELOCAL 0x0400
#define IPv6_ADDR_COMPATv4 0x0080
#define IPv6_ADDR_SCOPE_MASK 0x00f0
#define IPv6_ADDR_MAPPED 0x10000
#define IPv6_ADDR_RESERVED 0x20000 /* reserved address space */
```
