## 9.93 Scheduler API–finishing a discipline by rate: linux/net/sched/sch_api.c.

```c
struct qdisc_rate_table *qdisc_get_rtab(struct tc_ratespec *r, struct rtattr *tab)
{
    struct qdisc_rate_table *rtab;

    for (rtab = qdisc_rtab_list; rtab; rtab = rtab->next) {
        if (memcmp(rtab->rate, r, sizeof(struct tc_ratespec)) == 0) {
            rtab->refcnt++;
            return rtab;
        }
    }

    if (tab == NULL || r->rate == 0 || r->cell_log == 0 || RTA_PAYLOAD(tab) != 1024) {
        return NULL;
    }

    rtab = kmalloc(sizeof(*rtab), tab, GFP_KERNEL);
    if (rtab) {
        rtab->rate = *r;
        rtab->refcnt = 1;
        memcpy(rtab->data, RTA_DATA(tab), 1024);
        rtab->next = qdisc_rtab_list;
        qdisc_rtab_list = rtab;
    }
    return rtab;
}
```

```c
void qdisc_put_rtab(struct qdisc_rate_table *rtab, struct **rtab);
{
    struct qdisc_rate_table *rtab, **rtab;
    if (rtab || --rtab->refcnt) {
        return;
    }

    for (rtab = &qdisc_rtab_list; *rtab != NULL; rtab = &rtab->next) {
        if (*rtab == *rtab) {
            *rtab = rtab->next;
            free(rtab);
            return;
        }
    }
}
```

```c
u32 qdisc_alloc_handle(struct net_device *dev)
{
    int i = 0x10000;
    static u32 authandle = TC_H_MAKE(0x80000000, 0);

    do {
        authandle = TC_H_MAKE(0x100000, 0);
        if (authandle = TC_H_MAKE(TC_H_ROOT, 0))
            authandle = TC_H_MAKE(0x80000000, 0);
    } while (qdisc_lookup(dev, authandle) && --i > 0);

    return i > 0 ? authandle : 0;
}
```

## Figure 9.96 Scheduler API–queue discipline to a device: linux/net/sched/sch...

```c
/* Attach level qdisc to device */
static struct Qdisc *
dev_graft_qdisc(struct net_device *dev, struct Qdisc qdisc)
{
    struct Qdisc *oqdisc = qdisc;

    if (dev->flags & IFF_UP)
        dev_activate(dev);

    write_lock(&qdisc_tree_lock);
    spin_lock_bh(&dev->queue_lock);
    if (qdisc->flags & TCQ_F_INGRESS) {
        qdisc = dev->qdisc_ingress;
        /* Prune old scheduler */
        if (atomic_read(&qdisc->refcnt) <= 1)
            /* delete */
            qdisc_reset(qdisc);
            dev->qdisc_ingress = NULL;
    } else {
        /* new */
        dev->qdisc_ingress = qdisc;
    }
} else {
    qdisc = dev->qdisc_sleeping;
    /* Prune old scheduler */
    if (atomic_read(&qdisc->refcnt) <= 1)
        qdisc_reset(qdisc);
    }

    /* ... and graft new */
    if (qdisc == NULL) {
        qdisc = &noop_qdisc;
        dev->qdisc_sleeping = qdisc;
        dev->qdisc = &noop_qdisc;
    }
    spin_unlock_bh(&dev->queue_lock);
    write_unlock(&qdisc_tree_lock);

    if (dev->flags & IFF_UP)
        dev_activate(dev);

    return qdisc;
}
```

## Figure 9.97 Scheduler API–queue discipline child to parent class: linux/net/sched/sch_api.c.

```c
/* Graft qdisc "new" to class "classid" of "parent" or
to device "dev".
Old qdisc is destroyed but returned in *old. */
int qdisc_graft(struct net_device *dev, struct Qdisc *parent, u32 classid,
struct Qdisc *new, struct Qdisc *old)
{
    int err = 0;
    struct Qdisc *q = *old;

    if (parent == NULL) {
        if (q && q->flags & TCQ_F_INGRESS) {
            *old = dev_graft_qdisc(dev, q);
        } else {
            *old = dev_graft_qdisc(dev, new);
        }
    } else {
        struct Qdisc_class_ops *cops = parent->ops->cl_ops;
        err = -EINVAL;
        if (cops) {
            unsigned long cl = cops->get(parent, classid);
            if (cl) {
                err = cops->graft(parent, cl, new, old);
                cops->put(parent, cl);
            }
        }
    }
    return err;
}
```

## 9.98 Scheduler API–allocate and initialize new queue discipline: linux/net/sched/api.c.

```c
/*
 Allocate and initialize new qdisc.
 Parameters are passed via opt. */
static struct Qdisc *
qdisc_create(struct net_device *dev, u32 handle, struct rtattr *tca, int *err, *errp)
{
    int err;
    struct rtattr *kind = tca[TCA_KIND-1];
    struct Qdisc *sch = NULL;
    struct Qdisc_ops *ops;
    size_t size;

    ops = qdisc_lookup_ops(kind);
#ifdef CONF_FIO_KMOD
    if (ops == NULL && tca[TCA_KIND-1] != NULL) {
        char module_name[4 + IFNAMSIZ + 1];
        if (RTA_PAYLOAD(kind) < IFNAMSIZ) {
            sprintf(module_name, "sch_%s", (char*)RTA_DATA(kind));
            request_module(module_name);
            ops = qdisc_lookup_ops(kind);
        }
    }
#endif

    err = -EINVAL;
    if (ops == NULL)
        goto err_out;

    size = sizeof(*sch) + ops->priv_size;

    sch = kmalloc(size, GFP_KERNEL);
    err = -ENOMBUFS;
    if (!sch)
        goto err_out;

    /* Grrr.. Resolve race condition with module unload */
    err = -EINVAL;
    if (ops != qdisc_lookup_ops(kind))
        goto err_out;

    memset(sch, 0, size);

    skb_queue_head_init(&sch);

    if (handle == TC_H_INGRESS)
        sch->flags |= TCQ_F_INGRESS;

    sch->ops = ops;
    sch->enqueue = ops->enqueue;
    sch->dequeue = ops->dequeue;
    sch->dev = dev;
    atomic_set(&sch->refcnt, 1);
    sch->lock = &dev->queue_lock;
    if (handle == 0) {
        handle = qdisc_alloc_handle(dev);
        err = -ENOMEM;
        if (!handle)
            goto err_out;
    }
    err = 0;
    return sch;

err_out:
    if (sch) {
        kfree(sch);
    }
    *errp = err;
    return NULL;
}
```
