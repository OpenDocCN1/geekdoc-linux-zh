```c
    { 1, 2, 2, 2, 1, 2, 0, 0, 1, 1, 1, 1, 1, 1, 1 };

/* 3-band PFIFO queue: old style, but should be a bit faster than
   generic prio+fifo combination. */

static int
    pfifo_fast_enqueue(struct sk_buff *skb, struct qdisc *qdisc)
{
    struct sk_buff_head *list;

    list = ((struct sk_buff_head*)qdisc)->data +
           prioband[skb->priority&TC_PRIO_MAX];

    if (list->qlen < skb->dev->tx_queue_len) {
        __skb_dequeue_tail(list, skb);
        qdisc->qlen++;
        return 0;
    }

    qdisc->stats.drops++;
    kfree(skb);
    return NET_XMIT_DROP;
}
```

```c
struct qdisc *qdisc_create(struct net_device *dev, struct qdisc_ops *ops)
{
    struct qdisc *sch;
    int size = sizeof(*sch) + ops->priv_size;
    sch = kmalloc(size, GFP_KERNEL);
    if (!sch)
        return NULL;
    memset(sch, 0, size);
    skb_queue_head_init(&sch->qsch);
    sch->enqueue = ops->enqueue;
    sch->dequeue = ops->dequeue;
    sch->dev = dev;
    sch->stats.lock = &dev->queue_lock;
    atomic_set(&sch->refcnt, 1);
    if (ops->init || ops->init(sch, NULL) == 0)
        return sch;
    kfree(sch);
    return NULL;
}
```

# 912 Class-Based Queuing

CBQ is a framework, or a filter metadiscipline, within which sub-disciplines are placed. Initially, if a filter runs to classify a packet, which will then be handed to the CBQ discipline—within this discipline, the same filter/classify/queue steps are rerun to decide which subclass the packet belongs to. This can go on as long as you like. The CSZ framework discussed later is similar.
