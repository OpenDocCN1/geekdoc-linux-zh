# The Actual Classifier

# Figure 9.153 CBQ–classifier: linux/net/sched/cbq.c.

/* Classify packet. The procedure is pretty complicated, it allows us to combine link sharing and priority transparently. Namely, you can put link sharing rules (e.g. route based) at root of CBQ, so that it resolves to split nodes. Then packets are classified by logical priority, or a more specific classifier may be attached to the split node. */

```c
static struct cbq_class cbq_classifier(struct sk_buff *skb, struct qdisc *sch)
{
    struct cbq_class_data *q = (struct cbq_class_data*)sch->data;
    struct cbq_class *head = head->next_link;
    struct cbq_class **defmap = NULL;
    struct cbq_class *cl = NULL;
    u32 prio = skb->priority;
    struct tc_result *res;

    /* Step 1. If skb->priority points to one of our classes, use it. */
    if (TC_HJ(prio) == 0 && (cl = cbq_class_lookup(q, prio)) != NULL)
        return cl;

    for (;;) {
        int result = 0;
        defmap = head->defaults;

        /* Step 2. Apply classifier. */
        if (head->filter_list (res = tc_classify(skb, filter_list, &res)) < 0)
            goto fallback;

        if ((cl = (void*)res->class) != NULL) {
            if (TC_HMJ(res->class))
                cl = cbq_class_lookup(q, res->class);
            else if ((cl = defmap[res->class&TC_PRIO_MAX]) != NULL)
                cl = defmap[TC_PRIO_BESTEFFORT];
        }

        if (cl == NULL || cl->level >= head->level)
            goto fallback;
    }
}
```

```c
static struct cbq_class *cbq_reclassify(struct sk_buff *skb, struct cbq_class *this)
{
    struct cbq_class *cl, *new;

    for (cl = this->tparent; cl; cl = cl->tparent)
        if ((new = cl->defaults[TC_PRIO_BESTEFFORT]) != NULL && new != this)
            return new;

return NULL;
}
```

# Using a Class

# Figure 9.156 CBQ–activate or deactivate a class: linux/net/sched/sch_cbq.c.

/* A packet has just been enqueued on the empty class. cbq_activate_class adds it to the tail of active class list for its priority band. */
static __inline_ void cbq_activate_class(struct cbq_class *cl)
{
    struct cbq_sched_data *q = (struct cbq_sched_data*)cl->qdisc->data;
    int prio = cl->priority;
    struct cbq_class *cl_tail;

    cl_tail = q->active[prio];
    q->active[prio] = cl;

    if (cl_tail != NULL) {
        cl->next_alive = cl_tail->next_alive;
        cl_tail->next_alive = cl;
    } else {
        cl->next_alive = cl;
        q->activemask |= (1<<prio);
    }
}
```

# Figure 9.157 CBQ–activate or deactivate a class: linux/net/sched/sch_cbq.c.

/* Unlink class from active chain. Note that this procedure is done directly in round-robin procedure. */
static void cbq_deactivate_class(struct cbq_class *this)
{
    struct cbq_sched_data *q = (struct cbq_sched_data*)this->qdisc->data;
    int prio = cl->priority;
    struct cbq_class *cl;
    struct cbq_class *cl_prev = q->active[prio];

    do {
        cl = cl_prev->next_alive;
        if (cl == this) {
            cl_prev->next_alive = cl->next_alive;
            cl->next_alive = NULL;

            if (cl == q->active[prio]) {
                q->active[prio] = cl_prev;
                if (cl == q->active[prio]) {
                    q->active[prio] = NULL;
                    q->activemask &= ~(1<<prio);
                    return;
                }
            }
            cl = cl_prev->next_alive;
            return;
        }
    } while (cl_prev != q->active[prio]);
}
```

# Marking a Class

# Figure 9.158 CBQ–mark top level: linux/net/sched/sch_cbq.c.

```c
static void cbq_mark_toplevel(struct cbq_sched_data *q, struct cbq_class *cl)
{
    int tolevel = q->toplevel;
    if (toplevel && cl->level && (cl->flags&TC_F_THTTLED)) {
        psched_time_t now;
        psched_diff_t incr;

        PSCHED_GET_TIME(now);
        incr = PSCHED_DIFF_ADD(now, q->rt);
        PSCHED_DIFF_ADD(q->now, incr, now);

        do {
            if (PSCHED_LESS(cl->underflow, now)) {
                q->toplevel = cl->level;
                return;
            }
        } while ((cl->borrow) != NULL && tolevel > cl->level);
    }
}
```
