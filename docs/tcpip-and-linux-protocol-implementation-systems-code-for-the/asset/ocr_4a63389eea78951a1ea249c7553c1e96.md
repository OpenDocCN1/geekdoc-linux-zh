if (icmp->type==ICMP_DEST_UNREACH) {
    switch (icmp->code & 15) {
        case ICMP_NET_UNREACH:
            break;
        case ICMP_HOST_UNREACH:
            break;
        case ICMP_PROTO_UNREACH:
            break;
        case ICMP_PORT_UNREACH:
            break;
        case ICMP_FRAG_NEEDED:
            if (ipv4_config.no_pmtu_disc) {
                if (net_ratelimit()) {
                    printk(KEERN_INFO "ICMP: Xu.Xu.Xu.Xu: fragmentation\nneeded and DF set.\n",
                           NIPQUAD(iph->addr));
                }
            } else {
                unsigned short new_mtu;
                new_mtu = ip_rt_frag_needed(iph, ntohs(icmp->frag_mtu));
                if (new_mtu) {
                    return;
                }
                icmp->frag_mtu = htons(new_mtu);
            }
            break;
        case ICMP_SR_FAILED:
            if (net_ratelimit()) {
                printk(KEERN_INFO "ICMP: Xu.Xu.Xu.Xu: Source Route Failed.\n\t\t\t\t                       NIPQUAD(iph->addr));
            }
            break;
        default:
            break;
    }
    if (icmp->code>NR_ICMP_UNREACH) {
        return;
    }
}
