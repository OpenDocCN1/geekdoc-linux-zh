224        if (d > 1) {
225            sfq_index x = q->dep[d+SFQ_DEPTH].next;

228            kfree_skb(skb);
229            sfq_dec(q, d);

231            sch->stats.drops++;
232            return 1;
233        }
234
235        if (d == 1) {
236            /* It is difficult to believe, BUT ALL SLOTS HAVE LENGTH 1. *//
237            d = q->next(q)->tail;
238            q->next(tail) = q->next[d];
239            q->alloc(q->next[d]) += q->quantum;
240            skb = q->qs[d].prev;
241            _skb_unlink(skb, &q->qs[d]);
242            kfree_skb(skb);
243            sfq_dec(q, d);
244            sch->q_len--;
245            q->ht(q->hash[d]) = SFQ_DEPTH;
246            sch->stats.drops++;
247            return 1;
248        }
249
250        return 0;
251    }
