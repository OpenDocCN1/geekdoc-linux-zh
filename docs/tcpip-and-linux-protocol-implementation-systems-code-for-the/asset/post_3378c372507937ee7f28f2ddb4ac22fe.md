```c
static int cbq_delete(struct qdisc *sch, unsigned long arg)

    struct cbq_sched_data *q = (struct cbq_sched_data*)sch->data;
    struct cbq_class *cl = (struct cbq_class*)arg;

    if (cl->filters || cl->children || cl == &q->link)
        return -EBUSY;

    sch_tree_lock(sch);

    if (cl->next_alive)
        cbq_deactivate_class(cl);

    if (q->tx_borrowed == cl) {
        q->tx_borrowed = q->tx_class;
    }
    if (q->tx_class == cl) {
        q->tx_class = NULL;
        q->tx_borrowed = NULL;
    }
#ifdef CONFIG_NET_CLS_POLICE
    if (q->rx_class == cl)
        q->rx_class = NULL;
#endif

    cbq_unlink_class(cl);
    cbq_adjust_levels(cl->tparent);
    cl->defmap = 0;
    cbq_sync_defmap(cl);

    cbq_mprio(q, cl);
    sch_tree_unlock(sch);

    if (--cl->refcnt == 0) {
        cbq_destroy_class(cl);
    }

    return 0;
}
```

# Clark/Shenker/Zhang Scheduling

Clark, Shenker, and Zhang [19] suggested a different structure for a hierarchical classification architecture.

```c
/*
 * Clark/Shenker/Zhang
 * ============================
 * Source: David D. Clark, Scott Shenker and Lix Zhang
 * "Supporting Real-Time Applications in
 * Integrated Services: Architecture and Mechanism"
 *
 * CSZ presents a round-robin approach
 * the main idea is WFQ for each class
 * with priority services and the rest
 * to best effort traffic. It is proposed that shaping
 * at intermediate hops will introduce delays and jitter.
 *
 * Note that temporary overlimits are inevitable because
 * real links are not ideal, and the real algorithms must take
 * into account.
 */
```
