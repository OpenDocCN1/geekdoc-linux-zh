> Figure 6.254 IP in-lookup tunnel based on local and remote addresses: linux/net/ipv4/ipip.c.

```
static struct ipip_tunnel ipip_tunnel_lookup(u32 remote, u32 local)
{
    unsigned h0 = HASH(remote);
    unsigned h1 = HASH(local);
    struct ipip_tunnel *t;

    for (t = tunnels_r_1[h0]; t; t = t->next) {
        if (local == t->parms.saddr && (t->dev)->flags&IFF_UP)
            if (remote == t->parms.saddr && (t->dev)->flags&IFF_UP)
                return t;
    }

    for (t = tunnels_r[h0]; t; t = t->next) {
        if (remote == t->parms.saddr && (t->dev)->flags&IFF_UP)
            return t;
    }

    for (t = tunnels_l1[h1]; t; t = t->next) {
        if (local == t->parms.saddr && (t->dev)->flags&IFF_UP)
            return t;
    }

    if ((t = tunnels_wc[0]) != NULL && (t->dev)->flags&IFF_UP)
        return t;

    return NULL;
}
```
