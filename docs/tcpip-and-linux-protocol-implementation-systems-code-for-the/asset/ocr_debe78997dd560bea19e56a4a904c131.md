880
881     u32 ip = ((struct sockaddr_in *)&r->arp_pa)->sin_addr.s_addr;
882     struct neighbour *neigh;
883     int err = -ENXIO;
884
885     neigh = neigh_lookup(arp_tbl, ip, dev);
886     if (neigh) {
887         read_lock_bb(&neigh)->lock);
888         memcpy(r->arp_ha.sa_data, neigh->ha_dev, dev->ha_len);
889         r->arp_flags = arp_state_to_flags(neigh);
890         unlock_bb(&neigh)->lock);
891         r->arp_ba.sa_family = dev->type;
892         strncpy(r->arp_dev, dev->name, sizeof(r->arp_dev));
893         neigh_release(neigh);
894         err = 0;
895     }
896     return err;
