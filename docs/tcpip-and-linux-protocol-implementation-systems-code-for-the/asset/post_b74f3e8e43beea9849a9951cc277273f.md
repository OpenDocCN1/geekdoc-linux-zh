/*
 * This function is called by a protocol handler that wants to
 * remove its address family, and have unlinked from the
 * SOCKET module.
 */
int sock_unregister(int family)
{
    if (family < 0 || family >= NPROTO)
        return -1;

    net_family_lock();
    net_families[family]=NULL;
    net_family_unlock();
    return 0;
}
