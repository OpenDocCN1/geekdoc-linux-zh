# Figure 9.291 SFQ-queue decrement/increment functions: linux/net/sched/sch_
sfq.c.

```
extern __inline_ void sfq_link(struct sfq_sched_data *q, sfq_index x)
{
    sfq_index p, n;
    int d = q->qs[x].qlen + SFQ_DEPTH;

    p = d;
    n = q->dep[d].next;
    q->dep[x].next = n;
    q->dep[x].prev = p;
    q->dep[p].next = q->dep[n].prev = x;
}

extern __inline_ void sfq_dec(struct sfq_sched_data *q, sfq_index x)
{
    sfq_index p, n;
    n = q->dep[x].next;
    p = q->dep[x].prev;
    q->dep[p].next = n;
    q->dep[n].prev = p;

    if (n == p && q->max_depth == q->qs[x].qlen + 1) {
        q->max_depth--;
        sfq_link(q, x);
    }
}

ext __inline_ void sfq_inc(struct sfq_sched_data *q, sfq_index x)
{
    sfq_index p, n;
    int d;

    n = q->dep[x].next;
    p = q->dep[x].prev;
    q->dep[p].next = n;
    q->dep[n].prev = p;
    d = q->qs[x].qlen;
    if (q->max_depth < d) {
        q->max_depth = d;
    }
    sfq_link(q, x);
}
```
