static void ingress_destroy(struct qdisc *sch)
{
    struct ingress_qdisc_data *p = PRIV(sch);
    struct tcf_proto *tp;

    D_PRINTK("ingress_destroy(%xp, [qdisc %xp])\n", sch, p);
    while (p->filter_list) {
        tp = p->filter_list;
        p->filter_list = tp->next;
        tp->ups->destroy(tp);
