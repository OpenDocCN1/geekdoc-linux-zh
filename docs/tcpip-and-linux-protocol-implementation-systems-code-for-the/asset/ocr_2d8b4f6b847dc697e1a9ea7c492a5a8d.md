```
int ip_route_output_key(struct rtable **rp, const struct rt_key_key *key)
{
    unsigned hash;
    struct rtable *rth;
    hash = rt_hash_code(key->dst, key->src)(key->oif<5), key->tos);
    read_lock_bh(&rt_hash_table[hash].lock);
    for (rth=rt_hash_table[hash].head; rth; rth=rth->u.rt_next) {
        if (rth->key.dst == key->dst && rth->key.src == key->src &&
            rth->key.oif == 0 &&
            rth->key.oif == key->oif &&
#ifdef CONFIG_IP_ROUTE_FWMARK
            rth->key.fmark == key->fmark &&
#endif
            if ((key->tos& (ITOS|RT_MASK|RTO_ONLINK)) &&
                (key->tos&TPROXY) || (rth->rt_flags&RTCF_TPROXY)) {
                rth->u.dst.last = jiffies;
                dst_lookup(&rth->u.dst, u->dst);
                rth->u.dst..use++;
                read_unlock_bh(&rt_hash_table[hash].lock);
                *rp = rth;
                return 0;
            }}
    }
}
```
