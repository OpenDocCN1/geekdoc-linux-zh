static struct ip_tunnel **ipgre_bucket(struct ip_tunnel *t) {
    u32 remote = t->parms.iph.daddr;
    u32 local = t->parms.iph.saddr;
    u32 key = t->parms.i_key;
    unsigned h = HASH(key);
    int prio = 0;
    if (local) {
        prio = 1;
        if (remote && MULTICAST(remote)) {
            prio = 2;
        }
        h = HASH(remote);
    }
    return tunnels[prio][h];}
