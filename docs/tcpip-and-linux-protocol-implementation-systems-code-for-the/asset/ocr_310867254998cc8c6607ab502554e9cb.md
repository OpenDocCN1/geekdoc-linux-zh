```
struct qdisc *qdisc_create_dft(struct net_dev, struct qdisc_ops *ops)
{
    struct qdisc *sch;
    int size = sizeof(*sch) + ops->priv_size;
    sch = kmalloc(size, GFP_KERNEL);
    if (!sch)
        return NULL;
    memset(sch, 0, size);
    skb_queue_head_init(&sch->qsch);
    sch->ops = ops->enqueue;
    sch->dequeue = ops->dequeue;
    sch->dev = dev;
    sch->stats.lock = &dev->queue_lock;
    atomic_set(&sch->refcnt, 1);
    if (ops->init || ops->init(sch, NULL) == 0)
        return sch;
    kfree(sch);
    return NULL;
}
```

# 9.12 Class-Based Queuing

CBQ is a framework, or if you filter a metadiscipline, within which sub-disciplines are placed. Initially, if a filter will run to classify a packet, which will then be handed to the CBQ discipline—within this discipline, the same filter/classify/queue steps are rerun to decide which subclass the packet belongs to. This can go on as long as you like. The CSZ framework discussed later is similar.
