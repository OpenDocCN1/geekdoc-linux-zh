# Send an ARP

Figure 6.119 ARP--send an ARP message: linux/net/ipv4/arp.c.

```c
/*
 * Create and send an arp packet. if (dest_hw == NULL), we create a broadcast
 * message.
 */
void arp_send(int itype, int ptype, u32 dest_ip);
    struct net_device *dev, u32 src_ip;
    unsigned char *dest_hw, unsigned char *src_hw,
    unsigned char *target_hw
{
    struct sk_buff *skb;
    struct arphdr *arp;
    unsigned char *arp_ptr;
    /*
     * No arp on this interface.
     */
    if (dev->flags&IFF_NOARP)
        return;
    /*
     * Allocate a buffer
     */
    skb = alloc_skb(sizeof(struct arphdr) + 2*(dev->addr_len+4)
        + dev->hard_header_len + 15, GFP_ATOMIC);
    if (skb == NULL)
        return;
    skb_reserve(skb, dev->hard_header_len+15);
    skb->raw = skb->data;
    arp = (struct arphdr *)skb_put(skb, sizeof(struct arphdr) + 2*(dev->addr_len+4));
    arp->dev = dev;
    skb->protocol = constant_htons(ETH_P_ARP);
    if (src_hw == NULL)
        src_hw = dev->dev_addr;
    if (dest_hw == NULL)
        dest_hw = dev->broadcast;
```

/*
 * Fill out the arp protocol part.
 *
 * The arp hardware type should match the device type, except for FDDI,
 * which (according to RFC 1390) should always equal 1 (Ethernet).
 */

/*
 * Exceptions everywhere. AX.25 uses the AX.25 PID value not the
 * DIX code for the protocol. Make these device structure fields.
 */
switch (dev->type) {
    default:
        arp->ar_hr = htons(dev->type);
        arp->ar_pro = constant_htons(ETH_P_IP);
        break;

#if defined (CONFIG_AX25) || defined (CONFIG_AX25_MODULE)
    case ARPHD_AX25:
        arp->ar_hr = constant_htons(ARPHD_AX25);
        arp->ar_pro = constant_htons(AX25_P_IP);
        break;

#if defined (CONFIG_NETROM) || defined (CONFIG_NETROM_MODULE)
    case ARPHD_NETROM:
        arp->ar_hr = constant_htons(ARPHD_NETROM);
        arp->ar_pro = constant_htons(AX26_P_IP);
        break;
#endif
#endif

#if defined (CONFIG_FDDI)
    case ARPHD_FDDI:
        arp->ar_hr = constant_htons(ARPHD_ETHER);
        arp->ar_pro = constant_htons(ETH_P_IP);
        break;
#endif

#if defined (CONFIG_TR)
    case ARPHD_IEEE802_TR:
        arp->ar_hr = constant_htons(ARPHD_IEEE802);
        arp->ar_pro = constant_htons(ETH_P_IP);
        break;
#endif
}

int arp_rcv(struct sk_buff *skb, struct net_device *dev, struct packet_type pt)
{
    struct arphdr *arp = skb->data;
    unsigned char *arp_ptr = (unsigned char *)(arp+1);
    struct rtable *rt;
    unsigned char *sha, *tha;
    u32 ip, tip;
    u16 dev_type = dev->type;
    int addr_type;
    struct in_device *in_dev = in_dev_get(dev);
    struct neighbour *neigh;
    /*
     * The hardware length of the packet should match the hardware length
     * of the device. Similarly, the hardware types should match.
     * device should be ARP-able. Also, if len is not 4, then
     * is not an IP number. We can't currently handle this, so toss
     * it.
     */
    if (in_dev == NULL ||
        arp->ar_bln != dev->addr_len ||
        dev->flags & IFF_NOARP ||
        skb->pkt_type != PACKET_ARP ||
        skb->pkt_type != PACKET_LOOPBACK ||
        arp->ar_pl == 4)
        goto out;

    if ((skb_share_check(skb, GFP_ATOMIC)) == NULL)
        goto out_of_mem;
