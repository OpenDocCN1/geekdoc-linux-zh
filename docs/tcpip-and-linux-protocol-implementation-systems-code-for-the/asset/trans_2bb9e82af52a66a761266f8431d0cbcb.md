## 图 5.10 软中断——处理器 i：linux/kernel/softirq.c。

```c
static struct softirq_action softirq_vec[32] __cacheline_aligned;
asmlinkage void do_softirq() {
	int cpu = smp_processor_id();
	unsigned int active, mask;

if (in_interrupt())
    return;

local_bh_disable();

local_irq_disable();
    mask = softirq_mask(cpu);
    active = softirq_active(cpu) & mask;

if (active) {
    struct softirq_action *h;
    restart:

    /* 在启用中断前重置活动位 */
    softirq_active(cpu) &= active;
    local_irq_enable();

    h = softirq_vec;
    mask &= active;

    do {
        if (active & 1)
            h->action(h);
        h++;
        active >>= 1;
    } while (active);
}
```

这部分随后确保了我们不会嵌套硬中断和软中断，否则会导致糟糕的后果。

## 协议实现

本节启动了软中断向量。

图 5.12 软中断——设置：linux/kernel/softirq.c。

```c
static spinlock_t softirq_mask_lock = SPIN_LOCK_LOCKED;

void open_softirq(int nr, void (*action)(struct softirq_action), void *data) {
     unsigned long flags;
     int i;

     spin_lock_irqsave(&softirq_mask_lock, flags);
     softirq_vec[nr].data = data;
     softirq_vec[nr].action = action;

     for (i=0; i<NR_CPUS; i++)
        softirq_mask[i] |= (1<<nr);
     spin_unlock_irrestore(&softirq_mask_lock, flags);
}
```
