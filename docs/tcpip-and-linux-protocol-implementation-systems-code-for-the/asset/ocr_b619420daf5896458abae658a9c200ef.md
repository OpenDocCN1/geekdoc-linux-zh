static int teql_master_mtu(struct net_dev *dev, int new_mtu)
{
  struct teql_master *m = (void*)dev->priv;
  struct Qdisc *q;
  if (new_mtu < 68)
    return -EINVAL;
  q = m->slaves;
  if (q) {
    do {
      if (new_mtu > q->dev->mtu)
        return -EINVAL;
    } while ((q=NEXT_SLAVE(q)) != m->slaves);
  }
  dev->mtu = new_mtu;
  return 0;
}
