# Figure 9.272 tbf-reset: linux/net/sched/sch_tbf.c.

```
static void
btbf_reset(struct disc *sch)
{
  struct tbf_sched_data *q = (struct tbf_sched_data *)sch->data;
  skb_queue_purge(&sch->q);
  sch->stats.backlog = 0;
  PSCHED_GET_TIME(q->t,c);
  q->tokens = q->buffer;
  q->tokens = q->mtu;
  sch->flags &= TCQ_F_THROTTLED;
  del_timer(&q->timer);
}
```
