static void
fn_hash_select_default(struct fib_table *tb, const struct rt_key *key, struct fib_result *res)
{
    int order, last_idx;
    struct fib_node *fi;
    struct fib_info *fi = NULL;
    struct fib_info *last_resort;
    struct fn_hash *hash = (struct fn_hash)tb->hash_data;
    struct fn_zone *fz = t->fn_zones[0];

    if (fz == NULL)
        return NULL;

    last_idx = -1;
    last_resort = NULL;
    order = -1;

    read_lock(&fib_hash_lock);
    for (f = fz->fn_hash[0]; f = f->fn_next(f)) {
        struct fib_info *next_fi = FIB_INFO(f);

        if ((f->fn_state&FN_ZOMBIE) ||
            f->fn_scope != res->scope ||
            f->fn_type != RTN_UNICAST)
            continue;

        if (next_fi->fib_priority > res->fi->fib_priority)
            break;
        if (!next_fi->fib_nh[0].nh_gw || next_fi->fib_nh[0].scope != RT_SCOPE_LINK)
            continue;
        f->fn_state = FN_ACCESSED;

        if (fi == NULL)
            if (next_fi != res->fi)
                break;
        else if (fi->fib_detect_death(fi, &last_resort, &last_idx)) {
            if (res->fi)
                fib_info_put(res->fi);
            fi = fi;
            res->fi = fi;
            atomic_inc(&fi->clntref);
            fn_hash_last_dflit = order;
            goto out;
        }
        fi = next_fi;
        order++;
}
