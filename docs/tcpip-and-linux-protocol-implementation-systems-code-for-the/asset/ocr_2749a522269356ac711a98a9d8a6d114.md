int ipmr_mfc_delete(struct mfctl *mfc)
{
    int line;
    struct mfc_cache *c, **cp;

    line = HASH(mfc->mfc_mcastgrp.s_addr, mfc->mfc_origin.s_addr);
    for (cp = mfc_cache_array[line]; *cp != NULL; cp = &(*cp)->next) {
        if (c->mfc_origin == mfc->mfc_origin.s_addr &&
            c->mfc_mcastgrp == mfc->mfc_mcastgrp.s_addr) {
            write_lock_bh(mrt_lock);
            *cp = c->next;
            write_unlock_bh(mrt_lock);
        }
    }

    kmem_cache_free(mrt_cachep, c);
    return 0;
}
return -ENOENT;
