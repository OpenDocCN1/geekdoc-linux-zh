## Nonfragmented Transmission

It is common that the transport layer used MTU to find the maximum packet size tolerable on a path that fragmentation subsequently is not necessary. In this case, we want superfast code to handle the situation.

/*
 * Fast path for unfragmented packets.
 */
int ip_build_xmit(struct sock *sk,
    int getfrag(const void *,
    char *,
    unsigned int,
    unsigned int),
    const void *frag,
    unsigned length,
    struct ip_cookie *ipc,
    struct rtcam *rt,
    int flags);
{
    int err;
    struct sk_buff *skb;
    int df;
    struct iphdr *iphdr;

    /*
     * Try the simple case first. This leaves fragmented frames, and by
     * choice RAW frames within 20 bytes of maximum size (rare) to the long path
     */

    /* Check for slow path.
     */
    if (sk->protoinfo->af_inet->hdrnl) {
        length += sizeof(struct iphdr);
        if (length > rt->u.dest.mtu || ipc->opt != NULL)
            return ip_build_xmit_slow(sk, getfrag, frag, length, ipc, rt, flags);
    else {
        if (length > rt->u.dest.mtu)
            local_error(sk, EMGSIZE, rt->rt_dst, rt->sk_port, rt->u.dev->mtu);
