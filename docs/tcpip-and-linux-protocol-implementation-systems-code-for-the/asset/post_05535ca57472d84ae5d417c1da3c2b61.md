```c
/*
 * This IP datagram is too large to be sent in one piece. Break it up into
 * smaller pieces (each of size equal to IP header length
 * a block of the original IP data part) that will fit in
 * single device frame, and queue a frame for sending.
 * Yes this is inefficient, submit to a quicker one.
 */
int ip_fragment(struct sk_buff *skb, int (*output)(struct sk_buff *))
{
    struct iphdr *iphdr;
    unsigned char *raw;
    unsigned char *ptr;
    struct net_device *dev;
    struct sk_buff *skb2;
    unsigned int mtu, blen, left;
    int offset;
    int not_last_frag;
    struct rtable *rt = (struct rtable*)skb->dst;
    int err = 0;

    dev = rt->u.dst.dev;

    /*
     * Point into the IP datagram header.
     */
    raw = skb->nh_raw;
    iph = (struct iphdr*)raw;
```

```c
    /*
     * Setup starting values.
     */
    hlen = iph->ihl * 4;
    left = ntohs(iph->len) - hlen; /* Space per frame */
    mtu = rt->u.dst.mtu - hlen; /* Size of data */
    ptr = raw + hlen; /* Where to start */

    /*
     * Fragment the datagram.
     */
    offset = (ntohs(iph->frag_off) & IP_OFFSET) << 3;
    not_last_frag = iph->frag_off & htons(IP_MF);

    /*
     * Keep copying data until we out.
     */
    while(left > 0) {
        len = left;
        /* IP: it doesn't fit, use 'mtu' - the data left */
        if (len > mtu) {
            len = mtu;
            /* IP: we are not sending upto the packet end
               then align the next start on an eight byte boundary */
            if (len < left) {
                len &= ~7;
            }
        }
        /*
         * Allocate buffer.
         */
        if ((skb2 = alloc_skb(hlen+dev->hard_len+15, GFP_ATOMIC)) == NULL) {
            printk(KERN_INFO "IP: frag: no memory for new fragment!\n");
            err = -ENOMEM;
            goto fail;
        }

        /*
         * Set up data on packet
         */
        skb2->pkt_type = skb->pkt_type;
        skb2->priority = skb->priority;
        skb_reserve(skb2, (dev->hard_len-15) & ~15);
        skb2_put(skb2, len + hlen);
        skb2->nh_raw = skb2->data;
        skb2->h_raw = skb2->data + hlen;

        /*
         * Copy the memory for the fragment to any owner it might possess.
         */
        if (skb->sk)
            skb_set_owner_w(skb2, skb->sk);
        skb2->dst = dst_clone(skb->dst);
        skb2->dev = skb->dev;

        /*
         * Copy the packet header into the new buffer.
         */
        memcpy(skb2->nh_raw, raw, hlen);

        /*
         * Copy a block of the IP datagram.
         */
        memcpy(skb2->h_raw, ptr, len);
        len = len;
```

```c
/*
* Fetch data from kernel space and fill in checksum if needed.
 */
static int reply_glue_bits(const void *dptr, char *to, unsigned int offset,
                                unsigned fraglen)
{
    struct ip_reply_arg *dptr = (struct ip_reply_arg)dptr;
    u16 *pkpt = (u16)to;
    struct iov *iov;
    int len;
    int hdrflag = 1;
    iov = &dptr->iov[0];
    if (offset > iov->iov_len) {
        offset = iov->iov_len;
        iov++;
        hdrflag = 0;
    }
    len = iov->iov_len; // /* overlapping. */
    dp->csum = csum_partial_copy_nocheck(iov->iov_base+offset, to, len,
                               dp->csum);
    offset = 0;
    fraglen -= len;
    to += len;
    iov++;
    dp->csum = csum_partial_copy_nocheck(iov->iov_base+offset, to, fraglen,
                               dp->csum);

    if (hdrflag && dp->csumoffset) {
        *(pkpt + dp->csumoffset) = csum_fold(dp->csum); /* fill in checksum */
    }
    return 0;
}
```
