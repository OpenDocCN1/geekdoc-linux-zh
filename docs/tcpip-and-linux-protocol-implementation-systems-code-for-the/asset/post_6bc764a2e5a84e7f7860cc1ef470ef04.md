Every discipline has two major routines: enqueue and dequeue.

### ---dequeue ---

dequeue usually returns a skb to send. It is allowed to return NULL, but it does not mean that queue is empty, it just means that discipline does not want to send anything this time.

Queue is empty if q->qq.len == 0. For complicated disciplines with multiple q->qq.len is not real packet queue, however if q->qq.len must be valid.

### ---enqueue ---

enqueue returns 0, if packet was enqueued successfully. If packet (this one or another) was dropped, it returns zero error code.

However, modifications to data, participating in scheduling must be additionally protected by dev->queue_lock spinlock.

The idea is the following:
- enqueue, dequeue are serialized via top level device spinlock dev->queue_lock.
- tree walking is protected by read_lock(qdisc_tree_lock) and this lock is used only in process context.
- updates to tree are made only under rtl semaphore, hence this lock may be made without local bh disabling.

qdisc_tree_lock must be GRABBED BEFORE dev->queue_lock!

```c
int qdisc_restart(struct net_device *dev)
{
    struct qdisc *q = dev->qdisc;
    struct sk_buff *skb;

    /* Dequeue packet */
    if ((skb = q->dequeue(q)) != NULL) {
        if (spin_try_lock(&dev->xmit_lock)) {
            /* Remember that the driver is grabbed by us */
            dev->xmit_owner = processor_id();
            /* Release queue */
            spin_unlock(&dev->queue_lock);
            if (netif_queue_stopped(dev)) {
                dev_queue_xmit(skb, dev);

                if (dev->hard_start_xmit(skb, dev) == 0) {
                    dev->xmit_owner = -1;
                    spin_unlock(&dev->xmit_lock);
                    return -1;
                }
            }
        }
    }
    /* Release the driver */
    dev->xmit_owner = -1;
    spin_unlock(&dev->xmit_lock);
    spin_lock(&dev->queue_lock);
    q = dev->qdisc;
} else {
    /* So, someone grabbed the driver.
    /* It may be transient error,
    when hard_start_xmit() recurses. We detect it by checking xmit owner and drop
    packet when detected. */
    if (dev->xmit_owner == processor_id()) {
        kfree_skb(skb);
        if (!ratelimit())
            printk(KERN_DEBUG, "Dead loop on netdevice %s, fix it urgently!\n", dev->name);
        return -1;
    }
    dev->xmit_stat[processor_id()].collision++;
}
```

```c
static void dev_watchdog(unsigned long arg)
{
    struct net_device *dev = (struct net_device*)arg;

    spin_lock(&dev->xmit_lock);
    if (dev->qdisc != NULL_qdisc) {
        if (netif_dev_present(dev)) &&
        netif_running(dev) &&
        netif_carrier_ok(dev)) {
            if (dev_queue_stopped(dev) > dev->watchdog_timeout) {
                if (jiffies - dev->hard_start_xmit(dev) > dev->watchdog_timeout)
                    printk(KERN_INFO "NETDEV WATCHDOG: %s: timed out\n", dev->name);
                    dev_tx_timeout(dev);
            }
        }
        if (!mod_timer(dev->watchdog_timer, jiffies + dev->watchdog_timeout))
            dev_hold(dev);
    }
    spin_unlock(&dev->xmit_lock);
    dev_put(dev);
}

static void dev_watchdog_init(struct net_device *dev)
{
    init_timer(dev->watchdog_timer);
    dev->watchdog_timer.data = (unsigned long)dev;
    dev->watchdog_timer.function = dev_watchdog;
}

void __netdev_watchdog_up(struct net_device *dev)
{
    if (dev->xmit_timeout) {
        if (dev->watchdog_timeout <= 0)
            dev->watchdog_timeout = 5HZ;
        if (mod_timer(&dev->watchdog_timeout, jiffies + dev->watchdog_timeout))
            dev_hold(dev);
    }
}

static void dev_watchdog_up(struct net_device *dev) {
    spin_lock_bh(&dev->xmit_lock);
    __netdev_watchdog_up(dev);
    spin_unlock_bh(&dev->xmit_lock);
}

static void dev_watchdog_down(struct net_device *dev) {
    spin_lock_bh(&dev->xmit_lock);
    if (dev->watchdog_timer)
        __dev_put(dev);
    spin_unlock_bh(&dev->xmit_lock);
}
```
