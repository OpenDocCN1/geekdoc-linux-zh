```c
int qdisc_new_estimator(struct tc_stats *stats, struct rattr *opt)
{
    struct qdisc_estimator *est = *opt;
    struct tc_estimator *param = RTA_DATA(opt);

    if (RTA_PAYLOAD(opt) < sizeof(*param))
        return -EINVAL;

    if (param->interval < -2 || param->interval > 3)
        return -EINVAL;

    est = kmalloc(sizeof(*est), GFP_KERNEL);
    if (est == NULL)
        return -ENOMFS;

    memset(est, 0, sizeof(*est));
    est->interval = param->interval + 2;
    est->stats = stats;
    est->ewma_log = param->ewma_log;
    est->last_bytes = stats->bytes;
    est->avbps = stats->bps<<5;
    est->last_packets = stats->packets;
    est->avpps = stats->pps<<10;

    est->next = elist(&est->interval).list;
    if (est->next == elist(&est->interval).timer)
        init_timer(&elist[est->interval].timer);
    elist[est->interval].timer.data = est->interval;
    elist[est->interval].timer.expires = jiffies + ((HZ/4)<est->interval);
    elist[est->interval].timer.function = est_timer;
```

```c
int tcf_police(struct sk_buff *skb, struct tcf_police *sp)
{
  psched_time_t now;
  long toks;
  long ptoks = 0;
  spin_lock(&sp->lock);
  sp->stats->bytes += skb->len;
  sp->stats->packets++;
#ifdef CONFIG_NET_ESTIMATOR
  if (sp->stats->avbps >= sp->stats->limits)
    sp->stats->overlimits;
#endif
  if (skb->len <= sp->mtu)
    if (sp->R_tab == NULL) {
      spin_unlock(&sp->lock);
      return sp->action;
    }

  PSCHED_GET_TIME(now);

  toks = PSCHED_DIFF_SAFE(now, sp->t_c, sp->burst, 0);

  if ((toks-ptoks) >= 0) {
    sp->t_c = now;
    sp->toks = toks;
    sp->ptoks = ptoks;
    spin_unlock(&sp->lock);
    return sp->result;
  }

  if (sp->P_tab) {
    ptoks = toks + sp->ptoks;
    if (ptoks > (long)L2T_P(sp, sp->mtu))
        ptoks = (long)L2T_P(sp, sp->mtu);
    toks -= L2T(sp, skb->len);

    if ((toks-ptoks) >= 0) {
      sp->t_c = now;
      sp->toks = toks;
      sp->ptoks = ptoks;
      spin_unlock(&sp->lock);
      return sp->result;
    }
  }
  toks += sp->ptoks;
  if (toks > (long)sp->burst)
    toks = sp->burst;
  toks -= L2T(sp, skb->len);

  sp->stats->overlimits++;
  spin_unlock(&sp->lock);
  return sp->action;
}
```
