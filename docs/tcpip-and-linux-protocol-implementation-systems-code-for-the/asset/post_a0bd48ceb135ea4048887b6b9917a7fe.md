struct ip_ra_chain *ip_ra;

ip_ra_control(struct sock *sk, unsigned char ra, void (*destructor)(struct sock *))
{
	struct ip_ra_chain *new_ra, *rap;

	if (sk->type == SOCK_RAW || sk->num == IPPROTO_RAW)
		return -EINVAL;
	new_ra = kmalloc(sizeof(new_ra), GFP_KERNEL);

	write_lock(&ip_ra_lock);
	for (rap = &ip_ra_chain->ra, *rap != NULL; ra = &ra->next) {
		if (ra->sk == sk) {
			if (destructor) {
				unlock_bh(&ip_ra_lock);
				if (new_ra)
					kfree(new_ra);
				return -EADDRUSE;
			}
			rap = ra->next;
			unlock_bh(&ip_ra_lock);

		if (ra->destructor) {
			destructor(sk);
			sock_put(sk);
			kfree(sk);
			return 0;
		}
	}
	if (new_ra == NULL) {
		unlock_bh(&ip_ra_lock);
		return -ENOBUFS;
	}

	new_ra->sk = sk;
	new_ra->destructor = destructor;
	new_ra->next = ra;
	*rap = new_ra;
	sock_hold(sk);
	unlock_bh(&ip_ra_lock);

	return 0;
}

void ip_icmp_error(struct *sk, struct sk_buff *skb, int err,

	struct_externr_skb *serr;

	if (!sk->info.af_inet.recverr)
		return;

	skb = skb_clone(skb, GFP_ATOMIC);
	if (!skb)
		return;

	serr = SKB_EXT_ERR(skb);
	serr->ee.ee_errno = err;
	serr->ee.ee_origin = SO_EE_ORIGIN_ICMP;
	serr->ee_ee_type = skb->h->icmph->type;
	serr->ee.ee_code = skb->h->icmph->code;

	serr->ee.ee_info = info;
	serr->ee.ee_data = 0;
	serr->daddr_from = (unsigned *)&((struct iphdr *) (skb->h->icmph+1)) - skb->nh.raw;
	serr->port = port;

	skb->b.raw = payload;
	skb_pull(skb, payload - skb->data);
