  struct iphdr *old_iph = skb->nh_ip;
  struct iphdr *tip;

  int max_headroom;                           /* The extra header space needed */
  int gre_hlen;
  u32 dt;
  int mtu;

  if (tunnel->recursion++) {
    tunnel->stat.collisions++;
    goto tx_error;
  }

  if (dev->hard_header) {
    gre_hlen = 0;
    tip = (struct iphdr*)skb->data;
  } else {
    gre_hlen = tunnel->hlen;
    tip = tunnel->parms->iph;
  }

  if ((dst = tip->daddr) == 0) {
    /* NEMA tunnel */

      tunnel->stat.tx_fifo_errors++;
      goto tx_error;
    }
  }
