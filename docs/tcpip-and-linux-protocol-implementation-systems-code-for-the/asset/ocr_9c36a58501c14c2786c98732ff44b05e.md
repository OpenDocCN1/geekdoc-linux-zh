Figure 9.268 TBF—requeue and drop: linux/net/sched/sch_tbf.c.

```
static int
btbf_requeue(struct skb_buff *skb, struct Qdisc* sch)
{
  __skb_dequeue_head(&sch->q, skb);
  sch->stats.backlog += skb->len;
  return 0;
}

static int
btbf_drop(struct Qdisc* sch)
{
  struct skb_buff *skb;
  skb = __skb_dequeue_tail(&sch->q);
  if (skb) {
    sch->stats.backlog -= skb->len;
    sch->stats.drops++;
    kfree_skb(skb);
    return 1;
  }
}

```
static void tbf_watchdog(unsigned long arg)
{
  struct Qdisc *sch = (struct Qdisc*)*arg;

  sch->flags &= ~TCQ_F_THROTTLED;
  netif_schedule(sch->dev);
}
