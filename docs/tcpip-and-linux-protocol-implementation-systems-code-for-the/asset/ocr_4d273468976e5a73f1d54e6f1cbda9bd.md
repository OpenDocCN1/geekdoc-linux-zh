# Figure 6.227 IPMR–transmit encapsulated: linux/net/ipv4/ipmr.c.

```
/*
 * Encapsulate a packet by attaching a valid IPIP header to it.
 * This tunnel driver and other moves gives us the speed so
 * important for multicast video.
 */

static void _ip_encap(struct sk_buff *skb, u32 saddr, u32 daddr)
{
    struct iphdr *iph = (struct iphdr *)skb_push(skb,sizeof(struct iphdr));
    iph->version = 4;
    iph->tos = skb->nh->tos;
    iph->ttl = skb->nh->ttl;
    iph->frag_off = 0;
    iph->daddr = daddr;
    iph->saddr = saddr;
    iph->protocol = IPPROTO_IPIP;
    iph->id = 5;
    iph->tot_len = htons(skb->len);
    ip_select_ident(iph, skb->dst);
    ip_send_check(iph);
    skb->h->iph = skb->nh->iph;
    skb->nh->iph = iph;

#ifdef CONFIG_NETFILTER
    nf_conntrack_put(skb->nfct);
    skb->nfct = NULL;
#endif
```

# Figure 6.228 IPMR–forwarding: linux/net/ipv4/ipmr.c.

```
static inline int ipmr_forward_finish(struct sk_buff *skb)
{
    struct dst_entry *dst = skb->dst;

    if (skb->len <= dst->mtu)
        return dst->output(skb);
    else
        return ip_fragment(skb, dst->output);
}
```
