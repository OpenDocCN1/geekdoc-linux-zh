In the previous chapter we saw the generic socket layer for dealing with socket options. Here we the IP protocol-specific part of that function. Other protocol families would have different instances of this code:

/*
 * Socket option code for IP. This is the end of the line after any TCP,UDP options on
 * IP socket.
 */
ip_setsockopt(struct sock *sk, int level, int optname, char *optval, int optlen)
{
 int val=0,err;
 if (level != SOL_IP)
 return -ENOPROTOOPT;
 if ((i(optname & ((1<<IP_PKININFO) | (1<<IP_RECVTTL) |
 (1<<IP_RECVOPTS) | (1<<IP_REVTOS) |
 (1<<IP_RETOPS) | (1<<IP_TOS) |
 (1<<IP_TTL) | (1<<IP_HDRINCL) |
 (1<<IP_MTU_DISCOVER) | (1<<IP_RECKV)) ||
 optname == IP_MULTICAST_TTL |
 optname == IP_MULTICAST_LOOP)
 if (optlen >= sizeof(int)) {
 if (get_user(val, (int *) optval))
 return -EFAULT;
 } else if (optlen >= sizeof(char)) {
 unsigned char ucval;
 if (get_user(ucval, (unsigned char *) optval))
 return -EFAULT;
 val = (int)ucval;
}
