# Figure 8.67 FIB rules—lookup rule:linux/net/ipv4/fib_rules.c.

```
int fib_lookup(const struct rt_key *key, struct fib_result *res)
{
    int err;
    struct fib_rule *r, *policy;
    struct fib_table *tb;

    u32 daddr = key->dst;
    u32 saddr = key->src;

    FRprintk("Lookup %u.%u.%u.%u -> %u.%u.%u.%u\n",
        NIPQUAD(key->dst), NIPQUAD(key->src));
    read_lock(&fib_rules_lock);
    for (r = fib_rules; r; r = r->next) {
        if (((saddr-r->src) & r->src_mask) ||
            (daddr-r->dst) & r->dst_mask) ||
            #ifdef CONFIG_IP_ROUTE_TOS
            (r->tos && r->tos != key->tos) ||
            #endif
            #ifdef CONFIG_IP_ROUTE_FWMARK
            (r->fwmark && r->fwmark != key->fwmark) ||
            #endif
            (r->ifindex && r->ifindex != key->ifindex))
            continue;
    #endif
    FRprintk("tb %d r %d\n", r->table, r->action);
    switch (r->action) {
        case RTN_UNICAST:
        case RTN_NAT:
            policy = r;
            break;
        case RTN_UNREACHABLE:
            read_unlock(&fib_rules_lock);
            return -ENETUNREACH;
        default:
            case RTN_BLACKHOLE:
                read_unlock(&fib_rules_lock);
                return -EINVAL;
            case RTN_PROHIBIT:
                read_unlock(&fib_rules_lock);
                return -EACCES;
    }
}
```
