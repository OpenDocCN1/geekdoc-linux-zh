# Figure 9.204 FIFO queuing scheduler--requeue: linux/net/sched/sch_fifo.c.

```
static int
bfifo_dequeue(struct sk_buff *skb, struct Qdisc* sch)
{
    __skb_queue_head(&sch->q, skb);
    sch->stats.backlog += skb->len;
    return 0;
}

# Figure 9.206 FIFO queuing scheduler--drop: linux/net/sched/sch_fifo.c.

```
static int
fifo_drop(struct Qdisc* sch)
{
    struct sk_buff *skb;
    skb = __skb_dequeue_tail(&sch->q);
    if (skb) {
        sch->stats.backlog -= skb->len;
        kfree(skb);
        return 1;
    }
    return 0;
}
```

# Figure 9.207 FIFO queuing scheduler--reset: linux/net/sched/sch_fifo.c.

```
static void
fifo_reset(struct Qdisc* sch)
{
    __skb_queue_purge(&sch->q);
    sch->stats.backlog = 0;
}
