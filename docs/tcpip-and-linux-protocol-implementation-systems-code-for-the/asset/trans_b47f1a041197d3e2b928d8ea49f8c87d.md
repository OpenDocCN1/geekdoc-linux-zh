# 4.4 使用 Socket API

独立进程之间的通信涉及将数据从一个地址空间移动到另一个地址空间——在过去，这被称为跨越保护边界。由于由于，各种通信设施命名在“域（domain）”一词的混淆用法下（它与域名系统*完全没有*关系）。

图 4.2 中的示例 C 代码说明了 INET 域 socket 的使用，它具有流式语义。这基本上利用了内核中的 TCP 代码——在例例中，通信是在父子进程之间进行的，因此 socket 仅被用来替换管道（pipe）。

**图 4.2** Socket 编程 i—main: pix/sock.c。

```c
#include <stdio.h>
#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <sys/wait.h>
#include <netdb.h>

int main(int argc, char *argv[])
{
    int sockfd, child;
    struct sockaddr_in addr;
    char buffer[128];

    if (argc < 2) {
        printf("usage: %s <port>\n", argv[0]);
        exit(1);
    }

    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0) {
        perror("socket");
        exit(1);
    }

    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(atoi(argv[1]));

    if ((child = fork()) == 0) {
        /* 子进程：客户端 */
        if (connect(sockfd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
            perror("connect");
            exit(1);
        }
        write(sockfd, "Hello from client", 16);
        close(sockfd);
    } else {
        /* 父进程：服务器 */
        if (listen(sockfd, 1) < 0) {
            perror("listen");
            exit(1);
        }
        printf("Server listening on port %s...\n", argv[1]);
        
        socklen_t len = sizeof(addr);
        int client_fd = accept(sockfd, (struct sockaddr *)&addr, &len);
        if (client_fd < 0) {
            perror("accept");
            exit(1);
        }
        
        int n = read(client_fd, buffer, sizeof(buffer) - 1);
        if (n > 0) {
            buffer[n] = '\0';
            printf("Received: %s\n", buffer);
        }
        close(client_fd);
        close(sockfd);
        wait(NULL);
    }
    return 0;
}
```
