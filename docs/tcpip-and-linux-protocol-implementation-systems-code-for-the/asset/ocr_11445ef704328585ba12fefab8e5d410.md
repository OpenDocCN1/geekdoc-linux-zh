# Here's the global data part of the code to manipulate socket buffers.

**Figure 5.133** Copy socket buffer structure and data: linux/net/core/skbuff.c.

```
/**
 * skb_copy - copy an skb_buff
 * @skb_buff: buffer to copy
 * @gfp_mask: allocation priority
 *
 * Make a copy of both an skb_buff and its data. This is used when the
 * caller wishes to modify the data and needs a private copy of the
 * data to alter. Returns \NULL on failure or the pointer to the buffer
 * on success. The returned buffer has a reference count of 1.
 *
 * You must pass \GFP_ATOMIC as the allocation priority if this function
 * is called from an interrupt.
 */
struct skb_buff *skb_copy(const struct skb_buff *skb, int gfp_mask)
{
    struct skb_buff *n;
    /*
     * Allocate the copy buffer
     */
    n=alloc_skb(skb->end - skb->head, gfp_mask);
    if(n==NULL)
        return NULL;

    /* Set the data pointer */
    skb_reserve(n,skb->data-skb->head);
    /* Set the tail pointer and length */
    skb_put(n,skb->len);
    /* Copy the bytes */
    memcpy(n->data,skb->head,skb->end-skb->head);
    n->sum = skb->sum;
    copy_skb_header(n,skb);

    return n;
}
```
