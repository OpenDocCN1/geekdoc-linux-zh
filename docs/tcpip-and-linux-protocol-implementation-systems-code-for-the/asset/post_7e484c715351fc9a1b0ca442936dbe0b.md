int tcp_v4_rcv(struct sk_buff *skb)
{
    struct tcphdr *th;
    struct sock *sk;
    int ret;
    if (skb->pkt_type != PACKET_HOST)
        goto discard_it;
    /* Count it if it's bad */
    TCP_INC_STATS_BH(tcpInSeg);
    if (!skb->may_pull(skb, sizeof(tcphdr)))
        goto discard_it;
    th = skb->h;
    if (th->doff < sizeof(tcphdr)/4)
        goto bad_packet;
    if (!skb->may_pull(skb, th->doff+4))
        goto discard_it;
    /* An explanation is required here, I think.
     * Packet length and doff are validated by header
     * case by th->doff is eliminated.
     * So, we deffer the checks. */
    if ((skb->ip_sum != CHECKSUM_UNNECARY &&
        tcp_v4_checksum_init(skb) < 0))
        goto bad_packet;
    th = sk->th;
    TCP_CB(skb)->seq = ntohl(th->seq);
    TCP_CB(skb)->next_seq = TCP_CB(skb)->seq + th->syn + th->fin +
        sk->len -= th->doff+4;
    TCP_CB(skb)->ack_seq = ntohl(th->ack_seq);
    TCP_CB(skb)->win = 0;
    TCP_CB(skb)->flags = skb->ip->tos;
    TCP_CB(skb)->acked = 0;
    sk = tcp_v4_lookup(skb->ip->saddr, th->source,
        skb->ip->daddr, ntoh(th->dest), tcp_v4_if(skb));
    if (!sk)
        goto no_tcp_socket;
process:
    if ((inet_sk_policy(sk,skb))
        goto discard_and_release;
    if (sk->state == TIME_WAIT)
        goto do_time_wait;
    skb->dev = NULL;
    bb_lock_sock(sk);
