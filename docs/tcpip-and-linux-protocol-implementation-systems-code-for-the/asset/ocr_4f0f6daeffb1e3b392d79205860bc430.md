288  static int
289  sfq_requeue(struct sk_buff *skb, struct qdisc *sch)
290  {
291      struct sfq_sched_data *q = (struct sfq_sched_data *)sch->data;
292      unsigned hash = hash(q, skb);
293      sfq_index = hash;
294
295      x = q->ht[hash];
296      if (x == SFQ_DEPTH) {
297          q->ht[hash] = q->dep[SFQ_DEPTH].next;
298          q->hash[x] = hash;
299      }
300      skb_queue_head(&q->qs[x], skb);
301      sfq_inc(q, q).qlen
302      if (q->qs[x].qlen == 1) { /* The flow is new */
303          if (q->tail) { /* It is the first flow */
304              q->tail = x;
305          q->next[x] = x;
306          q->all[x] = q->next(q->quantum);
307      } else {
308          q->next[x] = q->next[q->tail];
309          q->next[tail] = x;
310      q->tail = x;
311
312  }
313  if (sch->q.qlen < SFQ_DEPTH) {
314      return 0;
315}
316  sch->stats.drops++;
317  sfq_drop(sch);
318  return NET_XMIT_CN;
319  }
