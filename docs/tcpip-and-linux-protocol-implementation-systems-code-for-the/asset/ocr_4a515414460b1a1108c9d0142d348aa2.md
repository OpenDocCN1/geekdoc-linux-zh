# Memory Allocation and Freeing for Fragments

**Figure 6.48** Fragmentation–memory tracking: linux/net/ipv4/ip_fragment.c.

```
atomic_t _ip_frag_mem = ATOMIC_INIT(0); /* Memory used for fragments */
/* Memory Tracking Functions. */
extern __inline__ void frag_queue_skb(struct sk_buff *skb)
{
    atomic_sub(sizeof(struct ipq)->truize, &ip_frag_mem);
    kfree_skb(skb);
}
extern __inline__ void frag_queue(struct ipq *qp)
{
    atomic_sub(sizeof(struct ipq), &ip_frag_mem);
    kfree(qp);
}
struct ipq *qp = malloc(sizeof(struct ipq), GFP_ATOMIC);
if(!qp)
    return NULL;
    atomic_add(sizeof(struct ipq), &ip_frag_mem);
    return qp;
};
```

**Figure 6.49** Fragmentation–free up ipq: linux/net/ipv4/ip_fragment.c.

```
/* Destruction primitives. */
/* Complete destruction of ipq. */
static void ip_frag_destroy(struct ipq *qp)
{
    struct sk_buff *fp;
    BUG_TRAP(qp->last_in&COMPLETE);
    BUG_TRAP(del_timer(&qp->timer) == 0);
    /* Release all fragment data. */
    fp = qp->fragments;
    while (fp) {
        struct sk_buff *xp = fp->next;
        frag_kfree_skb(fp);
        fp = xp;
    }
};
```
