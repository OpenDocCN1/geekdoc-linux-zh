## Figure 6.204 IPMR-delete vif: linux/net/ipv4/ipmr.c.

```c
236 static vif_delete(int vif)
{
    struct vif_device *vif;
    struct net_device *dev;
    struct in_device *in_dev;

    if (vif < 0 || vif > maxv)
        return -EADDRNOTAVAIL;

    vif = &vif_table[vif];

    write_lock_bh(mrt_lock);
    dev = vif->dev;
    vif->dev = NULL;

    if (dev) {
        write_unlock_bh(mrt_lock);
        return -EADDRNOTAVAIL;
    }

    #ifdef CONFIG_IP_PIM
    if (vif == reg_if_num)
        reg_if_num = -1;
    #endif

    if (vif+1 <= maxv) {
        int tmp;
        for (tmp=vif-1; tmp>=0; tmp--) {
            if (VIF_EXISTS(tmp))
                break;
        }
        maxvif = tmp+1;
    }

    write_unlock_bh(mrt_lock);
    dev_set_allmulti(dev, -1);

    if ((in_dev = __in_dev_get(dev)) != NULL) {
        in_dev->cnf_mc.forwarding = 0;
        ip_rt_multicast_event(in_dev);
    }

    if (dev->flags & (VIFF_TUNNEL|VIFF_REGISTER))
        unregister_net_device(dev);

    dev_put(dev);
    return 0;
}
```
