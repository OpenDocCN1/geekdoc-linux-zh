        return -EINVAL;

    table_id = rtm->rtm_table;
    if (table_id == RT_TABLE_UNSPEC) {
        struct fib_table *table;
        if (rtm->rtm_type == RTN_UNICAST || rtm->rtm_type == RTN_NAT) {
            if ((table = fib_empty_table()) == NULL)
                return -ENOBUFS;
