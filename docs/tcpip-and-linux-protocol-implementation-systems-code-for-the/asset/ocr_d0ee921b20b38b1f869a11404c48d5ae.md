static struct sk_buff *
tbf_dequeue(struct qdisc *q)
{
    struct tbf_sched_data *data = (struct tbf_sched_data *)q->sch->data;
    struct sk_buff *skb;

    skb = __skb_dequeue(q->sch->q);

    if (skb) {
        psched_time_t now;
        long toks;
        long ptoks = 0;

        PSCHED_GET_TIME(now);

        toks = PSCHED_TDIFF_SAFE(now, q->t_c, buffer, 0);

        if (q->P_tab) {
            ptoks = toks + q->ptoks;
            if (toks > (long)q->mtu) {
                toks = q->mtu;
            }
            ptoks -= L2T(q, skb->len);
        }
        toks += q->ptoks;
        if (toks > (long)q->buffer) {
            toks = q->buffer;
        }
        toks -= L2T(q, skb->len);
    }
}
