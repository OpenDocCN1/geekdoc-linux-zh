# TCP/IP and Linux Protocol Implementation

Figure 5.130 Free socket buffer: linux/net/core/skbuff.c.

```
/*
 * Free an skbuff by memory without cleaning the state.
 */
void kfree_skmem(struct sk_buff *skb)
{
    if (!skb->cloned || atomic_dec_and_test(skb_datarefp(skb)))
        kfree(skb->head);

    skb_head_to_pool(skb);
}
```
