```c
/*
 * Given (packet source, input interface) and optional (dst, oif, tos):
 * check that source is valid i.e. not broadcast or our local address.
 * figure out this packet arrived and calculate "specific destination".
 * check that packet arrived from expected interface.
 */
int fib_validate_source(u32 src, u32 dst, u8 tos, int oif,
    struct net_device *dev, u32 *spec_dst, u32 *itag)
{
    struct in_dev *dev;
    struct rt_key *key;
    struct fib_result res;
    int no_addr, rpf;
    int rst;

    key->dst = src;
    key->src = dst;
    key->tos = tos;
    key->oid = 0;
    key->lif = oif;
    key->scope = RT_SCOPE_UNIVERSE;

    no_addr = rpf = 0;
    read_lock(&inetdev_lock);
    in_dev = __in_dev_get(dev);
    if (in_dev) {
        no_addr = in_dev->ifa_list = NULL;
        rpf = IN_DEV_FILTER(in_dev);
    }
    read_unlock(&inetdev_lock);

    if (in_dev == NULL) {
        goto e_inval;
    }
    if (fib_lookup(key, &res)) {
        goto last_res;
    }
    if (res.type == RTN_UNICAST)
        goto e_inval_res;
```

# Figure 8.16 FIB front end--ioctl handler: linux/net/ipv4/fib_frontend.c.

```c
/*
 * Handle IP routing ioctl calls. These are used to manipulate the routing
 */
int ip_rt_ioctl(unsigned int cmd, void *arg)
{
    int err;
    struct kern_rta rta;
    struct rtentry r;
    struct {
        struct nlmsghdr nlh;
        struct rtmsg rtm;
    } req;

    switch (cmd) {
    /* Add a route */
    case SIOCADRT:
        /* Delete a route */
        if (!capable(CAP_NET_ADMIN))
            return -EPERM;
        if (copy_from_user(req, arg, sizeof(struct rtentry)))
            return -EFAULT;
        rtnl_lock();
        err = fib_convert_rtcmd(cmd, &req.rtm, &rta, &rta);
        if (err == 0) {
            if (cmd == SIOCDELRT) {
                struct fib_table *tb = fib_get_table(req.rtm_table);
                err = -ESRCH;
                if (tb)
                    err = tb->tb_delete(tb, &req.rtm, &rta, &req.nlh, NULL);
            } else {
                struct fib_table *tb = fib_new_table(req.rtm_table);
                err = -ENOMEM;
                if (tb)
                    err = tb->insert(tb, &req.rtm, &rta, &req.nlh, NULL);
            }
            if (rta.rta_mx)
                kfree(rta.rta_mx);
        }
        rtnl_unlock();
        return err;
    }
    return -EINVAL;
}
```
