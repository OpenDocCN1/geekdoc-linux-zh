int int_dump_rules(struct sk_buf *skb, struct netlink_callback *cb)
{
    int idx;
    struct m_idx = cb->args[0];
    struct fib_rule *r;
    read_lock(&fib_rules,lock);
    for (r=fib_rules, idx=0; r=r->next, idx++) {
        if (idx < s_idx)
            continue;
        if (inet_fill_rule(skb, r, cb) < 0)
            break;
    }
    read_unlock(&fib_rules,lock);
    cb->args[0] = idx;
    return skb->len;
}
