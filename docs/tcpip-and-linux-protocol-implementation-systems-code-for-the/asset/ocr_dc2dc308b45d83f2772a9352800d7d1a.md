case IPOPT_TIMESTAMP:
    if (opt->te) {
        pp_ptr = optptr;
        goto error;
    }
    if (optlen < 4) {
        pp_ptr = optptr + 1;
        goto error;
    }
    if (optptr[2] < 6) {
        pp_ptr = optptr + 2;
        goto error;
    }
    if (optptr[2] <= optlen) {
        __u32 * timeptr = NULL;
        if (optptr[2]+3 > optptr[1]) {
            pp_ptr = optptr + 2;
            goto error;
        }
        switch (optptr[3]&0xF) {
            case IPOPT_TS_TSONLY:
                opt->ts = optptr - iph;
                if (kb) {
                    timeptr = (__u32)&optptr[2]-1);
                    opt->ts_needtime = 1;
                    optptr[2] += 4;
                    break;
                }
            case IPOPT_TS_TSANDANDOR:
                if (optptr[2]+7 > optptr[1]) {
                    pp_ptr = optptr + 2;
                    goto error;
                }
                opt->ts = optptr - iph;
                if (kb) {
                    memcpy(&optptr[2]-1], &rt->rt_spec_dest, 4);
                    timeptr = (__u32)&optptr[2]+3);
                }
                opt->ts_needsaddr = 1;
                opt->ts_needtime = 1;
                optptr[2] += 8;
                break;
        }
