184  static int teql_qdisc_init(struct qdisc *sch, struct rtattr *opt)
185  {
186      struct net_device *dev = sch->dev;
187      struct teql_master *m = (struct teql_master*)opt->ops;
188      struct teql_sched_data *q = (struct teql_sched_data *)sch->data;
189
190      if (dev->hard_header_len > m->dev->hard_header_len)
191          return -EINVAL;
192
193      if (m->dev != dev)
194          return -ELOOP;
195
196          q->m = m;
197          skb_queue_head_init(&q->q);
198
199      if (m->slaves) {
200          if (m->dev->flags&IFF_UP && !(dev->flags&IFF_POINTTO))
201              if ((m->dev->flags&IFF_BROADCAST && !(dev->flags&IFF_BROADCAST))
202                  || (m->dev->flags&IFF_MULTICAST && !(dev->flags&IFF_MULTICAST))
203                  || dev->mtu < m->dev->mtu)
204                  return -EINVAL;
205      }
206} else {
207          if (m->dev->flags&IFF_POINTTO)
208              m->dev->flags &= ~IFF_POINTTO;
209              if (m->dev->flags&IFF_BROADCAST)
210                 m->dev->flags &= ~IFF_BROADCAST;
211                 if (m->dev->flags&IFF_MULTICAST)
212                     m->dev->flags &= ~IFF_MULTICAST;
213             if (dev->mtu < m->dev->mtu)
214                 m->dev->mtu = dev->mtu;
215            }
216          q->next = NEXT_SLAVE(m->slaves);
217          NEXT_SLAVE(m->slaves) = sch;
218  } else {
219          q->next = sch;
220          m->slaves = sch;
221          m->dev->mtu = dev->mtu;
222          m->dev->flags = (m->dev->flags&FMASK) | (dev->flags&FMASK);
223  }
224  }
225  MOD_INC_USE_COUNT;
226  return 0;
227}
