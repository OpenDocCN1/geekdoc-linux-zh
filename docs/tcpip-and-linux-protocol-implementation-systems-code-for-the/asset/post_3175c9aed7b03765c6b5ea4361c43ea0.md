```c
// This is what the sent packet queue engine uses to pass
* TCP per-packet control information to the transmission
* code. We also store the host-order sequence numbers
* here. This is 32-bit on 32-bit architectures,
* 40 bytes on 64-bit machines, if this grows
* skb->cb appropriately.
struct tcp_skb_cb {
    union {
        struct inket_ipv6_h4;
#if defined(CONFIG_IPV6) || defined(CONFIG_IPV6_MODULE)
        struct inket_skb_h6;
#endif /* For incoming frames */
        u32 seq;               /* Starting sequence number */
        u32 end_seq;           /* SEQ + FIN + SYN */
        u32 d;                /* used to compute rtt */
        u8 flags;              /* TCP flags. */
    };
    /* NOTE: These must match the flags in
    * real TCP header. */
#define TCPCB_FLAG_AIN 0x01
#define TCPCB_FLAG_SYN 0x02
#define TCPCB_FLAG_RST 0x04
#define TCPCB_FLAG_PSH 0x08
#define TCPCB_FLAG_ACK 0x10
#define TCPCB_FLAG_URG 0x20
#define TCPCB_FLAG_ECE 0x40
#define TCPCB_FLAG_CWR 0x80
    u8 sacked;               /* State for SACK/FACK. */
#define TCPCB_SACKED_ACK 0x01 /* SKB'd by a SACK block */
#define TCPCB_SACKED_RETRANS 0x02 /* SKB retransmitted */
#define TCPCB_LOST 0x04 /* SKB is lost */
#define TCPCB_7BITS 0x07 /* All tag bits */
#define TCPCB_RETRANS_EVER_RETRANS 0x80 /* Ever retransmitted */
#define TCPCB_RETRANS (TCPCB_SACKED_RETRANS|TCPCB_EVER_RETRANS)
#define TCPCB_URG 0x20 /* Urgent pointer advanced here. */
#define TCPCB_AT_TAIL (TCPCB_URG)
    u16 urg_ptr;            /* Valid/URG flags is set. */
    u32 ack_seq;            /* Sequence ACK'd */
};
```

```c
1008 int tcp_sendmsg(struct sock *sk, struct msghdr *msg, int size) {
    struct iovec *iov;
    struct tcp_opt *tp;
    struct sk_buff *skb;
    int iovlen, flags;
    int mss_now;
    int err, copied;
    long timeout;

    tp = &(sk->tp->pinfo.af_tcp);

    lock_sock(sk);
    TCP_CHECK_TIMER(sk);

    flags = msg->msg_flags;
    time = sock_mdtime(sk,MSG_DONTWAIT);

    /* Wait for a connection to finish. */
    if ((sk->state) & (TCPCF_ESTABLISHED | TCPF_CLOSE_WAIT))
        if ((err = wait_for_tcp_connect(sk, flags, &timeout)) != 0)
            goto out_err;

    /* This should be in poll */
    clear_bit(SOCK_ASYNC_NOSIGNACE, sk->socket->flags);

    mss_now = tcp_current_mss(sk);

    /* Ok commence sending. */
    iovlen = msg->msg_iovlen;
    iov = msg->msg_iovlen;
    copied = 0;

    err = -EPIPE;
    if (err || (sk->shutdown&SEND_SHUTDOWN))
        goto do_error;
```
