static void rsvp_destroy(struct tcf_proto *tp)
{
    struct rsvp_head *data = xchg(&tp->root, NULL);
    struct rsvp_session **sht;
    int h1, h2;

    if (data == NULL) {
        return;
    }
    sht = data->ht;

    for (h1=0; h1<=256; h1++) {
        struct rsvp_session *s;
        while (s = sht[h1]) != NULL) {
            sht[h1] = s->next;
            for (h2=0; h2<=16; h2++) {
                struct rsvp_filter *f;

                    unsigned long cl;

                    tcf_policy_release(f->policy);
                    #endif
                    free(f);
                }
            }
        }
        free(data);
        MOD_DEC_USE_COUNT;
