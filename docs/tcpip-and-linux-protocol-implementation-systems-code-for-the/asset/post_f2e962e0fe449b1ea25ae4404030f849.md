{
    int ret;
    struct dsmark_qdisc_data *p = PRIV(sch);
    printk("dsmark_requeue(%p, %p, [qdisc %p])\n",skb,sch,p);
    if ((ret = p->q->ops->requeue(skb, p->q)) == 0) {
        sch->q->qlen++;
        return 0;
    }
    sch->stats->drops++;
    return ret;

static int dsmark_drop(struct qdisc *sch)
{
    struct dsmark_qdisc_data *p = PRIV(sch);
    printk("dsmark_reset(%p, [qdisc %p])\n",sch,p);
    if (!(p->q->ops->drop))
        return 0;
    if (!p->q->ops->drop(p->q))
        return 0;
    sch->q->qlen--;
    return 1;
}

int dsmark_init(struct qdisc *sch, struct rtattr *opt)
{
    struct dsmark_qdisc_data *p = PRIV(sch);
    struct rtattr rattr[TCA_DSMARK_MAX];
    __u6 tmp;
    printk("dsmark_init(%p, [qdisc %p] opt %p)\n", sch, p, opt);
    if (rtattr_parse(tb, TCA_DSMARK_MAX, RTA_DATA(opt), RTA_PAYLOAD(opt)) < 0 ||
        !tb[TCA_DSMARK_INDICES-1] ||
        RTA_PAYLOAD(tb[TCA_DSMARK_INDICES-1]) < sizeof(__u6))
        return -EINVAL;
    memset(p->o, NULL, sizeof(p->o));
    if (!p->filter_list)
        p->indices = (__u6 *)RTA_DATA(tb[TCA_DSMARK_INDICES-1]);
    if (!p)
        return -EINVAL;
    p->default_index = 0;
    for (tmp = p->indices; tmp != 1; tmp += 1) {
        if (tmp & 1)
            return -EINVAL;
    }
    if (tb[TCA_DSMARK_DEFAULT_INDEX] < sizeof(__u6))
        return -EINVAL;
    p->default_index = (__u6 *)RTA_DATA(tb[TCA_DSMARK_DEFAULT_INDEX]);
    if (p->default_index >= p->indices)
        return -EINVAL;
    
    p->set_tc_index = &tb[TCA_DSMARK_SET_TC_INDEX];
    p->mask = kmalloc(p->indices, 2, GFP_KERNEL);
    if (!p->mask)
        return -ENOMEM;
    p->value = value;
    memset(p->mask, 0xff, p->indices);
    if ((p->q = qdisc_create_dlt(sch->dev, pfifo, qdisc_ops))
        p->q = &noop_qdisc;
    printk("dsmark_init: qdisc %p\n", &p->q);
    MOD_INC_USE_COUNT;
    return 0;
}

static void dsmark_destroy(struct qdisc *sch)
{
    struct dsmark_qdisc_data *p = PRIV(sch);
    struct tcf_proto *tp;
    
    printk("dsmark_destroy(%p, [qdisc %p])\n",sch,p);
    while (p->filter_list) {
        tp = p->filter_list;
        p->filter_list = tp->next;
        tp->ops->destroy(tp);
    }
    qdisc_destroy(p->q);
    p->q = knnoop_qdisc;
    kfree(p->mask);
    MOD_DEC_USE_COUNT;
}
