int qdisc_copy_stats(struct sk_buff *skb, struct tc_stats *st)
{
    spin_lock_bh(st->lock);
    RTA_PUT(skb, TCA_STATS, (char*)kst - (char*)st);
    spin_unlock_bh(st->lock);
    return 0;

trtattr_failure:
    spin_unlock_bh(st->lock);
    return -1;
}

static int tc_fill_qdisc(struct sk_buff *skb, struct qdisc *q, u32 clid,
    u32 pid, u32 seq, unsigned flags, int event)
{
    struct tcm *tcm;
    struct nlmsghdr *nlmh;
    unsigned char *b = skb->tail;

    nlmh = NLM_PUT(skb, pid, seq, sizeof(*tcm));
    nlmh->nlmsg_flags = flags;
    tcm = NLMSG_DATA(nlmh);
    tcm->tcm_family = AF_UNSPEC;
    tcm->tcm_index = q->dev ? q->dev->index : 0;
    tcm->tcm_parent = clid;
    tcm->tcm_handle = q->handle;
    tcm->tcm_info = atomic_read(&q->refcnt);
    RTA_PUT(skb, TCA_ALIGN, IFNAMIZE, q->ops->id);
    if (q->open_dump && q->open_dump(q, skb) -> dump(q, skb) < 0)
        goto rttattr_failure;
    q->state.qlen = q->qlen;
    if (qdisc_copy_stats(skb, &q->stats))
        goto rttattr_failure;
    nlmh->nlmsg_len = skb->tail - b;
    return skb->len;

rttattr_failure:
    skb_trim(skb, b - skb->data);
    return -1;
}
