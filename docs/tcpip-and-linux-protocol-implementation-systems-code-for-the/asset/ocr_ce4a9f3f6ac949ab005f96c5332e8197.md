static void
fsq_reset(struct qdisc *sch)
{
    struct sk_buff *skb;
    while (skb = sfq_dequeue(sch) != NULL)
        kfree_skb(skb);
}

static void sfq_perturbation(unsigned long arg)
{

q->perturbation = net_random()&0xF;
    q->perturb_timer.expires = jiffies + q->perturb_period;
