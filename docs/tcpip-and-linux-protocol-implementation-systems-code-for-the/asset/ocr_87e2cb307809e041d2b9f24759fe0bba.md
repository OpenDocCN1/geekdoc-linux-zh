cbcq_update(struct cbq_sched_data *q)
{
    struct cbq_class *this = q->tx_class;
    struct cbq_class *cl = this;
    int len = q->tx_len;

    q->tx_class = NULL;

    for ( ; cl = cl->share) {
        long avglide = cl->avgidle;
        long idle;

        cl->stats.packets++;
        cl->stats.bytes += len;

        /*
         (now - last) is total time between packet right edges.
         (last_pktlen/rate) is "virtual" busy time, so that
            idle = (now - last)-last_pktlen/rate
        */
        idle = PSCHED_DIFF(q->now, cl->last);
        if ((unsigned long)idle > 128+1024+1024)
            avglide = cl->maxwidth;
        else
            idle = L2T(cl, len);

        /*
         *true_avglide := (1-W)*true_avglide + W*idle;
             where W=2^(-ewma_log). But avglide is scaled:
                cl->avglide = true_avglide/W;
             hence;
            */
            avglide += idle - (avglide>>ewma_log);

    }
