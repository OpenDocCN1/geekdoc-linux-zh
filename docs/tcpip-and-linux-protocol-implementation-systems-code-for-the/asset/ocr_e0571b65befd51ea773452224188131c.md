# Figure 9.141 A generic scheduler–PFIFO operations: linux/net/sched/generic.c.

```
static const u8 prioband[TC_PRIO_MAX+1] =
    { 1, 2, 2, 2, 1, 2, 0, 0, 1, 1, 1, 1, 1, 1, 1 };

/* 3-band PFIFO queue: old style, but should be a bit faster than
   generic prio+fifo combination. */

static int
    pfifo_fast_enqueue(struct sk_buff *skb, struct qdisc *qdisc)
{
    struct sk_buff_head *list;

    list = ((struct sk_buff_head*)qdisc)->data) +
           prioband[skb->priority&TC_PRIO_MAX];

    if (list->qlen < skb->dev->tx_queue_len) {
        __skb_dequeue_tail(list, skb);
        qdisc->qlen++;
        return 0;
    }

    qdisc->stats.drops++;
    kfree(skb);
    return NET_XMIT_DROP;
}
```

static struct sk_buff *
pfifo_fast_dequeue(struct qdisc *qdisc)
{
    int prio;
    struct sk_buff_head *list = ((struct sk_buff_head*)qdisc)->data;
    struct sk_buff *skb;

    for (prio = 0; prio < 3; prio++) {
        skb = __skb_dequeue(list);
        if (skb) {
            qdisc->qlen--;
            return skb;
        }
    }
    return NULL;
}
