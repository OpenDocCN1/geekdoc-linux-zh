extern struct tcp_hashinfo {
    /* This is for sockets with full identity. Sockets here will
     * always be without wildcards and will have the following invariant:
     *
     TCP_ESTABLISHED <= sk->state < TCP_CLOSE
     *
     * First half of the table is for sockets in TIME_WAIT, second half
     * is for TIME_WAIT sockets only.
     */
    struct tcp_ehash_bucket __tcp_ehash__;
    /* Ok, let's try this, I give up, we need a local binding
     * TCP hash as well as the others for fast bind/connect.
     */
    struct tcp_bind_hashbucket __tcp_bbash;
    int __tcp_bbash_size;
    int __tcp_ehash_size;
    /* All sockets in TCP_LISTEN state will be in here. This is the only
     * table where wildcard'd TCP sockets can exist. Hash function here
     * is just local port number.
     */
    struct sock __tcp_listening_hash[TCP_LHTABLE_SIZE];
