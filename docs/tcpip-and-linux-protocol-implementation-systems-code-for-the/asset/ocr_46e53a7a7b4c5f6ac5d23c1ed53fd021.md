/* This checks if the data bearing SKB (usually tp->send_head)
* should be put on the right wire now.
*/
static __inline_ int tcp_scnd_test(struct tcp_opt *tp, struct skb_buf *skb,
    unsigned cur_ms, int nonagle)
{
    * RFC 1122 - section 4.2.3.4
    * We must queue if
        * The right edge of this frame exceeds the window
        * There are packets in flight and we have a small segment
            (SWS avoidance and Nagel algorithm)
            (part of SWS is done on packetization)
Minshall versions: there are no _small_
            segments in flight. (tcp_nagle_check)
        * We have too many packets 'in flight'
    Don't use the nagle rule for urgent data (or
        for the final frame -Dave)
    Also, Nagel rule does not apply to frames, which
        sit in the middle of queue (they have no chances
        to get data) and if room at tail of skb is
            not enough to save something seriously (<32 for now).
    /
/* Don't be strict about the congestion wind for
    * final frame -Dave)
*/
return (nonagle==1 || (!tcp_nagle_check(tp, skb, cur_ms, nonagle) &&
    (tcp_packets_in_flight(tp) < tp->snd_cwnd) ||
    TCP_SKB_CB(skb)->flags & TCPCB_FLAG_FIN) &&
    !after(TCP_SKB_CB(skb)->end_seq, tp->snd_una + tp->snd_snd));
}
