```
static struct tcinder_filter_result *lookup(struct tcinder_data *p, __u16 key)
{
    struct tcinder_filter *f;

    if (p->perfect)
        return p->perfect[key].res.classid ? p->perfect+key : NULL;
    if (p->h)
        return NULL;
    for (f = p->h[key % p->hash]; f = f->next)
        if (f->key == key)
            return f->result;

    return NULL;
```
