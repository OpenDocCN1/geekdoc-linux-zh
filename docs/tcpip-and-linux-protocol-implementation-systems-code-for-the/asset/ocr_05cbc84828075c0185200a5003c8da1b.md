static void rtmsg_fib(int event, struct fib_node* f, int z, int tb_id,
    struct nmsghdr *n, struct netlink_skb_params *req)
{
    struct skb_buff *skb;
    u32 pid = req ? req->pid : 0;
    int size = NLMSG_SIZE(sizeof(struct rtmsg)+256);

    skb = alloc_skb(size, GFP_KERNEL);
    if (!skb)
        return;

    if (fib_dump_info(skb, pid,nmsg_seq, event, tb_id,
        f->fn_type, f->fn_scope, kf->fn_key, f->fn_tos,
        FIB_INFO(f)) < 0)
        kfree_skb(skb);
        return;

    NETLINK_CB(skb)->dst_groups = RTGRP_IPV4_ROUTE;
    if (n->nmsg_flag&NLM_ECHO)
        atomic_inc(&skb->users);
    netlink_broadcast(rtnl, skb, pid, RTGRP_IPV4_ROUTE, GFP_KERNEL);
    if (n->nmsg_flag&NLM_ECHO)
        netlink_unicast(rtnl, skb, pid, MSG_DONTWAIT);
}
