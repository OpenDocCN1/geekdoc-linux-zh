# Listing B.1 Agent startup script

```
#!/sbin/sh
# Copyright 01/17/97 Sun Microsystems, Inc. All Rights Reserved.
#pragma ident "@(#)init.snmpdx 1.11 97/01/17 Sun Microsystems"
# 
# Return pid of named process in variable "pid"
pidproc() {
	id=`/usr/bin/ps -e | \n		/usr/bin/grep $1 | \n		/usr/bin/sed -e 's/^ *//' -e 's/ *//' -e '/\*'
}
# 
# Kill named process(es)
# killproc() {
	pidproc $1
	[ "$pid" -m 1 ] && kill -9 $pid
}
case "$1" in
	start')
		if [ -f /etc/snmp/conf/snmpdx.rsrc -a -x /usr/lib/snmp/snmpdx ]; then
			/usr/lib/snmp/snmpdx -y -c /etc/snmp/conf
		fi
	;;
	stop')
		killproc snmpdx
		killproc snmpv2d
		killproc mibiisa
	;;
	*)
		echo "Usage: /etc/init.d/init.snmpdx { start | stop }"
	;;
esac
exit 0
```

Like most startup scripts on Solaris, this script provides the ability to either start or stop the master agent and subagents. A case statement provides the logic to determine which argument is given
