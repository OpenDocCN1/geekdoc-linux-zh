```bash
return_value=$

# Close file descriptor 3
exec 3&-

# On the exit status
case $return_value in
($DIALOG_OK)
  echo "Result: $result";
($DIALOG_CANCEL)
  echo "Cancel pressed.";
($DIALOG_HELP)
  echo "Help pressed.";
($DIALOG_EXTRA)
  echo "Extra button pressed.";
($DIALOG_ITEM_HELP)
  echo "Item-help pressed.";
($DIALOG_ESC)
  if test -n "$result"; then
    echo "$result"
  else
    echo "ESC pressed."
  fi
esac
```

起初，这种重定向似乎毫无意义。首先，我们使用 `exec` 将文件描述符 1（stdout）复制并重定向到 3，以创建描述符 1 的备份副本。

下一步是执行命令替换，并将 `dialog` 命令的输出赋值给变量。命令将描述符 2（stderr）的重定向为描述符 1 的副本，最后，通过复制包含备份副本的描述符 3 将描述符 1 恢复到其原始值。可能无法立即看出为什么最后一步是必要的。在子过程中，标准输出（描述符 1）并不指向控制终端。相反，它指向一个将交付输出的管道。由于 `result` 需要标准输出到终端以便显示输入框，我们必须将标准错误重定向到标准输出（这样 `dialog` 的输出就会进入 `result` 变量），然后将标准输出重定向回控制终端。

那么，哪种方法更好？临时文件还是命令替换？可能是命令替换更好，因为它避免了文件的创建。

```bash
#!/bin/bash
# while-menu: a menu-driven system information program
DELAY=3 # 显示结果的秒数
while true; do
  clear
  cat << EOF
Please:
  1. 显示系统信息
  2. 显示磁盘空间
  3. 显示主目录空间利用
  0. 退出
EOF
read -p "Enter selection [0-3] " 
if [[ $REPLY =~ ^[0-3]$ ]]; then
  case $REPLY in
    1)
      echo "Hostname:$HOSTNAME"
      uptime
      sleep $DELAY
      continue
      ;;
    2)
      df -h
      sleep $DELAY
      continue
      ;;
    3)
      if [[ $(id -u) -eq 0 ]]; then
        echo "Home/Space Space Utilization (All Users)"
        du -sh /home/* 2>/dev/null
      else
        echo "Home Space Utilization ($USER)"
        du -sh $HOME 2>/dev/null
      fi
      sleep $DELAY
      continue
      ;;
    0)
      break
      ;
  esac
else
  echo "Invalid entry."
  sleep $DELAY
fi
done

echo "Program terminated."
```
