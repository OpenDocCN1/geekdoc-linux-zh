21 func (g GMFSSH) Expand(e GitMetaEntry) ([][]Cloneable,
	error) {
22 	clones := [][]Cloneable{}
23 	res, err := sshGetDirs(e.Host, e.Path)
24 	if err != nil {
25 		return clones, err
26 	}
27 
28 	for _, dir := range res {
29 	\lDir := strings.TrimSuffix(dir, ".git")
30 		clones = append(clones, Cloneable{
31 			URL: fmt.Sprintf("%s:%s", e.Host, e.Path, dir),
32 			Dir: lDir,
33 		})
34 	}
35 
36 	return clones, nil
37 }
38 
39 func sshGetDirs(host string, path string) ([][]string,
	error) {
40 	results := [][]string{}
41 	cmd := exec.Command("ssh", host, "ls", path+"/*/conf|g")
42 	output, err := cmd.Output()
43 	if err != nil {
44 		return results, err
45 	}
46 
47 	reader := bytes.NewReader(output)
48 	scanner := bufio.NewScanner(reader)
49 	for scanner.Scan() {
50 		line := scanner.Text()
51 		parts := strings.Split(line, "/")
52 		if len(parts) > 2 {
53 			results = append(results, parts[len(parts)-2])
54 		}
55 	}
56 
57 	return results, err
58 }
