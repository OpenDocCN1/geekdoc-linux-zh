FDataLink.Free;
FDataLink := nil;

end;

inherited Notification(AComponent, Operation);

(FDataLink <> nil) and
(AComponent = DataSource) then

Message.Result := Integer(FDataLink);

FDataLink.Edit;

  FDataLink.Edit;
