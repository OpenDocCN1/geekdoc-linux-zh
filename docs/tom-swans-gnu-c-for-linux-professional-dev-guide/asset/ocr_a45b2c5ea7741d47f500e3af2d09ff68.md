// The canvas object calls this to update the status
// bar's mouse x,y position information when the mouse

void TCmdWindow::UpdateMouseInfo(int x, int y)
{
    char buffer[8]; // Temporary text buffer

    sprintf(buffer, "%03d", x);
    SetString(m_XPosition, buffer); // Show X info
    sprintf(buffer, "%03d", y);
    SetString(m_YPosition, buffer); // Show Y info
}

    char buffer[8];
    IntToStr(n, buffer);
    SetString(m_NumLines, buffer);
