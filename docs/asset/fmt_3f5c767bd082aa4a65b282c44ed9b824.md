
```c
struct message {
  address to;
  address from;
  line subject;
  address cc;
  string body;
};
```

```c
struct message *getRecord(void) {
  struct message *msg;
  msg = allocate(sizeof(message));
  msg->to = getAddress(...);
  msg->from = getAddress(...);
  msg->cc = getAddress(...);
  msg->subject = getLine();
  msg->body = getString();
  return(msg);
}
```
