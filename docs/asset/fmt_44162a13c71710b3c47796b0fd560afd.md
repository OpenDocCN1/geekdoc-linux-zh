int main(void) {
  /* 首先你需要运行 mknod /dev/mydev 202 0 来创建 /dev/mydev */
  int my_dev = open("/dev/mydev", 0);
  if (my_dev < 0) {
    perror("Fail to open device file: /dev/mydev");
  } else {

    close(my_dev);
