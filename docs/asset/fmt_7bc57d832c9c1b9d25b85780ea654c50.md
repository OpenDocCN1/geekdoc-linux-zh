## 创建 float_subtract.ksh Shell 脚本

由于 `float_add.ksh` shell 脚本执行了一系列数字的加法，本节将详细介绍减法技术。`float_subtract.ksh` shell 脚本如列表 22.4 所示，我们将展示该脚本。

```bash
#!/bin/ksh

# SCRIPT: float_subtract.ksh
# AUTHOR: Randy
# DATE: 02/23/2001
# REV: 1.1

# PURPOSE: 此脚本用于减去一组数字。
# 数字可以是整数或浮点数。
# scale 是用户可以指定的数点后的位数。

# EXIT STATUS:
# 0 ==> 完成
# 1 ==> 错误
# 2 ==> 退出信号

trap 'exit_trap; exit 2' INT

function exit_trap
{
    echo "\n退出信号..."
}

function usage
{
    echo "\PURPOSE: 减去一组数字"
    echo "\USAGE: $SCRIPT_NAME [-scale_value] N1..N"
    echo "不带小数的整数"
}

SCRIPT_NAME=`basename $0`
SCALE="0"
COUNT=0
MAX_COUNT=$#

# 检查至少有两个命令行参数
if (( $# < 2 ))
then
    echo "\nERROR: 提供数字列表"
    usage
    exit 1
fi

# 解析命令行参数查找 scale 值（如果存在）。
while getopts ":s:S:" ARGUMENT
do
    case $ARGUMENT in
        s|S) SCALE=$OPTARG
        ;;
        \?) # 由于我们可能有负数，我们需要
            # 测试以连字符 (-) 开头的 ARGUMENT 是否为数字，而不是无效开关。
            for TST_ARG in "$@"
            do
                if [[ $(echo "$TST_ARG" | cut -c1) = '-' \
                && "$TST_ARG" != '-s' -a "$TST_ARG" != '-S' ]]; then
                    case "$TST_ARG" in
                        +([-0-9])) : # 无操作
                            ;;
                        +([-0-9]*)) : # 无操作
                            ;;
                        +([-0-9]*)) : # 无操作（注：此处原文逻辑重复）
                            ;;
                        *) echo "\nERROR: 命令行参数无效"
                            usage
                            exit 1
                            ;;
                    esac
                fi
            done
            ;;
    esac
done

# 遍历命令行参数并收集要减的数字列表。
while (( COUNT < MAX_COUNT ))
do
    (( COUNT = COUNT + 1 ))
    TOKEN=$1
    case $TOKEN in
        -s|-S) shift 2
            (( COUNT = COUNT + 1 ))
            ;;
        -s${SCALE}) shift
            ;;
        -S${SCALE}) shift
            ;;
        *) NUM_LIST="${NUM_LIST} $TOKEN"
            (( COUNT < MAX_COUNT )) && shift
            ;;
    esac
done

################################################################

# 确保精度是一个整数值

################################################################

# 检查提供的每个数字，以确保“数字”
# 是整数或浮点数。

for NUM in $NUM_LIST
do
    case $NUM in
        +([0-9])) # 检查是否为整数
            # 无操作，什么都不做。
            ;;
        +(-([0-9]))) # 检查是否为负整数
            # 无操作，什么都不做
            ;;
        +([0-9].[0-9]))
            # 检查是否为正浮点数
            # 无操作，什么都不做
            ;;
        +([0-9]|[.][0-9]))
            # 检查是否为带有 + 前缀的正浮点数
            # 无操作，什么都不做
            ;;
        +([0-9]|.[0-9])))
            # 检查是否为负浮点数
            # 无操作，什么都不做
            ;;
        +(-([-.][0-9])))
            # 检查是否为负浮点数
            # 无操作，什么都不做
            ;;
        +([.+0-9]))
            # 检查是否为正浮点数
            # 无操作，什么都不做
            ;;
        *) echo "ERROR: $NUM IS NOT a valid number"
            usage
            exit 1
            ;;
    esac
done

# 构建要相减的数字列表
SUBTRACT= # 将 SUBTRACT 变量初始化为空
MINUS= # 将 MINUS 变量初始化为空

# 遍历每个数字并构建一个数学语句
# 将减去列表中的数字。

for X in $NUM_LIST
do
    # 如果数字带有 + 前缀，将其移除！
    if [[ $(echo "$X" | cut -c1) = '+' ]]
    then
        X=$(echo "$X" | cut -c2-)
    fi

    SUBTRACT="${SUBTRACT} ${MINUS} ${X}"
    MINUS='-'
done

# 在此处通过文档向 bc 命令提供输入以进行数学计算。数字的差
# 被赋值给 DIFFERENCE 变量。

DIFFERENCE=$(bc <<EOF
scale=${SCALE}
(${SUBTRACT})
EOF
)

################################################################

# 向用户展示相减法的结果。

echo "\nThe difference of: ${SUBTRACT}"
echo "\nis: ${DIFFERENCE}\n"
```
