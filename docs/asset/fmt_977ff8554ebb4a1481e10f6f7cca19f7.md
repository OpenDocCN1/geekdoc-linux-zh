## `checkfs` 初始化脚本

```bash
#!/bin/sh
################################################
# 开始 checkfs
# 描述 : 文件系统检查
# 作者 : Gerard Beekemans - gerard@linuxfromscratch.org
DJ Lucas - dj@linuxfromscratch.org
# 更新 : Bruce Dubbs - dbububs@linuxfromscratch.org
# 版本 : LFS 7.0
################################################

### BEGIN INIT INFO
# Provides: checkfs
# Required-Start: modules
# Should-Start:
# Required-Stop:
# Default-Start: S
# Default-Stop:
# Short-Description: 在挂载前检查文件系统。
# Description: 在启动时，通过 fsck 检查文件系统。
# X-LFS-Provided-By: LFS
### END INIT INFO
```

```bash
/lib/lsb/init-functions

case "$1" in
  start)
    if [ -f /fastboot ]; then
      msg="fastboot 已启用，跳过文件系统检查。"
      log_info_msg "$msg"
      exit 0
    fi

    log_info_msg "以只读模式挂载根文件系统..."
    mount -o remount,ro /

    if [ $? -ne 0 ]; then
      _retval
      msg="无法挂载根文件系统 "
      msg="${msg}因为它处于只读模式。\n\n"
      msg="${msg}在你按下回车后，此系统"
      msg="${msg}将停止并关机。\n\n"
      log_failure_msg "${msg}"

      log_info_msg "按回车键继续..."
      wait_for_user
      /etc/rc.d/init.d/halt stop
    else
      log_success_msg

      if [ -f /forceck ]; then
        msg="/forceck 已启用，强制进行文件系统检查。"
        log_success_msg="${msg}"
        options="-f"
      else
        options=""
      fi

      log_info_msg "检查文件系统..."
      # 注意：使用 -p 参数但在 fsck 中会失败
      if is_true "VERBOSE_FSCK"; then
        fsck (${options}) -a -C
      else
        fsck (${options}) -A -C >/dev/null
      fi

      error_value=$?

      if [ "${error_value}\ -eq 0 ]; then
        log_success_msg2
      fi

      if [ "${error_value}\ -eq 1 ]; then
        msg="\n警告：发现并修复了 "
        msg="${msg}文件系统错误。\n"
        msg="${msg}你可能需要双检查 "
        msg="${msg}是否修复正确。"
        log_warning_msg "${msg}"
      fi

      if [ "${error_value}" -gt 3 -a "${error_value}" -lt 16 ]; then
        msg="${FAILURE:遇到了无法自动修复的 "
        msg="${msg}文件系统。此系统无法继续启动，"
        msg="${msg}将由系统管理员手动修复"
        msg="${msg}。\n\n"
        msg="${msg}在你按下回车后，此系统将"
        msg="${msg}停止并关机。\n\n"
        log_failure_msg "${msg}"

        log_info_msg "按回车键继续..."
        wait_for_user
        reboot -f
      fi

      if [ "${error_value}" -ge 16 ]; then
        msg="${FAILURE:\n运行 fsck 时发生意外故障。"
        msg="${msg}退出错误代码: ${error_value}"
        log_failure_msg $msg
        exit "${error_value}"
      fi
    fi
  fi
esac

exit 0
```

## 本地文件系统挂载脚本

```bash
### BEGIN INIT INFO

# Provides:              $local_fs
# Required-Start:       udev checks
# Should-Start:         swap
# Required-Stop:        
# Should-Stop:          S
# Default-Start:        
# Default-Stop:         0
# Short-Description:     挂载在 /etc/fstab 中定义的本地文件系统。
# Description:           在启动时重新挂载根文件系统为读写模式，并
                        挂载 /etc/fstab 中定义的所有文件。
                        停止时重新挂载根文件系统为只读并卸载
                        剩余的文件系统。

# X-LFS-Provided-By:    LFS
### END INIT INFO
```

```bash
/lib/lsb/init-functions

case "$1" in
start)
    log_info_msg "正在以读写模式挂载根文件系统..."
    mount -o remount,rw / >dev/null
    evaluate_retval

    # 移除 fsck 相关的文件系统标记。
    rm -f /fastboot /forcefsck

    # 确保 /dev/pts 存在
    mkdir -p /dev/pts

    # 这将挂载所有没有 _netdev 选项的文件系统
    # 它们的选项列表：_netdev 表示网络文件系统。

    log_info_msg "正在挂载剩余文件系统..."
    mount -a -o no_netdev /dev/null
    evaluate_retval
    exit $failed
    ;;

stop)
    # 不要卸载如 /run 等虚拟文件系统
    log_info_msg "正在卸载所有已挂载的文件系统..."
    umount -a -r -t notmpfs,nosysfs,nodevmpfs,nproc,nodevpts >dev/null
    evaluate_retval

    # 确保 / 为只读（除非 um 忙碌）
    mount -o remount,ro /

    # 在适当的情况下将所有 LVM 卷组设为不可用
    # 如果 swap 或 / 在 LVM 分区上，则会失败
    #if [ -x /sbin/vgchange ]; then /sbin/vgchange -an > /dev/null; fi
    ;;

*)
echo "用法: ${0} {start|stop}"
exit 1
;;
;;
esac
# End mountfs
```
