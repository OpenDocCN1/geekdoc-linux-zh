```bash
# 使用 df 命令获取当前磁盘空间使用的统计信息：
# -k $RESULTS_FROM_DF_COMMAND

# 读取循环（输入是来自 'df -k' 命令的数据）：
while read -r bytes used available capacity mounted_on
do
	# -d 选项允许我们删除一个字符；在这种情况下是百分号。
	capacity=$(echo $capacity | tr -d '%') # 详情请参见：tr.ksh
	((counter++)) # 记录计数器加 1。
	if [ $counter -eq 1 ]; then
	# 什么都不做；第一记录是标题。
	# 标题格式如下：
# 文件系统 字节 可用 容量 挂载点
print "Mounted_on \${capacity}%" >> $TOO_FULL
	fi
elif [ $capacity -gt 90 ]; then
	print "Mounted_on \${capacity}%" >> $TOO_FULL
	fi
done
```
