/* mkaddr.c
 *
 * Make a socket address:
 */
#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>
#include <errno.h>
#include <ctype.h>
#include <string.h>
#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <netdb.h>

/*
 * Create an AF_INET Address:
 *
 * ARGUMENTS:
 * 1. addr   Ptr to area
 *           where address is
 *           to be placed.
 * 2. addrlen Ptr to int that
 *           will hold the final
 *           address length.
 * 3. str_addr The input string
 *           format hostname, and
 *           port.
 * 4. protocol The input string
 *           indicating the
 *           protocol being used.
 *           NULL implies "tcp".
 * RETURNS:
 * 0   Success.
 * -1  Bad host part.
 * -2  Bad port part.
 *
 * NOTES:
 * "**" for the host portion of the
 * address implies INADDR_ANY.
 *
 * "**" for the port portion will
 * imply zero for the port (assign
 * a port number).
 *
 * EXAMPLES:
 * "www.lwn.net:80"
 * "localhost:telnet"
 * "*:21"
 * "**:*"
 * "ftp.redhat.com:ftp"
 * "sunsite.unc.edu"
 */

/* Listing 10.4: continued */
 * "sunsite.unc.edu:*"
 */
int
mkaddr(void *addr,
       int *addrlen,
       char *str_addr,
       char *protocol) {

    char *inp_addr = strdup(str_addr);
    char *host_part = strtok(inp_addr, ":");
    char *port_part = strtok(NULL, ":");
    struct sockaddr_in *ap =
        (struct sockaddr_in *) addr;
    struct hostent *hp = NULL;
    struct servent *sp = NULL;
    char *cp;
    long iv;

    /*
     * Set input defaults:
     */
    if ( !host_part )
        host_part = "*";
    if ( !port_part )
        port_part = "*";
    if ( !protocol )
        protocol = "tcp";

    /*
     * Initialize the address structure:
     */
    memset(ap, 0, *addrlen);
    ap->sin_family = AF_INET;
    ap->sin_port = 0;
    ap->sin_addr.s_addr = INADDR_ANY;

    /*
     * Fill in the host address:
     */
    if ( strcmp(host_part, "*") == 0 )
        ; /* Leave as INADDR_ANY */
    else if ( !isdigit(*host_part) ) {
        /*
         * Assume a hostname:
         */
        hp = gethostbyname(
             host_part);
        if ( !hp )
            return -1;
        if ( hp->h_addrtype != AF_INET )
            return -1;
        ap->sin_addr =
            *(struct in_addr *)
            hp->h_addr_list[0];
    } else {
        /*
         * Numeric IP address:
         */
        ap->sin_addr.s_addr =
            inet_addr(host_part);
        if ( ap->sin_addr.s_addr
             == INADDR_NONE )
            return -1;
    }

    /*
     * Process an optional port #:
     */
    if ( !strcmp(port_part, "*") )
        ; /* Leave as wild (zero) */
    else if ( isdigit(*port_part) ) {
        /*
         * Process numeric port #:
         */
        iv = strtol(port_part, &cp, 10);
        if ( cp != NULL && *cp )
            return -2;
        if ( iv < 0L || iv >= 32768 )
            return -2;
        ap->sin_port = htons(
            (short)iv);
    } else {
        /*
         * Lookup the service:
         */
        sp = getservbyname(
             port_part,
             protocol);
        if ( !sp )
            return -2;
        ap->sin_port =
            (short) sp->s_port;
    }

    /* Return address length */
    *addrlen = sizeof(*ap);

    free(inp_addr);
    return 0;
}

`mkaddr()` 函数在列表 10.4 中采用的基本步骤如下：

1. 在第 62 行使用 `strdup(3)` 复制输入字符串（这是为了稍后能够使用 `strtok(3)` 所必需的）。
2. 在第 63 行使用 `strtok(3)` 解析出主机字符串。
3. 在第 64 行使用 `strtok(3)` 解析出端口字符串。
4. 当提供 `NULL` 参数时，替换为默认字符串值（第 75 至 80 行）。
5. 初始化调用者的地址结构（第 85 至 88 行）。
6. 如果主机字符串匹配字符串 `"*"`，则保持值不变为 `INADDR_ANY`（第 94 行）。
7. 否则，如果主机名以数字开头，在第 99 至 103 行转换数字 IP 地址。
8. 否则，假设主机名是一个名称。查找它将第一个返回的 IP 编号放入地址中（第 108 至 116 行）。
9. 如果字符串匹配 `"*"`，则保持端口号为零（第 123 行）。
10. 否则，如果端口号以数字开头，提取数值并将其用作端口号（第 128 至 134 行）。
11. 否则，假设端口号是一个服务名称。查找它并将其分配给地址（第 139 至 145 行）。
12. 返回构建的地址长度（第 149 行）。
13. 释放步骤 1 中由 `strdup(3)` 创建的字符串（第 151 行）。
14. 返回零以指示地址转换成功。
