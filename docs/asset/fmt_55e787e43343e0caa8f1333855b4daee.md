```bash
length_of_string=${#string} #  NOTES。
reversed_string="" # reversed_string 初始为空（null）。
byte_counter=0

while [ $length_of_string -gt 0 ]
do
  ((byte_counter++)) # 为 byte_counter 加 1
  # 如果 byte_counter 是 9，那么就截取字符串的第 9 个字符：
  one_byte=$(echo $string | cut -c $byte_counter) # 截出一个字节
  reversed_string=$one_byte$reversed_string
  ((length_of_string--)) # 将 length_of_string 减 1
done
print "原始字符串: $string"
print "反转字符串: $reversed_string"
```
