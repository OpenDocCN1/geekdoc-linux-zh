  Move(S, P, Length(S));      // 将 'Sam' 移动到 P^

  Move(P, Temp[1], 1);       // 将 S 从 P 移动到 temp

  Move(P, Temp[2], 1);       // 将 a 从 P 移动到 temp

  DecChar(P, 2);              // 使 P 指向 Sam

end;
