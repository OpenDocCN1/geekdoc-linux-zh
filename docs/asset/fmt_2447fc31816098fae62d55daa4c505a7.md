```bash
#!/bin/sh
# 字符串比较
string1="abc"
string2="abd"

if [ $string1 = $string2 ]; then
    echo "string1 等于 string2"
else
    echo "string1 不等于 string2"
fi

if [ $string1 != $string2 ]; then
    echo "string1 不等于 string2"
else
    echo "string1 等于 string2"
fi

# 检查是否为空
if [ -z $string1 ]; then
    echo "string1 为空"
else
    echo "string1 不为空"
fi

if [ -n $string1 ]; then
    echo "string1 不为空"
else
    echo "string1 为空"
fi

# 检查类型
if [ -d $string1 ]; then
    echo "string1 是目录"
else
    echo "string1 不是目录"
fi

if [ -f $string1 ]; then
    echo "string1 是普通文件"
else
    echo "string1 不是普通文件"
fi
```

```text
dirl 是一个目录
filel 是一个普通文件
filel 有读取权限
filel 没有写入权限
dirl 有执行权限
```

```bash
#!/bin/sh
if [ -x filel -a -x dirl ]; then
    echo filel and dirl are executable
else
    echo at least one of filel or dirl are not executable
fi

if [ -w filel -o -w dirl ]; then
    echo filel or dirl are writable
else
    echo neither filel or dirl are executable
fi

if [ ! -w filel ]; then
    echo filel is not writable
else
    echo filel is writable
fi
```
