尽管可能有多个挂起的信号，但它们可以逐一个处理。图 18.31 所示的函数来自 `kernel/signal.c`，返回第一个未阻塞的信号。

```c
static int

next_signal(struct task_struct *tsk, sigset_t *mask)

{
    unsigned long i, x;

    int sig, *s, *m;

    s = tsk->pending;

    m = mask;

    switch (NSIG_WORDS) {

    default:

        for (i = 0; i < NSIG_WORDS; i++, ++s, ++m) {

            if (*s & ~m) {

                x = *s;

                sig = ffz(x) + i * NSIG_BW;

                goto out;

            }

        }

        return 0;

    case 2:

        if (s[0] & ~m[0]) {

            sig = ffz(s[0]);

        } else if (s[1] & ~m[1]) {

            sig = ffz(s[1]) + NSIG_BW + 1;

        } else {

            return 0;

        }

        break;

    case 1:

        if (s[0] & ~m[0]) {

            sig = ffz(s[0]);

        } else {

            return 0;

        }

        break;

    }

out:

    return sig;

}
```
