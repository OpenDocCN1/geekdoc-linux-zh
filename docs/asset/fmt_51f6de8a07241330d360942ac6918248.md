```bash
elif [ $FRUIT = STOP ]; then

	break

echo "You chose $FRUIT as your favorite."

	echo -e "That is choice number $REPLY.\n"

done
```

## 可选

下一个程序接受命令行上的两个文件名，对两者进行排序，并将输出发送到临时文件。然后程序将排序后的文件合并到标准输出，并在每行前面加上一个指示其来源的数字。

```bash
$ cat sortmerge

#!/bin/bash

usage ()

{

  if [ $ -ne 2 ]; then

    echo "Usage: $0 file1 file2" >&1

    exit 1

  fi

}

# 默认临时目录

: ${TMPDIR:=tmp}

# 检查参数计数

usage "$@"

# 为排序设置临时文件

file1=$TMPDIR/.$.file1

file2=$TMPDIR/.$.file2

# 排序

sort $1 > $file1

sort $2 > $file2

# 为读取打开 $file1 和 $file2。使用文件描述符 3 和 4。

exec 3<$file1

exec 4<$file2

# 从每个文件中读取第一行以确定如何开始。

read Line1<&3

status1=$?

read Line2<&4

status2=$?

# 策略：当两个文件中都有输入时：

# 输出应该先出现的行。

# 从该行所在的文件中读取新行。

while [ $status1 -eq 0 -a $status2 -eq 0 ];

do

    if [[ "$Line2" > "$Line1" ]]; then

        echo -e "1.\t$Line1"

        read Line1

        status1=$?

    else

        echo -e "2.\t$Line2"

        read Line2

        status2=$?

    fi

done

# 现在其中一个文件已到达末尾。

# 从每个文件读取到结束。

# 第一个 file1:

while [ $status1 -eq 0 ];

do

    echo -e "1.\t$Line1"

    read Line1<&3

    status1=$?

done

# 下一个 file2:

while [ $status2 -eq 0 ];

do

    echo -e "2.\t$Line2"

    read Line2<&4

    status2=$?

done

# 关闭并删除两个输入文件

exec 3<&-

rm -f $file1 $file2

exit 0
```

## 数组变量

Bourne Again Shell 支持一维数组变量。下标是整数，且从 0 开始索引（即数组的第一个元素下标为 0）。以下格式用于声明并为数组赋值值：

```bash
name=(element1 element2 ...)
```

以下示例为 `NAMES` 数组分配了四个值：

```bash
$ NAMES=(max helen sam zach)
```

你通过如下方式引用数组的单个元素：

```bash
$ echo ${NAMES[2]}
sam
```

下标 `[*]` 和 `[]` 都可以提取整个数组，但在引号内使用时工作方式不同。`@` 原始数组的副本；`*` 产生数组的单个元素（或普通变量），该元素包含 `IFS` 中的第一个字符（通常是空格）分隔的数组所有元素。在下面的示例中，数组 `A` 使用 `*` 用 `NAMES` 变量的元素填充，而 `B` 使用 `@` 填充。带有 `-` 选项的内置命令 `declare` 显示数组的值（并提醒你 `bash` 对数组使用从 0 开始的索引）：

```bash
$ A="${NAMES[*]}"
$ B="${NAMES[0]}"
```


