E_NOSTARTFILE=1

if [ !e "$startfile" ]; then

# 此脚本使用 10x10 网格（可以增加）
# 大网格会减慢执行。
# 修改变量以匹配大小。

DEAD=0

# 全局；持有当前代际。

display () {
    alive=0
    # 给定时间的活细胞。
    # 初始为 0。
    # 将参数转换为数组。
    arr=( `echo "$1"` )
    element_count=${#arr[@]}

    for ((i=0; i<$element_count; i++))
    do
        # 在行尾插入换行符。
        rowcheck=$((i % COLS))
        if [ "$rowcheck" -eq 0 ]; then
            # 换行。
            echo -n "    " # 缩进。
        fi

        cell=${arr[i]}

        if [ "$cell" = . ]; then
            alive += 1
        fi

        echo -n "$cell" | sed -e 's/_ /g' # 输出数组，将划线替换为空。
    done
    return
}

IsValid () { # 测试单元是否有效。
    if [ -z "$1" -o -z "$2" ]; then
        return FALSE
    fi
    local row
    local lower_limit=0 # 禁止负坐标。
    local upper_limit
    local left
    local right
    upper_limit=$((COLS-1)) # 单元总数。
    if [ "$1" -lt "$lower_limit" -o "$1" -gt "$upper_limit" ]; then
        return FALSE # 越数组边界。
    fi
    row=$2
    left=$((row * COLS)) # 左边界。
    right=$((left + COLS - 1)) # 右边界。

    if [ "$1" -lt "$left" -o "$1" -gt "$right" ]; then
        return FALSE # 越行边界。
    fi
    return $TRUE # 有效坐标。
}

IsAlive () # 测试单元是否存活。
    # 接收数组、单元编号和状态作为参数。
    GetCount "$1" $2
    local nhbd=$? # 获取邻居中的活细胞数量。

    if [ "$nhbd" -eq "$BIRTH" ]; then # 情况下存活。
        return ALIVE
    fi

    if [ "$3" = "." -a "$nhbd" -eq "$SURVIVE" ]; then # 只有之前存活时存活。
        return ALIVE
    fi

    return $DEAD # 默认死亡。
}

GetCount () # 计算通过单元邻居中的活细胞。
    # 需要两个参数：
    # $1) 持有数组的变量
    # $2) 单元编号
    local cell_number=$2
    local array
    local top
    local center
    local bottom
    local r
    local row
    local i
    local t_top
    local t_cen
    local t_bot
    local count=0
    local ROW_NHBD=3

    array=$(echo "$1")
    top=$((cell_number - $COLS - 1)) # 设置单元邻居。
    center=$((cell_number - *2))
    bottom=$((cell_number + COLS - 1))
    r=$((cell_number / COLS))

    for ((i=0; i<$ROW_NHBD; i++)) # 从左到右遍历。
    do
        t_top=$((top + $i))
        t_cen=$((center + $i))
        t_bot=$((bottom + $i))

        row=$r # 计算中心行。
        IsValid "$t_cen" $row # 有效的单元位置？
        if [ $? -eq "TRUE" ]; then
            if [ ${array[$t_cen]} ] = "ALIVE" ]; # 是活着的吗？
                # 如果是，那么
                count += 1
            fi
        fi

        row=$((r - 1)) # 计算上行。
        IsValid "$t_top" $row
        if [ $? -eq "TRUE" ]; then
            if [ ${array[$t_top]} ] = "ALIVE" ]; then # 此处冗余。
                # 可以优化吗？
                count += 1
            fi
        fi

        row=$((r + 1)) # 计算下行。
        IsValid "$t_bot" $row
        if [ $? -eq "TRUE" ]; then
            if [ ${array[$t_bot]} ] = "ALIVE" ]; then
                count += 1
            fi
        fi
    done

    if [ ${array[$cell_number]} ] = "ALIVE" ]; # 确保测试单元本身
        # 不被计入。
        count -= 1
    fi

    return $count
}

next_gen () { # 更新代际数组。
    local array
    local i=0
    array=( `echo "$1"` ) # 将解析的参数转换为数组。
    while [ "$i" -lt "$cells" ]

        isAlive "$1" ${array[$i]} # 单元活吗？
        if [ $? -eq "ALIVE" ]; then
            # 如果活，则
            array[$i]-=\. # 将单元表示为点。
        else
            array[$i]="__" # 否则下划线
            #（之后将转换为空格）。
        fi
        let i += 1
    done
}
