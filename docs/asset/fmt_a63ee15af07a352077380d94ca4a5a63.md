```bash
#!/bin/bash

command

command
# 不回显。
set -v
# 命令回显开启。
command
...
command
set +v
# 命令回显关闭。
command
exit 0
```

## 34.2. 封装器

“封装器”（wrapper）是一个封装系统命令或工具的脚本，它保存了传递给命令的一组参数。对于 `sed` 或 `awk` 特别有用。

`sed` 或 `awk` 通常从命令行调用。将此类脚本嵌入 Bash 脚本中可以简单地调用它，使其变得“可重用”。这允许组合 `sed` 和 `awk` 的功能，例如将 `sed` 的输出管道至 `awk`。

### 示例 34–1. 封装器

```bash
#!/bin/bash
# 这是一个删除文件中空行的简单脚本。
```

```bash
# 无参数检查。
# 与 `sed -e '^$/d' filename` 从命令行调用的方式相同。
sed -e '/$/d' "$"
# `-e` 意味着后面的“编辑”命令（此处）。`^` 是行开头，`$` 是行尾。匹配开头和结尾之间的空行。
引号命令允许文件名包含空格和特殊字符。
exit 0
```

### 示例 34–2. 稍微复杂的封装器

```bash
#!/bin/bash
# 在目标文件中添加指定的列。

ARGS=2
E_BADARGS=65

if [ $# -ne "$ARGS" ]; # 检查命令行参数是否正确
then
    echo "用法: $0 filename列数"
    exit $E_BADARGS
fi

old_pattern=$1
new_pattern=$2

if [ -f "$3" ]; then
    file_name="$3"
else
    echo "文件 $3 不存在。"
    exit $E_BADARGS
fi

# 这里是执行大量工作的地方。
sed -e "s/$old_pattern/$new_pattern/g" "$file_name"
# 's' 显然是 `sed` 中的替换命令，`/pattern/` 触发对 $old_pattern 的替换。
# "g" 全局标志会对每行的每一个出现处进行替换，而不只是第一个。
查阅 `sed` 文献以获取更深入的解释。

exit 0 # 脚本成功执行返回 0。
```
