static int
red_drop(struct qdisc *skb)
{
    struct sk_buff *skb;
    struct red_sched_data *q = (struct red_sched_data *)skb->data;
    skb = NULL; __skb_dequeue_tail(skb->q);
    if (skb) {
        skb->stats.backlog -= skb->len;
        skb->stats.drops++;
        q->stat.other++;
        kfree(skb);
        return 1;
    }
    PSCHED_GET_TIME(q->qidlestart);
    return 0;
}
