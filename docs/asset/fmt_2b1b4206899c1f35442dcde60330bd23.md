# 示例 A-6. Collatz 序列（碰撞猜想）

```bash
#!/bin/bash
# collatz.sh
# --------------------------------------------------
# 臭名的“hallstone”或 Collatz 序列。
# --------------------------------------------------
# 1) 从命令行获取整数“种子”。
# 2) NUMBER <-- 种子
# 3) 打印 NUMBER：
# 4) 如果 NUMBER 是偶数，除以 2；或者
# 5) 如果是奇数，乘以 3 并加 1。
# 6) NUMBER <-- 结果
# 7) 回到 3（循环指定的迭代次数）。

# 该理论是：每一个序列，
# 无论初始值有多大，
# 最终都会稳定在循环“4,2,1...”周期中，
# 即使在经历了大范围数值的波动之后。

# 这是“迭代”的一个实例，
# 这种操作将输出反馈回输入中。
# 有时结果是一个“混沌”序列。

ARGS=1
E_BADARGS=65

if [ $# -ne $ARGS ];    # 需要一个种子数字。
then
    echo "Usage: $0 NUMBER"
    exit $E_BADARGS
fi

MAX_ITERATIONS=200
# 对于较大的种子数字 (>32000)，请增加 MAX_ITERATIONS。

h=$1
# 种子
echo
echo "C($1) --- $MAX_ITERATIONS 次迭代"
echo

for ((i=1; i<=MAX_ITERATIONS; i++))
do
    echo -n "$h"
    #
    let remainder = h % 2
    if [ "$remainder" -eq 0 ]; then # 是偶数吗？
        let "h /= 2"          # 除以 2。
    else
        let "h = h*3 + 1"    # 乘以 3 并加 1
    fi

    COLUMNS=10 # 每行输出 10 个值。
    let "line_break = i % $COLUMNS"
    if [ "$line_break" -eq 0 ]; then
        echo
    fi

done
echo

# 关于此数学函数的更多信息，
# 请参阅 Pickover 的《Computers, Pattern, Chaos, and Beauty》，第 185 页及以后，
# 见参考文献列表。

exit 0
```
