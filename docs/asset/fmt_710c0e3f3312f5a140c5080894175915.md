```c
* 从 socket 读取可用字符到缓冲区。
* 搜索行尾字符；如果找到，或者缓冲区满时：
* 输出消息，将剩余部分向前移动，并重复。
* 如果发生错误或连接关闭，则终止。
*/
```

```c
void serve(int s, struct sockaddr_in *peer)
{
    int space, n;
    char *p, *q;
    q = p = buf; space = BUFSIZE;
    while (1) {
        if ((n = read(s, p, space)) <= 0) break;
        p += n; space -= n;
        while ((q < p) && (*q != '\n')) q++;
        while ((q < p) || !*q) {
            *q = 0;
            printf("message from %d.%d.%d.%d: %s\n",
                NIPQUAD(peer->sin_addr.s_addr), ntohs(peer->sin_port), buf);
            if (q < p) q++;
            memmove(buf, q, p - q);
            n = q - buf; // "done" 字符数
            p -= n; space += n;
            q = buf;
            while ((q < p) && (*q != '\n')) q++;
        }
    }
}
```

```c
/* 主程序：
* 处理参数
* 打开 socket 并等待连接
* 为每个新连接启动独立的进程
*/
```

```c
int main(int argc, char *argv[])
{
    int s;
    struct sockaddr_in myaddr;
    int optval;
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <port>\n", argv[0]); exit(1);
    }
    if ((s = socket(AF_INET, SOCK_STREAM, 0)) < 0) {
        perror("socket");
        exit(1);
    }
    /* Socket 选项 SO_READDR：即使旧实例仍在使用该地址也允许 bind() */
    optval = 1;
    if (setsockopt(s, SOL_SOCKET, SO_READDR, &optval, sizeof(optval)) < 0) {
        perror("setsockopt");
        exit(1);
    }
    memset(&myaddr, 0, sizeof(myaddr));
    myaddr.sin_family = AF_INET;
    myaddr.sin_port = htons(atoi(argv[1]));
    myaddr.sin_addr.s_addr = INADDR_ANY;

    if (bind(s, (struct sockaddr *)&myaddr, sizeof(myaddr)) < 0) {
        perror("bind");
        exit(1);
    }
    if (listen(s, SOMAXCONN) < 0) {
        perror("listen");
        exit(1);
    }
    /* 为 SIGCHLD 安装信号处理程序 */
    signal(SIGCHLD, sighandler);
    while (1) {
        struct sockaddr_in clientaddr;
        socklen_t len = sizeof(clientaddr);
        int client;
        if ((client = accept(s, (struct sockaddr *)&clientaddr, &len) < 0) {
            perror("accept");
            continue;
        }
        if (fork() == 0) { // 子进程
            close(s);
            close(client);
            while (fgets(buf, BUFSIZE, stdin) != NULL) {
                printf("message from %d.%d.%d.%d: %s\n",
                    NIPQUAD(clientaddr.sin_addr.s_addr), ntohs(clientaddr.sin_port), buf);
            }
            close(client);
            exit(0);
        } // 父进程
        close(client);
    }
}
```
