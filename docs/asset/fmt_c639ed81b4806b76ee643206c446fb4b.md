`.lib/sh/init-functions`

```bash
case "$1" in

start)

  log_info_msg "正在重试失败的 uevents（如果有）..."

  rundir=/run/udev

  # 来自 Debian："在 / 以读写模式挂载前生成的规则：”

  for file in ${rundir}/rules*; do

    dest=${file##*-}

    [ $dest ] || break

    cat "$file" >> /etc/udev/rules.d/$dest

    rm -f "$file"

  done

  # 重触发可能失败的 uevents，
  # 希望它们现在可以成功

  /bin/sed -e 's/.*//' /etc/sysconfig/udev_retry | /bin/grep -v '^$' | \
  while read line ; do
    udevadm trigger --subsystem=$line --action=add
  done

  # 现在等待 udevd 处理我们触发的 uevents

  if ! is_true "OMIT_UDEV_RETRY_SETTLE"; then
    /bin/udevadm settle
  fi

  evaluate_retval
  ;;

*)

  echo "用法: $0 (start)"
  exit 1
  ;;

esac

exit 0
```

# 在启动时创建文件/目录的函数

```bash
create_files()
{
  # 输入到文件描述符 9 输出到标准输入（重定向）
  exec 9>4 < /etc/sysconfig/createfiles

  while read name type grp gtype maj junk
  do
    # 忽略注释和空行
    case "${name}" in
      ""|#*) continue;
    esac

    # 忽略已存在的文件
    if [ ! -e "$(name)" ]; then
      # 根据类型创建内容
      case "${type}" in
        dir)
          mkdir "$(name)"
          ;;
        file)
          > "$(name)"
          ;;
        dev)
          case "${type}" in
            char)
              mkdn "$(name)" c ${maj} ${min}
              ;;
            block)
              mkdn "$(name)" b ${maj} ${min}
              ;;
            pipe)
              mkdn "$(name)" p
              ;;
          *)
            log_warning_msg "\n未知设备类型: ${type}"
            ;;
          esac
          ;;
        *)
          log_warning_msg "\n未知类型: ${type}"
          continue
          ;;
      esac

      # 同时设置权限
      chown $(usr):$(name)"
      chmod $(perm) "$(name)"
    fi
  done

  # 关闭描述符 9（结束重定向）
  exec 0>&4-9&-
  return 0
}
```
