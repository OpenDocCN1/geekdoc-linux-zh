/* * sockserv.c:
 *
 * 这个简单的服务器将把套接字
 * 提供给有效的接收者：
 */
#include "common.h"
#include <pwd.h>

/*
 * 检查用户的访问权限：
 *
 * 返回：
 *   ptr    如果授予，指向 (struct passwd *)。
 *   NULL   拒绝访问。
 */
static struct passwd *
check_access(
        int port,          /* 请求的端口 */
        struct ucred *pcred, /* 用户凭据 */
        char **uidlist) { /* 有效用户列表 */
        int x;
        struct passwd *pw; /* 用户 passwd 条目 */

        /*
         * 在 /etc/passwd 文件中
         * 查找用户的 uid 号：
         */
        if ( ( pw = getpwnuid(pcred->uid) ) == 0 ) {
/*
         * 确保请求来自
         * 可接受的用户之一：
         */
        for ( x=0; uidlist[x]; ++x )
            if ( !strcmp(uidlist[x],pw->pw_name) )
                break;
            if ( !uidlist[x] )
                pw = 0;    /* 拒绝访问 */
        }

        /*
         * 筛选端口号。对于这个演示，
         * 只允许 80 端口。
         */
        if ( port != 80 )
            pw = 0;         /* 拒绝访问 */

    return pw;  /* 授予则返回 NULL 或指针 */
}

/*
 * 访问已授予：将套接字
 * 发送给客户端。
 *
 * 参数：
 *    c      客户端套接字
 *    port   请求的端口
 *
 * 返回：
 *    0       成功
 *    -1      失败：检查 errno
 */
static int
grant_access(int c,int port) {
    int z;
    int fd = -1;             /* 新套接字 fd */
    int b = TRUE;            /* 布尔 TRUE */
    struct sockaddr_in addr; /* 工作地址  */

    /*
     * 创建一个新的 TCP/IP 套接字：
     */
    fd = socket(PF_INET,SOCK_STREAM,0);
    if ( fd == -1 ) {
       perror("socket(2)");
       goto errxit;
    }

    /*
     * 打开 SO_REUSEADDR：
     */
    z = setsockopt(fd,SOL_SOCKET,
                   SO_REUSEADDR,&b,sizeof b);
    if ( z == -1 )
       bail("setsockopt(2)");

    /*
     * 创建要绑定的地址：
     */
    memset(&addr,0,sizeof addr);
    addr.sin_family = AF_INET;
    addr.sin_port = ntohs(port);
  addr.sin_addr.s_addr = ntohl(INADDR_ANY);   

  /*
   * 绑定请求的地址：
   */
  z = bind(fd,
      (struct sockaddr *)&addr,
      sizeof addr);
  if ( z == -1 ) {
     fprintf(stderr,"s: binding port %d\n",    
            strerror(errno),port);
     goto errxit;
  }

  /*
   * 将 fd 发送回
   * 请求的客户端：
   */
  z = send_fd(c,fd,NULL,0);
  if ( z == -1 ) {
     perror("send_fd()");
     goto errxit;
  }

  close(fd);      /* 结束对 fd 的使用 */
  return 0;                  /* 成功*/

  errxit:
  z = errno;               /* 保存 errno */
  if ( fd )
    close(fd);         /* 释放套接字 */
    errno = z;           /* 恢复 errno */
    return -1;
  }

  /*
   * 处理已连接客户端的请求：
   */
  void
  proc_client(
     int c,             /* 客户端套接字 */
     char *uidlist      /* 有效用户列表  */
    ) {
    int z;
    int er;            /* 捕获的 errno 值 */
    int b = TRUE;      /* 布尔：真 */
 struct ucred cred; /* 客户端凭据 */
 short port; /* 请求的端口 */
 struct passwd *pw; /* 用户 passwd 条目 */

 /*
  * 现在确保我们可以
  * 在此套接字上接收凭据：
  */
 z = setsockopt(c,
               SOL_SOCKET,
               SO_PASSCRED,
               &b,
               sizeof b);
 if ( z )
     bail("setsockopt(2)");

 /*
  * 接收带有
  * 用户凭据的请求：
  */
 z = recv_cred(c, /* 套接字 */
               &cred, /* 返回的凭据 */
               &port, /* 返回的端口号 */
               sizeof port, /* 数据大小 */
               NULL, 0); /* 无套接字地址 */

 if ( z == -1 )
     perror("recv_cred()");

 /*
  * 现在检查访问权限。 如果 pw
  * 作为非 NULL 返回，则请求有效。
  */
 pw = check_access(port,&cred,uidlist);

 if ( pw ) {
     if ( grant_access(c,port) ) {
         close(c);
         return; /* 请求成功 */
     }
     /* 失败 */
     er = errno; /* 捕获原因 */
 } else {
     /*
     * UserId 未知，或不在
     * 特权列表中：
     */
     err = EACCES;    /* 权限被拒绝 */
 }
 
 /*
  * 如果请求失败
  * 或被拒绝，控制流到达此处：
  *
  * 这里我们只是发送错误
  * 代码，而不带文件
  * 描述符。这种缺少 fd
  * 将被客户端检测到。
  */
 do {
     z = write(c,&err,sizeof er);
 } while ( z == -1 && errno == EINTR );
 
     if ( z == -1 )
         perror("write(2)");
 }

  * 主程序：
  */
 int
 main(int argc,char **argv) {
     int z;
     int s;        /* 服务器 UDP 套接字 */
     int c;        /* 客户端套接字 */
     int alen;           /* 地址长度 */
     struct sockaddr_un asrv;  /* 服务器 @ */
     struct sockaddr_un acint; /* 客户端 @ */
 
 /*
  * 确保指定了 userid：
  */
     if ( argc < 2 ) {
         fputs("必须至少有 1 个 userid.\n",
             stderr);
         exit(1);
     }
 
 /*
