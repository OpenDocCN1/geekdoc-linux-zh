class Ball(Widget):
    def update(self, dt):
        self.pos_hint_x += self.velocity[0] * dt
        self.pos_hint_y += self.velocity[1] * dt
        if self.right > self.parent.right: # 从右侧反弹
            self.velocity[0] *= -self.velocity[0]
        if self.left < self.parent.left: # 从左侧反弹
            self.velocity[0] *= -self.velocity[0]
