    then

      echo -n "Starting professional ftp daemon: " --start --quiet --pidfile /var/run/$NAME.pid \
        --exec $DAEMON -- $proftd_options
      then
        echo "$NAME."
    else
        echo
      fi
    fi
    ;;

  stop)
    if [ $run_proftd = 1 ]
    then
      echo -n "Stopping professional ftp daemon: "
      if killall $NAME > /dev/null 2>&1
      then
        echo "$NAME."
      else
        echo
      fi
    fi
    ;;

  restart|reload)
    $0 stop
    $0 start
    ;;

  *)
    echo "Usage: /etc/init.d/ppd (start|stop|restart|reload)"
    exit 1
    ;;
esac

exit 0

## rc

`rc` 脚本是一个关重要的脚本，因为它负责在运行级别 `runlevel` 改变时启动和停止各种服务。

```bash
#! /bin/bash
#
# rc    此文件负责在运行级别改变时
#      启动/停止服务。
#
#      优化功能：
#      如果服务在之前的运行级别中正在运行，且在
#      运行级别转换过程中没有被停止，则不运行启动脚本
#      （大多数 Debian 服务）这些服务
#      在 rc(1,2,3,4,5) 中没有 K 链接。
#
# 作者: Miquel van Smoreenburg <miquels@cistron.nl>
#      Bruce Perens <Bruce@Pixar.com>
#
# 版本: @(#)rc 2.73 26-Nov-1997 miquels@cistron.nl
#
# 注消以下内容以进行调试。
# debug=echo
#
# 启动脚本或程序。
#
startup() {
  case "$1" in
    *.sh)
      $debug sh "$@"
      ;;
    *)
      $debug "$@"
      ;;
  esac
}

# 仅在此 shell 中忽略 CTRL-C，以便我们可以中断子进程。
trap ":*" QUIT TSTP

# 设置 onclcr 以避免阶梯效应。
stty onclcr 0>2&1

# 现在查找当前运行级别和之前的运行级别。

runlevel=$RUNLEVEL
# 获取第一个参数。将新的运行级别设置为此参数。
[ "$1" != "" ] && runlevel=$1

previous=$PREVLEVEL
export runlevel previous
```

```bash
# 是否存在这个新运行级别的 rc 目录？
if [ -d /etc/rc$runlevel.d ]
then
    # 首先，运行 KILL 脚本。
    if [ $previous != N ]
    then
        for i in /etc/rc$runlevel.d/$K[0-9][0-9]*
        do
            # 检查脚本是否存在。
            [ ! -f $i ] && continue
            # 停止服务。
            startup $i stop
        done
    fi
    # 现在运行此运行级别的 START 脚本。
    for i in /etc/rc$runlevel.d/$S*
    do
        [ ! -f $i ] && continue

        if [ $previous != N ]
        then
            # 在上一个运行级别查找启动脚本，
            # 在当前运行级别查找停止脚本。
            suffix=${i#/etc/rc$runlevel.d/$S[0-9][0-9]}
            stop=/etc/rc$runlevel.d/$K[0-9][0-9]$suffix
            previous_start=/etc/rc$previous.d/$S[0-9][0-9]
            # 如果上一个级别有启动脚本
            # 且当前级别没有停止脚本，我们不需要
            # 重新启动服务。
            [ -f $previous_start ] && [ -f $stop ] && continue
        fi
        case "$runlevel" in
            0)
                startup $i stop
                ;;
            *)
                startup $i start
                ;;
        esac
    done
fi
```
