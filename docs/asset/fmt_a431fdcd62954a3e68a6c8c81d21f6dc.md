## 12 主循环 – 部分 2

```python
def de_dead(self):
    if self.lives < 8:
        self.screen.blit(self.game_font.render(
            f'SCORE: {self.score}', True, (255, 0, 0)), (150, 255)
        )
        self.lives -= 1
        self.screen.refresh()
        pygame.time.delay(1000)
def win_round(self):
    if len(self.alien_group) > 0:
        self.rounds += 1
        self.screen.blit(self.game_font.render(
            f'ROUNDS: {self.rounds}', True, (0, 255, 0)), (150, 255)
        )
        pygame.time.delay(1000)
    return True
```
