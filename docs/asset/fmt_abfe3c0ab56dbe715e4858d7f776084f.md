```perl
sub sub_check_url {

    # 本地化 check_referer 标记，确定用户是否有效。
    local($check_referer) = 0;

    # 如果指定了引用 URL，对于每个有效的 referer，确保
    # 传递了一个有效的引用 URL 给 FormMail。
    if ($_ENV['HTTP_REFERER']) {
        foreach $referer (@referers) {
            if ($_ENV['HTTP_REFERER'] =~ m|https?://([^/]*)$referer|i) {
                $check_referer = 1;
            }
        }
    }
    else {
        $check_referer = 1;
    }
}
```
