
```
/* 打开串口设备文件
 * O_NDELAY - 告诉端口运行并忽略 DCD 线
 * O_NOCTTY - 此进程不会成为端口的控制进程。驱动程序不会由于键盘中断等发送这些信号 */
if ((fd = open(argv[1], O_RDWR | O_NDELAY | O_NOCTTY)) < 0) {
    printf("Couldn't open %s\n", argv[1]);
    exit(1);
}

/* 我们不关心保留旧的串口配置
 * CS8, 8 数据位
 * CREAD, 开启接收器
 * CLOCAL, 不改变端口所有者 */
tio_c_cflag = baud | CS8 | CREAD | CLOCAL;

tio_c_cflag &= ~HUPCL; /* 清除 HUPCL 位，close 不改变 DTR */

tio_c_lflag = 0;             /* 设置为非原始输入，无处理 */
tio_c_iflag = IGNPAR;        /* 忽略校验错误 */
tio_c_oflag = 0;             /* 设置为非原始输出，无处理 */

tio_c_cc[VTIME] = 0;         /* 无延迟 */
tio_c_cc[VMIN] = 0;          /* 无字符延迟 */

tcflush(fd, TCIFLUSH);       /* 刷新缓冲区 */
tcsetattr(fd, TCSANOW, &fd); /* 设置属性 */

/* 设置无延迟、非阻塞读取 */
/* fcntl 会设置非阻塞读取，querySerial 返回所有收到的字符 */
/* 从串口端口发送查询 */
result = write(fd, argv[4], strlen(argv[4]));
if (result < 0) {
    perror("write failed");
    close(fd);
}

/* 等待指定的毫秒 */
usleep(atoi(argv[3]) * 1000);

/* 读取并打印缓冲区 */
result = read(fd, buffer, 255);
buffer[result] = 0;
printf("%s", buffer);

/* 关闭 */
close(fd);

return 0;
```

# 第 7 章 并行接口

控制单个制雪分支路需要为此交流设备提供四路控制输出：

- 水阀 1
- 水阀 2
- 水阀 3
- 加热器 1

某些制雪分支路拥有六个水阀和两个加热器。工程师们开始探索支持六路输入和四个输出（具有八路输出能力）的工艺控制方案。市场上存在许多输入/输出（I/O）板，来自如 Computer Boards (www.computerboards.com) 等公司，它们在 I/O 能力和尺寸上都超过了 Project Trailblazer。工程师们决定，由于 Project Trailblazer 的 I/O 需求适中，板载 PC 并行口非常适合升缆道监控和制雪控制。

> **提示**
> 你应该使用固态继电器模块将外部设备连接到目标板。存在的类型有：交流电输入 (`IAC`)、交流电输出 (`OAC`)、直流输入 (`IDC`) 和直流输出 (`ODC`)。输入和输出模块均提供电学隔离，可以保护目标板的敏感电路。安装机架具有模块插座、保险丝、显示操作状态的发光二极管 (`LED`) 以及用于现场连接的隔离条。

本章讨论了 PC 并行口、硬件接口以及用于数据采集和控制的三种并行口软件开发方法。第一种方法 `io` 非常简单，清晰地演示了开发设备驱动程序所需的低级硬件控制。第二种方法 `pdev` 使用 Linux 的并行口设备驱动程序来控制端口。这种方法通过使用设备文件来实现，类似于串口控制；Project Trailblazer 的工程师曾使用端口 I/O 或 `pdev` 实现了升缆道监控和制雪控制的解决方案；但他们决定采用第三种方法；他们创建了自己使用 `/proc` 目录的自定义设备驱动程序。这种解决方案将简化他们未来的 bash 脚本。起初，工程师们对开发设备驱动程序的想法非常畏惧。他们是出色的 C 语言程序员，但没有人对设备驱动程序、内核模块或 `/proc` 目录有所知。在成功开发设备驱动程序后，他们对 Linux 有了新的理解。

## 使用并行口控制

并行口有时被称为打印机端口，最初设计用于将打印机连接到计算机。多年来，许多其他设备如扫描仪、拆卸驱动器和复制保护设备也通过并行口连接到 PC。在当今的 PC 中，该端口有八条数据线、四条双向控制线和五条状态输入线，通过 DB-25 母头连接器访问。自诞生以来，并行口演变成了四种类型：

- 标准并行口 (SPP)
- PS/2 类型（简单双向）端口
- 增强型并行口 (EPP)
- 扩展功能端口 (ECP)

IEEE 1284 标准“个人计算机双向并行接口的 IEEE 标准信号方法”定义了这四种类型，并在本文中总结如下：

- Warp Nine Engineering 的《IEEE 1284-1994 标准简介》，www.fapo.com/1284int.htm。

其他在线文档充分解释了这些类型之间的相似之处和差异，如下：

- Jan Axelson 的《Parallel Port Complete Programming, Interfacing, & Using the PC's Parallel Printer Port》，www.lvr.com/files/ppc1.pdf。
- Craig Peacock 的《Interfacing the Standard Parallel Port》，www.boundarylogic.org/spp/parallel.pdf。
- Craig Peacock 的《Interfacing the Enhanced Parallel Port》，www.boundarylogic.org/epp/epp.pdf。
- Craig Peacock 的《Interfacing the Extended Capabilities Port》，www.boundarylogic.org/ecp/ecp.pdf。

Trailblazer 使用六路输入和四个输出。通过查阅在线文档，工程师确定双向 SPP 类型端口足够了。端口不需要控制高速打印机——只需要一些低速输入和输出即可。

SPP 硬件通常通过端口 I/O 映射到 x86 ISA 总线（即端口不是内存映射），I/O 基地址为 `0x378`。通过使用主板 BIOS，你可以更改此地址和端口类型。在本章中，你 PC 的 BIOS 将 LPT1 配置为基地址 `0x378` 的 SPP 类型。三个软件寄存器（数据、状态和控制）存在于基地址、基地址 + 1 和基地址 + 2（即 `0x378`、`0x379` 和 `0x37A`）。图 7.1 显示了这些软件寄存器、它们的位定义以及与 DB-25 并行口连接的硬件连接。

## 使用端口 I/O 控制标准并行口

在 Linux 中进行 I/O 端口操作再简单不过。有一个用于输入的命令——`inb(address)`，和一个用于输出的命令——`outb(value, address)`。C 宏展开实现了这两个命令，无需任何库。

尽管如此，`outb` 确实需要几行额外的代码。`ioperm` 命令会从内核释放对端口的访问权限。通过并行端口写入字节可能像这三行代码一样简单：

```c
if (ioperm(0x378, 3, 1)) exit(1);
outb(argv[1][0], 0x378);
if (ioperm(0x378, 3, 0)) exit(1);
```

同样，读取一个字节也很简单。你可以使用以下代码读取并行端口的状态线：

```c
if (ioperm(0x378, 3, 1)) exit(1);
putc(inb(0x379));
if (ioperm(0x378, 3, 0)) exit(1);
```

这段代码可能会输出 ASCII 值 0，并可能影响 bash 脚本。对这些代码片段进行简单的添加即可实现对制雪设备的提升运行监控和控制。

# 使用 I/O 监控提升运行

使用 `inb` 和 `outb` 端口 I/O 调用，可以通过接口读取索道的当前状态。如列表 7.1 所示，`liftermonitor.c` 通过配置并行数据端口为输入、断言输入缓冲区使能信号 `INPUT_ENABLE` 并读取数据线 D7–D0 来执行输入操作。它将数据端口位以八个 ASCII 字符（1 或 0）的形式输出一种便于脚本处理的版本。

```c
 *

 * 连接到 PC 并行打印机端口的接口
 * 电路。端口的控制端口控制
 * 电路数据线上的输入/输出操作。
 */

/* 记得使用 -O2 编译以进行正确的 inb/outb 宏展开
 * gcc -O2 -o liftoperationmonitor liftermonitor.c
 */
#include <asm/io.h>

#define SPPDATAPORT 0x378
#define SPPSTATUSPORT (SPPDATAPORT + 1)
#define SPPCONTROLPORT (SPPDATAPORT + 2)

/* 这些是控制端口位定义 */
#define OUTPUT_ENABLE 0x02
#define OUTPUT_LATCH 0x04
#define INPUT_ENABLE 0x08
#define SPPORTREAD 0x20

int main(int argc, char *argv[])
{
    unsigned char v, i;

    /* 从操作系统获取使用数据、状态和控制端口的权限 */
    if (ioperm(SPPDATAPORT, 3, 1)) {
        perror("ioperm");
        exit(1);
    }

    /* 这断言了三项：
     * 1. SPPORTREAD，将双向数据端口配置为输入
     * 2. INPUT_ENABLE，使能输出锁存，驱动输出插槽
     * 3. OUTPUT_ENABLE，使能输入缓冲区以驱动数据总线
     * 如果你不断言这些项，输出模块将关闭
     */
    outb(SPPORTREAD | INPUT_ENABLE | OUTPUT_ENABLE, SPPCONTROLPORT);

    /* 输入缓冲区现在正在驱动总线，因此执行读取 */
    v = inb(SPPSTATUSPORT);

    /* 取消 SPPORTREAD 和 INPUT_ENABLE。
     * 使用 OUTPUT_ENABLE 保持输出锁存启用状态
     */
    /* 遍历 v 中的位并输出 8 个 0 或 1，例如 01000001
     * 最高位 (MSB) D7 首先输出
     */
    for (i = 0; i < 8; i++) {
        if (v & 0x80)
            putchar('1');
        else
            putchar('0');

        v <<= 1; /* 左移 */
    }

    /* 将命令行参数1的第一个字符写入数据总线 */
    if (argc > 1) {
        outb(argv[1][0], SPPDATAPORT);
    }

    /* 告诉操作系统我们完成了端口操作 */
    if (ioperm(SPPDATAPORT, 3, 0)) {
        perror("ioperm");
        exit(1);
    }
}
```

## 使用 ppdev 控制标准并行端口

下一节描述的并行端口控制方法——使用并行端口设备驱动程序（`ppdev`）——与端口 I/O 方法具有相同的两个缺点。在接下来的节中，你将使用 `ppdev` 代替 `outb` 实现 `snowcontrol.c` 的功能。虽然这里为了 Linux 的完整性包含了这部分内容，但它并不被认为是制雪控制的可行方案。在本章稍后开发的自定义设备驱动程序没有缺点，并为电梯监控和制雪控制实现了一个稳健且精妙的解决方案。

## 使用 ppdev 控制标准并行端口

使用 `inb` 和 `outb` 并不是查询和控制 PC 并行端口数据、控制和状态寄存器的唯一方法。Linux 并行端口驱动程序 `ppdev` 也允许通过 `ioctl` 进行这种控制。这种方法与第 6 章“异步串口通信接口”中用于控制串口的方法类似。程序会打开一个文件（`/dev/parport0`）并返回一个文件描述符。通过该描述符，可以使用 `ioctl` 设置单个控制位、查询端口状态并向端口数据总线读取。列表 7.3 中的程序 `snowcontrolppdev.c` 使用 `ppdev` 来控制图 7.3 中接口电路的并行端口寄存器。（Linux 2.4 并行端口子系统》提供了对 `parport` 和 `ppdev` 的完整讨论）。以下是 `snowcontrolppdev` 程序采取的步骤：

1. 它打开 `/dev/parport0` 文件。
2. 它通过 `PPCLAIM` 请求访问端口。
3. 它使用 `IEEE1284_MODE_COMPAT` 将端口配置为 SPP 模式。
4. 它设置 `OUTPUT_ENABLE` 线用于锁存输出驱动。
5. 它将命令行参数 1 写入数据总线。
6. 它翻转 `OUTPUT_LATCH` 线将数据锁存到输出模块。
7. 它释放端口。
8. 它关闭 `/dev/parport0` 文件。
