# UPS 连接并工作正常。
pf::powerfail:/sbin/shutdown -f -h x2 "Power Failure; System Shutting Down"

# 如果在关机启动前恢复电源，则取消。
pr:12345:powerwick:/sbin/shutdown -c "Power Restored, Shutdown Cancel"

# 在标准运行级别下运行 getty
