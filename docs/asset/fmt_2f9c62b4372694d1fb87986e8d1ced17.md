### basic.c - 源代码

```c
#include <stdio.h>
#include <pthread.h>
#include <stdlib.h>
#include <semaphore.h>

void* thread_a(void* arg) {
    // 等待信号量
    semaphore_wait(&sem);
    printf("Thread A finished\n");
    return NULL;
}

void* thread_b(void* arg) {
    // 等待信号量
    semaphore_wait(&sem);
    printf("Thread B finished\n");
    return NULL;
}

semaphore_t sem;

int main() {
    pthread_t t1, t2;
    sem_init(&sem, 0, 1);
    pthread_create(&t1, NULL, thread_a, NULL);
    pthread_create(&t2, NULL, thread_b, NULL);
    pthread_join(&t1, NULL);
    pthread_join(&t2, NULL);
    sem_destroy(&sem);
    return 0;
}
```
