IsValid ()    # 测试单元坐标是否有效。

{

  if [ -z "$1" -o -z "$2" ]    # 缺少强制参数？

  then

    return FALSE

  fi

  local row

  local lower_limit=0

  local upper_limit

  local left

  local right

  let "upper_limit" = $ROWS * $COLS - 1  # 单元格总数。

  if [ "$1" -lt "$lower_limit" -o "$1" -gt "$upper_limit" ]

  then

    return FALSE    # 越数组边界。

  fi

}

# - 是“默认替换”操作符。
# 记住如果 'n' 尚未初始化，则将其设置为 1。

  PASS=$SPASS${($RANDOM%${#MATRIX})):1}
  # --> 非常聪明，但很复杂。

  # --> ${#MATRIX} 返回数组 MATRIX 的长度。
  # --> ${RANDOM%${#MATRIX}} 返回 1 到

  # --> 看到 PASS 的构建，
