
# Makefile 配置

```make
CDEBUG = -g
CFLAGS = $(CDEBUG) -I -I$(srcdir) $(DEFS) \
-DDEF_AR_FILE=\$(DEF_AR_FILE) \
-DEFBLOCKING=$(DEFBLOCKING)
LDFLAGS = -g
prefix = /usr/local
# 每个安装程序的前缀，
# 通常为空。
binprefix =
```

```make
# 安装 tar 的目录。
bindir = $(prefix)/bin
```

```make
# 安装 info 文件的目录。
infodir = $(prefix)/info
#### 系统配置部分结束 ####
```

```make
SRC1 = tar.c create.c buffer.c
```

```make
SRC2 = getoldopt.c update.c gnu.c
version.c list.c diffarch.c
port.c wildmat.c getoldopt.c \
SRC3 = getopt.c getdate.y
SRCS = $(SRC1) $(SRC3)
OBJ1 = tar.o create.o buffer.o
getoldopt.o update.o gnu.o
OBJ2 = version.o list.o diffarch.o
port.o wildmat.o getoldopt.o
OBJ3 = getopt.o getdate.o $(RTAPELIB)
OBJS = $(SRC1) $(OBJ2) $(SRC3)
AUX = README ChangeLog.in
configure.in
tar.texinfo *.tex
tar.port.h regex.h
rmt.h rtapelib.c allocate.c
dir.h cexpang.c
level-1 backup-specs testpad.c
all:
tar.info
tar:\ $(OBJS)
$(CC) $(LDFLAGS) -o $@ $(OBJS) $(LIBS)
rmt:
rmt.c
$(CC) $(LDFLAGS) -o $@ rmt.c
tar.info: tar.texinfo
install: all
$(INSTALL) tar $(bindir)/$(prefix)tar
-f rmt || $(INSTALL) rmt /etc
$(srcdir)/tar.info $(infodir)
$(OBJS): tar.h port.h
regex.o buffer.o tar.o: regex.h
# getdate.y has shift/reduce conflicts
testpad.h: testpad
./testpad
testpad: testpad.o
clean:
rm -f *.o rmt testpad.h
core

distclean: clean
rm -f TAGS Makefile config.status
realclean: distclean
rm -f tar.info*
```

```make
shar: $(SRCCS) $(AUX) | compress
  shar $(SRCCS) -sed -e '/version_string/!d' \
  -e 's/[0-9]*\([0-9]*).*/\1/' \
  -e q
  version.c'.shar.Z
```

```make
dist: $(SRCCS) $(AUX)
echo tar-'sed '\'
  -e '/version_string/!d' \
  -e 's/\\[0-9]*\([0-9]*).*/\1/' \
  -e q
  version.c'.filename
rm -rf .cat .filename
mkdir .cat .filename
ln $(SRCCS) $(AUX) 'cat.filename'
tar zf 'cat.filename'.tar.Z 'cat.filename'
rm -rf 'cat.filename' .filename
```

```make
tar.zoo: $(SRCCS) $(AUX)
-rm -rf tmp.dir
mkdir tmp.dir
-rm tar.zoo
for X in $(SRCCS) $(AUX) ; do \
  echo $$X; '\'
  sed 's/\$/\M/', $$X '\'
  > tmp.dir/$$X; done
cd tmp.dir; zoo AM ../tar.zoo *
-rm -rf tmp.dir
```
