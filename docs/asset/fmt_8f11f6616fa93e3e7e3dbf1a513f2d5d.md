function twirl
  TCOUNT="0"  # 每마다 TCOUNT，行旋转一个增量
  while : # 无限循环...直到跳出循环
  do
    TCOUNT=$((TCOUNT) + 1) # 增加 TCOUNT
    case ($TCOUNT) in
      "1"
        echo "\-'''\b\c"
        sleep 1
        ;;
      "2"
        echo "\\\''\\\b\c"
        sleep 1
        ;;
      "3"
        echo "\b\c"
        sleep 1
        ;;
      "4"
        echo "/b\c"
        sleep 1
        ;;
      *)
        TCOUNT="0" ;; # 将 TCOUNT 重置为 "0"，零。
    esac
  done
# twirl function 结束
