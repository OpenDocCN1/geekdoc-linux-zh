### `__vma_adjust()` 初始状态建立

```c
struct file *file = vma->vm_file;
bool start_changed = false, end_changed = false;
long adjust_next = 0;
int remove_next = 0;

if (next && insert) {
    struct vm_area_struct *exporter = NULL, *importer = NULL;

    if (end && next->vm_end) {
        /*
         * vma 与 next 重叠，并且
         * 可能与下一个重叠 (mprotect 情况 6)。
         * 进入这里的其他情况是
         * 情况 1、情况 7 和情况 8。
         */
        if (next == expand) {
            /*
             * 我们不扩展 "vma"
             * 而是扩展 "next" 的唯一情况是情况 8。
             */
            VM_WARN_ON(end != next->vm_end);
            /*
             * remove_next == 3 意味着我们
             * 正在移除 "vma"，并且
             * 交换了 "vma" 和 "next"。
             */
            remove_next = 3;
            VM_WARN_ON(file);
            swap(vma, next);
        } else {
            VM_WARN_ON(expand);
            /*
             * 交换了 vma 和 next。
             */
            remove_next = 3;
            VM_WARN_ON(file);
            swap(vma, next);
        }
    }
    exporter = next;
    importer = vma;
}
```

> **注**：原文结尾存在大量乱码和截断，上述代码仅根据上下文逻辑对可识别的部分进行了结构性处理。

### 匿名 VMA 导入/导出

```c
/*
 * 如果在扩展 vma 时，确保其设置了 anon_vma，
 * 如果在缩小 vma 时，以覆盖任何导入的匿名页。
 */
if (exporter && exporter->anon_vma && !importer->anon_vma) {
    int error;
    importer->anon_vma = exporter->anon_vma;
    error = anon_vma_clone(importer, exporter);
    if (error)
        return error;
}
```

这确保了 `struct anon_vma` 对象从合并源（导出者）传播到合并后的目标（导入者）。关于这些对象以及它们与反向映射的关系，请参考第 7 章。

### `__vma_adjust()` VMA 更新 (`mm/vma.c`)

现在我们已经建立了 VMA 调整的初始状态，可以执行重型工作。

此处以 `again` 标签为开头，用于在情况 6 中执行多次通过以合并上一个和下一个 VMA。

```c
again:
if (file) {
    mapping = file->f_mapping;
    root = &mapping->i_mmap;
    i_mmap_lock_write(mapping);
    if (insert) {
        /*
         * 放入区间树，使得实例化的节点
         * 对 anon/parisc 刷新 dcache 页可见
         * 贯穿；但在 vma start 或 end 更新之前，我们无法插入到 anon
         * 空间。
         */
        vma_link_file(insert);
    }
}

anon_vma = vma->anon_vma;
if (!anon_vma && adjust_next)
    anon_vma = next->anon_vma;

if (anon_vma) {
    VM_WARN_ON(adjust_next && next->anon_vma &&
               anon_vma != next->anon_vma);
    anon_vma |= next->anon_vma;
    vma_lock_write(anon_vma);
    anon_vma_interval_tree_pre_update_vma(vma);
    anon_vma_interval_tree_pre_update_vma(next);
}

if (file) {
    flush_dcache_mmap_lock(mapping);
    vma_interval_tree_remove(vma, root);
    if (adjust_next) {
        vma_interval_tree_remove(next, root);
    }
}

if (start != vma->vm_start) {
    vma->vm_start = start;
    start_changed = true;
}
if (end != vma->vm_end) {
    vma->vm_end = end;
    end_changed = true;
}
vma->vm_pgoff = pgoff;

if (adjust_next) {
    next->vm_start += adjust_next;
    next->vm_pgoff += adjust_next >> PAGE_SHIFT;
}

if (file) {
    if (adjust_next) {
        vma_interval_tree_insert(next, root);
        vma_interval_tree_insert(vma, root);
        flush_dcache_mmap_unlock(mapping);
    }
}
```

让我们检查这里发生了什么：

- 如果 vma 是文件支持的，我们将文件的 `struct address_space` 页缓存对象映射（即 VMA 区间树的根）放入 `root` 中，然后通过 `i_mmap_lock_write()` 获取其锁（获取 `mapping->i_mmap_rw` 信号量的锁）。
- 如果是插入而非合并，我们处理了一个针对架构的边界情况。当 VMA 经过调整时，我们必须重构与反向映射相关的区间树，这些树用于从 folio 查找 VMA 作为反向映射的一部分（深入讨论请参见第 7 章）。这是必要的，因为它们是基于 `vm_start` 键的，而它即将改变。`struct anon_vma` 对象跟踪匿名映射，因此为了进行调整，我们通过 `anon_vma_interval_tree_pre_update_vma()` 移除 `anon_vma` 区间树中的所有项。还通过 `anon_vma_lock_write()` 获取 `anon_vma` 树（维护以考虑写时拷贝映射）写锁，指示下一个 VMA 将被调整，然后我们对此 VMA 执行相同操作。
- 如果 vma 缺少任何 `anon_vma` 对象且 `adjust_next` 表明下一个 VMA 将被更新并拥有非空 `anon_vma` —— 在这种情况下我们将 `anon_vma` 设置为该对象。同样，我们必须对与文件支持 VMA 关联的页缓存对象执行等操作（注意 `MAP_PRIVATE` 映射可能存在于两个树中）。这通过 `vma_interval_tree_remove()` 执行。与匿名情况类似，如果 `adjust_next` 不为零，我们对下一个 VMA 执行相同操作。
- 在此处我们继续对 vma 进行更改 —— 如果 `vm_start` 更新则设置 `start_changed`；同样，如果 `vm_end` 改变则设置 `end_changed`。我们更新 `vm_pgoff`。如果设置了 `adjust_next`，我们通过偏移 `next->vm_start` 和 `next->vm_pgoff` 执行此调整。
- 最后，我们将重新插入（如果 `adjust_next` 表明我们要把 VMA 调整到页缓存）`i_mmap` 区间树。注意某些需要手动更新硬件缓存的架构。对于 x86 并非如此，但这告知了某些决策。

### `__vma_adjust()` VMA 移除/插入 (`mm/mmap.c`)

一旦此过程完成，我们按照以下代码处理 VMA 移除。

```c
if (remove_next) {
    /*
     * vma 已合并到 vma，
     * 需要我们在释放锁之前移除 next。
     */
    if (remove_next != 3)
        vma_unlink(next);
    else
        /*
         * 如果它们已交换，则在 vma 之前是 next。
         * next->vm_start 已减小，
         * 告诉 validate_vm_rb 忽略交换后的
         * "next"（存储在交换后的
         * "vma" 中）
         */
        vma_unlink(next, vma);

    if (file) {
        remove_shared_vm_struct(next, mapping);
    }
} else if (insert) {
    /*
     * vma 分自 vma，
     * 需要我们在释放锁之前插入它
     * (它可能在 vma 之后或之前)。
     */
    insert_vm_struct(mm, insert);
} else {
    if (start_changed) {
        vma_gap_update(vma);
    }
    if (end_changed) {
        if (!next)
            mm->highest_vm_end = vm_end_gap(vma);
        else if (adjust_next)
            vma_gap_update(next);
    }
    if (anon_vma) {
        anon_vma_interval_tree_post_update_vma(vma);
        if (adjust_next)
            anon_vma_interval_tree_post_update_vma(next);
        anon_vma_unlock_write(anon_vma);
    }
    if (file) {
        i_mmap_unlock_write(mapping);
    }
}
```

> **注意**：此函数是通过 `mm/interval_tree.c` 中的 `INTERVAL_TREE_DEFINE` 宏生成的。

### `__vma_unlink()` 实现

我们使用 `__vma_unlink()` 执行下一个 VMA 的移除。

```c
static __always_inline void __vma_unlink(struct mm_struct *mm,
                                         struct vm_area_struct *vma,
                                         struct vm_area_struct *ignore)
{
    vma_rb_erase_ignore(vma, mm->mm_rb, ignore);
    __vma_unlink_list(mm, vma);
}
```

`vma_rb_erase_ignore()` 最终移除 VMA 红黑树节点，`__vma_unlink_list()` 将 VMA 从链表中移除。

```c
void __vma_unlink_list(struct mm_struct *mm, struct vm_area_struct *vma)
{
    struct vm_area_struct *prev, *next;

    next = vma->vm_next;
    prev = vma->vm_prev;
    if (prev)
        prev->vm_next = next;
    else
        mm->mmap = next;

    if (next)
        next->vm_prev = prev;
}
```

`__vma_unlink_list()` 正确处理了 `vm_next` 或 `vm_prev` 为空的情况。如果 VMA 是文件映射的，则通过 `__remove_shared_vm_struct()` 从区间树移除 VMA。

```c
/*
 * 需要 inode->i_mapping->i_mmap_rwsem
 */
static void __remove_shared_vm_struct(struct vm_area_struct *vma,
                                      struct file *file, struct address_space *mapping)
{
    if (vma->vm_flags & VM_SHARED)
        mapping_unmap_writable(mapping);

    flush_dcache_mmap_lock(mapping);
    vma_interval_tree_remove(vma, &mapping->i_mmap);
    flush_dcache_mmap_unlock(mapping);
}
```

对于在映射时需要手动维护数据缓存的架构，会执行 `flush_dcache_mmap_lock()` / `flush_dcache_mmap_unlock()` 操作；但对于像 x86-64 这样的架构，这是一个空操作。调用 `mapping_unmap_writable()` 是为了跟踪页缓存对象的可写映射（假设 `i_mapping` 可能会在任何时间被写入）。

`struct address_space->i_mmap` 间隔树的实际移除由 `vma_interval_tree_remove()` 完成。

如果该操作是插入而非合并，则会调用 `insert_vm_struct()`。我们将在第 5.1.4 节关于拆分 VMA 的讨论中检查该函数。

如果我们删除或插入一个 VMA，我们会进行一些计数操作——`vma_gap_update()` 更新用于红黑树内部的内部状态，使用通用的增强红黑树内核代码来跟踪 VMA 与最近子树 VMA 之间的间隙，并根据需要更新 `struct mm_struct->highest_vm_end` 字段。

如果我们有一个匿名 vma 对象，我们会通过 `anon_vma_interval_tree_post_update_vma()` 为 vma 重新启动间隔树，如果 `adjust_next` 为非零，则在 `anon_vma_unlock_write()` 解锁 anon_vma 之前进行处理。

最后，如果 vma 是文件支持的，我们需要通过 `i_mmap_unlock_write()` 解锁页缓存对象的间隔树。

在完成任何的移除操作后，在调用 `validate_mm()` 之前，我们必须处理一些最后的清理任务，以确保 `struct mm_struct` 对象处于有效状态。
