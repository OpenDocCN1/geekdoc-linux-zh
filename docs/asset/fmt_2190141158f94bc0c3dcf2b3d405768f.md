### 隐藏和显示工具栏项

虽然工具栏是静态定义为 `GnomeUIInfo` 结构体数组的，但可以选择显示哪些工具栏项以及不显示哪些。当某个工具栏项未显示时，工具栏会自动调整自身，就好像该隐藏项未包含在定义数组中一样。

以下示例创建一个包含三个按钮的工具栏。右侧按钮使中间按钮消失，左侧按钮使中间按钮重新出现。

```c
/** tbhide.c **/
#include <gnome.h>

static void dummyCallback(GtkObject *, gpointer);
static void showProperties(GtkObject *, gpointer);
static void hideProperties(GtkObject *, gpointer);

gint eventDelete(GtkWidget *widget,
          GdkEvent *event, gpointer data);
gint eventDestroy(GtkWidget *widget,
           GdkEvent *event, gpointer data);

#define PROPERTIES_INDEX 1
static GNOMEUIInfo showToolbar[] = {
  { GNOME_APP_UI_ITEM, "Show Properties",
    "Enable the Properties toolbar selection",
    showProperties, NULL, NULL,
    GNOME_APP_PIXMAP_STOCK, GNOME_STOCK_PIXMAP_NEW,
    0, 0, NULL },
  { GNOME_APP_UI_ITEM, "Properties",
    "Configure application properties",
    dummyCallback, NULL, NULL,
    GNOME_APP_PIXMAP_STOCK, GNOME_STOCK_PIXMAP_PROPERTIES,
    0, 0, NULL },
  { GNOME_APP_UI_ITEM, "Hide Properties",
    "Hide the Properties toolbar selection",
    hideProperties, NULL, NULL,
    GNOME_APP_PIXMAP_STOCK, GNOME_STOCK_PIXMAP_SAVE_AS,
    0, 0, NULL },
  GNOMEUIINFO_END
};

int main(int argc, char *argv[])
{

  GtkWidget *app;

  gnome_init("tbhide", "1.0", argc, argv);
  app = gnome_app_new("tbhide", "Toolbar Hide");
  gtk_signal_connect(GTK_OBJECT(app), "delete_event",
                GTK_SIGNAL_FUNC(eventDelete), NULL);
```
