1. 一个正确的实现如下：

```
#!/bin/sh

USAGE='Usage: basename $0 [-c|-t] [files|directories]'
```

```
if [ $# -lt 2 ] ; then
  echo 'USAGE' ;
  exit 1 ;
fi

case "$1" in
  -t|-v)
    TARGS="$@"; shift
    for i in "$@" ; do
      if [ -f "$i" ] ; then
        FILES=`tar "$i" 2>/dev/null`
        if [ $? -eq 0 ] ; then
          echo "$i"
          echo 'FILES'
        else
          echo 'ERROR: $i is not a tar file.'
        fi
      else
        echo 'ERROR: $i not a file.'
      fi
    done
    ;;
  -c) shift ; TARGS="-cvf" ;
    tar $TARGS archive.tar "$@"
    ;;
  *) echo 'USAGE'
    exit 0
    ;;
esac
exit $?
```
