## 函数 `list_of_disks`

```bash
function list_of_disks
{
# 开启/关闭单个灯光：
# 遍历传递给 $1 的每个磁盘，即 $2, $3, $4...
# 通过大于 $1 的位置参数传递，即 $2, $3, $4...
# 我们首先判断每个参数是 pdisk 还是 hdisk。
# 对于传递给脚本的每个 hdisk，我们首先需要
# 将 hdisk 定义转换为 pdisk 定义。此脚本
# 设置为接受 hdisks 和 pdisks 的组合。

# 我们将根据 $SWITCH 变量为每个 pdisk 
# 开启或关闭标识符灯：

# 开启灯光：-y
# 关闭灯光：-n

# 将 $SWITCH 的值传递给 "ssaidify" 命令。
# 传递给此脚本的磁盘可以是全 hdisk、全 pdisk
# 或者 pdisk 与 hdisks 的组合；这并不重要。
# 我们将每个 hdisk 转换为关联的 pdisk(s)。

echo "\n正在处理单个 SSA 磁盘灯光...\n"

for PDISK in $(echo $PDISKLIST)
do
# 这是一个真实的 pdisk 吗？
if [ -c /dev/${PDISK} ] 2>/dev/null
then 
    # 是的，对其操作 ...
    /usr/bin/ssaidentity -1 $PDISK -${SWITCH}
    if [ $? -eq 0 ];
    then
        /usr/bin/ssaxlate -1 $PDISK -${SWITCH}
        if ((($? == 0))
        then
            echo "$PDISK 的灯光状态为 $STATE"
        else
            echo "执行 $STATE $PDISK 失败"
        fi
    else
        echo "\ERROR: $PDISK 不是 $THISHOST 上定义的设备\"
    fi
done
echo "\n...任务完成...\n"
```

### 对灯光进行操作

```bash
if (($# == 1)) && [[ $MODE = "DEFINED_DISKS" ]]
then
    # 此函数将开启/关闭所有灯
    all_defined_pdisks
elif [[ $MODE = "VARIED_ON" ]] && (($# == 2))
then
    # 此函数将开启/关闭
    # 当前变异组 (varied groups) 中的 SSA 磁盘
    all_varied_on_pdisks
# 现在检查 hdisk 和 pdisk 参数
elif [[ $MODE = DEFINED_DISKS ]] && ((echo $@ | grep disk >/dev/null) && (($# >= 2))
then
    # 如果我们在这里，我们必须有一组 hdisk
    # 和/或 pdisk

    # 在命令行参数中查找 hdisk 和 pdisk
    for DISK in $(echo $@ | grep disk)
    do
        case $DISK in
        hdisk*) HDISKLIST="$HDISKLIST $DISK"
            ;;
        pdisk*) PDISKLIST="$PDISKLIST $DISK"
            ;;
        *)     # 无操作 - 不执行
            ;;
        esac
    done

    if [[ -n "$HDISKLIST" ]]; # 检查需要转换为 pdisk 的 hdisk
    then
        # 我们有一些 hdisk 需要转换为 pdisk
        # 所以开始将 hdisk 转换为 pdisk

        # 给用户提供反馈
        echo "\n正在将 hdisk 转换为 pdisk 定义"
        echo "\n...请等待...\n"

        # 开始将 hdisk 转换为 pdisk
        for HDDISK in $(echo $HDISKLIST)
        do
            PDISK=$(ssaxlate -1 $HDDISK)
            if (($? == 0)); then
                echo "$HDDISK 转换为 $PDISK"
            else
                echo "ERROR: hdisk 到 pdisk 转换失败"
            fi
            # 构建 pdisk 列表
            # 将 pdisk 添加到 pdisk 列表中
            PDISKLIST="$PDISKLIST $PDISK"
        done
    fi
    
    if [[ -z "$PDISKLIST" ]]
    then
        echo "\nERROR: 必须指定一个 hdisk 或 pdisk\n"
        man_page
        exit 1
    else
        # 开启/关闭 SSA 识别灯
        list_of_disks
    fi
fi
```
