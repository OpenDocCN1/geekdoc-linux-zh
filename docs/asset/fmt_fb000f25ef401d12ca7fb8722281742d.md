#include <stdio.h>

#define kIntSize 32  // *** 取决于机器 !!! ***

// 函数：将无符号整数向左或向右旋转

unsigned int rotate (unsigned int value, int n)
{
    unsigned int result, bits;

    /* 将移位计数缩减到定义的范围 */

    if (n > 0)
        n = n % kIntSize;
    else
        n = -(n % kIntSize);

    if (n == 0)
        result = value;
    else if (n > 0)     /* 左旋转 */
    {
        bits = value >> (kIntSize - n);
        result = value << n | bits;
    }
    else                /* 右旋转 */
    {
        n = -n;
        bits = value << (kIntSize - n) ;
        result = value >> n | bits;
    }

    return result;
}

{
   float   liters, gallons;
