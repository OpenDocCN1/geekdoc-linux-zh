void
Log::buf_dump(void *data, int len)
{
    size_t a, b;
    unsigned char c, *dat;
    dat = (unsigned char *)data;
    if (ilen || len < 0) return;
    raw(` Hexadecimal ASCII\n*);
    raw(` -------------------------------*);
    raw(` *);
    for (a = 0; a < len; a++) {
        c = (unsigned char)dat[a];
        if (a && !(a % 8)) raw(`` *);
        if (a && !(a & 16)) {
            raw(` *);
            for (b = a - 16; b < a; b++) {
                c = (unsigned char)dat[b];
                if (!isspace(dat[b])) {
                    if (isprint(dat[b])) {
                        raw(`%c", c);
                    } else {
                        raw(`*-` *);
                    }
                }
            }
            raw("\n %02x", c);
        } else {
            raw("%02x", c);
        }
    }
    for (b = a; (b & 16); b++) raw(" ");
    raw(" ");
    for (b = a - 16; b < a; b++) {
        c = (unsigned char)dat[b];
        if (isprint(dat[b])) raw("%c", c);
        else raw("-");
    }
    for (; (b & 16); b++) raw(" ");
    raw("\n");
    raw(" -------------------------------------"
        "--------------------------------------\n");
}
