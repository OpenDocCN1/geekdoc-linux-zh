```cpp
// 列表 15.1: String 类 - string.hpp - 列表 15.2 使用

#include <iostream.h>
#include <string.h>

class String {
public:
    // 构造函数
    String();
    String(const char *const);
    String(const String &);
    ~String();

    // 重载运算符
    char& operator[](int offset);
    char operator[](int offset) const;
    String operator+(const String&);
    void operator+=(const String&);
    String& operator=(const String &);

    // 通用访问器
    int getLen() const { return itsLen; }
    const char* getString() const { return itsString; }
    static int ConstructorCount;

private:
    String(int); // 私有构造函数
    char *itsString;
    unsigned short itsLen;
};

// 默认构造函数创建 0 字节的字符串
String::String() {
    itsString = new char[1];
    itsString[0] = '\0';
    itsLen = 0;
    cout << "\tdefault constructor\n";
    ConstructorCount++;
}

// 私有（辅助）构造函数，仅用于类方法以创建所需大小的新字符串。空填充。
String::String(int len) {
    itsString = new char[len+1];
    for (int i = 0; i <= len; i++)
        itsString[i] = '\0';
    itsLen = len;
    // cout << "\tString(int) constructor\n";
    // ConstructorCount++;
}

// 将字符数组转换为 String
String::String(const char * const cString) {
    itsLen = strlen(cString);
    itsString = new char[itsLen+1];
    for (int i = 0; i < itsLen; i++)
        itsString[i] = cString[i];
    itsString[itsLen] = '\0';
    // cout << "\tString(char) constructor\n";
    // ConstructorCount++;
}

// 拷贝构造函数
String::String(const String & rhs) {
    itsLen = rhs.GetLen();
    itsString = new char[itsLen+1];
    for (int i = 0; i < itsLen; i++)
        itsString[i] = rhs[i];
    itsString[itsLen] = '\0';
    // cout << "\tString(String&) constructor\n";
    // ConstructorCount++;
}

// 析构函数，释放分配的内存
String::~String() {
    delete itsString;
    itsLen = 0;
    // cout << "\tString destructor\n";
}

// 赋值运算符，释放现有内存然后拷贝字符串和大小
String& String::operator=(const String & rhs) {
    if (this == &rhs)
        return *this;
    delete itsString;
    itsLen = rhs.GetLen();
    itsString = new char[itsLen+1];
    for (int i = 0; i < itsLen; i++)
        itsString[i] = rhs[i];
    itsString[itsLen] = '\0';
    // cout << "\tString operator=\n"
    return *this;
}

// 非 const 偏移运算符，返回字符引用以便可以被修改
char& String::operator[](int offset) {
    if (offset > itsLen)
        return itsString[itsLen-1];
    else
        return itsString[offset];
}

// const 偏移运算符，用于 const 对象（见拷贝构造函数！）
char String::operator[](int offset) const {
    if (offset > itsLen)
        return itsString[itsLen-1];
    else
        return itsString[offset];
}

// 通过将当前字符串相加到 rhs 创建新字符串
String String::operator+(const String& rhs) {
    int totalLen = itsLen + rhs.GetLen();
    String temp(totalLen);
    int i, j;
    for (i = 0; i < itsLen; i++)
        temp[i] = itsString[i];
    for (j = 0; j < rhs.GetLen(); j++)
        temp[i] = rhs[j];
    temp[totalLen] = '\0';
    return temp;
}

// 修改当前字符串，无返回值
void String::operator+=(const String& rhs) {
    unsigned short rhsLen = rhs.GetLen();
    unsigned short totalLen = itsLen + rhsLen;
    String temp(totalLen);
    int i, j;
    for (i = 0; i < itsLen; i++)
        temp[i] = itsString[i];
    for (j = 0; j < rhs.GetLen(); j++, i++)
        temp[i] = rhs[i-itsLen];
    temp[totalLen] = '\0';
    *this = temp;
}
```
