
```c
void *pmx_context;          /* PMU 上文指针 */

unsigned long pfn_needs_checking; /* 内核退出时检查 */

#endif

u64 dbr[IA64_NUM_DBG_REGS];

u64 dbr[IA64_NUM_DEG_REGS];

struct ia64_fpreg fph[96];   /* 按需保存/加载 */

};
```

IA-64 具有监控子系统。如果内核配置为运行该子系统，则必须保存寄存器。

### A.7.3 ARM

ARM 系统有两个版本：处理器设计为 26 位和 32 位字长。由于大多数系统都使用 32 位，因此附录中包含了此机器的定义。

`pt_regs` 结构的定义仅由一个包含内核模式所有寄存器值的数组组成：

```c
include/asm-arm/ptrace.h

struct pt_regs {
	long uregs[18];
};
```

寄存器的符号名称及其在数组中的位置通过预处理器常量定义：

```c
include/asm-arm/ptrace.h

#define ARM_cps uregs[16]
#define ARM_pc uregs[15]
#define ARM_lr uregs[14]
#define ARM_sp uregs[13]
#define ARM_ip uregs[12]
#define ARM_fp uregs[11]
#define ARM_r10 uregs[10]
#define ARM_r9 uregs[9]
#define ARM_r8 uregs[8]
#define ARM_r7 uregs[7]
#define ARM_r6 uregs[6]
#define ARM_r5 uregs[5]
#define ARM_r4 uregs[4]
#define ARM_r3 uregs[3]
#define ARM_r2 uregs[2]
#define ARM_r1 uregs[1]
#define ARM_r0 uregs[0]
#define ARM_ORIG_r0 uregs[17]
```

没有必要保存任何浮点寄存器，因为 ARM 处理器仅提供浮点操作软件支持：

```c
include/asm-arm/processor.h

struct thread_struct {
	unsigned long /* 异常信息 */
	unsigned long trap_no;
	unsigned long error_code;
	struct debug_info debug;
};
```

然而，机器指令（操作码）可以与内存地址一起保存以便调试，如下：

```c
include/asm-arm/processor.h

union debug_insn {
	u32 arm;
	u16 thumb;
};

struct debug_entry {
	u32 address;
	union debug_insn insn;
};

struct debug_info {
	int nsaved;
	struct debug_entry bp[2];
};
```

### A.7.4 Sparc64

Sparc64 处理器也使用 `pt_regs` 结构中的数组为各个寄存器提供内存空间。寄存器通过预处理器常量命名并分配空间如下：

```c
include/asm-sparc64/ptrace.h

struct pt_regs {
    unsigned long u_regs[16]; /* 全局变量和指令 */
    unsigned long tstate;
    unsigned long tpc;
    unsigned long tnpc;
    unsigned int y;
    unsigned int fprs;
};

#define UREG_G0 0
#define UREG_G1 1
#define UREG_G2 2
#define UREG_G3 3
#define UREG_G4 4
#define UREG_G5 5
#define UREG_G6 6
#define UREG_G7 7
#define UREG_I00 8
#define UREG_I01 9
#define UREG_I02 10
#define UREG_I03 11
#define UREG_I04 12
#define UREG_I05 13
#define UREG_I06 14
#define UREG_I07 15
#define UREG_FP UREG_I6
#define UREG_RETPC UREG_I7
```

与其他架构不同，Sparc64 尝试将未保存的寄存器存储在 `thread_info` 中而不是 `thread_struct` 中。按照习惯，必须能够与其他部分区分开来。如果内核没有编译自旋锁，该结构可能是空的。然而，由于早期的 GCC 错误（如后续源代码所示，Sparc 的作者并不喜欢），该结构包含一个虚拟元素：

```c
include/asm-sparc64/processor.h
/* Sparc64 处理器特定的线程结构。 */
/* XXX 这应该删了，现在所有内容都可以进入 thread_info。 */
struct thread_struct {
#ifdef CONFIG_DEBUG_SPLOCK
    /* 此线程持有的自旋锁数量。
     * 用于自旋锁调试，以捕获在持有锁的情况下
     *非法睡眠的任务。
     */
    int sm_lock_count;
    unsigned int sm_lock_pc;
```

作为经典的 RISC 机器，Alpha CPU 使用一个大型寄存器，其成员主要根据有序原则通过数字进行标识。如前所述，Alpha 架构使用 PAL（特权体系级别）代码来执行系统任务。该代码也用于系统调用的实现。

## 附录 A：架构细节

内核的 C 或汇编代码不需要保存 `pt_regs` 中列出的所有寄存器，因为某些寄存器是由 PAL 代码程序自动放在栈上的。需要保存的是以下：

```c
include/asm-alpha/ptrace.h

struct pt_regs {
    unsigned long r0;
    unsigned long r1;
    unsigned long r2;
    unsigned long r3;
    unsigned long r4;
    unsigned long r5;
    unsigned long r6;
    unsigned long r7;
    unsigned long r8;
    unsigned long r19;
    unsigned long r20;
    unsigned long r21;
    unsigned long r22;
    unsigned long r24;
    unsigned long r25;
    unsigned long r26;
    unsigned long r27;
    unsigned long hae;
    /* 这些是由 PALCode 提供的 a0-a2 的值 */
    unsigned long trap_a0;
    unsigned long trap_al1;
    unsigned long trap_a2;
    /* 这些由 PAL-code 保存： */
    unsigned long ps;
    unsigned long pc;
    unsigned long gp;
    unsigned long r16;
    unsigned long r17;
    unsigned long r18;
};
```

该架构同样使用了 `thread_struct` 结构：

```c
include/asm-alpha/processor.h
/* 这已失效。所有内容已移动到 thread_info。 */
struct thread_struct { };
```

浮点寄存器 `f0` 到 `f31` 的内容并不保存在 `thread_info` 中，但少数整数寄存器也会像浮点寄存器一样被保存；通常内核不需要这些整数寄存器：

```c
include/asm-alpha/ptrace.h

struct switch_stack {
    unsigned long r9;
    unsigned long r10;
    unsigned long r11;
    unsigned long r12;
    unsigned long r13;
    unsigned long r14;
    unsigned long r15;
```
