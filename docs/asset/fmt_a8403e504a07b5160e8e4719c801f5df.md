### Example 16.33

`Label` widget:

```
from Tkinter import *
def label():
    master = Tk()
    w = Label(master, text="Label")
    w.pack()
    mainloop()
label()
```

### Example 16.35

`Canvas` widget:

```
from Tkinter import *
def canvas():
    master = Tk()
    master.title("Canvas")
    w = Canvas(master, width=200, height=100)
    w.pack()
    w.create_line(0, 0, 200, 100)
    w.create_line(0, 100, 200, 0, fill="red", dash=(4, 4))
    w.create_rectangle(50, 25, 150, 75, fill="blue")
    mainloop()
canvas()
```

## 示例 16.38

`Frame` 部件：

```
from Tkinter import *
def frame():
    master = Tk()
    master.title("Frame")
    Label(text="above Frame separator").pack()
    separator = Frame(height=2, bd=1, relief=SUNKEN)
    separator.pack(fill=X, padx=5, pady=5)
    Label(text="below Frame separator").pack()
    mainloop()
frame()
```

## 示例 16.40 Menu 部件

```
from Tkinter import *
def menu():
    root = Tk()
    root.title("Menu")
    def hello():
        print "I am here"
    # create a toplevel menu
    menubar = Menu(root)
    menubar.add_command(label="1stMenu", command=hello)
    menubar.add_command(label="2ndMenu", command=root.quit)
    root.config(menu=menubar)
    mainloop()
menu()
```

## 示例 16.41 RadioButton 部件

```
from Tkinter import *
def radiobutton():
    master = Tk()
    master.title("RadioButton")
    v = IntVar()
    Radiobutton(master, text="One", variable=v, value=1).pack(anchor=W)
    Radiobutton(master, text="Two", variable=v, value=2).pack(anchor=W)
```

## 示例 16.42

`Scale` 部件

```
from Tkinter import *
def scale():
    master = Tk()
    master.title("Scale")
    w = Scale(master, from_=0, to=100)
    w.pack()
    w = Scale(master, from_=0, to=200, orient=HORIZONTAL)
    w.pack()
    mainloop()
scale()
```
