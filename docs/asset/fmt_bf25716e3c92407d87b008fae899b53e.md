$ORIGIN example.com.

    1 ; serial
    21600 ; refresh after 6 hours
    3600 ; retry after 1 hour
    60480 ; expire after 1 week
    86400 ; minimum TTL of 1 day
@ NS dns1.example.com.

@ MX 10 mail.example.com.
mail A 192.168.1.1
server1 A 192.168.1.2
server2 A 192.168.1.3
