    vga_setmode(G320x200x256);
    vga_setcolor(vga_white());

    book_polygon1(VERTICES, xarray, yarray);

    vga_getch();

    vga_setmode(TEXT);
    return 0;
}

void book_polygon1(int n, int *xarray, int *yarray)
{
    int x1, y1, x2, y2, i;

    for (i = 0; i < n; i++)
    {
        x1 = xarray[i];
        y1 = yarray[i];
        if (i != (n - 1))
        {
            x2 = xarray[i + 1];
            y2 = yarray[i + 1];
        }
        else
        {
            x2 = xarray[0];
            y2 = yarray[0];
        }
        vga_drawline(x1, y1, x2, y2);
    }
}
