import numpy as np

import matplotlib.pyplot as plt

myPoly = np.poly1d(np.array([3, -1, 11]).astype(float))

y = myPoly(x)

ticks = np.arange(0, 100, 10)

plt.yticks(ticks)

plt.plot(x)
