ax = plt.subplots(figsize=(15, 12), constrained_layout=True)
fig.patch.set_facecolor('#F8F8F8')

places = {
    'Carroupilha, Porto Alegre': ['#EEB4E1', '#E7D8C9', '#E6EBEA'],
    'Cadeira Baixa, Porto Alegre': ['#49295C', '#77625C', '#B2F1CF', '#E1F2F6', '#98D2EB'],
    'Bom Fim, Porto Alegre': ['#BA8D6E', '#72BA9B', '#F7D95C'],
}

for i, (place, palette) in enumerate(places.items()):
    plot(
        place,
        # ... 其他参数
        drawing_kwargs={
            'perimeter': {'fill': False, 'lw': 0, 'zorder': 0},
            'streets': {'fc': '#F6F6F6', 'lw': 1.5, 'zorder': 3},
            'building': {'palette': palette, 'ec': '#F2F7F7', 'lw': 1, 'zorder': 4},
        }
    )
