# 第 15 章

- 1. 示例实现如下：

```bash
lspids() {
    USAGE="Usage: lspids [-h] process"
    HEADER=false
    PSCMD="/bin/ps -ef"

    case "$1" in
        -h) HEADER=true ; shift ;;
    esac

    if [ -z "$1" ]; then
        echo $USAGE ;
        return 1 ;
    fi

    if [ "$HEADER" = "true" ]; then
        $PSCMD 2>/dev/null | head -1 ;
    fi

    $PSCMD 2>/dev/null | grep "$1" | grep -v grep
}
```

对于 Linux 或 FreeBSD，将变量 `PSCMD` 从

```bash
PSCMD="/bin/ps -ef"
```

改为

```bash
PSCMD="/bin/ps -auwx"
```

- 2. `string` 是另一种可能的实现方式：

```bash
lspids () {
    USAGE="Usage: lspids [-h|-s] process";
    HEADER=false;
    SORTCMD=-ef";
    PSCMD=/bin/ps
    SORTCMD="sort -rn k 2,2";
    for OPT in $$;do
      case "$OPT" in
        -h)
          HEADER=true;
          shift
          ;;
        -s)
          SORT=true;
          shift
          ;;
        -*)
          echo $USAGE;
          return 1
          ;;
      esac
    done
    if [ -z "$1" ]; then
      echo $USAGE;
      return 1;
    fi
    if [ "$HEADER" = "true" ]; then
      $PSCMD 2>/dev/null | head -1;
    fi
    if [ "$SORT" = "true" ]; then
      $PSCMD 2 > /dev/null | grep "$1" | grep -v grep | $SORTCMD;
    else
      $PSCMD 2 > /dev/null | grep "$1" | grep -v grep;
    fi
}
```

对于 LINUX 和 FreeBSD，将变量 `SORTCMD` 改为

`SORTCMD="sort -rn"`

而不是

`SORTCMD="sort -rn k 2,2"`

# 第16章

- 1. 一种可能的实现方式是

```bash
sgrep() {
  if [ $ -lt 2 ]; then
    echo "USAGE: sgrep pattern files" >&2
    exit 1
  fi

  PAT="$1"; shift ;

  for i in $@; 
  do
    if [ -f "$i" ]; then
      sed -n "/$PAT/p" "$i"
    else
      echo "ERROR: $i not a file." >&2
    fi
  done

  return 0
}
```

- 2. 下面的命令可以完成这项任务：

```
$ uptime | sed 's/.* load/load/'
```

- 3. 有两种可能的解决方案：

```
$ df -k | sed -n '^\//p'
$ df -k | sed '^\//d'
```

- 4. 下面的命令将解决此问题：

```
/bin/ls -al | sed -e '^\//d' -e 's/ *[0-9]* * / /'
```

# 第17章

- 1. 一种可能的实现如下：

```bash
#!/bin/sh

if [ $# -lt 1 ]; then
  echo "USAGE: `basename $0` files"
  exit 1
fi

awk '{
  for (i=NF;i>=1;i--) {
    printf("%s ",$i);
  }
  printf("\n");
}' $@
```

- 2. 一种可能的解决方案是：

```bash
#!/bin/sh
awk 'BEGIN {FS=":"} {
  $1 == "B" {
    BAL=$NF;
    next;
  }
  $1 == "D" {
    BAL += $NF;
  }
  ($1 == "C") || ($1 == "W") {
    BAL=$NF;
  }
  ($1 == "C") || ($1 == "W") || ($1 == "D") {
    printf(" %10s %.2f\n",BAL,$2);
  }
}' account.txt;
```
