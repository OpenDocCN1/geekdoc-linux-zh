/** canvasImage.c **/
#include <gnome.h>

#include "logo.xpm"
#include "foot.xpm"

gint eventDelete(GtkWidget *widget,
                GdkEvent *event, gpointer data);
gint eventDestroy(GtkWidget *widget,
                GdkEvent *event, gpointer data);
static gint dragEvent(GnomeCanvasItem *item,
                 GdkEvent *event, gpointer data);

int main(int argc, char *argv[])
{
    GtkWidget *app;
    GtkWidget *canvas;
    GnomeCanvasGroup *rootGroup;
    GnomeCanvasItem *item;
    GdkPImlibImage *image;

    gnome_init("canvasImage", "1.0", argc, argv);
    app = gnome_app_new("canvasImage",
                        "GnomeCanvas Image");
    gtk_signal_connect(GTK_OBJECT(app), "delete_event",
                GTK_SIGNAL_FUNC(eventDelete), NULL);
    gtk_signal_connect(GTK_OBJECT(app), "destroy",
                GTK_SIGNAL_FUNC(eventDestroy), NULL);

    canvas = gnome_canvas_new();
    gtk_widget_set_usize(canvas, 600, 360);
    gnome_canvas_set_scroll_region(GNOME_CANVAS(canvas),
            0.0, 0.0, 600, 360);
    rootGroup = gnome_canvas_root(GNOME_CANVAS(canvas));
    gnome_app_set_contents(GNOME_APP(app), canvas);

    image = gdk_imlib_create_image_from_xpm_data(logo);
    item = gnome_canvas_item_new(rootGroup,
                gnome_canvas_image_get_type(),
                "image", image,
                "x", (double)100,
                "y", (double)100,
                "width", (double)257,
                "height", (double)303,
                "anchor", GDK_ANCHOR_CENTER,
                NULL);
    gtk_signal_connect(GTK_OBJECT(item),"event",
                (GtkSignalFunc)dragEvent,NULL);
    gdk_imlib_destroy_image(image);

    image = gdk_imlib_create_image_from_xpm_data(foot);
    item = gnome_canvas_item_new(rootGroup,
                gnome_canvas_image_get_type(),
                "image",image,
                "x",(double)300,
                "y",(double)100,
                "width",(double)172,
                "height",(double)151,
                "anchor",GDK_ANCHOR_CENTER,
                NULL);
    gtk_signal_connect(GTK_OBJECT(item),"event",
                (GtkSignalFunc)dragEvent,NULL);

    item = gnome_canvas_item_new(rootGroup,
                gnome_canvas_image_get_type(),
                "image",image,
                "x",(double)200,
                "y",(double)150,
                "width",(double)200,
                "height",(double)200,
                "anchor",GDK_ANCHOR_CENTER,
                NULL);
    gtk_signal_connect(GTK_OBJECT(item),"event",
                (GtkSignalFunc)dragEvent,NULL);
    gdk_imlib_destroy_image(image);

    gtk_widget_show_all(app);
    gtk_main();
    exit(0);
}
