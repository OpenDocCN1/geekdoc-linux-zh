
```bash
#####################
print "示例 2: 接下来让我们读取一个小型文件（两条记录），"
print "按字节处理每条记录，并打印反转后的记录。"
IFS='' # 将内部字段分隔符设为空。详情见：IFS=ksh。

# 首先，创建一个小的（两条记录）测试文件：
INPUT=/tmp/input.$$ $$ 是当前唯一的 PID/进程 ID。
print "abc 123" > $INPUT
print "DEF 456" >>$INPUT

while read -r record
do
  reversed_record="" # reversed_record 初始为空（null）。
  length=${#record} #  NOTES。
  byte_counter=0
  # 现在通过按字节重建来反转记录：
  while [ $length -gt 0 ]
  do
    ((byte_counter++)) # 为 byte_counter 加 1
    # 如果 byte_counter 是 9，那么就截取记录的第 9 个字符：
    one_byte=$(echo $record | cut -c $byte_counter) # 截出一个字节
    reversed_record=$one_byte$reversed_record
    ((length--)) # 将 length 减 1
  done
  print "原始记录: $record"
  print "反转后的记录: $reversed_record"
done < $INPUT

# 删除/删除测试文件：
rm $INPUT 2>/dev/null
```
