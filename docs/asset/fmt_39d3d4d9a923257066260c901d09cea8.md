
## D.17. `/etc/rc.d/init.d/network`

```sh
#!/bin/sh

################################################################

# 开始 network

# Description：网络控制脚本

# Authors：Gerard Beekmans - gerard@linuxfromscratch.org

#          Nathan Coulson - nathan@linuxfromscratch.org

#          Kevin Fleming - fleming@linuxfromscratch.org

#          DJ Lucas - dj@linuxfromscratch.org

#          Bruce Dubbs - bdubbs@linuxfromscratch.org

# Update

# Version：LFS 7.0

################################################################

### BEGIN INIT INFO
# Provides: $network
# Required-Start: $local_fs
# Should-Start: $syslog
# Required-Stop: $local_fs
# Should-Stop: $syslog
# Default-Start: 3 4 5
# Default-Stop: 0 6
# Short-Description:启动并配置。
# Description:启动并配置接口。
# X-LFS-Provided-By: LFS
### END INIT INFO

case "$1" in
start)
  # 启动网络
  for file in /etc/sysconfig/network/*
  do
    interface=${file##*/}
    # 如果接口为空则跳过
    if [ -z "$interface" ]; then
      continue
    fi
    /sbin/ifup $interface
  done
  ;;
stop)
  # 停止网络
  net_files=""
  for file in /etc/sysconfig/network/*
  do
    interface=${file##*/}
    # 如果接口为空则跳过
    if [ -z "$interface" ]; then
      continue
    fi
    net_files="$net_files $interface"
  done
  # 关闭所有接口
  for interface in $net_files
  do
    /sbin/ifdown $interface
  done
  ;;
restart)
  $0 stop
  sleep 1
  $0 start
  ;;
*)
  echo "Usage: ${0} {start|stop|restart}"
  exit 1
  ;;
esac

exit 0
```

## D.26. `/sbin/ifup`

```bash
RELEASE="7.0"
用法: $0 [-hV] [--help] [--version] interface
VERSTR=LFS ifup, version ${RELEASE}
while [ $# -gt 0 ]; do
  case "$1" in
    --help | -h) help="y"; break ;;
    --version | -V) echo "${VERSTR}"; exit 0 ;;
    -*) echo "ifup: ${1}; invalid option" >&2
         echo "${USAGE}" >&2
         exit 2 ;;
    break ;;
  esac
done

if [ -n "$help" ]; then
  echo "${VERSTR}"
  echo "${USAGE}"
  cat << EOF
ifup is used to bring up a network interface
parameter, e.g. eth0 or eth0:2, must match the
interface specifications file, e.g. /etc/sysconfig/ifconfig.eth0:2.
EOF
  exit 1
fi

file=/etc/sysconfig/ifconfig.${1}
# 跳过备份文件
[ "${file}" = "${file}*" ] || exit 0
/lib/lsb/init-functions

if [ ! -r "${file}" ]; then
log_warning "${file}" is missing or cannot accessed.
exit 1
fi

. ${file}

if [ "${IFACE}" = "" ]; then
log_failure_msg "${file}" does not define an interface.
exit 1
fi

# 应当启动
if [ "${INBOOT}" != "yes" ]; then
  log_info_msg "Starting..."
  ip link ${IFACE} up
else
  log_info_msg "Starting..."
  ip link ${IFACE}
fi

# 设置 MTU
if [ -n "$MTU" ]; then
  if [[ "$MTU" =~ ^[0-9]+$ ]]; then
    ip link set dev ${IFACE} mtu ${MTU}
```
