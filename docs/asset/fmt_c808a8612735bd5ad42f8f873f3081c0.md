**列表 14-10** `chg_base_bc.Bash` 脚本

```bash
# 从 ibase_num 变量中提取进制和数字
IBASE=$(echo $IBASE_NUM | awk -F '#' '{print $1}')
NUMBER=$(echo $IBASE_NUM | awk -F '#' '{print $2}')
# 测试确保输入进制在 2 到 16 之间
if (( ABASE < 2 || ABASE > 16 ))
then
    echo -e "\n错误：输入进制必须在 2 到 16 之间\n"
    exit 1
fi
# bc 工具要求所有大于
# 10 的进制使用大写字母
# 所有非数字字符数字，即十六进制。
# 我们使用 tr 命令将所有小写
# 字符大写。
if (( ABASE > 10 ))
then
    NUMBER=$(echo $NUMBER | tr '[a-z]' '[A-Z]')
fi
# 询问用户输出进制
echo -e "\n你想将 $IBASE_NUM 转换为什么进制？
注意：2 到 16 进制都是有效的\n"
echo -e " 输出进制: \c"
read OBASE
# 测试确保输出进制为整数
case $OBASE in
    [0-9]*) : 什么都不做
        ;;
    *) echo -e "\n错误：$OBASE 不是有效数字\n"
        exit 1
        ;;
esac
# 测试确保输出进制在 2 到 16 之间
if (( OBASE < 2 || OBASE > 16 ))
then
    echo -e "\n错误：输出进制必须在 2 到 16 之间\n"
    exit 1
fi
```
