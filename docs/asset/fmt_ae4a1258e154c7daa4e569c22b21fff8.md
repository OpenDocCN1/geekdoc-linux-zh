
```yaml
version: '3'

services:

  db:
    image: postgres:12
    environment:
      POSTGRES_PASSWORD: password
      POSTGRES_USER: ansible_user
      POSTGRES_DB: awx
    volumes:
      - postgres_data:/var/lib/postgresql/data

  redis:
    image: redis:6
    volumes:
      - redis_data:/data

  awx_server:
    image: ansible/awx:latest
    ports:
      - "8000:8080"
    depends_on:
      - db
      - redis
    environment:
      POSTGRES_URL: postgres://ansible_user:password@db:5432/awx
      REDIS_URL: redis://redis:6379

volumes:
  postgres_data:
  redis_data:
```
