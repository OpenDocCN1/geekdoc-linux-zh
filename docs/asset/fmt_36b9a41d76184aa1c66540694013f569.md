
# timespec

## 目的
用于格式化日志文件时间戳的内部工具

## 设置
`STAMP` 变量。

## 返回值
未使用

```bash
timespec()
{
    STAMP=$(date +"%b %d %T:%Z `hostname`")
}
```

# log_success_msg

## 用法
`log_success_msg` `message`

## 目的
向屏幕和日志文件打印成功消息。

## 输入
消息

## 返回值
未使用

```bash
log_success_msg()
{
  echo -n "${BMPREFIX}$@"
  echo -e "${CURS_ZERO}${SUCCESS_PREFIX}${SUCCESS_SUFFIX}"
  # 从日志文件中剥离不可打印字符
  logmessage=$(echo "$@" | sed 's/\033[a-zA-Z]//g')
  timespec
  echo -e "${STAMP} ${logmessage}" >> "${BOOTLOG}"
  return 0
}
```

```bash
log_success_msg2()
{
  echo -n "${BMPREFIX}$@"
  echo -e "${CURS_ZERO}${SUCCESS_PREFIX}${SUCCESS_SUFFIX}"
  echo "OK" >> "${BOOTLOG}"
  return 0
}
```

# evaluate_retval

## 用法
评估返回值并根据情况打印成功或失败

## 目的
用于终止信息消息的便利函数

## 返回值
未使用

```bash
evaluate_retval()
{
    local error_value="${?}"

    if [ ${error_value} -eq 0 ]; then
        log_success_msg
    else
        log_failure_msg
    fi
}
```

# check_signal

## 用法
`check_signal()` `[-signal ]`

## 目的
检查是否有效有效。这在 LSB 草案中没有定义，但需要检查信号以确定所选的信号是否为函数的无效参数。

## 输入
接收格式为 `{-signal}` 的单个字符串值

## 返回值
`0` - 成功（信号有效）
`1` - 信号无效

```bash
check_signal()
{
    local valsig

    # 为无效信号添加错误处理
    valsig="-ALARM -HINT -KILL -PILL -POLL -TERM -USRI2"
    valsig="${valsig} -VTERM -STKLFT -PWR -WINCH -CHLG -URG -TSTP -TTPIN"
    valsig="${valsig} -TTOU -STOP -CONT -ABRT -FPE -ILL -QUIT -SEGV -TRAP"
    valsig="${valsig} -SYS -ENT -BUS -XCPU -XFSE -0 -2 -3 -4 -5 -6 -8 -9*"
    echo "${valsig}" | grep -E "${1}" > /dev/null
    if [ "${1}" -eq "0" ]; then
        return 0
    else
        return 1
    fi
}
```
