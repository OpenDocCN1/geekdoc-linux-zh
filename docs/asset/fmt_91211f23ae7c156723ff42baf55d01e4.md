
# `page_remove_rmap()` 函数分析

## 函数签名

```c
void page_remove_rmap(struct page *page,
    struct vm_area_struct *vma, bool compound)
```

## 核心逻辑

### 匿名页处理

```c
if (!PageAnon(page)) {
    page_remove_file_rmap(page, compound);
    goto out;
}
```

### 映射计数检查

```c
/* page stage mapped by someone else? */
if (!atomic_negative(-1, &page->mapcount))
    goto out;
```

### 统计信息更新与锁定处理

```c
/*
 * 我们使用 irq-unsafe_{inc|mod}_zone_page 是因为
 * 这些计数器不会在中断上下文中修改，
 * 并且持有了 pte lock(一个自旋锁)，这意味着禁用了抢占。
 */
_dec_lru_page_state(page, NR_AN_MAPPED);

/*
 * 如果在这里重置 PageAnon 映射，
 * 可能会覆盖正在竞争的 page_anon_rmap，
 * 该映射会在我们添加映射之后增加 mapcount：
 * 因此，我们将重置操作交给 free_unref_page，
 * 并记住它只有在被映射时才是可靠的。
 * 这样做也有助于 swapoff 更快地恢复
 * 仍留在 swapcache 中的页的 ptes。
 */
munlock_vma_page(page, vma, compound);
```

## `page_remove_file_rmap()` 函数

```c
static void page_remove_file_rmap(struct page *page, bool compound) {
    int i, nr = 0;

    if (atomic_negative(-1, &page->mapcount))
        nr++;

    if (nr)
        _mod_lru_page_state(page, NR_FILE_MAPPED, -nr);
}
```

## TLB 批处理机制

### `_tlb_remove_page_size()`

```c
_tlb_remove_page_size(struct mmu_gather *tlb, struct page *page, int page_size) {
    struct mmu_gather *batch;

    VM_BUG_ON(!tlb->end);

    #ifdef CONFIG_MMU_GATHER_PAGE_SIZE
    VM_WARN_ON(tlb->page_size != page_size);
    #endif

    batch = tlb->active;

    /*
     * 将页添加到活动批次，并检查我们是否已满。
     * 如果是，强制刷新。
     */
    batch->pages[batch->nr++] = page;

    if (batch->nr == batch->max) {
        if (!tlb_next_batch(tlb))
            return true;
        batch = tlb->active;
    }

    VM_BUG_PAGE(batch->nr > batch->max, page);
    return false;
}
```

### `tlb_next_batch()`

```c
static bool tlb_next_batch(struct mmu_gather *tlb) {
    struct mmu_gather_batch *batch;

    batch = tlb->active;

    if (batch->next) {
        tlb->active = batch->next;
        return true;
    }

    if (tlb->batch_count == MAX_GATHER_BATCH_COUNT)
        return false;

    batch = (void*)_get_free_pages(GFP_NOWIT | __GFP_NOWARN, 0);
    if (!batch)
        return false;

    tlb->batch_count++;
    batch->next = NULL;
    batch->nr = 0;
    batch->max = MAX_GATHER_BATCH;
    tlb->active->next = batch;
    tlb->active = batch;
    return true;
}
```

**关键参数**：
- `MAX_GATHER_BATCH_COUNT`：包含 1000 个页的批次数（x86 系统）
- `MAX_GATHER_BATCH`：每个批次的最大页数，x86-64 上等于 509

## `tlb_end_vma()`

```c
static inline void tlb_end_vma(struct mmu_gather *tlb, struct vm_area_struct *vma) {
    if (tlb->full) {
        return;
    }

    /*
     * VM_PFNMAP 更脆弱，因为核心 mm 不跟踪
     * pagecount——这些 PFN 可能没有页帧。
     * 强制刷新此类范围的 TLB 以防止 unmap() 与
     * mapping_range() 竞态。
     */
    if (tlb->vma_pfn || !ENABLED(CONFIG_MMU_GATHER_MERGE_VMAS)) {
        /*
         * 执行 TLB 刷新并在 VMA 边界重置范围；
         * 避免随着连续 VMA 之间未使用空间的增长。
         */
        tlb_flush_mmu();
    }
}
```

```c
         * aarch64 在使用 fullmm 时产生更好的性能，
         * 避免了多个 CPU 滥用 TLBI 消息
         */

        343         * same time.
        344         *
        345         * 在 x86 上，非-fullmm 与 fullmm 相比
        346         * 没有显著差异。
        347         */

        349 tlb_reset_range(tlb);
        350 tlb->freed_tables = 1;
        351

```

**执行流程**：
1. 通过 `mm_tlb_flush_nested()` 检查是否有并发 TLB 刷新（检查 `struct mmu_struct->tlb_flush_pending > 1`）
2. 若有并发，切换到全进程空间刷新（`tlb->fullmm = 1`）
3. 调用 `tlb_flush_mmu()` 执行实际刷新
4. 通过 `tlb_batch_list_free()` 释放批量列表
5. 通过 `dec_tlb_flush_pending()` 减少并发刷新计数

## `free_pgtables()`

```c
void free_pgtables(struct mmu_gather *tlb, struct vm_area_struct *vma,
    unsigned long floor, unsigned long ceiling)
{
    while (vma) {
        struct vm_area_struct *next = vma->vm_next;
        unsigned long addr = vma->vm_start;

        /*
            在释放页表之前，从 rmap 和 truncate_pagecache 中隐藏 vma
            *
            */
        unlink_anon_vma(vma);
        unlink_file_vma(vma);

        /*
        * 优化：将相邻的 vmas 合并到一个调用
        while (next && next->vm_start < vma->vm_end +
        PMD_SIZE
            && is_vm_hugetlb_page(next)) {
                vma = next;
                next = vma->vm_next;
                unlink_anon_vmas(vma);
                unlink_file_vma(vma);
            }
            free_pgd_range(tbl, addr, vma->vm_end,
                floor, next ? next->vm_start : ceiling);

            vma = next;
        }
    }
```

## `unlink_file_vma()`

```c
/*
 * 从其区间树中取消基于文件的 vm 结构的链接，以便在释放页表
 * 之前从 rmap 和 vmruntrace 中隐藏 vma。
 */
void unlink_file_vma(struct vm_area_struct *vma)
{
    struct file *file = vma->vm_file;

    if (file) {
        struct address_space *mapping = file->f_mapping;
        i_mmap_lock_write(mapping);
        remove_shared_vm_struct(vma, file, mapping);
        i_mmap_unlock_write(mapping);
    }
}
```

---

## `__remove_shared_vm_struct()`

```c
/*
 * 需要 inode->i_mapping->i_mmap_rwsem
 */
static void __remove_shared_vm_struct(struct vm_area_struct *vma,
                             struct file *file, struct address_space *mapping)
{
    if (vma->flags & VM_SHARED)
        unmap_writable(mapping);
}
```
