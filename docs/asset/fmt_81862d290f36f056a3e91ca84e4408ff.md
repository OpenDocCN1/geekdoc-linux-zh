```bash
echo
# 读取并验证目录的名称

read CHOICE

    cpio -i </dev/rmt0;
  2) echo "Archiving..."
    ls | cpio -o >/dev/rmt0;
  3) echo "Unloading..."
    ls | cpio -o >/dev/rmt0;
  *) echo "Sorry, $CHOICE is not a valid choice"
    exit 1
esac
# 检查 cpio 错误
if [ $? -ne 0 ]
then
    echo "A problem has occurred during the process"
    if [ $CHOICE = 3 ]
    then
        echo "The directory will not be erased"
    fi

    exit 2

    then
        rm *
    fi
fi
# 询问用户是否继续

while [ $ANSWER = Y ]
do
    # 在此处重复菜单逻辑
done
```

`until` 结构在条件为真时停止循环。`until` 结构是 `while` 的互补；你通常可以使用任何一种。

`until` 结构的一般形式如下：

```
until command
do
  command
  ...
done
```

在 `for` 循环中重复：shell 中的第三种循环结构是 `for` 循环。`for` 循环与其他的区别在于它不是基于真假条件。相反，`for` 循环为它提供的参数列表中的每个单词执行一次。在循环的每次迭代中，命令行指定的变量赋值为参数列表中下一个单词的值。

`for` 循环的般语法如下：

```
for variable in arg1 arg2 ... argn
do
  command
  ...
done
```

下面的简单示例说明了该结构：

```
$ for LETTER in a b c d; do echo $LETTER; done
a
b
c
d
```

由于参数列表中四个单词，循环精确执行了四次。`for` 命令中的参数列表不必是字面常量；它可以来自变量替换。

你也可以将 `shift` 命令用于另一个用途。shell 预定义九个位置参数，`$1` 到 `$9`。为了访问更多参数，如果你知道已经处理了前三个位置参数（例如），并想开始循环来处理剩余参数，可以使用以下命令 `shift 3`：

```bash
shift 3
```

## 从中间位置退出循环

通常情况下，循环结构会执行 `do` 语句和 `done` 语句之间的所有命令。有两个命令可以让你绕过这一局限性：`break` 命令导致程序立即退出循环，`continue` 命令导致程序跳过循环中剩余的命令但仍留在循环中。

在 Shell 编程中，经常使用的一种技术是启动一个无限循环，这种循环在执行 `break` 或 `continue` 命令之前不会结束。无限循环通常用 `true` 或 `false` 命令启动。`true` 命令始终返回退出状态 0，而 `false` 命令始终返回非零的退出状态。该循环如下：

```bash
while true
do
command
...
command
done
```

循环会一直执行，直到你的程序执行了 `break` 或用户发起了中断。你也可以按照如下方式编写无限循环：

```bash
until false
do
command
...
command
done
```

我们可以使用这种技术来让列表 9.7 中的交互式归档程序变得稍微好用。修改后的程序位于列表 9.10。

## 列表 9.10。归档程序的另一个版本。

```bash
# 用于恢复、备份或卸载目录的交互式程序
echo "Welcome to the menu driven Archive program"
while true
do
## 显示菜单
echo
echo "Make a Choice from the Menu below"
echo
echo "1 Restore Archive"
echo "2 Backup directory"
echo "3 Unload directory"
echo "4 Quit"
echo
## 读取用户的选择
echo "Enter Choice: \c"
read CHOICE
case $CHOICE in
[1-3] ) echo
## 读取并验证目录名称
echo "What directory do you want? \c"

read WORKDIR
if [ -d "$WORKDIR" ]
then
    echo "Sorry, $WORKDIR is not a directory"
    continue
fi
## 将目录设置为当前工作目录
cd $WORKDIR;
4) ;;
*) echo "Sorry, $CHOICE is not a valid choice"
esac

case "$CHOICE" in
  1) echo "Restoring..."
     cpio -i </dev/rmt0;
  2) echo "Archiving..."
     ls | cpio -o >dev/rmt0;
  3) echo "Unloading..."
     ls | cpio -o >dev/rmt0;
  4) echo "Quitting"
     break;
esac
# 检查 cpio 错误
if [ $? -ne 0 ]
then
    echo "A problem has occurred during the process"
    if [ $CHOICE = 3 ]
    then
        echo "The directory will be erased"
    fi
    echo "Please check the device and try again"
    continue
else
    if [ $CHOICE = 3 ]
    then
        rm *
    fi
fi
done
```

在程序 9.1 中，只要 `true` 返回退出状态 0，循环就会继续，或者直到用户选择了 4，这会执行 `break` 命令并终止循环。同样注意，如果用户在选择选项或输入目录名称时出错，执行的是 `continue` 语句而不是 `exit` 语句。这样，使用户在输入数据时出错，他们也可以留在程序中，但该错误是无法撤销的。

同样注意两个 `case` 语句的使用。第一个 `case` 语句仅在选择 1、2 或 3 时才要求操作员输入目录名。这个示例说明了 `case` 语句中的匹配与命令行上的匹配相似。在第一个 `case` 语句中，如果用户选择选项 4，则执行空命令 `())`。第一个 `case` 检查无效选择，如果进行了无效选择则执行 `continue`，第二个 `case` 语句则不需要检查任何有效选择。

## 使用函数的结构化 Shell 编程

高级编程语言中的一个共同特性是能够将指令组合成函数，这些函数可以从程序的任何地方被调用。这些函数有时被称为子程序。Bourne shell 也提供了这种能力。

函数定义的通用语法如下：

```bash
funcname ()
  command
  ..
  command;
}
```

一旦定义了函数，就可以使用 `funcname` 作为命令从 shell 的任何地方调用它。你可能想将命令分组为函数的原因有两个。其中一个很好的理由是将复杂的程序拆分成更易于管理的段，创建一个结构化的程序。一个结构化程序可能采取以下形式：

```bash
## 程序开始
setup ()
  command list ;
do_data ()
  command list ;
cleanup ()
  command list ;

setup
do_data
cleanup
## 程序结束
```

在示例中，`setup`、`do_data`、`cleanup` 是函数。当你观察一个结构良好的程序时，函数的名称让你对这些函数的作用有一个大致的了解。如果你试图分析它，你可能会假设 `setup` 和 `cleanup` 函数集中在 `do_data` 部分。

> **提示**：始终给变量和函数赋予有意义的名字。看起来看起来你正在写一个你会记得变量和函数用途的程序，但经验证明随着时间的推移，情况就不再那么清晰了。你还应该记住，可能会有其他人查看你的程序，那个人会感谢描述性强的名称。

将命令分组为函数的另一个合理原因是，你可能想从程序中的几点执行命令序列。在列表 9.10 的交互式程序中的几处，发生了非致命错误并执行了 `continue` 命令。你可以给用户在这些点上一个继续的选项。

`rm $TMPDIR/*; exit`

被放置在陷获（trap）表中。当检测到信号时，表中的命令字符串会被解释，然后进行变量替换。在第一次，`$TMPDIR` 和 `$$` 具有它们在执行陷获时所具有的值。在第二次，`$TMPDIR` 和 `$$` 假设了它们在检测到信号时所具有的值。请确保你清楚究竟是哪一个。

`trap` 命令的命令字符串几乎总是包含一个 `exit` 语句。如果你不包含 `exit` 语句，当检测到信号时将执行 `rm` 命令，程序将在信号发生并退出后，从它上次停止的地方继续运行。例如，如果你不想让程序在终端断开连接时停止，你可以捕获挂断（hangup）信号，如下文所示：

```
trap : 1
```

你可以通过执行不带命令字符串的 `trap` 命令将陷获恢复为默认值，如下：

```
trap 1
```

以下命令的效果是使用户需要按两次中断键来终止程序：

```
trap 'trap 2' 2
```

## 使用 And/Or 构造进行条件命令执行

正如你已经看到的，你可以以多种方式编写脚本，而不改变程序的结果。例如，`until` 语句仅仅是使用 `while` 语句的反向方式。你可以使用 `if-then-else` 构造使命令根据条件执行，但你也可以使用 `&& ||` 运算符来实现条件执行。在 C 语言中，这些符号分别表示逻辑与（and）和逻辑或（or）操作。在 Bourne shell 中，`&&` 连接两个命令，使得只有在第一个命令成功时，才执行第二个命令。

`command && command` 的通用格式如下：

```
command && command
```

例如，在语句中

```
rm $TMPDIR/* && echo "Files successfully removed"
```

`echo` 命令只有在 `rm` 命令成功时才会执行。你也可以像下面这样的 `if-then` 语句中进行这种编程：

```
if rm $TMPDIR/*
then
echo "Files successfully removed"
fi
```

相反，`||` 连接命令，使得只有在第一个命令执行失败时才执行第二个命令，如下所示所示：

```
rm $TMPDIR/* || echo "Files were not removed"
```

上述内容等同于编程中的：

```
if rm $TMPDIR/*
then
else
  echo "Files were not removed"
fi
```

你还可以级联运算符。在接的命令行中，只有当 `command1` 和 `command2` 都成功时，才会执行 `command3`：

```
command1 && command2 && command3
```

你还可以级联不同类型的运算符。在下面的命令行中，只有当 `command1` 成功且 `command2` 失败时，才会执行 `command3`：

```
command1 && command2 || command3
```

`&` 和 `|` 是条件命令执行的简单形式，通常只用于执行单个命令的情况。虽然命令可以是复合的，但如果这种格式中命令过多，程序将变得难以阅读。如果你使用的命令多于两个，`if-then` 构造似乎会更清晰。

## 读取 UNIX 风格选项

关于 Unix 的几点是，大多数标准命令具有相似的命令行格式：

```
command -options parameters
```

如果你在为他人编写脚本，使用相同的约定会很好。为了帮助你这样做，Bourne shell 中提供了 `getopts` 命令，用于读取和处理这种格式的选项：`getopts` 命令具有以下格式：

```
getopts option_string variable
```

其中 `option_string` 包含有效的单字符选项。如果 `getopts` 在命令输入流中看到连字符（-），它会将该连字符与 `option_string` 中的字符进行比较。如果发生匹配，`getopts` 将 `variable` 设置为该选项；如果连字符不匹配 `option_string` 中的任何字符，则将 `variable` 设置为问号（?）。如果 `getopts` 在连字符后没有找到更多字符，它将允许你在循环中使用 `getopts`。

第 9.11 节的程序展示了如何使用 `getopts` 处理 `date` 命令的选项。该程序创建了一个符合标准 UNIX 的 `date` 版本，并添加了一些选项。

## 列表 9.11. 标准化的日期函数 newdate。

```bash
#newdate
if [ $# -lt 1 ]
then
  date
fi

while getopts mdyHMSTjJahr OPTION
do
  case $OPTION
    in
      m) date '+%m'; # Month of Year
      d) date '+%d'; # Day of Month
      y) date '+%y'; # Year
      D) date '+%d'; # MM/DD/YY
      H) date '+%H'; # Hour
      M) date '+%M'; # Minute
      S) date '+%S'; # Second
      T) date '+%T'; # HH:MM:SS
      j) date '+%j'; # day of year
      J) date '+%y%j'; # 5 digit Julian date
      w) date '+%w'; # Day of the Week
      a) date '+%a'; # Day abbreviation
      h) date '+%h'; # Month abbreviation
      r) date '+%r'; # AM-PM time
  esac
    echo "Invalid option $OPTION"
done
```

在程序 9.11 中，每个选项被依次处理。当 `getopts` 处理完所有选项后，它返回一个非零退出状态，`while` 循环终止。注意 `getopts` 允许选项堆叠在一个单个连字符后面，这也是一个常见的特性。以下示例说明了 `newdate` 如何工作：

```
$ newdate -J
$ newdate -a -h
$ newdate -d
```

有时选项需要一个参数，`getopts` 还会通过冒号解析 `option_string` 中的标记。当 `getopts` 看到冒号时，它会查找选项后面的值。如果值存在，`getopts` 将该值存储在特殊变量 `OPTARG` 中。如果找不到值，`getopts` 将问号存储在 `OPTARG` 中并向标准错误写入一条消息。程序 9.12 复制文件并提供所创建的文件名。`-c` 选项接受一个指定副本数量的参数，而 `-v` 选项指示程序进入详细模式（verbose）。
