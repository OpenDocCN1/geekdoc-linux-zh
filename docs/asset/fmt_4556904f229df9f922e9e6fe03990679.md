```c
* "超时"读取 - timeout 指定了每次调用在放弃前等待的秒数
* 的，精确读取 n 字节。
* 返回读取的字节数，出错时返回 -1。
* 锁：
*/
size_t
ntread(int fd, void *buf, size_t nbytes, unsigned int timeout)
{
    size_t nleft;
    size_t read;

    nleft = nbytes;
    while (nleft > 0) {
        if ((read = tread(fd, buf, nleft, timeout)) < 0) {
            if (nleft == nbytes)
                return(-1); /* 出错，返回已读取量 */
            else
                break; /* 出错，返回已读取量 */
        } else if (read == 0)
            break; /* EOF */
        }
        nleft -= read;
        buf += read;
    }
    return(nbytes - nleft); /* 返回 >= 0 */
}
```
