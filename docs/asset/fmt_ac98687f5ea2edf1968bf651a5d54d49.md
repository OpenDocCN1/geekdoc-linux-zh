`build_default_conf` () {
    #在此处定义文件和变量
    `HN`="$1" # 要添加的服务器主机名
    `BANK`="$2" # 用于备份的备份库 (Bank)
    `DAYS`="$3" # 备份过期前的天数
    `TREE`="$4" # 要备份的目录
    `IGNORE_LIST`="$5" # 要忽略的文件和目录
    `OUTFILE`="$DEFAULT_CONF" # 输出文件名
    # 以下所有输出均用于构建
    # 此备份的新的默认 default.conf 文件
    echo "client: $HMN"
    echo "bank: $BANK"
    echo "vault: $HMN"
    echo "server: $DIRVISH_SERVER"
    echo "expire: $DAYS"
    echo "index: gzip"
    echo "log: gzip"
    echo "image: $(HMN)-%y%m%d%H%M%S"
    echo "tree: $TREE"
    echo -e "exclude:"
    echo "$IGNORE_LIST" | sed /^\$/d | while read Q
    do
        echo -e \t$Q"
    done
}}OUTFILE
}
