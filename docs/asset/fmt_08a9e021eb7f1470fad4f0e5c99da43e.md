```python
class Stack:
    def __init__(self):
        self._items = list()

    def push(self, item):
        self._items.append(item)

    def pop(self):
        return self._items.pop()

    def __len__(self):
        return len(self._items)
```

```python
class Stack:
    def __init__(self, container=None):
        if container is None:
            container = list()
        self._items = container

    def push(self, item):
        self._items.append(item)

    def pop(self):
        return self._items.pop()

    def __len__(self):
        return len(self._items)
```

```python
class Stack:
    def __init__(self):
        self._items = None
        self._size = 0

    def push(self, item):
        self._items = (item, self._items) # 注意：此逻辑在原始 OCR 中是错误的
        self._size += 1

    def pop(self):
        (item, self._items) = self._items
        self._size -= 1

    def __len__(self):
        return self._size
```
