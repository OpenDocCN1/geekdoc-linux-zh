
```bash
#!/bin/bash

FILE_NAME=/var/log/vmstat.log

NR_OF_SEC=2 # 读取间隔

WAIT_TIME=10 # 数字为秒。

echo "######################################" >> $FILE_NAME

echo "######################################" >> $FILE_NAME

echo "" >> $FILE_NAME

date >> $FILE_NAME

echo "--------------------------------" >> $FILE_NAME

vmstat $NR_OF_SEC >> $FILE_NAME

echo "" >> $FILE_NAME

sleep $WAIT_TIME

for i in $(ps ax|grep vmstat | awk '{print $1}')

do

kill -9 $i > /dev/null

done
```
