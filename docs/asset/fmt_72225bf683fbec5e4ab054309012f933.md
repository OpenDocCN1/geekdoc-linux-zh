    for ( p = buf; *p != '\0'; )
        ++p++;

        if (p == buf+block_size)
            zeros += block_size;

}

if (zeros > 0)
    printf("%ld %s\n", zeros, filename);

if (ferror(f)) {
    errormsg(0, -1, "read failed for %s", filename);
    return -1;
}

return 0;
