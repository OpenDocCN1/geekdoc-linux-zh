```bash
killproc() {
    local pidfile=""
    local killsig=TERM 默认信号为 SIGTERM
    pidlist=""

    # 这将确保与之前的 LFS 引导程序兼容
    if [ -n "${PIDFILE}" ]; then
        pidfile="${PIDFILE}"
    fi

    while true
    do
        case "${1}" in
            -p)
                pidfile="${2}"
                shift 2
                ;;
            *)
                log_failure_msg "未知选项: ${1}"
                return 2
                ;;
            *)
                break
                ;;
        esac
    done

    if [ "${#}" = "2" ]; then
        killsig="${2}"
    elif [ "${#}" = "1" ]; then
        shift 2
        log_failure_msg "用法: killproc [-p pidfile] pathname [signal]"
        return 2
    fi

    # 进程正在运行？
    if [ -z "${pidfile}" ]; then
        pidofproc -s "${1}"
    else
        pidofproc -s -p "${pidfile}" "${1}"
    fi

    # 过期的 pidfile
    if [ "$?" -ne 1 ]; then
        boot_msg "正在移除过期的 pid 文件: ${pidfile}." "${WARNING}"
        rm -f"${pidfile}"
    fi

    # 如果正在运行，则发送信号
    if [ -n "${pidlist}" ]; then
        for pid in ${pidlist}
        do
            kill -s${killsig} ${pid} 2>/dev/null
        done
    fi

    # 等待最多 3 秒直到 $(pid) 终止
    case ${killsig} in
      SIGTERM|SIGKILL|SIGKILL)
        # 睡眠 0.1 秒
        # 将 KILDELAY 乘以 10
        local dtime="${KILDELAY}0"
        while [ "${dtime}" -ne "0" ];
        do
            kill -0 ${pid} 2>/dev/null || break
            sleep 0.1
            dtime=$(( ${dtime} - 1 ))
        done
        # 如果 $pid 仍在运行，杀死它
        kill -0 ${pid} 2>/dev/null || kill -KILL ${pid} 2>/dev/null
        ;;
    esac

    # 如果我们过停止它，检查进程是否仍在运行
    case ${killsig} in
      SIGTERM|SIGKILL|SIGKILL)
        if [ -z "${pidfile}" ]; then
          pidofproc -s "${1}"
        else
          pidofproc -s -p "${pidfile}" "${1}"
        fi

        # 程序已终止
        if [ "$?" -ne "0" ]; then
          # 如果需要，移除 pidfile
          if [ -f "${pidfile}" ]; then
            rm -f"${pidfile}"
          fi
          echo_ok
          return 0
        else # 程序仍在运行
          echo echo_failure
          return 4 # 未知状态
        fi
        ;;
      )
        # 检查 kill 是否成功返回
        evaluate_retval
        ;;
    esac
  else # 进程未运行
    print_status warning not running
    fi
}
```
