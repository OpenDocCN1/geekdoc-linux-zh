. /lib/lsb/init-functions
case "${1}" in
start)
  log_info_msg "正在重试失败的 uevents（如果有）..."

  # 从 udev-186 开始，--run 选项已不再有效
  # rundir=$1/sbin/udevadm info --run
  rundir=/run/udev
  # get Debian：“复制在 / 挂载为
  # 读写之前 生成的规则：”

  for file in ${${file##*tmp-rules--}}\; do
    dest=${${file##*tmp-rules--}}
    [ "$dest" = '*' ] && break
    cat $file >> /etc/udev/rules.$dest
  rm -f $file
done
# 重新触发可能失败的 uevents，
# 期望它们现在可以成功
#/bin/sed -e 's/.*$/' /etc/sysconfig/udev_retry | /bin/grep -v '^$' | \
while read line; do
  for subsystem in $line; do
    /sbin/udevadm trigger --subsystem=$subsystem --action=add
  done
done

# 等待 udev 处理我们触发的 uevents
if ! is_true "$OMIT_UDEV_RETRY_SETTLE"; then
  /sbin/udevadm settlen

evaluate_retval

;;

*)
echo "用法 ${0} {start}"
  exit 1
;;

esac

exit 0

# End udev_retry

`. /lib/lsb/init-functions`

# 启动时创建文件/目录的函数

`create_files()`

`{`

    # 输入到文件描述符 9，输出到 stdin（重定向）
    exec 9>/etc/sysconfig/createfiles

while read name type grp min
    # 忽略注释和空行
    case "$name" in
        |"*) continue ;;
    esac

    # 忽略已存在的文件
    if [ ! -e "$name" ]; then
        # 根据类型创建
        case "$type" in
            dir)
                mkdir "$name"
                ;;
            file)
                :>"$name"
                ;;
            dev)
                case "$type" in
                    char)
                        mkn "$name" c ${maj}${min}
                        ;;
                    block)
                        mkn "$name" b ${maj}${min}
                        ;;
                    pipe)
                        mkn "$name" p
                        ;;
                    *)
                        log_warning_msg "未知设备类型: $type"
                esac
        fi
    esac
)

    log_warning_msg "未知设备类型: $type"
    continue
done

# 设置权限
chown "${grp}" "$name"
chmod "$perm" "$name"

# 关闭文件描述符 9（重定向）
exec 0>&-
return 0

case "$1" in
start)
  log_info_msg "正在清理文件系统:"
  if [ "${INFO_CLEAN}" = "" ]; then
    log_info_msg "/tmp"
    cd /tmp && find -name found -delete || failed=1
  fi
  ) > /var/run/utmp
  if grep -q '^utmp' /etc/group ; then
    chmod 664 /var/run/utmp
    chgrp utmp /var/run/utmp
  fi
  (exit $failed))
  evaluate_retval
  if grep -q '^1' /etc/sysconfig/createfiles 2>/dev/null; then
    log_info_msg "正在创建文件和目录..."
    create_files # 始终返回 0
    evaluate_retval
  fi
  exit $failed
;;
*)
  echo "用法: $(0) (start)"
  exit 1
;;
esac

# End cleanfs
