
```bash
/lib/lsb/init-functions

case "$1" in
  start)
  log_info_msg "激活所有交换文件/分区..."
  swapon -a
  evaluate_retval
  ;;

stop)
  log_info_msg "取消激活所有交换文件/分区..."
  swapoff -a
  evaluate_retval
  ;;

restart)
  $0 stop
  sleep 1
  $0 start
  ;;

status)
  log_success_msg "正在获取交换状态。"
  swapon -s
  ;;

*)
  echo "用法: $0 {start|stop|restart|status}"
  exit 1
  ;;
esac

exit 0
# 结束 swap
```

```bash
#!/bin/sh
################################################################
# 开始 setclock
# 描述 : 设置 Linux 时钟
# 作者 : Gerard Beekmans - gerard@linuxfromscratch.org
DJ Lucas - dj@linuxfromscratch.org
# 更新 : Bruce Dubbs - dbubbs@linuxfromscratch.org
# 版本 : LFS 7.0
################################################################

### BEGIN INIT INFO
# Provides:
# Required-Start: modules
# Should-Start:
# Required-Stop:
# Default-Start: S
# Default-Stop:
# Short-Description: 从硬件时钟存储并恢复时间
# Description: 在启动时，系统时间从 hwclock 获取。
# 硬件时钟也可以在关机时设置。
# X-LFS-Provided-By: LFS
### END INIT INFO
```

```bash
/lib/lsb/init-functions

[ -r /etc/config/clock ] && . /etc/config/clock

case "$UTC" in
  true)
    CLOCKPARAMS="utc"
  ;;
  false)
    CLOCKPARAMS="localtime"
  ;;
esac

case "$1" in
  start)
    hwclock ${CLOCKPARAMS} >/dev/null
    ;;

  stop)
    log_info_msg "设置时钟..."
    hwclock ${CLOCKPARAMS} >/dev/null
    ;;

  *)
    echo "用法: $0 {start|stop}"
    exit 1
    ;;
esac

exit 0
```

```bash
### BEGIN INIT INFO
# Provides:          udev $time
# Required-Start:    modules
# Should-Start:
# Required-Stop:
# Should-Stop:
# Default-Start:     S
# Default-Stop:
# Short-Description: 在 /dev 中填充设备节点。
# Description:       将 temps 挂载到 /dev 并启动 udev 守护进程。
                    设备节点根据 udev 的定义创建。
# X-LFS-Provided-By:  LFS
### END INIT INFO
```

```bash
/lib/lsb/init-functions
case "$1" in
  start)
  log_info_msg "正在 /dev 中填充设备节点..."
  if grep -q '[[[:space:]]*sysfs/proc/modules'; then
    log_failure_msg2
    msg="失败:\n\n无法创建 SysFS 文件系统\n\n"
    msg="${msg}缺少没有 SysFS 文件系统的设备\n\n"
    msg="${msg}在你按下回车后，此系统"
    msg="${msg}将停止并关机。\n\n"
    log_info_msg "$msg"
    log_info_msg "按回车键继续..."
    wait_for_user
    /etc/rc.d/init.d/halt stop
fi

# 启动 udev 守护进程以持续监控并处理
# uevents
/sbin/udevd --daemon

# 现在遍历 /sys 以“冷插拔”已经发现的设备
/sbin/udevadm trigger --action=add --type=subsystems
/sbin/udevadm trigger --action=add --type=devices
/sbin/udevadm trigger --action=change --type=devices

# 等待 udevd 处理我们触发的更改
if ! is_true "$OMIT_UDE_SETTLE"; then
  /sbin/udevadm settle
fi

# 如果系统上有任何 LVM 分区，确保它们
# 已被激活以供使用。
if [ -x /sbin/vgchange ]; then /sbin/vgchange -a y >/dev/null; fi

log_success_msg2
;;

*)
echo "用法 $0 {start}"
exit 1
esac

exit 0

# 结束 udev
```
