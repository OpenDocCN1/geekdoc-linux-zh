```perl
# 如果字段名在 %Config 数组中已指定，则
# defined($Config[$name]) 将返回 1，我们应该将
#此值与相应的配置变量关联。如果这
# 不是配置字段，则将其放入关联
#  %Form 中，如果已经存在值，则在值后追加逗号。
# 我们在 #Field_Order 数组中也保存了表单字段的顺序，
# 我们可以使用此顺序进行通用排序。
if (defined($Config[$name])) {
    $Config[$name] = $value;
} else {
    if ($Form[$name] && $value) {
        $Form[$name] = '$Form[$name], $value';
    }
    elsif ($value) {
        push($Field_Order, $name);
        $Form[$name] = $value;
    }
}
```
