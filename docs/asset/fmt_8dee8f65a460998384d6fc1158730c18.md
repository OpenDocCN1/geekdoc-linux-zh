### 函数

#### allocate_init

```assembly
# PURPOSE: 调用此函数来初始化
# 函数（特别是，这设置了 heap_begin 和
# current_break）。它没有参数也没有
# 返回值。
.glob allocate_init
.type allocate_init,@function
```

```assembly
allocate_init:
pushl %ebp
## 标准函数内容
movl %esp, %ebp
## 如果在 %ebx 为 0 的情况下调用 brk 系统调用，
# 它返回最后一个有效的可用地址
movl $SYS_BRK, %eax # 找出 break 在哪里
movl $0, %ebx
int $LINUX_SYSCALL
```

```assembly
incl %eax #%eax 现在拥有最后一个有效的
# 地址，而我们想要
# 之后的内存位置
movl %eax, current_break # 存储当前的 break
movl %eax, heap_begin # 存储当前的 break
## 作为我们的起始地址。
## 第一次运行时
movl %ebp, %esp # 退出函数
popl %ebp
```

```assembly
ret
```

#### allocate

```assembly
# PURPOSE: 此函数用于获取一段内存。
## 它检查是否存在空闲块，如果没有，
## 它会向 Linux 请求新的内存。
## PARAMETERS: 此函数有一个参数 -
## 要分配的内存大小。
## RETURN VALUE:
# 此函数在 %eax 中返回分配的
# 地址。如果没有可用内存，
## 它将在 %eax 中返回 0。
```

```assembly
## 使用的变量：
## %ecx - 保存请求内存的大小
#（唯一的）
## %eax - 当前内存区域
## %ebx - 当前 break
## %edx - 当前内存的大小
allocate:
pushl %ebp
## 标准函数内容
movl %esp, %ebp
pushl %ecx
pushl %edx
movl heap_begin, %eax # eax 持有位置
movl current_break, %ebx # ebx 持有 break
alloc_loop_begin:
cmpl %ebx, %eax # 如果这些相等
je move_break # 需要更多内存
movl HDR_SIZE_OFFSET(%eax), %edx # 获取大小
cmpl $UNAVAILABLE, HDR_AVAIL_OFFSET(%eax) # 如果可用
je next_location
cmpl %edx, %ecx # 比较大小和需求
jle allocate_here # 如果足够，跳转
next_location:
add $HEADER_SIZE, %eax # 总大小求和
add $edx, %eax # 加上 8 字节
jmp alloc_loop_begin # 继续查找
allocate_here:
## 标记空间为不可用
movl $UNAVAILABLE, HDR_AVAIL_OFFSET(%eax)
add $HEADER_SIZE, %eax # 将 eax 移过头部
popl %edx
popl %ecx
movl %esp, %ebp # 从函数返回
popl %ebp
```

```assembly
ret
```

```assembly
move_break:
## 如果我们到了这里，
# 意味着我们已经耗尽了所有
# 可寻址内存，并且
# 我们请求更多。
## %ebx 持有当前的
```

## 第 9 章 中级内存主题

数据的端点，

并且 `%ecx` 存储它的大小

我们需要增加 `%ebx`，

直到我们 *想要* 内存结束的地方，所以我们

```assembly
addl $HEADER_SIZE, %ebx #为头部信息留出空间
#structure
```

```assembly
addl %ecx, %ebx #为请求的数据留出空间
```

```assembly
# 现在是时候向 Linux
索要更多内存了
pushl %eax #保存需要的寄存器
pushl %ecx
pushl %ebx
```

```assembly
movl $SYS_BRK, %eax #重置断点（%ebx 存储了
# 所请求的断点位置）
```

```assembly
int $LINUX_SYSCALL #在正常情况下，这应该
在 %eax 中返回新的断点，它
如果失败则为 0，或者
将等于或大于
## 我们请求的。在这个程序中，我们不在乎
它实际上在哪里
设置了断点，所以只要 %eax
不为 0，我们就不在乎它是什么
```

```assembly
cmpl $0, %eax #检查错误条件
je error
```

```assembly
popl %ebx #恢复保存的寄存器
popl %ecx
```

```assembly
popl %eax
```

```assembly
#将此内存标记为不可用，因为我们正准备
把它出去
mov $UNAVAILABLE, HDR_AVAIL_OFFSET(%eax)
#设置内存的大小
mov %ecx, HDR_SIZE_OFFSET(%eax)
```

```assembly
#将 %eax 移动到可用内存的实际起始位置
#%eax 现在持有返回值
addl $HEADER_SIZE, %eax
```

```assembly
mov %ebx, current_break    #保存新的断点
mov %ebp, %esp               #返回函数
popl %ebp
ret
```

```assembly
error:
mov $0, %eax                #出错时，我们返回零
mov %ebp, %esp
popl %ebp
ret
```

#### 解分配 deallocate

```assembly
## own:
## 该函数的目的是在我们停止使用
## 某内存区域后，将其还给内存池。
## 参数：
# 唯一的参数是我们要返回内存池的
# 内存地址。
## 返回值：
## 没有返回值
# 处理过程：
#   如果你还记得，我们实际上是给程序
#   提供了它们可以使用的内存起始，即
#   内存区域实际起始位置之后的 8 个存储单元。我们
#   做的只是回退 8 个位置并将该内存标记为可用，
#   这样分配函数就知道可以使用它了。
.globl deallocate
.type deallocate,@function
#要释放的内存区域在栈上的位置
.equ ST_MEMORY_SEG, 4
deallocate:
#由于该函数非常简单，我们
#不需要任何花哨的函数操作
```

```assembly
#获取要释放的内存地址
#(通常是 8(%ebp) 或将 %esp 移动到
#我们没有推送 %ebp 或将 %esp 移动到
# ebp，我们可以直接使用 4(%esp)
movl  ST_MEMORY_SEG(%esp), %eax
```

```assembly
#获取内存真实起始位置的指针
subl $HEADER_SIZE, %eax
```
