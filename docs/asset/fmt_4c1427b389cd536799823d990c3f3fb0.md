player_score = 0
computer_score = 0

def start():
    print("这是一个剪刀石头布游戏")
    while True:
        player = input("你的选择: ")
        computer = random.randint(1, 3)
        print(f"电脑出了: {computer}")
        if player == computer:
            print("平局")
        elif rules[player] == computer:
            print("你赢了！")
            player_score += 1
        else:
            print("你输了！")
            computer_score += 1
        answer = input("再玩一次吗？(y/n): ")
        if answer.lower() != 'y':
            break
    print(f"最终得分：你 {player_score} - 电脑 {computer_score}")
    print("感谢游戏！")

if __name__ == "__main__":
    start()
