package main

import "fmt"

func main() {
	var myArray = []int{2, 0, 4}
	length := len(myArray)
	for i := 0; i < length; i++ {
		fmt.Printf("%d, ", myArray[i])
	}
	fmt.Printf("\n")

	var otherArray = []int{2, 3, 5, 0, 7, 0}
	for i := 0; i < len(otherArray); i++ {
		fmt.Printf("%d, ", otherArray[i])
	}
	fmt.Printf("\n")

	// 打印两个数组 (2D)
	twoD := make([][]int, 3)
	for i := 0; i < len(twoD); i++ {
		twoD[i] = make([]int, 3)
		for j := 0; j < len(twoD[i]); j++ {
			twoD[i][j] = i + j
		}
	}

	// 打印二维数组
	for i := 0; i < len(twoD); i++ {
		fmt.Printf("%d: ", i)
		for j := 0; j < len(twoD[i]); j++ {
			fmt.Printf("%d, ", twoD[i][j])
		}
		fmt.Printf("\n")
	}

	// 使用 range 打印
	for _, value := range myArray {
		fmt.Printf("%d, ", value)
	}
	fmt.Printf("\n")

	// 使用 range 打印二维数组
	for i, row := range twoD {
		fmt.Printf("%d: ", i)
		for _, value := range row {
			fmt.Printf("%d, ", value)
		}
		fmt.Printf("\n")
	}
}
