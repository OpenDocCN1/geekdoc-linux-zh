```bash
print new Processing "$INPUT_DIR/"$filename"
IFS=' # 将字段分隔符设为空；默认值为空格
while read -r line; do
  col9=$(echo "$line"|cut -c 9)
  cols1to4=$(echo "$line"|cut -c 1-4)
  # 如果记录的第 1-4 列是 ACTG 且第 9 列是 'B'
  # 则将第 1-4 列更改为 WXYZ并将第 9 列更改为 'z'
  if [[ $cols1to4 == "ACTG" && $col9 == "B" ]]; then
    cols5to8=$(echo "$line"|cut -c 5-8)
    col9="z"
    cols10end=$(echo "$line"|cut -c 10-)
    edited_line=$(echo "$line"|cut -c 1-4)$cols5to8}${col9}${cols10end})
    echo "$edited_line" >>"$filename_new"
  else
    # 即使记录没有变化也进行编辑：
    echo "$line" >>"$filename_new"
  fi
done < "$filename"
# 让组中的每个人都对文件可读可写：
chmod 0664 "$filename_new" 2>/dev/null
done END DO-LOOP #1
```
