/*
* CreateMissile
* 导弹是外星人发射的。它们被赋予
* 固定的方向并在消失前持续有限的时间。
*/
typUnit *CreateMissile(typUnit *alien, typUnit *hero) {
    float length;
    typUnit *missile;
    /* 分配内存 ... */
    missile = (typUnit *) g_malloc(sizeof(typUnit));
    /* 初始化结构 ... */
    missile->bDestroy = FALSE;
    missile->Firing = 0;
    missile->type = MISSILE;
    missile->x = alien->vx;
    missile->y = alien->y;
    /* 计算导弹速度 ... */
    missile->vx = (float) DistanceBetween (missile, hero) *
    Direction (missile, hero);
    missile->vy = (float) (hero->y - alien->y);
    /*
     * 导弹速度
     */
    length = sqrt (missile->vx * missile->vx + missile->vy * missile->vy);
    if (length < .1) length = .1;
    length /= 3;
    missile->vx /= length;
    missile->vy /= length;
    missile->lockunit = NULL;
    /*
     * --- 该导弹有一个生命值；当它归零时，
     * 就会被销毁。
     */
    missile->life = 60;
    return (missile);
}

/*
* DistanceBetween
*
* 计算两个单位之间的距离 - 但
* 只使用 X 方向计算距离。
* 距离不仅仅是单位 X 坐标的差值，
而是因为世界
* 是环形的，我们必须考虑这样一个事实：
* 1 和 X_RANGE - 1 之间的距离并不是
* (X_RANGE-1) - 1
*/
int DistanceBetween(typUnit *u1, typUnit *u2) {
    int nDistance;
    int nDistance2;
    /* -- 找出哪个更大 ... */
    if (u1->x < u2->x) {
        /* --- 分两种方式计算距离 --- */
        nDistance = u2->x - u1->x;
        nDistance2 = X_RANGE + u1->x - u2->x;
    } else {
        /* --- 分两种方式计算距离 --- */
        nDistance = u1->x - u2->x;
        nDistance2 = X_RANGE + u2->x - u1->x;
    }
    /* -- 选取较短的距离 -- */
    return (nDistance < nDistance2) ? nDistance : nDistance2;
}
/*
* Direction
*
* 单位之间的 -x- 方向是什么？
* 它将是 -1, 1 或 0。
*/
int Direction(typUnit *u1, typUnit *u2) {
    int nDistance;
    int nDistance2;
    if (u1->x < u2->x) {
        /* ... 两种方向的距离 ... */
        nDistance = u2->x - u1->x;
        Distance2 = X_RANGE + u1->x - u2->x;
    } else {
        /* ... 两种方向的距离 ... */
        Distance2 = u1->x - u2->x;
        nDistance = X_RANGE + u2->x - u1->x;
    }
    /* ... 方向取决于较近的距离 ... */
    return ((nDistance < nDistance2) ? 1 : -1);
}
