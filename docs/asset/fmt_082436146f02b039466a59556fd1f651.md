```
[ -e /proc/modules ] || exit 0
/lib/sh${1}-init-functions

case "${1}" in

start)

  # 如果没有模块文件或没有有效的
  # 条则退出
  [ -r /etc/sysconfig/modules ] || exit 0
  egrep -qv '^#|($)$' /etc/sysconfig/modules || exit 0

  log_info_msg "加载模块:"

  # 只有当用户确实给了我们要加载的
  # 模块时才尝试加载。
  while read module args; do
    # 忽略注释和空行。
    case "${module}" in
      ""|"#") continue;
    esac

    # 尝试加载模块，传递任何提供的参数。
    modprobe ${module} ${args} >/dev/null

    # 如果成功则打印模块名，否则记录。
    if [ $? -eq 0 ]; then
      log_info_msg2 "${module}"
    else
      failedmod="${failedmod} ${module}"
    fi
  done < /etc/sysconfig/modules

  # 在正确的行上打印关于成功加载模块的消息。
  log_success_msg2

  # 打印失败消息，并列出任何
  #加载失败的模块。
  if [ -n "${failedmod}" ]; then
    log_failure_msg "加载模块失败:${failedmod}"
    exit 1
  fi

  ;;

*)
  echo "用法: $0 {start}"
  exit 1
  ;;

esac

# End modules
```

## D.6. /etc/rc.d/init.d/swap

```bash
#!/bin/sh

################################################################
# 开始 swap
#
# 描述：交换分区控制
# 作者：Gerard Beekmans - gerard@linuxfromscratch.org
# DJ Lucas - dj@linuxfromscratch.org
# 更新：Bruce Dubbs - bdubbs@linuxfromscratch.org
# 版本：LFS 7.0
################################################################

### BEGIN INIT INFO
# 提供: swap
# 需要启动: udev
# 应该启动: modules
# 需要停止:
# 应该停止:
# 默认启动: S
# 默认停止:
# 简描述: 挂载和卸载交换分区。
# 描述: 挂载和卸载 /etc/fstab 中定义的交换分区。
# X-LFS-Provided-By: LFS
### END INIT INFO

/lib/lsb/init-functions

[ -r /etc/sysconfig/clock ] && . /etc/sysconfig/clock

case ${UTC} in
yes|1)
  CLOCKPARAMS="${CLOCKPARAMS} --utc"
  ;;
false|0)
  CLOCKPARAMS="${CLOCKPARAMS} --localtime"
  ;;
esac

case $1 in
start)
  log_info_msg "激活交换分区..."
  swapon -a
  ;;
stop)
  log_info_msg "停用交换分区..."
  swapoff -a
  ;;
restart)
  $0 stop
  sleep 1
  $0 start
  ;;
status)
  log_success_msg "获取交换分区状态。"
  sw -s
  ;;
*)
  echo "用法: $0 {start|stop|restart|status}"
  exit 1
  ;;
esac

exit 0
```

```bash
######################################################################
# 开始 setclock
# 描述：设置 Linux 时钟
# 作者：Gerard Beekmans - gerard@linuxfromscratch.org
# DJ Lucas - dj@linuxfromscratch.org
# 更新：Bruce Dubbs - bdbubbs@linuxfromscratch.org
# 版本：LFS 7.0

### BEGIN INIT INFO
# 提供: setclock
# 需要启动:
# 应该启动: modules
# 需要停止:
# 应该停止: $syslog
# 默认启动: S
# 默认停止:
# 简描述: 从硬件时钟存储和恢复时间
# 描述：启动时，从硬件时钟获取系统时间。关机时也可以设置硬件时钟。
# X-LFS-Provided-By: LFS BLFS
#### END INIT INFO

/lib/lsb/init-functions

[ -r /etc/sysconfig/clock ] && . /etc/sysconfig/clock

case $1 in
start)
  hwclock --hctctys $(CLOCKPARAMS) >/dev/null
  ;;
stop)
  log_info_msg "设置硬件时钟..."
  hwclock --systohc $(CLOCKPARAMS) >/dev/null
  evaluate_retval
  ;;
esac
```
