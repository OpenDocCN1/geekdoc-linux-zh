static void tcp_delack_timer(unsigned long data)
(
    struct sock *sk = (struct sock *)data;
    struct tcp_opt *tp = &(sk->tp_pinfo.af_tcp);

    bh_lock_sock(sk);
    if (sk->lock_users) {
        /* 稍后试。 */
        tp->ack_blocked = 1;
        NET_INC_STATS_BH(DelayedACKLocked);
        if (mod_timer(&tp->delack_timer, jiffies + TCP_DELACK_MIN))
            sock_hold(sk);
            goto out_unlock;
        }
    }

    tcp_mem_reclaim(sk);

    if (sk->state == TCP_CLOSE && (tp->ack_pending & TCP_ACK_TIMER)) {
        goto out;
    }
    if ((long)(tp->ack_timeout - jiffies) > 0) {
        if (mod_timer(&tp->delack_timer, tp->ack_timeout)) {
            sock_hold(sk);
            goto out;
        }
    }
    tp->ack_pending &= ~TCP_ACK_TIMER;
    if (skb_queue_len(&sk->ucopy_prequeue)) {
        struct skb *sb = skb;

        net_statistics(smp_processor_id(), TCPSchedulerFailed += skb_queue_len(&sk->ucopy_prequeue));
        while ((sb = skb_dequeue(&(tp->ucopy_prequeue))) != NULL) {
            sk->backlog_rcv(sk, sb);
        }
    }
    tp->ucopy_memory = 0;
}
    if (tcp_ack_scheduled(tp)) {
        if (!tp->ack_pingpong(tp)) {
            /* 延迟确认确认已确认：放大 ATO。 */
            tp->ack_ato = min(tp->ack_ato << 1, tp->rto);
        } else {
            /* 延迟确认确认已确认：退出 pingpong 模式并
             * 缩小 ATO。 
             */
            tp->ack_pingpong = 0;
            tp->ack_ato = TCP_ATO_MIN;
        }
    }
    tcp_send_ack(sk);
    NET_INC_STATS_BH(DelayedACKs);
}
    TCP_CHECK_TIMER(sk);
out:
    if (tcp_memory_pressure) {
        tcp_mem_reclaim(sk);
} out_unlock:
    bh_unlock_sock(sk);
    sock_put(sk);
}

static inline void tcp_reset_xmit_timer(struct sock *sk, int what, unsigned long when)
{
    struct tcp_opt *tp = &sk->tp_pinfo.af_tcp;

    if (when > TCP_RTO_MAX) {

        when = TCP_RTO_MAX;
    }

    switch (what) {
        case TCP_TIME_RETRAN:
        case TCP_TIME_PROBE0:
            tp->pending = what;
            tp->timeout = jiffies+when;
            if (mod_timer(&tp->retransmit_timer, tp->timeout))
                sock_hold(sk);
                break;

        case TCP_TIME_DACK:
            tp->pending |= TCP_ACK_TIMER;
            tp->timeout = jiffies+when;
            if (mod_timer(&tp->delack_timer, tp->ack_timeout))
                sock_hold(sk);
                break;

        default:
            printk(KERN_DEBUG, "bug: unknown timer value'%u\n");
    }
}

## 10.4.9 `tcp_write_timer()`

这是一个用于 RTO 和窗口探测定时器的回调例程。我们在通过 `bh_lock_sock()` 持有套字锁的情况下处理定时器。接下来，我们检查套字是否被其他线程访问（`sk->lock_users != 0`）。如果是，我们不会继续处理定时器，而是通过第 424 行的 `mod_timer` 将定时器推迟 `HZ/20`。如果定时器在第 425 行尚未安装，我们需要持有套字的额外引用，并在第 429 行检查套字是否关闭或没有待处理的定时器（`tp->pending == 0`）。如果这些条件成立，我们返回定时器。如果定时器在第 432 行提前过期，我们使用过期时间 `tp->timeout` 重置定时器。如果定时器在第 434 行尚未安装，持有套字的额外引用。如果我们在这里，说明该执行定时器。RTO 或窗口探测超时。`tp->pending` 字段存储了哪个定时器过期。根据挂起的定时器，我们调用回调例程。在每次退出例程时，我们分别通过 `bh_unlock_sock()` 和 `sock_put()` 释放锁和套字的额外引用。
