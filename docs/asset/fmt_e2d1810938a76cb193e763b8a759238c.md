function doHangCollect
{
    echo >> $invlog
    echo "################################################################" >> $invlog
    echo "## Hang Collect ##" >> $invlog
    echo "################################################################" >> $invlog

    # 注意: $pid 包含挂起进程的进程 ID
    ## 检查该进程是否实际存在
    kill -0 $pid 2>/dev/null 1>/dev/null
    if [ ! $? -eq 0 ];
    then
        echo "Process ID \$pid not found."
        exit 3
    fi

    # 在此处添加特定命令
}
