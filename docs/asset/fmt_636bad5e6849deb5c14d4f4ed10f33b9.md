```apache
AddIcon /icons/layout.gif .html .shtml .htm .pdf

AddIcon /icons/f.gif .for

AddIcon /icons/bomb.gif .core

AddIcon /icons/blank.gif ^^BLANKICON^^

DefaultIcon /icons/unknown.gif

Name README.html

HeaderName HEADER.html

IndexIgnore ??* -* HEADER* README RCS CVS *,v *,t

AddEncoding x-compress Z

AddEncoding x-gzip gz tgz

AddLanguage in .en

AddLanguage fr .fr

AddLanguage de .de

AddLanguage da .da

AddLanguage el .el

AddLanguage it .it

LanguagePriority en fr de

<IfModule mod_php4.c>
AddType application/x-httpd-php .php4 .php3 .phtml .php
</IfModule>
```

# 列表 21-2（继续）

```apache
Options Indexes Includes FollowSymLinks
AllowOverride None
Order allow,deny
Allow from all

<Directory>
UserDir public_html
DirectoryIndex index.html index.htm index.shtml index.php index.php4
index.php3.cgi
AccessFileName .htaccess
<Files ~ "^\.ht">
  Order allow,deny
  Deny from all
</Files>
UseCanonicalName On
TypesConfig /etc/mime.types
DefaultType text/plain
<IfModule mod_magic.c>
  MIMagFile conf/magic
</IfModule>
HostnameLookups Off
ErrorLog /var/log/httpd/error_log
LogLevel warn
LogFormat "%h %l %u %r* %s %b \"(Referer)i\" \"(User-Agent)i\" d
LogFormat "%h %l %u %r* %s %b" common
LogFormat "%%(Referer)i -> %%U" refer
LogFormat "%%(User-agent)i" agent
CustomLog /var/log/httpd/access_log combined
ServerSignature On
Alias /icons/ "/var/www/icons/"
<Directory "/var/www/icons/">
  Options Indexes MultiViews
  AllowOverride None
  Order allow,deny
  Allow from all
</Directory>
ScriptAlias /cgi-bin/ "/var/www/cgi-bin/"
<Directory "/var/www/cgi-bin">
  AllowOverride None
  Options ExecCGI
  Order allow,deny
  Allow from all
</Directory>
```
