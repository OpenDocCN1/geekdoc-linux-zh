由于提供的对象接受任何 Unicode 文本，因此您通常需要验证用户输入的内容在特定上下文中是否有效。未经修改的对象提供将 `QValidator` 对象附加到部件的功能。这对大多数目的来说是足够的，但并非所有。

通常通过子类化此部件并添加所需的更改，您可以确切地从该部件获得所需的功能。以下示例演示了一个虚构的部件号部件，它是 `QLineEdit` 的简单子类。我们的设计规格为：

- 接受范围在 A0000 到 Z9999 之间的五个字符标识符

- 第一个字符是“A”和“Z”之间的字母值

- 其余字符是 0 到 9 范围内的数字

- 提供布尔值测试以确定有效性

- 当有效状态改变时生成通知，适合连接到 `QButton::setEnabled( bool )` 插槽，例如，一个 OK 按钮

- 可以编程设置值

- 不接受无效输入

- 小写开头字母会自动大写

```cpp
#ifdef _PARTNUMH
#define _PARTNUMH

#include <qlineedit.h>
#include <ctype.h>

class PartNum : public QLineEdit
{
  Q_OBJECT
public:
  PartNum( QWidget *p=0, const char *n=0 );
  virtual PartNum();

  bool       isValid( { return _valid } );

signals:
  // We'll emit this whenever the validity
  // changes
  void       valid( bool );

protected slots:
  // This slot will perform all the input
  // validation
  virtual void verify( const QString & );

private:
  bool       _valid;
};

PartNum.h

#include <PartNum.h>

PartNum::PartNum( QWidget *parent,
                   const char *name )
  : QLineEdit( parent, name )
{
  _valid = false;
  connect( this,
           SIGNAL(textChanged( const QString & ) ),
           SLOT(verify( const QString & ) ) );
}
```

```cpp
PartNum::~PartNum()
{
}

void
PartNum::verify( const QString &s )
{

    static bool busy = false;
    char bp[6];  // OK-range invalid for 118n
    int i;

    if(busy == true) return;
    memset( bp, 0, 6);
    strncpy(bp, s, 5);
    if(strlen(s) > 5) {

        busy = true;
        setText( bp );
        busy = false;
    }
    if(bp[0] >= 'a' && bp[0] <= 'z') {
        bp[0] = toupper(bp[0]);
        busy = true;
        setText( bp );
        busy = false;
    }
    if(bp[0] >= 'A' && bp[0] <= 'Z') {
        for(i=1;i<5;i++) {
            if(!isdigit(bp[i])) break;
        }
        if(i == 5) {
            if(_valid == false) {
                _valid = true;
                emit valid( _valid );
            }
        }
        return;
    } else {
        bp[i] = '\0';
        busy = true;
        setText( bp );
        busy = false;
    }
    }
    if(_valid == true) {
        _valid = false;
        emit valid( _valid );
    }
    return;
}

#include <PartNum.moc>
```
