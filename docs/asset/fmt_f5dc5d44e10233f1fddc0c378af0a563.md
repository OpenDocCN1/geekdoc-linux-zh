列表 3-49: `mm/vmalloc.c`: `enum fit_type`

`fit_type` 为：

- `NOTHING_FIT`: 完全不拟合。
- `FL_FIT_TYPE`: 所需块完美地放入空闲块（即它们的大小和对齐方式完全相同）。
- `LE_FIT_TYPE`: 空闲块在其“右侧”（高地址处）有一定的空间，但左侧完美拟合。
- `RE_FIT_TYPE`: 空闲块在其“左侧”（低地址处）有一些额外空间，但右侧完美拟合。
- `NE_FIT_TYPE`: 所需块位于空闲块内，但在两侧都留下了“间隙”。

执行此分类的逻辑定义在 `classify_va_fit_type()` 中，如列表 3-50 所示。

```c
static __always_inline enum fit_type
classify_va_fit_type(struct vmap_area *va,
    unsigned long nva_start_addr, unsigned long size)
{
    enum fit_type type;
    /* 检查是否在 VA 范围内。 */
    if (nva_start_addr < va->va_start ||
        nva_start_addr + size > va->va_end)
        return NOTHING_FIT;
    /* 现在进行分类。 */
    if (va->va_start == nva_start_addr) {
        if (va->va_end == nva_start_addr + size)
            type = FL_FIT_TYPE;
        else
            type = LE_FIT_TYPE;
    } else if (va->va_end == nva_start_addr + size) {
        type = RE_FIT_TYPE;
    } else {
        type = NE_FIT_TYPE;
    }
    return type;
}
```

列表 3-50: `mm/valloc.c`: `classify_va_fit_type()`

`classify_va_fit_type()` 指示了我们需要做什么。如果是 `NOTHING`，我们直接报错；对于 `LE_FIT_TYPE` 或 `RE_FIT_TYPE`，我们只需要调整空闲块的 `va_start` 或 `va_end`。在这些情况下，没有删除或添加节点。

在 `FL_FIT_TYPE` 情况下，我们通过 `unlink_va()` 从树中删除节点，并通过 `kmem_cache_free()` 释放区域。在 `NE_FIT_TYPE` 情况下，我们需要分配一个新的块来描述这两个独立的块，我们通过从每个 CPU 的值 `ne_fit_preload_node` 获取块来实现，该值通过 `preload_this_cpu()` 填充，并在 `alloc_vmap_area()` 中调用。新分配的块作为“左侧”块。

最后，对于 `FL_FIT_TYPE` 以外的所有情况，都会更新树，通过 `propagate()` 重新计算 `subtree_max_size` 值，并通过 `insert_vmap_area()` 添加一个新节点。

最后，我们可以查看 `alloc_vmap_area()` 中的最后一步，在新分配的 `vmap_area` 上调用 `insert_vmap_area()`，如列表 3-51 所示。

```c
      struct rb_root *root, struct list_head *head) {
```

列表 3-51: `mm/valloc.c`: `insert_vmap_area()`

这里它实际上是使用内核的红黑树将节点插入树中（以及添加到列表）。相反，我们关注 `find_va_links()`，它找到插入节点的位置，并返回父节点的左侧或右侧节点以及父节点本身，如列表 3-52 所示。

```c
/*
 * 此函数返回父节点及其左侧或右侧链接以便后续处理。
 * 否则返回 NULL。在这种情况下，所有关于冲突范围的
 * 插入步骤都必须被拒绝并被视为错误。
 */

static __always_inline struct rb_node **
find_va_links(struct rb_root *root, struct vmap_area *va,
    struct rb_node **parent)
{
    struct vmap_area *tmp_va;
    struct rb_node **link;

    link = &root->rb_node;
    if (unlikely(!link)) {
        *parent = NULL;
        return link;
    }

    /*
     * 转到树的底部。当我们到达最后一个点时，
     * 我们得到了 rb_node 和正确的方向，我将其命名
     * 为 link，新的 va->rb_node 将附加在那里。
     */
    do {
        tmp_va = rb_entry(*link, struct vmap_area, rb_node);

        /*
         * 在遍历过程中我们也进行了一些合理性检查。
         * 如果存在侧（左/右）或完全重叠，则触发 BUG()。
         */
        if (va->va_end <= tmp_va->va_start)
            link = &(*link)->rb_left;
        else if (va->va_start >= tmp_va->va_end)
            link = &(*link)->rb_right;
        else {
            WARN(1, "malloc bug: 0x%lx-0x%lx overlaps with 0x%lx\n",
                  va->va_start, va->va_end, tmp_va->va_start);
            return NULL;
        }
    } while (*link);

    *parent = rb_entry(*link, struct vmap_area, rb_node); // 逻辑修正：通常 parent 在循环中更新
    return link;
}
```

## 3.5.3 物理分配

`alloc_vmap_area()` 的最后部分通过 `__vmalloc_area_node()` 分配并映射物理内存。该函数由 `gfp_mask` 参数化，定义了将分配哪些页，以及 `prot`，定义了页表标志，因此调用者可以控制这些页的映射方式。

让我们查看列表 3-53 中该函数的第一个部分。

```c
static void * __vmalloc_area_node(struct vm_struct *area, unsigned long size, gfp_t gfp_mask, int node, void *caller) {
    const gfp_t nested_gfp = (gfp_mask & GFP_RECLAIM_MASK) | __GFP_ZERO;
    bool warn = gfp_mask & __GFP_NOWARN;
    unsigned long addr = (unsigned long)area->addr;
    unsigned long area_size = get_vm_area_size(area);
    unsigned long array_size;
    unsigned int nr_small_pages = area_size >> PAGE_SHIFT;
    unsigned int page_order;
    unsigned int flags;

    array_size = (unsigned long)nr_small_pages * sizeof(struct page);
    gfp_mask |= __GFP_NOWARN;
    if (!(gfp_mask & (GFP_DMA | GFP_DMA32))) {
        gfp_mask |= __GFP_HIGHMEM;
    }

    /* 请注意，递归是严格限制的。 */
    if (array_size > PAGE_SIZE) {
        area->pages = __vmalloc_area_node(array_size, 1, nested_gfp, node, area->caller);
    } else {
        area->pages = kmalloc_node(array_size, nested_gfp, node);
    }

    if (!area->pages) {
        warn_alloc(gfp_mask, NULL,
            "vmalloc error: size %lu, failed to allocate array size %lu (%d pages of size %lu",
            nr_small_pages * PAGE_SIZE, array_size);
        free_vm_area(area);
        return NULL;
    }
}
```

我们首先通过 `get_vm_area_size()` 确定要分配的页数，该函数获取不包含保护页（guard page）的区域大小，如列表 3-54所示。

```c
static inline size_t get_vm_area_size(const struct vm_struct *area) {
    if (!(area->flags & VM_NO_GUARD)) {
        /* 返回不包含保护页的实际大小 */
        return area->size - PAGE_SIZE;
    }

}
```
