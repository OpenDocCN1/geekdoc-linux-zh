
每次采样都会收集有关函数调用和计时的信息。要收集逐行的使用情况，还要指定调试标记 `-g`。要统计基本块的执行次数（例如 do 循环的迭代次数），请使用 `-a`。

第二步是运行程序。在程序运行时，分析数据会被收集到一个名为 `gmon.out` 的文件中，但仅针对已执行的代码部分。你可能只想对特定部分进行分析。程序必须正常终止，分析文件才会被写入。

## A.3.4 gprof 如何收集数据

当可执行文件运行时，每当调用函数时，其计数就会增加。此外，`gprof` 会定期中断可执行文件以确定当前执行的函数。这些采样决定了执行时间。由于 Linux 的时钟中断间隔为 0.01 秒，这些中断最多发生 0.01 秒。因此，对于快速执行的程序或不频繁执行函数的分析是不准确的。为了避免这些不准确性，请长时间运行可执行文件，或汇总多次执行的数据。在 `gprof` 页面中阅读关于 `-s` 选项以汇总分析数据。

int main () {
  char command_line[1000];
  char* command_to_parse;
  char* token;
  Stack number_stack = create_stack();
  while (1) {
    printf("Please enter a postfix expression:\n");
    command_to_parse = fgets(command_line, sizeof(command_line), stdin);
    if (command_to_parse == NULL)
      return 0;
    token = strtok(command_to_parse, " \t\n");
    command_to_parse = NULL;
    while (token != 0) {
      if (isdigit(token[0])) {
        push_stack(&number_stack, string_to_number(token));
      } else if (((strcmp(token, "+") == 0) &&
        apply_binary_function(&add, &number_stack)) ||
        (strcmp(token, "-") == 0) &&
        apply_binary_function(&substract, &number_stack)) ||
        (strcmp(token, "*") == 0) &&
        apply_binary_function(&product, &number_stack)) ||
        (strcmp(token, "even") == 0) &&
        apply_unary_function(&even, &number_stack)) ||
        (strcmp(token, "odd") == 0) &&
        apply_unary_function(&odd, &number_stack))
        return 1;
      }
      token = strtok(command_to_parse, " \t\n");
    }
    if (empty_stack(number_stack))
      return 1;
    else {
      number answer = pop_stack(&number_stack);
      printf("%%u\n", number_to_unsigned_int(answer));
      destroy_number(answer);
      clear_stack(&number_stack);
    }
  }
  return 0;
}

## 列表 A.4 (number.c) 一进制数字实现

```c
/* 对一进制数字进行操作。 */
#include <assert.h>
#include <stdlib.h>
#include <limits.h>
#include "definitions.h"

/* 创建代表零的数字。 */
number make_zero ()
{
    return 0;
}

/* 如果数字代表零，则返回非零。 */
int zerop (number n)
{
    return n == 0;
}

/* 将一个正数减 1。 */
number decrement_number (number n)
{
    number answer;
    assert (!zerop(n));
    answer = n->one_less_;
    free(n);
    return answer;
}

/* 给数字加 1。 */
number add_one (number n)
{
    number answer = malloc(sizeof (struct ListNode));
    answer->one_less_ = n;
    return answer;
}

/* 销毁一个数字。 */
void destroy_number (number n)
{
    while (!zerop (n))
        n = decrement_number (n);
}

/* 拷贝一个数字。此函数仅由于内存分配需要。 */
```

```c
number copy_number(number n)
{
    number answer = make_zero();
    while (!zerop(n)) {
        answer = add_one(answer);
        n = n->one_less;
    }
    return answer;
}
```

/* 两个数字相加。 */

```c
number add(number n1, number n2)
{
    number answer = copy_number(n2);
    addend = n1;
    while (!zerop(addend)) {
        answer = add_one(answer);
        addend = addend->one_less;
    }
    return answer;
}
```

/* 从另一个数字中减去一个。 */

```c
number subtract(number n1, number n2)
{
    number answer = copy_number(n1);
    number subtrahend = n2;
    while (!zerop(subtrahend)) {
        assert (!zerop(answer));
        answer = decrement_number(answer);
        subtrahend = subtrahend->one_less;
    }
    return answer;
}
```

number product(number n1, number n2)
{
    number answer = make_zero();
    number multiplicand = n1;
    while (!zerop(multiplicand)) {
        number answer2 = add(answer, n2);
        destroy_number(answer);
    }
}
