    unsigned char *target_hw;
    u32 target_ip;
    arp = (struct arphdr *)skb->data;
    if (skb->len < sizeof(struct arphdr) + 2*(dev->addr_len+4))
        goto out;
    arp_ptr = (unsigned char *)(arp+1);
