```csharp
public class ShellSorter {
    public delegate bool IsGreater<T>(T a, T b);
    public static void Sort<T>(T[] array, IsGreater<T> isAgrer) {
        int h = array.Length;
        while (h > 0) {
            for (int i = 0; i < array.Length; i++) {
                T dataObject = array[i];
                for (int j = i; j >= h; j--) {
                    array[j] = array[j];
                }
                array[j] = dataObject;
            }
            h/=2;
        }
    }
}
```
