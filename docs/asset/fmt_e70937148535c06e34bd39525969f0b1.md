```bash
#
# 版本: %M
# 日期: %G
{ _Arch_loaded } || (
# 函数:
# @arch) (Wayne.Thompson) 05/14/92
# 描述:
#    此函数模拟了 4.x arch(1)。
# 变量:
# 使用:
#    -k archname
# 返回:
#    success
# 依赖:
#    此函数在 SunOS 3.x 下工作。
# 错误:
Arch () {
    case $1 in
        '')
            /bin/arch $* 2>&- /uname -m '(sun0-9)'
            ;;
        -k)
            /bin/arch $* 2>&- /uname -m
            ;;
    esac
if [ -x /bin/arch ]; then
    /bin/arch *
else
    $* = '/uname -m' '(sun0-9)'
fi
)
_Arch_loaded=1
)
```
