void module_generate (int fd) {
  pid_t child_pid;
  int rval;

  /* 写入页面开头。 */
  write(fd, page_start, strlen(page_start));
  /* 分叉子进程。 */
  child_pid = fork();
  if (child_pid == 0) {
    /* 这是子进程。 */
    /*为调用 df 设置一个参数。 */
    char* argv[] = { "/bin/df", "-h", NULL };
    
    /* 复制 stdout 和 stderr 以将数据发送到客户端套接字。 */
    rval = dup2(fd, STDOUT_FILENO);
    if (rval == -1)
      system_error("dup2");
    rval = dup2(fd, STDERR_FILENO);
    if (rval == -1)
      system_error("dup2");
    
    /* 运行 df 显示挂载文件系统的空闲空间。 */
    execv(argv, argv);
    /* 除非发生错误，否则 exec 调用不会返回。 */
    system_error("exec");
  } else if (child_pid > 0) {
    /* 这是父进程。等待子进程完成。 */
    rval = waitpid(child_pid, NULL, 0);
    if (rval == -1)
      system_error("waitpid");
  } else {
    /* fork 调用失败。 */
    system_error("fork");
  }

  /* 写入页面结尾。 */
  write(fd, page_end, strlen(page_end));
}
