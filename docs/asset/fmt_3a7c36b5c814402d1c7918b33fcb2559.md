
```perl
#!/usr/bin/perl -w
###############################################
#
# Extensible Agent Script Example - Maintenance of the
# /var/log/messages file. It
# provides a way to delete the contents of the file to zero bytes
# and also shows the
# the last 50 lines of the file.
# Steve Maxwell
# December '99
#
#
$MESSAGESFILE = "/var/log/messages";
$OID_PATHBASE = "1.3.6.1.4.1.2021.255";
$SET_OID    = "1.3.6.1.4.1.2021.255.51";
$SET_OID_OBJ   = 51;
$DEF_LINES   = 50;
$LOG_NAME     = "agent-log";
$LOG_PATH     = "/etc/snmp/";
$DATE        = "/bin/date '+Date: %m/%d/%y  Time: %H:%M:%S':";
$TAIL        = "/usr/bin/tail -n";
$LOGGING     = 1;
##
##  Creates logfile.  On success, return 0, else 1.
##
sub CreateLogFile {

    $lf = sprintf("%s%s", $LOG_PATH, $LOG_NAME);
    unless (open (LFP, ">$lf")) {
        print "Can't open log file: $lf\n";
        exit 1;
    }
    return 0;
}
##
##  Logs any string passed to it.
##
sub Log {
    if ($LOGGING) {
        print LFP " $DATE  $_[0]\n";
    }
}
##
##  Close open log file and exit program with supplied
##  error code.
##
sub Leave {
    close LFP;
    exit $_[0];
}
##
##  We need to purge the log file, but we need to make
##  sure the correct OID string is provided to us.
##
sub SnmpSet {
    ($oid, $type, $value) = @_;


##  ##  Make sure OID from agent matches are OID set object.
##  ##  if so, then we know to execute the SET operation.
##
if($oid eq $SET_OID) {
        Log "SnmpSet: oid...$oid, type...$type, and value...$value";
    $comm = sprintf("%s %s %s", "/bin/cp", "/dev/null",
        $MESSAGESFILE);
    Log "Executing... $comm";
    $rc = system ($comm);
    $stat = $rc/256;
    if ( $stat ) {
        Log "SnmpSet: Command executes successfully";
        Leave 0;
    }
    }
    Leave 1;
    }
##
## Get the next OID and line from the messages
## file.
##
sub SnmpGetNext {
($Reqoid, $master_oid) = @_;

@data = `$TAIL $DEF_LINES $MESSAGESFILE`;

if ($Reqoid eq $master_oid) {
    Log "SnmpGetNext: OIDs are equal\n";
    print "$Reqoid.0\n";
    print "string\n";
    print "$data[0]";
    Log "SnmpGetNext: $Reqoid.0, string, $data[0]";
    Leave 0;
} else {
$newstring = "";
$oidstring = $Reqoid;
(@oids) = split (/\./, $oidstring);
$count = @oids-1;
for ($i = 1; $i <= $count; $i++) {
    $newstring = sprintf("%s.%s", $newstring,
    $oids[$i]);
}
$lastoid = $oids[$#oids];
$oid_int = sprintf("%i", $lastoid);
$oid_int++;
print "$newstring.$oid_int\n";
print "string\n";
print "$data[$oid_int]";
Log "SnmpGetNext: $newstring.$oid_int, string,
    $data[$oid_int]";
}
Leave 0;
}
##
## Get an individual line of the messages file
##
sub SnmpGet {
$Reqoid = $_[0];

Log "SnmpGet: $Reqoid";
@lines = `$TAIL $DEF_LINES $MESSAGESFILE`;

(@oids) = split (/\./,$Reqoid);

$lastoid = $oids[$#oids];
if ( $lastoid =~ /$SET_OID_OBJ/ ) {
print "$Reqoid\n";
print "integer\n";
print "0";
Leave 0;
}
if ( $lastoid > 0 && $lastoid < 50 ) {
print "$Reqoid\n";
print "string\n";
print "\n$lines[$lastoid]";
Log "SnmpGet: $Reqoid, string, $lines[$lastoid]";
Leave 0;
} else {
Leave 1;
}
}
##
## Main program starts here
##

CreateLogFile;

($arg1, $arg2, $arg3, $arg4) = @ARGV;

if ($arg1 eq "-s") {
SnmpSet $arg2, $arg3, $arg4;
} elsif ($arg1 eq "-n") {
SnmpGetNext $arg2, $OID_PATHBASE;
} else {
SnmpGet $arg2;
}
```
