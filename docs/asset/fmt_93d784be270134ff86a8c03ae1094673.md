#!/bin/bash
{
  output="022"
  perm_mask=$(printf '%o' 077)
  valid_shells="$(sed -rn '^\/.*\///g;p)' /etc/shells | paste -s d ' '
  awk -v pat="$valid_shells" -F: pat '{ print $1 " $(NF) }' /etc/passwd | while read -r user home; do
    for dfile in $(find "$home" -type f -name ".*"); do
      mode=$(stat -c '%a' "$dfile")
      if [ $((mode & perm_mask)) -gt 0 ]; then
        output="$output\n- User \$user file: \$dfile is too permissive (should be \$maxperm)"
      fi
    done
    if [ -n "$output" ]; then
      echo -e "\n- Failed:\n$output"
    else
      echo -e "\n- Passed:\n- All user dot files are mode:\maxperm"
    fi
  done
}

# 修复

在不通知社区的情况下对用户文件进行全局修改可能会导致意外的停机和用户不满意。因此，建议建立一种监控策略来报告用户权限并根据站点策略确定要采取的行动。以下脚本将删除交互式用户主目录内点文件的过度权限。

```bash
#!/usr/bin/env bash
{
perm_masks='022'
valid_shells="$(sed -rn '^\/.*\///g;p)' /etc/shells | paste -s d ' ')"
awk -v pat="$valid_shells" -F ':' $NF ~ pat '{ print $1 " $(NF) }' /etc/passwd | while read -r user home; do
  find "$home" -type f -name ".*" | while read -r dfile; do
    mode=$(stat -c '%a' "$dfile")
    if [ $((mode & perm_masks)) -gt 0 ]; then
      echo "\n- Modifying \$user file: \$dfile\n-removing group-w and other permissions"
      chmod go-w "$dfile"
    fi
  done
done
}
```
