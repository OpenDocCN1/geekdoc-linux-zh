
```c
// 打印一个月天数的程序
#include <stdio.h>

int main (void)
{
    enum month { January = 1, February, March, April, May, June,
                 July, August, September, October, November, December };
    enum month aMonth;
    int days;

    printf ("输入月份编号: ");
    scanf ("%i", &aMonth);

    switch (aMonth) {
    case January:
    case March:
    case May:
    case July:
    case August:
    case October:
    case December:
        days = 31;
        break;
    case April:
    case June:
    case September:
    case November:
        days = 30;
        break;
    case February:
        days = 28;
        break;
    default:
        printf ("无效的月份编号\n");
        days = 0;
        break;
    }

    if ( days != 0 )
        printf ("天数为 %i\n", days);

    if ( aMonth == February )
        printf ("...如果是闰年则为 29 天\n");

    return 0;
}
```

```c
// 用于处理日期的头文件

#include <stdbool.h>

// 枚举类型
enum kMonth { January=1, February, March, April, May, June,
              July, August, September, October, November, December };
enum kDay { Sunday, Monday, Tuesday, Wednesday, Thursday, Friday };

struct date
{
    enum  kMonth month;
    enum  kDay    day;
    int           year;
};

// 日期类型
typedef struct date Date;

// 处理日期的函数
Date  dateUpdate (Date today);
int   numberOfDays (Date d);
bool  isLeapYear (Date d);

// 用于在结构中设置日期的宏
#define setDate(s,mm,dd,yy)  s = (Date) {mm, dd, yy}

// 外部变量引用
extern Date todaysDate;
```

```c
#include "date.h"

// 计算明天日期的函数
Date dateUpdate (Date today)
{
    Date tomorrow;

    if ( today.day != numberOfDays(today) )
        setDate (tomorrow, today.month, today.day + 1, today.year);
    else if ( today.month == December )    // 年底
        setDate (tomorrow, January, 1, today.year + 1);
    else                    // 月底
        setDate (tomorrow, today.month + 1, 1, today.year);
    
    return tomorrow;
}
```
