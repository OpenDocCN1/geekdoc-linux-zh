```bash
if [ -z "${pidfile}" ]; then
    # 通过发现确定 pid
    pidlist=pidofproc "${1}"
    retval="${?}"
```

# 用: `statusproc [-p pidfile] pathname`
#set: 此函数向标准输出打印特定守护进程的状态
# 输入: `-p pidfile`, 使用指定的 pidfile 代替 pidof
# pathname, 指定程序的路径
# 返回值:
# 0 - 状态已打印
# 1 - 输入错误。未指定要检查的守护进程。
################################################################################
