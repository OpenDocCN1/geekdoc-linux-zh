
# 脚本代码

```
clear # 清空用户的屏幕。
print "Comment: This script is $0."
print "PURPOSE: 列出包含用户想要查找字符串的文件。\n"
# 如果用户在命令行上提供了搜索目录 (`$where_to_search`)
# 和要查找的字符串 (`$what_to_find`)，则使用它们；
# 否则，提示用户输入。

if [ $ -eq 2 ]; # 如果命令行上的参数数量为2，则
    where_to_search=$1
    shift # 在 shift 之后，$2 变为 $1
    what_to_find=$2
else

if [ ! -d "$where_to_search" ]; # 如果它不是目录或不可读/可执行，则
    if [ ! -r "$where_to_search" || ! -x "$where_to_search" ]; then
        print "无效目录: \$where_to_search"
        exit
    else
        # 提示用户输入 (`$where_to_search` 和 `$filename`)。
        print "\n输入你想搜索的目录："
        print " -意味着系统。警告！这将花费很长时间！"
        print " -意味着你的当前目录。"
        print "默认起始目录是 /home/larry:"
        print "\n输入名称或按回车使用默认值 /home/larry:"
        read dir_name
        where_to_search=${dir_name:-/home/larry}
    fi
fi

# 删除双引号（八进制 \042）和单引号（八进制 \047）
revised=$(echo "\$where_to_search" | tr -d "[\042,\047]")
where_to_search="$revised"

# 验证它是否为有效目录：
if [[ ! -d "$where_to_search" || ! -x "$where_to_search" ]]; then
    print "无效目录: \$where_to_search"
    exit
fi

print "\n用法示例:"
print "./find_string.ksh # 你将被提示 (1) 起点 (2) '字符串'。"
print "./find_string.ksh /home/larry 'Mary had a little lamb stew'"
print "./find_string.ksh /home/larry 'Mary had a little lamb stew' >tmp/results & # 在后台运行。"

print "\n输入你想查找的单词或字符串。"
print -n"要查找什么"
read -r what_to_find

# 删除双引号（八进制 \042）和单引号（八进制 \047）
revised=$(echo "\$what_to_find" | tr -d "[\042,\047]")
what_to_find="$revised"

if [ -z "$what_to_find" ]; # 如果用户只是按了回车，则退出
    print "\n错误：你必须指定单词或字符串。重试。退出。"
    exit
fi

fi # 结束_if_then_else

date_and_time=$(date +%b%d.%Y) # 示例：May12.T0905 (上午09:05)
print "\n开始时间: $date_and_time\n"

print "\n以下文件包含你的字符串:\n"
find $where_to_search -type f -exec "$what_to_find" {} \; 2>/dev/null

date_and_time=$(date +%b%d.%Y) # 示例：May12.T0905 (上午09:05)
print "\n结束时间: $date_and_time\n"
print "\n搜索结束。脚本名称: $0"

print "\n完成。脚本: $0\n"
```
