
来撤销。对于叶子函数，不需要执行标准的 2 条指令序言，也不需要 `leave` 指令。虽然通常可以省略它们，但这会防止 `gdb` 能够反向追溯栈帧。

当栈帧中存在局部变量时，使用名称而不是在 `rsp` 上增加 8 或 16 来访问这些变量更为合理。这可以使用 `yasm` 的 `equ` 伪操作来实现。以下代码为两个局部变量设置了 0 和 8 的符号名称。

`x` `equ` `0`
`y` `equ` `8`

现在我们可以在函数调用之前轻松地在 `x` 和 `y` 中保存两个寄存器

```asm
mov [rsp+x], r8
mov [rsp+y], r9
```

在任何函数协议中，你必须指定哪些寄存器必须在函数中保留。对于 `System V ABI`（Linux 和 OSX），必须保留寄存器 `rbx`、`rbp` 和 `r12-15`，而 Windows 调用约定要求保留寄存器 `rbx`、`rbp`、`rsi`、`rdi` 和 `r12-15`。

下面的程序调用名为 `print_max` 函数来打印作为参数传递的两个 `long` 的最大值。它调用了 `printf`，因此使用 `extern` 伪操作告知 `yasm` 和 `ld` 它将从库加载。

```asm
segment .text

global main

extern printf

; void print_max ( long a, long b )

{

    a equ 0

    b equ 8

print_max:

    push rbp;          ; 普通栈帧

    mov rbp, rsp

    sub rsp, 32        ; 为 a, b 和 max 留空间

    max equ 16

    mov [rsp+a], rdi   ; 保存 a

    mov [rsp+b], rsi   ; 保存 b

    ; max = a;

    mov [rsp+max], rdi

    if (b > max) max = b;

    cmp rsi, rdi

    jng skip

    mov [rsp+max], rsi

skip:

    printf("%d,%d", a, b, max);

}

segment .data

    db fmt "max(%d,%d) = %d", 0xa0

segment .text

    fmt:

    lea rdi, [fmt]

    mov rsi, [rsp+a]

    mov rdx, [rsp+b]

    mov rcx, [rsp+max]

    xor eax, eax      ; 0 点参数

    call printf

}

    leave

    ret

main:

    push rbp;

    mov rbp, rsp;

    print_max(100, 200);

    mov rdi, 100      ; 第一个参数

    mov rsi, 200      ; 第二个参数

    call print_max;

    xor eax, eax      ; 返回 0

    leave

    ret
```

## 9.6 递归

计算机编程中的基本技术之一是递归。调用自身的函数被称为递归函数。递归函数必须有一个边界情况以防止无限递归。

非负整数 $n$ 的阶乘是所有小于等于 $n$ 的正整数的乘积。例如，$5! = 5 \times 4 \times 3 \times 2 \times 1 = 120$。边界情况是 $0! = 1$。我们可以递归地定义阶乘函数：

```c
int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n-1);
}
```

以下是对应的汇编代码：

```asm
segment .text

global main

extern printf

main:

    push rbp

    mov rbp, rsp

    mov rdi, 5          ; n = 5

    call factorial

    mov rdi, rax       ; 结果

    lea rsi, [fmt_res]

    mov rdx, 0

    call printf

    xor eax, eax

    leave

    ret

fact:

    push rbp

    mov rbp, rsp

    sub rsp, 16

    cmp rdi, 1

    jle base

    mov eax, 1

    leave

    ret

    mov [rsp+8], rdi

    dec rdi

    call fact

    mov rdi, [rsp+8]

    imul rax, rdi

    leave

    ret

base:

    mov eax, 1

    leave

    ret

segment .data

    fmt_res db "Fact(%d) = %d", 0
```

## 10 数组
