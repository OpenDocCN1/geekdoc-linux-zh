      [String]
    $HostPrefix

      ValueFromPipelineByPropertyName=$true)

    $NetworkAssociation
    [parameter (ValueFromPipelineByPropertyName=$true)]
[String]

[parameter(ValueFromPipelineByPropertyName=$true)]
[String]
$IPv4Netmask
[parameter(ValueFromPipelineByPropertyName=$true)]

    }
    'IPv6IpPoolEnabled' {
      $pool.Ipv6Config.PoolEnabled = $IPv6IpPoolEnabled
    }
  }
  $IPPoolManager.UpdatePool($Datacenter.Id, $pool)
}
