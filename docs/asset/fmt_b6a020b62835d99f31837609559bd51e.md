## Win32 示例：用标准输入输出的管道

### 生产者

```c
#include <windows.h>
#include <stdio.h>
#include <io.h>
#include <fcntl.h>
#include <process.h>
#include <string.h>
int main() {
    DWORD data_processed;
    HANDLE file_pipes[2];
    const char *some_data[] = "123\n";
    DWORD tstat;
    SECURITY_ATTRIBUTES sa;
    STARTUPINFO si;
    PROCESS INFORMATION pi;
    sa.length = sizeof(SECURITY_ATTRIBUTES);
    sa.lpSecurityDescriptor = NULL; // 默认描述符
    sa.bInheritHandle = TRUE; // 可继承
    // 创建管道
    if (CreatePipe(&file_pipes[0], &file_pipes[1], &sa, &0)) {
        // 创建写端的副本以便保留原始的
        if (!DuplicateHandle(GetCurrentProcess(),
            file_pipes[0],
            GetCurrentProcess(),
            NULL,
            // 不创建新句柄
            0,
            FALSE,
            DUPLICATE_SAME_ACCESS)) {
            CloseHandle(file_pipes[0]);
            CloseHandle(file_pipes[1]);
            exit(EXIT_FAILURE);
        }
    }
    // 现在为 CreateProcess 填充启动信息
    GetStartupInfo(&si);
    si.dwFlags = START_USEHANDLES;
    si.hStdInput = file_pipes[0];
    si.hStdOutput = file_pipes[0];
    si.hStdError = GetStdHandle(STD_ERROR_HANDLE);
    // 启动进程
    if (CreateProcess(
        "C:\\Users\\Debug\\Consumer.exe", // 命令行
        NULL,    // 参数
        NULL,    //进程安全
        NULL,    // 线程安全
        TRUE,    // 继承句柄
        0,       // 创建标志
        NULL,    // 环境块
        NULL,    // 当前目录
        &si,     // 启动信息
        &pi))   // 进程信息 (out)
    {
        CloseHandle(file_pipes[1]);
        exit(EXIT_FAILURE);
    } else {
        WriteFile(file_pipes[1], some_data, strlen(some_data), &data_processed, NULL);
        printf("%d - wrote %d bytes\n", getpid(), data_processed);
        CloseHandle(file_pipes[1]);
        // 等待启动的程序完成
        WaitForSingleObject(pi.hProcess, INFINITE);
        // 获取进程退出码
        GetExitProcess(pi.hProcess, &tstat);
        CloseHandle(pi.hThread);
        CloseHandle(pi.hProcess);
        if (tstat & 0x0) {
            printf("Consumer failed\n");
        }
    }
    exit(EXIT_SUCCESS);
}
```

# 附录 9.7：等待启动进程

以下显示了如何将等待子进程的 UNIX 代码迁移到 Win32。

## UNIX 示例：等待启动进程

```c
#include <unistd.h>
#include <stdio.h>
#include <sys/types.h>
#include <sys/wait.h>
int main() {
    pid_t pid;
    int tstat;
    printf("Running ps with fork and execclp\n");
    pid = fork();
    switch(pid) {
        case -1:
            perror("fork failed");
            exit(1);
        case 0:
            if (execclp("ps", NULL) < 0) {
                perror("execclp failed");
                exit(1);
            }
            break;
        default:
            break;
    }
    waitpid(pid, &tstat, 0);
    printf("Child process %d terminated with code %d.\n", pid, tstat);
    exit(0);
}
```

### CreateProcess

```c

int main() {
```
