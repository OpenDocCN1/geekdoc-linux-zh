#include <stdio.h>
#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <signal.h>
#include <string.h>

#define READ_BUFFER 500

int main(int argc, char *argv[]) {
    int sockfd, newsfd, port, len, childpid;
    char buffer[READ_BUFFER];
    struct sockaddr_in serv_addr, cli_addr;
    int n;
    signal(SIGCHLD, SIG_IGN);
    if (argc < 2) {
        perror("ERROR, no port provided");
        exit(1);
    }

    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0)
        perror("ERROR opening socket");
    
    port = atoi(argv[1]);
    serv_addr.sin_family = AF_INET;
    serv_addr.sin_addr.s_addr = INADDR_ANY;
    serv_addr.sin_port = htons(port);
    
    if (bind(sockfd, (struct sockaddr *)&serv_addr, sizeof(serv_addr)) < 0)
        perror("ERROR binding");
        exit(2);

    listen(sockfd, 5);
    while (1) {
        len = sizeof(cli_addr);
        newsfd = accept(sockfd, (struct sockaddr *)&cli_addr, (socklen_t *)&len);
        if (newsfd < 0)
            perror("ERROR on accept");
        if ((childpid = fork()) < 0) {
            perror("error: fork error");
            exit(3);
        } else if (childpid == 0) {
            close(sockfd);
            memset(buffer, 0, READ_BUFFER);
            n = read(newsfd, buffer, READ_BUFFER);
            if (n < 0)
                perror("ERROR reading from socket");
            n = write(newsfd, buffer, n);
            if (n < 0)
                perror("ERROR writing to socket");
        }
    }
    close(newsfd);
}

### 1.7.2 客户端应用程序

客户端程序必须确定要联系的服务器。要联系服务器，它必须知道关于服务器的两点：

- 服务器监听的端口号
- 运行此服务器的宿机的 IP 地址
