print "$num_arguments_on_command_line"
while [ $num_arguments_on_command_line -gt 0 ]; do
((counter++)) # 计数器加 1。
argument=$1
shift # shift 从参数列表中移除一个参数。
print "参数：$argument"
# 从“命令行参数数量”中减去 1：
((num_arguments_on_command_line--))
done
