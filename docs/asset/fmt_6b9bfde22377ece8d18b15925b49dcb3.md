
```c
#include <R.h>

// 参数：
// m: 矩阵
// n: 行数
// k: 对角线索引
void subdiag(double *m, int *n, int *k, double *result) {
  int val = *n, kval = *k;
  int stride = val + 1;
  for (int i = 0, j = kval; i < n; ++i, j+= stride)
    result[i] = m[j];
}
```
