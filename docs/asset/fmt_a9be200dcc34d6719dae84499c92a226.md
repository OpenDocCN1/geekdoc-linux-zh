```python
from functools import wraps

def trace(message):

    def decorate(func):

        @wraps(func)

        def wrapper(*args, **kwargs):

            print(message.format(func=func))

            return func(*args, **kwargs)

        return wrapper

    return decorate

```

```python

def reduce(func, items, initial):

    result = initial

    for item in items:

        result = func(result, item)

    return result

```

```python

def countdown(start):

    n = start

    while n > 0:

        yield n

        n -= 1

```

```python

def up_and_down(n):

    yield from countup(n)

    yield from countdown(n)

```
