void urltranslate(char *out_line, const char *in_line)
{
    int in_length, loop_index, out_index;
    in_length = strlen(in_line);
    for (loop_index = 0, out_index = 0; loop_index < in_length; loop_index++)
    {
        if (in_line[loop_index] == '%') /* 需要翻译
        {
            /* 如果你的系统使用符号字符，使用 strtol() */
            /* 你可能需要对字符进行有效性检查 */
            out_line[out_index] = strtoul(in_line+loop_index + 1, NULL, 16);
            out_index++;
            loop_index += 2; /* 跳过十六进制值 */
        }
        else if (in_line[loop_index] == '+') /* 转换为空格
        {
            out_line[out_index] = ' ';
            out_index++;
        }
        else if (in_line[loop_index] == '&') /* 转换为换行
        {
            out_line[out_index] = '\n';
            out_index++;
        }
        else /* 直接复制 */
        {
            out_line[out_index] = in_line[loop_index];
            out_index++;
        }
    }
    out_line[out_index] = '\0'; /* 终止字符串 */
}
