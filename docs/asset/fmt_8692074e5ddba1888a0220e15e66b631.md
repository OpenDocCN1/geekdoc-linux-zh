记住 `popen` 第一个参数的一种方法是记住：与 `fopen` 一样，如果类型为 `"r"`，则返回的文件指针是可读的；如果类型为 `"w"`，则该指针是可写的。`pclose` 函数关闭标准 I/O 流，等待命令终止，并返回 `shell` 的退出状态。（我们已经在第 8.6 节中描述了退出状态。第 8.13 节中描述的系统函数也返回退出状态。）如果无法执行 `shell`，`pclose` 返回的退出状态将如同 `shell` 执行了 `exit(127)`。

```sh
sh -c cmdstring
```

这意味着 `shell` 会展开 `cmdstring` 中的任何特殊字符。这允许我们说例如：

- `fp = popen("ls *.", "r")`

或者

- `fp = popen("cmd 2>&1", "r")`

## 示例 — `popen` 和 `pclose` 函数

图 15.12 显示了我们版本的 `popen` 和 `pclose`。

```c
#include "apue.h"
#include <errno.h>
#include <fcntl.h>
#include <sys/wait.h>

/*
* 指向运行时分配的数组的指针。
 */
static pid_t *childpid = NULL;

/*
* 来自我们的 open_max()，图 2.17。
 */
static int    maxfd;

FILE *
open(const char *cmdstring, const char *type) {
    int i;
    int pfd[2];
    pid_t pid;
    FILE *fp;

    /* 仅允许 type[0] = 'r' 或 "w" */
    if ((type[0] == 'r' && type[0] != 'w') || type[1] != 0) {
        errno = EINVAL;
        return(NULL);
    }

if (childpid == NULL) { /* 第一次通过 */
    /* 为子进程 pid 分配零值数组 */
    maxfd = open_max();
    if ((childpid = calloc(maxfd, sizeof(pid_t))) == NULL) {
        return(NULL);
    }
}

if (pipe(pfd) < 0)
    return(NULL); /* errno 由 pipe() 设置 */
    if (pfd[0] >= maxfd || pfd[1] >= maxfd) {
        close(pfd[0]);
        close(pfd[1]);
        errno = EFILE;
        return(NULL);
    }

if ((pid == fork()) < 0)
    return(NULL); /* errno 由 fork() 设置 */
} else if (pid == 0) {
    if (*type == 'r') {
        close(pfd[0]);
        if (pfd[1] != STDOUT_FILENO) {
            dup2(pfd[1], STDOUT_FILENO);
        }
        close(pfd[1]);
    }

} else {
    close(pfd[1]);
    if (pfd[0] != STDIN_FILENO) {
        dup2(pfd[0], STDIN_FILENO);
        close(pfd[0]);
    }
}

/* 关闭 childpid[] 中的所有描述符 */
for (i = 0; i < maxfd; i++)
    if (childpid[i] > 0)
        close(childpid[i]);

execcl("/bin/sh", "sh", "-c", cmdstring, NULL);

/* 关闭 childpid[] 中的所有描述符 */
for (i = 0; i < maxfd; i++)
    if (childpid[i] > 0)
        close(childpid[i]);

execcl("/bin/sh", "sh", "-c", cmdstring, NULL);

/* 关闭 childpid[] 中的所有描述符 */
for (i = 0; i < maxfd; i++)
    if (childpid[i] > 0)
        close(childpid[i]);

execcl("/bin/sh", "sh", "-c", cmdstring, NULL);

int
pclose(FILE *fp) {
    int fd, stat;
    pid_t pid;

    if (childpid == NULL) {
        errno = EINVAL;
        return(-1);
    }
    fd = fileno(fp);
    if (fd >= maxfd) {
        errno = EINVAL;
        return(-1);
    }
    if (childpid[fd] == 0) {
        errno = EINVAL;
        return(-1);
    }
    fclose(fp);
    pid = childpid[fd];
    childpid[fd] = 0;

    if (waitpid(pid, &stat, 0) < 0) {
        if (errno != EINTR) {
            return(-1);
        }
    }
    return(stat);
}
```

```c
#include "apue.h"
#include <ctype.h>

int
main(void)
{
    int c;

    while ((c = getchar()) != EOF) {
        if (isupper(c))
            c = tolower(c);
        if (putchar(c) == EOF)
            err_sys("output error");
        if (c == '\n')
            fflush(stdout);
    }
    exit(0);
}
```

```c
#include "apue.h"
#include <sys/wait.h>

int
main(void)
{
    char line[MAXLINE];
    FILE *fpin;

    if ((fpin = fopen("myuclc", "r")) == NULL)
        err_sys("popen error");
    for ( ; ) {
        fputs("prompt", stdout);
        fflush(stdout);
        if (fgets(line, MAXLINE, fpin) == NULL) /* 从管道读取 */
            break;
        if (fputs(line, stdout) == EOF)
            err_sys("fputs error to pipe");

    if (pclose(fpin) == -1)
        err_sys("pclose error");

    exit(0);
}
```

## 15.4 协进程

`UNIX` 系统过滤器是一个从标准输入读取并向标准输出写入的程序。过滤器通常在 `shell` 管道中线性地连接。当同一个程序既生成过滤器的输入又读取过滤器的输出时，该过滤器就变成了协进程。

`shell` 提供了协进程 [Borky and Korn 1995]。`Bourne` shell、`Bourne`-again `shell` (`bash`) 和 `C` shell 并不提供将进程作为协进程的方式连接在一起的方法。协进程通常在 `shell` 的后台运行，其标准输入和输出通过管道连接到另一个程序。尽管启动协进程并将其输入和输出连接到其他进程的 `shell` 语法相当复杂（细节请参阅 `Borky` and `Korn` 1995 第 62–63 页），但协进程对于协进程本身也是有用的。

`pop` 使用单向管道连接到另一个进程的标准输入或从其标准输出读取，而在协进程中，我们有两个指向另一个进程的单向管道：一个指向它的标准输入，另一个从它的标准输出出来，让它处理数据，然后从它的标准输出读取。

**图 15.16** 通过向其标准输入写入和读取其标准输出来驱动协进程

图 15.17 中的程序是一个简单的协进程，它从其标准输入读取两个数字，并将和写入到其标准输出。（协进程通常的工作比我们在这里演示的要有趣得多。这个例子诚然是人为设计的，以便我们研究连接进程所需的管道结构。）

```c
#include "apue.h"

int
main(void) {
    int n, int1, int2;
    char line[MAXLINE];
    while ((n = read(STDIN_FILENO, line, MAXLINE)) > 0) {
        line[n] = 0; /* 空字符终止 */
        if (sscanf(line, "%d%d", &int1, &int2) == 2) {
            sprintf(line, "%d\n", int1 + int2);
            n = strlen(line);
            if (write(STDOUT_FILENO, line, n) != n)
                err_sys("write error");
        } else {
        if (write(STDOUT_FILENO, "invalid args\n", 13) != 13)
            err_sys("write error");
    }
    exit(0);
}
```
