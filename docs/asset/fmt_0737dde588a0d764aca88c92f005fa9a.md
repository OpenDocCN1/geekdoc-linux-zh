## 初始化信息

### 脚本功能
该部分涉及硬件时钟的初始化配置。

```bash
./lib/lsb/init-functions

[ -r /etc/sysconfig/clock ] && . /etc/sysconfig/clock

case "$(UTC)" in
yes|true|1)
    CLOCKPARAMS="$(CLOCKPARAMS) --utc"
    ;;
no|false|0)
    CLOCKPARAMS="$(CLOCKPARAMS) --localtime"
    ;;
esac

case $(1) in
start)
    hwclock --hctys $(CLOCKPARAMS) >/dev/null
    ;;
stop)
    log_info_msg "正在设置硬件时钟..."
    hwclock --systohc $(CLOCKPARAMS) >/dev/null
    evaluate_retval
    ;;
*)
    echo "用法: ${0} (start|stop)"
    exit 1
    ;;
esac

exit 0
```

## D.8. /etc/rc.d/init.d/checkfs

```bash
#!/bin/sh
################################################################################
# 开始 checkfs
#
# Description : 文件系统检查
# Authors : Gerard Beekmans - gerard@linuxfromscratch.com
#     A. Luecke - luebeke@users.sourceforge.net
#     DJ Lucas - dj@linuxfromscratch.org
# Update：Bruce Dubbs - bdubbs@linuxfromscratch.org
#
# 版本：LFS 7.0
# 基于 LFS-3.1 及更早版本的脚本
#
#  自 man fsck
# 0 - 无错误
# 1 - 已修复文件系统错误
# 2 - 系统应重启
# 4 - 文件系统错误未修复
# 8 - 运行错误
# 16 - 使用方法或语法错误
# 32 - 用户请求取消了操作
# 128 - 共享库错误
################################################################################

### BEGIN INIT INFO
# Provides: checkfs
# Required-Start: udev swap $time
# Should-Start:
# Required-Stop:
# Should-Stop:
# Default-Start: S
# Default-Stop:
# Short-Description: 在挂载前检查本地文件系统。
# Description: 在挂载前检查本地文件系统。
# X-LFS-Provided-By: LFS
### END INIT INFO

/lib/lsb/init-functions

case "${1}" in
start)
    if [ -f /fastboot ]; then
        msg="发现 fastboot，将按照要求跳过系统检查。\n"
        log_info_msg "${msg}"
        exit 0
    fi

    log_info_msg "正在以只读模式挂载根系统..."
    mount -n -o remount,ro / /dev/null

    if [ $? -ne 0 ]; then
        log_failure_msg
        msg="\无法检查根目录，因为无法以只读模式挂载。\n\n"
        msg="在按下回车键后，系统将停止并关机。\n\n"
        log_failure_msg "${msg}"
        log_info_msg="按回车键继续..."
        wait_for_user
        /etc/rc.d/init.d/halt stop
    else
        if [ -f /force ]; then
            msg="发现 force，将按照要求强制系统检查。\n"
            log_success_msg "${msg}"
            options="-f"
        else
            options=""
        fi

        log_info_msg "正在检查系统..."
        # 注意：-a 选项曾用于 -p；但这在 fsck 等情况下会失败。
        if [ "$VERBOSE" = true ]; then
            fsck ${options} -a -C -T
        else
            fsck ${options} -a -C -T >/dev/null
        fi

        error_value=$?

        if [ "${error_value}" -eq 0 ]; then
            log_success_msg
        fi

        if [ "${error_value}" -eq 1 ]; then
            msg="\发现错误并已修复，你可能需要检查文件系统。"
            log_warning_msg
        fi

        if [ "${error_value}" -eq 2 ] || [ "${error_value}" -eq 3 ]; then
            msg="\发现错误并已修复，但系统需要重启。\n"
            msg="在按下回车键后，系统将重启。\n"
            log_failure_msg "${msg}"
            log_info_msg="按回车键继续..."
            wait_for_user
            reboot -f
        fi

        if [ "${error_value}" -eq 4 ]; then
            msg="\发现错误但未修复。\n"
            msg="你需要手动修复这些错误。\n\n"
            msg="在按下回车键后，系统将停止并关机。\n\n"
            log_failure_msg "${msg}"
            log_info_msg="按回车键继续..."
            wait_for_user
            /etc/rc.d/init.d/halt stop
        fi

        if [ "${error_value}" -eq 8 ] || [ "${error_value}" -eq 16 ]; then
            msg="\运行错误或语法错误。错误代码：${error_value}。\n"
            log_failure_msg $msg
            exit ${error_value}
        fi

        if [ "${error_value}" -eq 128 ]; then
            msg="\发现共享库错误。错误代码：${error_value}。\n"
            log_failure_msg $msg
            exit ${error_value}
        fi

        if [ "${error_value}" -eq 32 ]; then
            msg="\用户请求取消了操作。错误代码：${error_value}。\n"
            log_failure_msg $msg
            exit ${error_value}
        fi

        if [ "${error_value}" -gt 0 ]; then
            msg="\发生意外错误。错误代码：${error_value}。\n"
            log_failure_msg $msg
            exit ${error_value}
        fi

        log_success_msg "检查完成。"
        exit 0
    fi
    ;;
esac
```
