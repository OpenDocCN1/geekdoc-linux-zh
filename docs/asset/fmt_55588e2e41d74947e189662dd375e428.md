function AIX_SYSV_printing
(
  LOOP=0 # 循环计数：每次获取三行
  lpc status all | egrep '|printing|queueing' | while read LINE
  do
    # 每次加载三不同的行
    case $LINE in
      *) Q=$(echo $LINE | cut -d':' -f-1)
          ;;
      printing*)
        PSTATUS=$(echo $LINE | awk '{print $3}')
          ;;
      queueing*)
        QSTATUS=$(echo $LINE | awk '{print $3}')
          ;;
    esac

    ## 增加 LOOP 计数器
    (( LOOP = LOOP + 1 ))

    if ((LOOP == 3)) # 是否获取了所有三行数据？
    then
        # 检查 PSTATUS
        case $PSTATUS in
          disabled) 
            lpc start $Q >/dev/null
            (( $? == 0)) && echo "\nQ 打印已重启\n"
            ;;
          enabled*) # 无操作
            ;;
        esac

        # 检查队列状态
        case $QSTATUS in
          disabled) 
            lpc enable $Q >/dev/null
            (( $? == 0)) && echo "\nQ 队列已重新启用\n"
            ;;
          enabled*) # 无操作
            ;;
        esac

        LOOP=0 # 将循环计数器重置为零
    fi
  done
)
