```c
        /* 父进程：服务器 */
        if (listen(sockfd, 1) < 0) {
            perror("listen");
            exit(1);
        }
        printf("Server listening on port %s...\n", argv[1]);
        socklen_t len = sizeof(addr);
        int client_fd = accept(sockfd, (struct sockaddr *)&addr, &len);
        if (client_fd < 0) {
            perror("accept");
            exit(1);
        }
        int n = read(client_fd, buffer, sizeof(buffer) - 1);
        if (n > 0) {
            buffer[n] = '\0';
            printf("Received: %s\n", buffer);
        }

        close(sockfd);
```

```c
#include    <signal.h>
#include    <fcntl.h>
#define BUFFERSIZE 4096
int     sigflag; /* 必须是全局变量！ */
int sigio_handler()
{
    sigflag = 1;      /* 只设置标志并返回 */
}
int main()
{
    int     n, fret;
    char    buff[BUFFERSIZE];
    signal(SIGIO, sigio_handler);
    fret=fcntl(0, F_SETOWN, getpid());
    check(fret, "F_SETOWN error");
    fret=fcntl(0, F_SETFL, FASYNC);
    check(fret, "F_SETFL FASYNC error");
/* 这将异步变成了一个在 sigpause ... 中阻塞的循环 ... */
    for (;;) {
        sigblock(sigmask(SIGIO));
        while (sigflag == 0)
            pause(0);      /* 等待信号 */
        /*
        * 如果 sigflag != 0，我们就在这里。此外，我们知道
        * SIGIO 信号当前处于阻塞状态。
        */
        if ((n = read(0, buff, BUFFERSIZE)) > 0) {
            if (write(1, buff, n) != n)
                check(-1, "write error");
        } else if (n < 0)
            check(fret, "read error");
        } else if (n == 0)
            exit(0);      /* EOF */
        sigflag = 0;      /* 关闭我们的标志 */
        sigsetmask(0);    /* 并重新启用信号 */
    }
}
```
