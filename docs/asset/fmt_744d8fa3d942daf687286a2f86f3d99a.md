```c
#include <unistd.h>
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <sys/io.h>

#define MSR 0x3FE // 调制解调器状态寄存器 = base + 6, bit 4 = CTS
    // 用于 /dev/ttyS0

char bRead;
void *counter(void);

main() {
    pthread_t thread1;
    int tHandle;

    iopl(3);
    // 创建独立线程
    tHandle = pthread_create(&thread1, NULL, counter, NULL);

    // 在主程序继续执行之前等待线程完成。除非
    // 等待，否则我们将面临执行 exit 的风险，这将在线程
    // 完成之前终止进程和所有线程。
    pthread_join(thread1, NULL);
    exit(0);
}

void *counter(void) {
    char tmp = 0x0;
    while (1) {
        do {
            bRead = inb(MSR) & 0x10;
        } while (tmp == bRead);
        tmp = bRead;
        printf("CTS = 0x%X\n", bRead);
    }
}
```
