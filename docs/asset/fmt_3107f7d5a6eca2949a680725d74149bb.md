# 将状态返回给 `/etc/init.d/inetinit`

在某些情况下你必须修改此文件。例如，要在启动时安装静态路由，你需要编辑 `rc.init` 文件以包含支持这些路由（包括默认网关）所需的路由命令。此外，你可能需要更改与 `named` 等守护相关的配置文件路径规范。

```sh
#!/bin/sh

# 确保可以找到启动此阶段

# 所需的库文件。

LD_LIBRARY_PATH=/etc/lib; export LD_LIBRARY_PATH

#

# 配置软件环回驱动程序。早期完成的网络初始化

#是为了抑制无盘无数据配置。

# /sbin/ifconfig 127.0.0.1 2>/dev/null

# 对于由内核配置的接口（例如无盘机器上的接口），

# 使用本地 `/etc/netmasks` 文件重置子掩码，如果

# 该文件存在的话。

# /sbin/ifconfig -au netmask + broadcast + 2>/dev/null

# 获取需要配置的网络接口列表，通过将

# /etc/hostname.* 按照冒号 ":" 作为 shell 分隔符 拆分为独立的参数，

# 然后遍历参数并对每隔参数执行 ifconfig。

# 在此过程中使用本地 `/etc/netmasks` 文件设置子掩码。

# 这也为接口设置了管道结构。

# 如果 /etc/hostname 文件为空，则仅设置流管道，

# 允许 ifconfig auto-reverse 命令尝试设置地址。

interface_names=`echo /etc/hostname.*[0-9]` 2>/dev/null

if test "$interface_names" != "/etc/hostname*" ; then

	echo "configuring network interfaces:"

	IFS="$IFS"

	set echo `echo /etc/hostname.*.*[0-9]`

	while test $# -ge 2; do

		shift

		if [ "$1" != "xx0" ]; then

			addr=$(cat /etc/hostname.$1)

			/sbin/ifconfig $1 up

			if test -n "$addr" ; then

				/sbin/ifconfig $1 inet $addr netmask + \

					broadcast + up \

					2>&1 > /dev/null

				echo "$1"

			fi

		fi

		shift

	done

	echo "."

fi

# 自动静静地配置其余接口。

/sbin/ifconfig -ad auto-reverse netmask + up \

2>&1 >/dev/null
```

    # 为了支持与远程子网服务器上的
    # "/usr" 文件系统 NFS 进行操作，
    # 我们必须小心替换默认路由器。
    # 我们希望 "/etc/defaultrouter" 文件中列出的

    pass=1

      if [ $pass -eq 1 ]; then
        /usr/sbin/route -f add $router 1
      else
        /usr/sbin/route add $router 2
      fi
    done
  fi

```bash
# 只有在我们还没有安装默认路由时，才运行 routed/router 发现。
if [ -z "$defrouters" ]; then
  # 上面的 "route" 命令没有设置默认路由 - 检查
  # 内核路由表中是否有其他默认路由。
  defrouters=$(netstat -rn | grep default)
  if [ -z "$defrouters" ]; then
    # 确定活动接口的数量以及点对点
    # 接口的数量。如果接口超过 2 个
    #（包括环回接口）或有一个或多个点对点
    # 接口，则作为路由器运行。如果 /etc/gateways 存在，也作为路由器运行。
    # 如果 /etc/nteroute 存在，则不要作为路由器。
    numifs=$(ifconfig -au | grep inet | wc -l)
    numptifs=$(ifconfig -au | grep inet | egrep -e '--' | wc -l)
    if [ ! -f /etc/nteroute ] && { [ $numifs -gt 2 ] || [ $numptifs -gt 0 ] || [ -f /etc/gateways ]; }; then
      # 机器是路由器：开启 ip_forwarding，运行 routed，
      # 并使用路由发现将我们宣告为路由器
      echo "machine is a router."
      nd set /dev/ip_forwarding 1
      if [ -f /usr/sbin/in.route ]; then
        /usr/sbin/in.route -s
      fi
      if [ -f /usr/sbin/in.rdisc ]; then
        /usr/sbin/in.rdisc -r
      fi
    else
      # 机器是主机：如果路由发现了一个路由器，
      # 我们依赖路由发现。如果没有路由
      # 通过路由发现来宣告自己，
      # 则在节省空间的模式下运行 routed。
      # 关闭 ip_forwarding
      nd set /dev/ip_forwarding 0
      if [ -f /usr/sbin/in.route ] && /usr/sbin/in.rdisc -s; then
        echo "starting router discovery."
      elif [ -f /usr/sbin/in.route ]; then
        /usr/sbin/in.route -q
        echo "starting router daemon."
      fi
    fi
  fi
fi
```

- `/etc/sysconfig/network-scripts/*`
