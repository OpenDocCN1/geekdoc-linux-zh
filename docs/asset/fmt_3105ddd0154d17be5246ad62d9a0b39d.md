
## D.17. `/etc/rc.d/init.d/sendssignals`

```bash
#!/bin/sh

############################################################
# Begin sendsignal
# Description : Sendssignals 脚本
# Authors : Gerard Beemans - gerard@linuxfromscratch.org
#          DJ Lucas - dj@linuxfromscratch.org
# Update : Bruce Dubbs - dbubbs@linuxfromscratch.org
# Version : LFS 7.0
############################################################################
### BEGIN INIT INFO
# Provides: sendsignal
# Required-Start: 
# Should-Start:
# Required-Stop: $local_fs swap localnet
# Should-Stop:
# Default-Start:
# Default-Stop: 0 6
# Short-Description: 尝试杀死剩余进程。
# Description: 尝试杀死剩余进程。
# X-LFS-Provided-By: LFS
### END INIT INFO

. /lib/lsb/init-functions

case "$1" in

stop)

  log_info_msg "Sending all processes the TERM signal..."
  killall5 -15
  error_value=$?
  sleep ${KILLDELAY}

  if [ "${error_value}" = 0 -o "${error_value}" = 2 ]; then
    log_success_msg
  else
    log_failure_msg
  fi

  log_info_msg "Sending all processes the KILL signal..."
  killall5 -9
  error_value=${KILLDELAY}
  sleep ${KILLDELAY}

  if [ "${error_value}" = 0 -o "${error_value}" = 2 ]; then
    log_success_msg
  else
    log_failure_msg
  fi

;;

*)

    echo "Usage: $0 {stop}"
    exit 1

;;

esac

exit 0
```

## D.20. `/etc/rc.d/init.d/template`

```bash
#!/bin/sh

################################################################
# Begin template
# Description :
# Authors :
# Version : 
################################################################

### BEGIN INIT INFO
# Provides:
# Required-Start:
# Should-Start:
# Required-Stop:
# Should-Stop:
# Description:
# Description:
# X-LFS-Provided-By: LFS
### END INIT INFO

. /lib/lsb/init-functions

case "${1}" in
start)
  log_info_msg "Starting..."
  start_daemon fully_path
  ;;

stop)
  log_info_msg "Stopping..."
  killproc fully_path
  ;;

restart)
  $0 stop
  sleep 1
  $0 start
;;

*)
  echo "Usage: $0 {start|stop|restart}"
  exit 1
;;
esac

exit 0

# End template
```
