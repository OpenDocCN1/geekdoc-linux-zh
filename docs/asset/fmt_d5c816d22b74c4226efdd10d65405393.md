for number in $list_of_numbers; do
if $(echo $number | grep '[^0-9.\-]' > /dev/null 2>/dev/null); # 参见下文说明。
then print "${number} 包含了除了数字、句号（点）或减号以外的 *OTHER THAN* 字符"
elif $(echo $number | grep '[^0-9.\-]+\$' > /dev/null 2>/dev/null); # 如果它以...结尾
then print "${number} 不得以加号结尾。"
elif $(echo $number | grep '-$' > /dev/null 2>/dev/null); # 如果它以...结尾
then print "${number} 不得以减号结尾。"
elif $(echo $number | grep '[^0-9]' > /dev/null 2>/dev/null); # 如果它以句号结尾
then print "${number} 不得以句号结尾"
elif $(echo $number | grep '\.' > /dev/null 2>/dev/null); # 如果它包含
then print "${number} 不得包含连续的句号"
elif $(echo $number | grep '[^0-9]\+' > /dev/null 2>/dev/null); # 如果它包含+
then print "${number} 不得包含减号后跟加号。"
elif $(echo $number | grep '[^0-9]-+' > /dev/null 2>/dev/null); # 如果它包含
then print "${number} 不得包含加号后跟减号。"
elif $(echo $number | grep '[^0-9]+\.[0-9]+\$' > /dev/null 2>/dev/null); # 如果它包含
then print "${number} 不得包含多个句号。"
else print "无效数据：${number}"
fi
done
