# 添加新日历条目

```bash
function addentry {
  $CLEAR
  ENTRY="$DATE"

  # 对于已存在的条目，仅添加更多数据
  COUNT=$(grep -c "^$DATE" $CALFILE)

  if ((COUNT > 0)); then
    changeentry
    return
  fi

  # 提示输入
  print "Enter info for $DATE: (enter <Return> by itself finished)"

  while true
  do
    read LINE
    if [[ -n $LINE ]]; then
      ENTRY="$ENTRY,$LINE"
    else
      break
    fi
  done

  # 追加到日历文件
  print $ENTRY>>$CALFILE

  # 排序日历文件
  sort -o $CALFILE $CALFILE
}
```

function formatentry {
  $CLEAR
  typeset IFS="," \
  BORDER="********************" \
  BORDER1="*"

  FILE=$1

  if [[ -s $FILE ]]; then
    # 打开日历文件读取并
    # 格式化输出
    (exec 0<$FILE
    while read -r ENTRY
    do
      print "$BORDER\n$BORDER1"
      set $ENTRY
      typeset -l35 LINE="DATE: $1"
      print "* $LINE**"
      shift
      print "$BORDER1"
      for i
      do
        LINE="$i"
        print "* $LINE**"
      done
      print "$BORDER"
    done
    ) | $PAGER
  else
    print "No entries found."
  fi

  # 提示继续
  until [[ $REPLY = "" ]]
  do
    read REPLY?"Enter <Return> to continue..."
  done
}

# 附录 D：示例 Korn Shell 脚本

```bash
# 查找特定项目
function findentry {
  $CLEAR
  # 检查条目 -将其放入临时找到文件
  grep $DATE $CALFILE >/tmp/.FOUND$

  # 格式化找到的条目
  formatentry /tmp/.FOUND$
}
```

```bash
# 修改条目
function changeentry {
  # 查找特定条目 -将其放入临时文件
  grep $DATE $CALFILE | tr ', ' '\012' >/tmp/.FOUND$

  # 如果未找到条目，则返回
  if [ ! -s /tmp/.FOUND$ ]
  then
    $CLEAR
    read TMP?"未找到 $DATE 的条目 - 按 <Return> 继续"
    return
  fi

  # 再次提示修改
  while [[ $REPLY != "" ]]
  do
    read REPLY?"是否修改/添加 $DATE 的条目?"
    case $REPLY in
      [yY]*)
        break ;;
      [nN]*)
        print "好的，取消条目修改"
        return ;;
      *)
        print "无效回复 - 请重试。"
    esac
  done

  # 编辑临时找到的文件
  ${EDITOR:-vi} /tmp/.FOUND$$$$
}
```

# 删除特定条目

```bash
function delentry {
  # 查找条目
  grep $DATE $CALFILE >/tmp/.FOUND$$

  # 如果未找到，则返回
  if [[ ! -s /tmp/.FOUND$$ ]]
  then
    CLEAR
    read TMP?"未找到 $DATE 的条目 - 按 <Return> 继续"
    return
  fi

  # 提示删除
  while [[ $REPLY != "" ]]
  do
    read REPLY?"删除 $DATE 的条目吗?"
    case $REPLY in
      [yY]* | "")
        break ;;
      [nN]*)
        print "好的，取消删除"
        return ;;
      * )
        print "无效回复 - 请重试。"
        ;;
    esac
  done

  # 合并更改 -将其放入临时文件
  grep -v $DATE $CALFILE > /tmp/.DEL$

  # 写回新文件
  cat /tmp/.DEL$ > $CALFILE

  # 清理临时文件
  rm -rf /tmp/.DEL$ /tmp/.FOUND$$
}
```
