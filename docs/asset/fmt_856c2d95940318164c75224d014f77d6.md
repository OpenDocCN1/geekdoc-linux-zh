### 示例 11–6. 使用 `let` 进行算术运算。

```bash
#!/bin/bash

echo

let a=11      # 等同于 'a=11'

let a=a+5     # 等同于 let 'a = a + 5'

              # (引号和空格使其更具可读性)

echo 11 + 5   $a

let a <<=3    # 等同于 let 'a = << 3'

echo "$a"    # (16) 左移 3 位 $a

let a /=4     # 等同于 let 'a = a / 4'

echo 128 / 4  $a

let a -=5     # 等同于 let 'a = a - 5'

echo 32 - 5   $a

let a *=10    # 等同于 let 'a = a * 10'

echo 27 * 10  $a

let a %=8      # 等同于 let 'a = a % 8'

echo 270 modulo 8 = $a (270 / 8 = 33, 余数为 $a)

echo

exit 0
```
