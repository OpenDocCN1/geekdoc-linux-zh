```c
        );
        if (hPipe == INVALID_HANDLE_VALUE) {
            fprintf(stderr, "Could not create pipe %s\n", FIFO_NAME);
            exit(EXIT_FAILURE);
        }
    }
```

## 附录 9.10：使用 FIFO 进行进程间通信

以下示例演示了你如何将 UNIX FIFO 进程间通信转换为 Win32 代码。

### UNIX 示例：使用 FIFO 进行进程间通信

提供者 (Provider)

```c
#include <unistd.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <limits.h>
#include <sys/types.h>
#include <sys/stat.h>
```

```c
#define FIFO_NAME "tmp/my_fifo"
#define BUFFER_SIZE PIPE_BUF
#define TEN_MEG (1024 * 1024 * 10)
```

```c
int main() {
    int pipe_fd;
    int res;
    int open_mode = O_WRONLY;
    int bytes_sent = 0;
    char buffer[BUFFER_SIZE + 1];
```

```c
if (access(FIFO_NAME, F_OK) == -1) {
    mkfifo(FIFO_NAME, 07777);
    if (res != 0) {
        fprintf(stderr, "Could not create fifo %s\n", FIFO_NAME);
        exit(EXIT_FAILURE);
    }
}
```

```c
printf("Process %d opening FIFO_WRONLY\n", getpid());
pipe_fd = open(FIFO_NAME, open_mode);
printf("Process %d result %d\n", getpid(), pipe_fd);
```

```c
if (pipe_fd != -1) {
    while (bytes_sent < TEN_MEG) {
        res = write(pipe_fd, buffer, BUFFER_SIZE);
        if (res == -1) {
            fprintf(stderr, "Write error on pipe\n");
            exit(EXIT_FAILURE);
        }
        bytes_sent += res;
    }
    (void)close(pipe_fd);
} else {
    exit(EXIT_FAILURE);
}
```

```c

exit(EXIT_SUCCESS);
```

```c
{
    while (bytes_sent < TEN_MEG) {
        if (WriteFile(
            hPipe,
            // 文件句柄
            buffer,
            // 数据缓冲区
            BUFFER_SIZE,
            // 写入字节数
            &NumberOfBytesWritten, // 写入字节数
            NULL // 重叠缓冲区
        )) {
            fprintf(stderr, "Write error on pipe\n");
            exit(EXIT_FAILURE);
        }
        bytes_sent += NumberOfBytesWritten;
    }
printf("Process %d finished\n", getpid());
} else {
    // 客户端无法连接，因此关闭管道。
    CloseHandle(hPipe);

exit(EXIT_SUCCESS);
```
