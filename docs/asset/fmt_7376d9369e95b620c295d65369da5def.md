
```python
def __init__(self, thelist):
    self.thelist = thelist

def __enter__(self):
    self.workingcopy = list(self.thelist)
    return self.workingcopy

def __exit__(self, type, value, tb):
    if type is None:
        self.thelist[:] = self.workingcopy
    return False
```

    assert n > 0, "必须提供一个正值"
    result = 1
    while n > 1:
        result *= n
        n -= 1
    return result

class FRange:
    def __init__(self, start, stop, step):
        self.start = start
        self.stop = stop
        self.step = step

    def __iter__(self):
        x = self.start
        while x < self.stop:
            yield x
            x += self.step
