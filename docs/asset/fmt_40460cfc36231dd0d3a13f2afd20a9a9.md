- 来自 `swap -s` 命令的输出

```bash
SW_USED=$(swap -s | awk 'print $9' | cut -dk -f1)
SW_FREE=$(swap -s | awk 'print $11' | cut -dk -f1)
```

((SW_TOTAL = SW_USED + SW_FREE))

function Linux_swap_mon
{
free -m | grep -i swap | while read junk SW_TOTAL SW_USED SW_FREE
do
# 使用 here document 中的 bc 工具来计算
# 空闲和已使用交换空间占用的百分比。
PERCENT_USED=$(bc <<EOF
scale=4
($SW_USED / $SW_TOTAL) * 100
EOF)
PERCENT_FREE=$(bc <<EOF
scale=4
($SW_FREE / $SW_TOTAL) * 100
EOF)
# 输出剩余的对页空间（paging space）报告：
echo "\nTotal Amount of Swap Space:\t${SW_TOTAL}MB"
echo "Total KB of Swap Used:\t${SW_USED}MB"
echo "Total KB of Swap Free:\t${SW_FREE}MB"
echo "\nPercent of Swap Space:\t${PERCENT_USED}%"
echo "\nPercent of Swap Free:\t${PERCENT_FREE}%"
# 提取已使用百分比的整数部分，以
# 测试是否超过了限制阈值
INT_PERCENT_USED=$(echo $PERCENT_USED | cut -d. -f1)
if (( PC_LIMIT <= INT_PERCENT_USED ));
then
tput rmso
