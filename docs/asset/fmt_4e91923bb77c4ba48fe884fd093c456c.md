
# 第 7 章：启动过程

> 在 `/etc/rc.d/` 下的所有文件在启动期间都以 `root` 身份运行（在 Debian 系统上，这些将是 `/rc.boot/rc[0-6].d` 以及 `init.d/` 子目录直接 `/etc/`）。被跟随的链接无论是否留在 `/etc/rc.d/` 内，都以 `root` 身份执行程序。任何新脚本或对现有脚本的修改将在下次重启或运行级别运行时运行。初始化期间使用的目录和文件除 `root` 外的人无法写入。

以下是 `start-stop` 脚本的关键脚本的简要概述。这并不是为了教你编写脚本。更多信息请参考关于 shell 脚本的书。查看这些脚本将让你了解编写恶意程序有多容易。第一个关键脚本是 `/etc/rc.d/rc`（见列表 7-3）。行号没有出现在脚本中；它们仅供参考。

true() { return 0; }
false() { return 1; }

Echo() {
    local a=$1; shift
    local o=$1; shift
    local i

    echo -n "$a" > $LOG
    for i in "$@"; do
        echo -n '$i' > $LOG
    done
    echo "$o" > $LOG

if $bootlogger; then

    CMDS="start"

    # 将 STDOUT 和 STDERR 重定向到
    exec - >LOG 2>&1

    # 调试
    Booter() {
        local c=$1; shift
        local s
        local i

        case "$c" in
            add)
                s="$1"; shift
                eval "$s" $c
                ;;
            start)
                s="$1"; shift
                eval "$s" $c "$@"

                    0) $bootlogger ok; ;;
                    1) $bootlogger fail; ;;
                    2) $bootlogger skip; ;;
                    *) $bootlogger "N/A"; ;;
                esac

            stop)
                s="$1"; shift

                eval "$s" $c "$@"
                ;;

    local c="$1"; shift
