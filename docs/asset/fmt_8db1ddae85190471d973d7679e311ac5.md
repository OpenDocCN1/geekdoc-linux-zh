char *XMSBuf; /* 用于传输的常规内存缓冲区 */

char XMS_init()
{
    /*
      如果 XMS 存在，返回 0；
      如果 XMS 不存在，返回 1；
      如果无法分配传输缓冲区，返回 2
    */
    unsigned char status;
    _AX=0x4300;

    geninterrupt(0x2F);
    status = _AL;
    if (status==0x80)
    {
      GetXMSEntry();
      XMSBuf = (char far *) farmalloc(BLOCKSIZE);
      if (XMSBuf == NULL)
        return 2;
      return 0;
    }
    return 1;
}

void GetXMSEntry(void)
{
    /*
        GetXMSEntry 将 XMSFunc 设置为 XMS 管理程序的入口点，
        以便我们稍后可以调用它
    */
    _AX=0x4310;
    geninterrupt(0x2F);
    XMSFunc = (void (far *) (void)) MK_FP(_ES,_BX);
}

void XMSSize(int *kbAvail, int *largestAvail)
{
    /*
        XMSSize 返回可用的总千字节数，
        以及最大可用块的千字节大小
    */
    _AH=8;
    (*XMSFunc)();
    *largestAvail=_DX;
    *kbAvail=_AX;
}

char AllocXMS(unsigned long numberBytes)
{
    /*
        分配一块长度为 numberBytes 的 XMS 内存
    */
    _DX = (int)(numberBytes / 1024);
    _AH = 9;
    (*XMSFunc)();
    if (_AX==0)
    {
        return FALSE;
    }
    XMSHandle=_DX;
    return TRUE;
}

void XMS_free(void)
{
    /*
        释放已使用的 XMS
    */
    _DX=XMSHandle;
    _AH=0x0a;
    (*XMSFunc)();
}
