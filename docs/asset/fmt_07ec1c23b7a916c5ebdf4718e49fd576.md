#include <qapplication.h>
#include <qtabbar.h>

int main(int argc, char **argv) {

  tabbar->addTab("First");
  tabbar->addTab("Second");
  tabbar->addTab("Third");
  tabbar->addTab("Fourth");
  tabbar->show();
  app.setMainWindow(tabbar);
  return(app.exec());
}
