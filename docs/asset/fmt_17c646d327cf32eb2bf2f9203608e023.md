# 示例 27-5. 将脚本加载到数组

```bash
#!/bin/bash

# script-array.sh: 将此脚本加载到数组。

# 灵感来自 `Chris Martin` 的邮件（感谢！）。

script_contents=$(cat "$0")      # 将此脚本的内容 ($0)
                                  # 存储在数组中。

for element in $(seq 0 $((${#script_contents[@]} - 1)))
do
  # ${script_contents[$element]}
  # 给出数组中的元素个数。

  echo -n "$script_contents[$element]"
  # 在单行内输出脚本的每个字段。
  # "$script_contents[$element]" 同样适用于 ${ ... }。
  echo -n "# 使用 " - " " 作为字段分隔符。
done
```

# 示例 27-6. 数组的某些特殊属性

```bash
#!/bin/bash

declare -a colors

# 此脚本中后续的所有命令都将
# 将变量 "colors" 视为数组。

echo "输入你喜爱的颜色（用空格隔开）"
read -a colors # 输入至少 3 种颜色以演示下面的功能。
# 'read' 命令的特殊选项，
# 允许为数组元素进行赋值。

echo
element_count=${#colors[@]}
# 提取数组元素数量的特殊语法。
# element_count=${#colors[*]} 也可以。
# "*" 变量允许在引号内进行单词拆分
# (提取由空白字符分隔的变量)。
# 这对应于位置参数中的
# "$@" 和 "*" 的行为。

index=0
while [ "$index" -lt "$element_count" ]; do
    # 列出数组中的所有元素。
    echo ${colors[$index]} # 同样有效，因为它在 ${ } 括号内。
    let index=$((index + 1))
    # 或者：
    # ((index++))
done

# 每个数组元素单独占一行列出。
# 如果不需要这样做，使用 echo -n "${colors[$index]}" ""
# 使用 "for" 循环来实现：
# for i in "${colors[@]}"; do
#     echo "$i"
# done
# （感谢 S.C.）
```

```bash
# "unset" 命令删除数组的元素或整个数组。

unset colors[1]      # 数组的第 2 个元素。
                     # 效果与 colors[1]= 相同。

echo ${colors[@]}      # 再次列出数组，缺少第 2 个元素。

unset colors                     # 删除数组。
                     # unset colors[*] 和
                     # unset colors[@] 同样有效。

echo -n "Colors are gone."
echo ${colors[@]}      # 再次列出数组，现在为空。

exit 0
```

如上一个示例所示，`${array_name[@]}` 或 `${array_name[*]}` 指数组的所有元素。要获取数组中的元素数量，使用 `${#array_name[@]}` 或 `${#array_name[*]}`。`${#array_name}` 是 `${array_name[0]}`（数组的第一个元素）的长度（字符数）。

`${array_name[@]}` 和 `${array_name[*]}` 之间的关系类似于 `$@` 和 `$*`。这种强大的数组表示法有很多用途。

```bash
# 将文件内容加载到数组。

# 感谢 S.C.。
# ${#...} 操作在命令替换的帮助下，使得将文本文件的内容加载到数组成为可能。

#!/bin/bash

filename=sample_file

cat sample_file

#
# 1 a b c
# 2 d e fg
#

declare -a array

array=`cat "$filename"`   # 将 $filename 的内容
                          # 加载到数组。
                          # 将文件列到标准输出

array=`cat "$filename" | tr '\n' ' '`
# 将文件中的换行符改为空格。
# 没有必要，因为 Bash 会进行单词拆分，
# 将换行符改为空格。

# 复制数组。
array2="${array1[@]}"

# 然而，这对于“稀疏”数组会失效，
# 即包含洞（缺失元素）的数组，
# 如 Jochen DeMeester 指出的。

array1[0]=0
# array1[1] 未赋值
array1[2]=2

array2="${array1[@]}"   # 复制它？
echo ${array2[0]}           # 0
echo ${array2[2]}           # 2
# (null)，应该是 0 2

# 文件中每个由空格分隔的“单词”
# 都已赋值给数组中的一个元素。
element_count=${#array[*]}
echo $element_count
```

```bash

bigThree=("0...")
```
