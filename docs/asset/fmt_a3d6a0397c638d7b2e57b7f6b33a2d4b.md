
## `__rmqueue_fallback()` 函数分析

### 代码片段 1：初始化与标志检查

```c
/*
 * 不要从属于其他页块的空闲列表中偷取页面
 * 即 orders < pageblock_order。如果没有空闲区域，
 * 将在不带 ALL_NOFRAGMENT 的情况下遍历区域列表。
 */

if (alloc_flags & ALLOC_NOFRAGMENT)
    min_order = pageblock_order;
```

### 代码片段 2：查找最大可用页面

```c
/*
 * 在另一个列表中寻找最大的可用页面。这
 * 类似于寻找带有空闲页面的页块，
 * 但那样做的成本太高。
 */

for (current_order = MAX_ORDER; current_order >= min_order; --current_order) {
    area = &(zone->free_area[current_order]);
    fallback_mt = find_suitable_fallback(area, current_order,
                 start_migratetype, false, &can_steal);
    if (fallback_mt == -1)
        continue;

    /*
     * 我们不能从页块中偷取所有空闲页面，
     * 这种情况下更好
     * 请求的迁移类型是可移动的。
     * 偷取并拆分最小的可用页面，而不是
     * 最大可用页面，即使下一个可移动
     * 分配落到了与当前不同的页块，
     * 它也不会导致永久性碎片化。
     */

    if (!can_steal && start_migratetype == MIGRATE_MOVABLE
        && current_order > order)
        goto find_smallest;

    goto do_steal;
}

return false;

find_smallest:
for (current_order = order; current_order < MAX_ORDER; current_order++) {
    area = &(zone->free_area[current_order]);
    fallback_mt = find_suitable_fallback(area, current_order,
                 start_migratetype, false, &can_steal);
    if (fallback_mt != -1)
        break;
}

/*
 * 这不应该发生——我们在寻找最大页面时
 * 已经找到了合适的回退。
 */
VM_BUG_ON(current_order == MAX_ORDER);

do_steal:
page = get_from_free_area(area, fallback_mt);
steal_suitable_fallback(zone, page, flags, start_migratetype,
            can_steal);
trace_mm_page_alloc_extfrag(page, order, current_order,
            start_migratetype, fallback_mt);
return true;
}
```

### 功能描述

如果调用 `__rmqueue_fallback()`，说明我们未能找到所需迁移类型的页面，或者找到高于所需阶数的页面，因此我们回退到从另一种迁移类型移动页面到目标类型。

页面的迁移类型通过其所属的页块通过 `get_pageblock_migratype()` 确定，但这仅仅是一种启发式方法，因为我们以小于此粒度的度在迁移块之间“偷取”页面。当页面被释放时，它最终将返回页块。

水位线提升通过 `boost_watermark()` 应用，该函数仅在 `steal_suitable_fallback()` 中调用。查看列表 2-66 所示的 `find_suitable_fallback()` 及其调用的函数 `can_steal_fallback()`（列表 2-67）。

### 列表 2-67: `can_steal_fallback()`

```c
static bool can_steal_fallback(unsigned int order, int start_mt)
{
    /* 保留这个 order 是刻意的，尽管在后续中有
       松弛的 order 检查。原因是
       如果满足此条件，我们实际上可以窃取页块，
       但下面的检查并不能保证这一点，这只是一个启发式方法，
       可以随时更改。
    */
    if (order >= pageblock_order)
        return true;

    if (order > pageblock_order / 2 ||
        start_mt == MIGRATE_RECLAIMABLE ||
        start_mt == MIGRATE_UNMOVABLE ||
        page_group_by_mobility_disabled)
        return false;

    return true;
}
```

### 表 2-8: 迁移类型回退

| 迁移类型 | 主回退方案 | 次级回退方案 |
| :--- | :--- | :--- |
| `MIGRATE_UNMOVABLE` | `MIGRATE_RECLAIMABLE` | `MIGRATE_MOVABLE` |
| `MIGRATE_MOVABLE` | `MIGRATE_RECLAIMABLE` | `MIGRATE_UNMOVABLE` |
| `MIGRATE_RECLAIMABLE` | `MIGRATE_UNMOVABLE` | `MIGRATE_MOVABLE` |

> 注意：其他迁移类型没有回退方案。

### `find_suitable_fallback()` 逻辑

1. 如果空闲列表不包含页面，返回 `-1`，表示未找到合适的回退方案。
2. 尝试主回退方案，通过 `free_area_empty()` 检查所需的 order 和迁移类型是否存在页面。如果不存在，则尝试次级回退。如果两者都失败，则返回 `-1`，表示未找到合适的回退方案。
3. 将 `can_steal` 输出参数设置为 `can_steal_fallback()` 的结果。这检查了 order 是否大于或等于一个页块的一半（此时我们将该页块视为具有所需的迁移类型一次），或者如果分配是 `MIGRATE_RECLAIMABLE` 或 `MIGRATE_UNMOVABLE`，我们将其视为比可移动页块中的不可移动页更少。`page_group_by_mobility_disabled` 也会导致窃取的发生，但这仅与内存极低的系统相关。
4. 如果 `only_stealable` 未被设置（任何 `page_alloc.c` 分配调用的情况），则我们返回选定的迁移类型，否则我们返回 `can_steal` 是否已设置。

### 列表 2-68: `steal_suitable_fallback()`

```c
/*
 * 此函数实现了实际的窃取行为。如果 order 足够大
 * 我们尝试窃取整个页块。
 */

if (boost_watermark && (alloc_flags & ALLOC_KSWAPD))
    set_bit(ZONE_BOOSTED_WATERMARK, &zone->flags);

/* 我们不允许尝试从整个块窃取 */
if (!whole_block)
    goto single_page;

free_pages = move_freepages_block(zone, page, start_type,
    &movable_pages);

/*
 * 确定有多少页面与我们的分配兼容。
 * 对于可移动分配，它是我们刚刚获得的可移动页数量。
 * 对于其他类型，要复杂一些。
 */
if (start_type == MIGRATE_MOVABLE) {
    alike_pages = movable_pages;
} else {
    /* 如果我们将一个 RECLAIMABLE 或 UNMOVABLE 分配
     * 回退到 MOVABLE 页块，考虑将所有不可移动页面
     * 视为兼容。如果是 UNMOVABLE 回退到 RECLAIMABLE
     * 反之亦然，要保持保守，因为我们无法区分
     * 不移动页的精确迁移类型。
     */
    if (old_block_type == MIGRATE_MOVABLE)
        alike_pages = pageblock_nr_pages - (free_pages + movable_pages);
    else
        alike_pages = 0;
}

/* 移动整个块可能会因为区域边界条件而失败 */
if (!free_pages)
    goto single_page;

/*
 * 如果块中有足够数量的页面是空闲或
 * 与我们的分配相同的迁移性，则占用整个块。
 */
if (free_pages + alike_pages >= (1 << (pageblock_order - 1)) ||
    page_group_by_mobility_disabled)
    set_pageblock_migratetype(page, start_type);

return;

single_page:
move_to_free_list(page, zone, current_order, start_type);
}
```

### `steal_suitable_fallback()` 检查逻辑

1. 如果由于竞态条件，现有的页块迁移类型被设置为 `MIGRATE_HIGHATOMIC`，我们不尝试窃取整个块，而是进入 `single_page` 分支。
2. 接下来我们计算 `alike_pages`，即页块内与所需的迁移类型兼容的已分配页面计数：
    - 如果目标类型是可移动的，则等于页块内的可移动页数量。
    - 如果源类型是 `MIGRATE_MOVABLE`，则等于页块内的不可移动页数量。
    - 如果源和目标类型都不是可移动的，则保持保守，因为我们无法区分不可移动页的精确迁移类型。
3. 如果移动整个块由于区域边界条件失败，则进入 `single_page` 分支。
4. 否则，通过求和 `free_pages` 和 `alike_pages` 检查是否大于或等于 `(1 << (pageblock_order - 1))` 或 `page_group_by_mobility_disabled`，来设置页块的类型。
5. `single_page` 分支调用 `move_to_free_list()`。

### 列表 2-69: `__page_smallest()`

```c
static inline struct page *__page_smallest(struct zone *zone,
                                           unsigned int order,
                                           int migratetype)
{
    unsigned int current_order;
    struct free_area *area;
    struct page *page;

    /* 遍历空闲列表查找一个 order 的页面 */
    for (current_order = order; current_order < MAX_ORDER; ++current_order) {
        area = &(zone->free_area[current_order]);
        page = get_page_from_area(area, migratetype);
        if (!page)
            continue;

        del_page_from_list(page, zone, current_order);
        expand(zone, page, current_order, migratetype);
        set_pageblock_migratetype(page, migratetype);

        if (pcp_allowed(order) &&
            migratetype < MIGRATE_PCP)
            return page;
    }

    return NULL;
}
```

### 高阶原子分配预留

内核包含一种机制，用于预留一个页块，专门用于高阶原子分配，这些分配是在内核无法睡眠的情况下执行的（例如在中断上下文中），如列表 2-71 所示。

### 列表 2-71: `reserve_highatomic_pageblock()`

```c
/*
 * 预留一个页块供高阶原子分配独占使用
 * 这里没有包含合适阶数的空页块
 */
static void reserve_highatomic_pageblock(struct page *page, struct zone *zone,
                                         unsigned int alloc_order)
{
    int mt;
    unsigned long max_managed, flags;

    /* 将预留数量限制为 1 个页块或约一个区域的 1%。
     * 检查存在竞态风险但并无害。
     */
    max_managed = (zone->managed_pages / 100) + pageblock_nr_pages;

    if (zone->nr_reserved_highatomic >= max_managed)
        return;

    spin_lock_irqsave(&zone->lock, flags);

    /* 在锁保护下重新检查 nr_reserved_highatomic 限制 */
    if (zone->nr_reserved_highatomic >= max_managed)
        goto out_unlock;

    /* 拿走！ */
    mt = get_pageblock_migratetype(page);

    /* 仅预留普通页块（即它们可以与其他页合并） */
    if (migratetype_is_mergeable(mt)) {
        zone->nr_reserved_highatomic += pageblock_nr_pages;
        set_pageblock_migratetype(page, MIGRATE_HIGHATOMIC);
        move_freepages_block(zone, page, MIGRATE_HIGHATOMIC, NULL);
        return;
    }

out_unlock:
    spin_unlock_irqrestore(&zone->lock, flags);
}
```

**列表 2-71: mm/page_alloc.c: `reserve_highatomic_pageblock()`**

如果分配阶数大于 0 且设置了 `ALLOC_HARDER` 标志（通常用于内存压力下的原子分配），则由 `get_page_from_freelist` 调用。只有当 `migratetype` 为 `migratetype_is_mergeable()` 时，页块才会移动到 `MIGRATE_HIGHATOMIC`，这表示该迁移类型是否有回退机制。
