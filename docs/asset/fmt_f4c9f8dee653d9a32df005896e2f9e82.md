```cpp
/* showlineeditdlg.cpp */
#include <app.h>
#include <klineeditdlg.h>
int main(int argc, char **argv)
{
    bool OK;
    QString str;
    KApplication app(argc,argv,"showlineeditdlg");
    str = KLineEditDlg::getText(
        "The caption",
        "The editable text",
        &OK
    );
    if(OK) {
        // 选择了 OK 按钮
    } else {
        // 选择了 CANCEL 按钮
        return(app.exec());
    }
}
```
