# 示例 15-12. 使用 eval 在变量间进行选择

```bash

#!/bin/bash

## arr-choice.sh

# 将参数传递给函数，以便从一组变量中

# 选择一个特定的变量。

arr0=(10 11 12 13 14 15 )

arr1=(20 21 22 23 24 25 )

arr2=(30 31 32 33 34 35 )

#      0 1 2 3 4 5       索引（从0开始）

choose_array() {

  eval array_member=\${arr\$array_number}[$element_number]

  # 使用 eval 构建变量名，

  # 在这种特殊情况下，是一个数组名。

  echo "Element $element_number of array $array_number is $array_member"

}

# 函数可以重写为接收参数。

array_number=0      # 第一个数组。

element_number=3

choose_array        # 13

array_number=2      # 第三个数组。

element_number=4

choose_array        # 34

array_number=3      # 空数组（未分配 arr3）。

element_number=4

choose_array        # (null)

# 感谢 Antonio Macchi 指出这一点。

```
