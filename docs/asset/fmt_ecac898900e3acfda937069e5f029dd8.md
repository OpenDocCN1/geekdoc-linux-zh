  if [ "$BASH*" ] && [ "$BASH" != "/bin/sh" ]; then
    # bash.bashrc 文件已经设置了默认的 PS1。
    PS1=\h:\w\$
    if [ -f /etc/bash.bashrc ]; then
      . /etc/bash.bashrc
    fi
  else
    if [ "$($id -u)" -eq 0 ]; then
      PS1='#'
    else
      PS1='$'
    fi
  fi
  if [ -d /etc/profile.d ]; then
    for i in /etc/profile.d/*.sh; do
      if [ -r $i ]; then
        . $i
      fi
    done
    unset i
  fi
fi
