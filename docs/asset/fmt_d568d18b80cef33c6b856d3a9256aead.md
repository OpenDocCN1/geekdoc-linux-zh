void freehand(int x, int y, int color)
{
    vga_waitretrace();
    gl_putbox(x, y, 11, 19, box);
    gl_setpixel(x, y, color);
    gl_getbox(x, y, 11, 19, box);
    draw_cursor(x, y);

    while (button & MOUSE_LEFTBUTTON)
    {
        mouse_update();
        x = mouse_getx();
        y = mouse_gety();
        if (x < LEFT)
        {
            x = LEFT;
            mouse_setposition(x, y);
        }
        if (y < TOP)
        {
            y = TOP;
            mouse_setposition(x, y);
        }
        if (x > RIGHT)
        {
            x = RIGHT;
            mouse_setposition(x, y);
        }
        if (y > BOTTOM)
        {
            y = BOTTOM;
            mouse_setposition(x, y);
        }
        button = mouse_getbutton();

        if ((x != ox) || (y != oy))
        {
            vga_waitretrace();
            gl_putbox(ox, oy, 11, 19, box);
            gl_setpixel(x, y, color);
            gl_getbox(x, y, 11, 19, box);
            draw_cursor(x, y);

            ox = x;
            oy = y;
        }
    }
}
