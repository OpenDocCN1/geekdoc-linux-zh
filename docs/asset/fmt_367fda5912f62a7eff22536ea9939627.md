
# kgrep - Korn Shell grep 程序

```ksh
#!/bin/ksh
#
# kgrep - Korn Shell grep 程序
#

# 声明默认标志
CFLAG= NFLAG= LFLAG= SFLAG= VFLAG=
integer LNUM=0 COUNT=0 TOT_COUNT=0

# 禁用文件名生成
set -f

# 检查用法
if [[ ($# < 2 )); then
    print "Usage: $0 [options] expression files"
    exit 1
fi

# 解析命令行选项
while true
do
    case $1 in
        -b*) ) print "b option not supported";;
        -c*) ) CFLAG=1;
        -i*) ) IFLAG=1;
        -l*) ) LFLAG=1;
        -n*) ) NFLAG=1;
        -s*) ) SFLAG=1;
        -v*) ) VFLAG=1;
        -*) ) print "$0: unknown flag $1"; exit 2 ;;
        *) )
            PATTERN=$1
            shift
            break ;;
    esac
    shift
done
```

```ksh
# 设置不打印标志
NOPRINT=$VFLAG$CFLAG$LFLAG
VOPRINT=$CFLAG$LFLAG
```

```ksh
# 设置大/小写模式
typeset -u UPATTERN=$PATTERN
typeset -l LPATTERN=$PATTERN
```

```ksh
# 检查文件参数
if (( ($# == 0)); then
    print "Must have file argument"
    exit 1
fi
```

```ksh
# 处理文件
for FILE
do
  # 为标准输入打开文件
  exec 0<$FILE
```

```ksh
  # 读取文件中的每一行
  while read -r LINE
  do
    # 增加行号计数器
    ((LNUM+=1))

    # 检查每一行的模式
    case $LINE in
      # 查看 PATTERN 是否匹配输入
      $PATTERN*)
        if [[ $VFLAG = "" ]]; then
          [[ $NOPRINT = "" ]] && \
            print -r "${NFLAG+:$LNUM}:$LINE"
          ((COUNT+=1))
        fi
        ;;
      # 对于 -i 选项：查看
      # 大/小写模式是否匹配
      * $UCPATTERN | * $LCPATTERN *)
        if [[ $LFLAG == "" ]]; then
          if [[ $VFLAG = "" ]]; then
            if [[ $NOPRINT = "" ]] && \
              print -r "${NFLAG+:$LNUM}:$LINE"
            ((COUNT+=1))
          fi
        else
          if [[ $VFLAG != "" ]]; then
            if [[ $NOPRINT = "" ]] && \
              print -r "${NFLAG+:$LNUM}:$LINE"
            ((COUNT+=1))
          fi
          fi
        fi
        ;;
      # 对于 -v 选项：查看
      # 模式不匹配
      !($PATTERN*)
        if [[ $VFLAG != "" ]]; then
          [[ $NOPRINT = "" ]] && \
            print -r "${NFLAG+:$LNUM}:$LINE"
          ((COUNT+=1))
        fi
        ;;
    esac
  done

  # 处理 -l 标志
  # 显示文件名
  # 如果文件中有匹配项
  if [[ $LFLAG != "" ]]; then
    print $FILE
  fi

  # 增加总匹配计数
  # 多个文件
  ((TOT_COUNT+=COUNT))

  # 重置下一个文件的行号计数器
  LNUM=0
done

# 重置下一个文件的匹配计数
COUNT=0

# 处理 -c 标志
# 显示找到的匹配总数
if [[ $CFLAG != "" ]]; then
    print $TOT_COUNT
fi

# 成功退出
exit 0
```

# 日历程序 - kcal

这是一个实现菜单驱动日历的 Korn shell。它支持添加、删除和列出日历条目。它提供了查找当前日期所有日历条目的功能。

```ksh
#!/bin/ksh
#
# kcal - Korn Shell 日历程序
# 处理错误
function error {
    print "${1:-"error encountered"}"
    exit 1
}
# 检查参数
if (($# > 0)); then
    error "Usage: $0"
fi
# 使用环境变量设置或默认
CALFILE=$HOME/.calfile
# 如果文件不存在则创建日历文件
if [[ ! -f $CALFILE ]]; then
    print "Creating default $CALFILE"
    touch $CALFILE || error "$CALFILE"
fi
# 变量声明/赋值
typeset DATE
ENTRY="yes"
PAGER=$(whence more)
CLEAR=$(whence clear)
# 设置 trap 允许中断
trap 'CLEAR; print "Interrupt ignored to quit. Press <Return> to continue."' INT
# 设置 EXIT trap - 执行清理
trap 'rm -rf /tmp/$$' EXIT
# 检查日期
function checkdate {
    while true
    do
        # 提示日期
        read DATE?"Enter date in mmdd[yy] format (default today):"
        case $DATE in
            # 使用今天
            "") DATE=$(date +%m-%d-%y)
                break ;;
            # 检查给定的日期
            +([0-9]{2})
                case ${DATE#??} in
                    # 设置月份为前两个字符
                    4|)
                        typeset -L2 MO=$DATE
                        # 检查年份长度
                        # 4 = mmdd, 6 = mmddyy
                        if (${#DATE} == 6)); then
                            # 设置 TMP 获取日期
                            typeset -L4 TMP=$DATE
                            # 获取日
                            typeset -R2 DA=$TMP
                            # 获取年份
                            typeset -R2 YR=$DATE
                        else
                            # 获取日
                            typeset -R2 DA=$DATE
                            # 设置当前年份
                            YR=$(date +%y)
                            DATE=$DATE
```
