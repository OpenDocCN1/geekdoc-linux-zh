```python
class DecoderBase:
    _registry = { }

    @classmethod
    def __init_subclass__(cls):
        for mt in cls.mimetypes:
            DecoderBase._registry[mt.mimetype] = cls
```

```python
# 使用注册表的工厂函数
def create_decoder(mimetype):
```

```python
import inspect
class Base:
    @classmethod
    def __init_subclass__(cls):
        # 创建一个 __repr__ 方法
        args = list(inspect.signature(cls).parameters)
        argvals = ', '.join('{self:%s!r}' % arg for arg in args)
        code = 'def __repr__(self):\n'
        code += f' return f"{cls.__name__}({argvals})"\n'
        locs = { }
        exec(code, locs)
        cls.__repr__ = locs['__repr__']
```

```python
import time
class Date:
    def __init__(self, year, month, day):
        self.year = year
        self.month = month
        self.day = day

    @classmethod
    def today(cls):
        t = time.localtime()
        self = cls.__new__(cls) # 创建实例
        self.year = t.year
        self.month = t.month
        self.day = t.day
        return self
```

```python
class Date:
    _cache = { }
    @staticmethod
    def __new__(cls, year, month, day):
        selfref = Date._cache.get((year, month, day))
        if not selfref:
            self = super().__new__(cls)
            self.year = year
            self.month = month
            self.day = day
            Date._cache[(year, month, day)] = weakref.ref(self)
        else:
            self = selfref
        return self

    def __init__(self, year, month, day):
        pass

    def __del__(self):
        del Date._cache[self.year, self.month, self.day]
```

```python
# 示例
d = Date(2012, 12, 21)
e = Date(2012, 12, 21)
assert d is e # 同一个对象
```

```python
class SomeClass:
    def __init__(self):
        self.resource = open()

    def __del__(self):
        self.resource.close()

    def close(self):
        self.resource.close()

    def __enter__(self):
        return self

    def __exit__(self, ty, val, tb):
        self.close()
```

```python
# 通过 __del__() 关闭

del s
```
