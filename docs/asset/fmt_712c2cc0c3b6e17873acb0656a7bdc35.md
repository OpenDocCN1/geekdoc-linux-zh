我们还看到，当通过 `SUB` 指令分配存储空间时，局部变量默认不会被初始化，且在尾言执行期间不会被清除。分配之前内存中是什么内容，分配后就会变成局部变量的值，即所谓的“垃圾值”。

## 带注释的项目反汇编代码

这里是 `main` 和 `arithmetic` 的带注释反汇编代码，为了视觉清晰已除了内存地址：

### 第 11 章 函数参数

`main`:

```
stp     x29, x30, [sp,#-48]!    // 建立栈
                                     frame for
mov     x29, sp                     // 参数和局部
                                     variables
str     w0, [x29,#28]               // 保存第一个 main
                                     parameter
str     x1, [x29,#16]               // 保存第二个 main
                                     parameter
mov     w0, #0x1                   // 设置第一个
                                     parameter
mov     w1, #0x1                   // 用于 arithmetic
                                     function
                                     // 设置第二个
                                     parameter
bl     0X405e4 <Z10arithmeticii>  // 用于 arithmetic
                                     function
str     w0, [x29,#44]               // 设置结果
mov     w0, #0x0                   // 局部变量
ld     x29, x30, [sp],#48         // main应该返回 0
                                     // 恢复之前的
                                     stack frame,
                                     // frame and link
                                     registers
sub     sp, sp, #0x10               // 从 main 返回
                                     // 建立栈
                                     frame for
```

```
// parameters and local variables
str     w0, [sp,#12]
// 保存第一个 arithmetic 参数 (a)
str     w1, [sp,#8]
// 保存第二个 arithmetic 参数 (b)
ldr     w1, [sp,#8]
// w1 <- [b]
ldr     w0, [sp,#12]
// w0 <- - [a]
add     w0, w1, w0
// w0 <- w1 + w0
str     w0, [sp,#8]
// [b] - w0
ldr     w0, [sp,#12]
// w0 <- - [a]
add     w0, #0X1
// w0 <- - w0 + 1
str     w0, [sp,#8]
// [a] - w0
ldr     w1, [sp,#8]
// w1 <- - [b]
ldr     w0, [sp,#12]
// w0 <- - [a]
mul     w0, w1, w0
// w0 <- - w1 * w0
str     w0, [sp,#8]
// [b] <- w0
ldr     w0, [sp,#8]
// - w0
```

```
add     sp, sp, #0x10
// 恢复之前的栈帧
ret                               // 从 arithmetic 返回
```
