{

    int i = 0;

/* 返回 str 在字符串数组 strs 中的索引 */

/* 如果不存在，则返回 NOT_FOUND */

    while (strcmp(strs[i], " ") != 0) {

        if (strcmp(strs[i], str) == 0)

            return (i);

    } else {

        i++;

    }

    return (NOT_FOUND); /* 未找到 */

}
