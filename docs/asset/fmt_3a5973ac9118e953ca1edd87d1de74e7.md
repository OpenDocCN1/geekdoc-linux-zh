```bash
#!/usr/bin/env bash

{
  # 检查是否安装了 GNOME Desktop Manager。如果未安装包，建议为“不适用”
  # 确定系统的包管理器

  l1_pkgoutput=""

  if command -v dpkg-query > /dev/null 2>&1; then
    l1_pq="dpkg-query -W"
  elif command -v rpm > /dev/null 2>&1; then
    l1_pq="rpm -q"
  fi

  # 检查是否安装了 GDM
  l1_pcl="gdm3" # 要检查的包列表

  for ln in $l1_pcl; do
    $l1_pq "$ln" >/dev/null 2>&1 && l1_pkgoutput="y" && echo -e "\nPackage: $ln" 于系统 - 需要修复
  done

  # 检查配置（如果适用）
  if [ -n "$l1_pkgoutput" ]; then
    # 查找 idle-delay 以确定正在使用的配置，测试需要
    l1_kfd=$(grep -l "idle-delay=uint32" /etc/dconf/db/* | awk -F'/' '{split($NF-1,a); print a[1]}') # 设置键文件目录
    # 查找 lock-delay 以确定正在使用的配置，测试需要
    l2_kfd=$(grep -l "lock-delay=uint32" /etc/dconf/db/* | awk -F'/' '{split($NF-1,a); print a[1]}') # 设置键文件目录

    if [ -d "$l1_kfd" ]; then # 如果目录不存在，无法锁定
      fi

    if grep -q "/org/gnome/desktop/session/idle-delay"; then
      echo "idle-delay 已锁定在 /org/gnome/desktop/session/$l1_kfd"
    else
      echo "正在创建锁以 idle-delay"
      if [ ! -d "$l1_kfd/locks" ]; then
        echo "创建目录 $l1_kfd/locks"
        mkdir "$l1_kfd/locks"
        echo -e "\n锁定桌面屏幕保护程序 idle-delay 设置"
        echo "/org/gnome/desktop/session/idle-delay"
        >> "$l1_kfd/locks/00-screensaver"
      fi
    else
      echo -e "idle-delay 未设置，可以被锁定\n"
      请遵循建议="当用户闲置 idle-delay 时确保 GDM 屏幕锁定"并遵循该建议
    fi

    if [ -d "$l2_kfd" ]; then # 如果目录不存在，无法锁定
      if grep -q "/org/gnome/desktop/screensaver/lock-delay" "$l2_kfd"; then
        echo "lock-delay 已锁定在 \"" -F'/'($grep -l /org/gnome/desktop/screensaver/lock-delay) \"$l2_kfd\"
      fi
    fi
  fi
}

else
  echo "正在创建锁 \"lock-delay\"条目"
  { ! -d "$1_kfd2/locks" } && echo "正在创建目录 $1_kfd2/locks"; mkdir "$1_kfd2"
  {
    echo -e '\n# 桌面屏幕保护程序锁屏延迟设置'
    echo /org/gnome/desktop/screensaver/lock-delay'
  } >> "$1_kfd2/locks/00-screensaver"
  fi
else
  echo -e "\n -未设置\"lock-delay\"，因此无法锁定\n"
  请遵循建议“确保在用户空闲时 GDM 屏幕锁屏”，并再次遵循此建议"
fi
else
  echo -e " -系统未安装 GNOME 桌面管理器包\n - 建议不适用"
)
```

## 1.8.6 确保禁用 GDM 自动挂载可移动介质（自动化）

适用范围：

- 等级 1 - 服务器
- 等级 2 - 工作站

描述：

默认，GNOME 会在插入可移动介质时自动进行挂载，以方便用户使用。

原理：

启用自动挂载后，任何具有物理访问权限的人都可以连接 USB 驱动器或光盘，即使他们没有权限自行挂载，系统也能访问其内容。

影响：

便式硬盘对于工作站用户非常常见。如果您的组织允许在工作站上使用便式存储或介质，并且认为工作站的物理访问控制是足够的，那么关闭自动挂载的不大。

审计：

运行以下脚本以验证自动挂载已禁用：

```bash
#!/usr/bin/bash
{
  1_pkgoutput=""
  1output2=""
  # 检查是否安装了 GNOME 桌面管理器。如果未安装包，则建议不适用
  # 确定系统的包管理器
  if command -v dpkg-query > /dev/null 2>&1; then
    1_pkg="dpkg-query -w"
  elif command -v rpm > /dev/null 2>&1; then
    1_pkg="rpm -q"
  fi
  # 检查 gdm 是否安装
  1_cl="gdm3" # 检查的包列表
  for lp in $1_cl; do
    if $1_pkg "$lp" > /dev/null 2>&1; then
      1_pkgoutput="\n系统上存在包=\"$lp\"\n 正在检查配置"
    fi
  done
  # 检查配置（如果适用）
  if [ -n "$1_file" ]; then
    echo -e "$1_pkgoutput"
    # 查找存在的设置和变量
    1_kfile="$(grep -rip '^auttomount-behavior' /etc/dconf/db*/.d)"
    1_k2="$(grep -rip '^auttomount-behavior' /etc/dconf/db*/.d)"
    # 根据 dconf 目录 (${PROFILE_NAME}) 设置包名
    if [ -f "$1_kfile" ]; then
      1_gpname=$(awk -F'/' '{split($NF-1,",");print $a[1]}')
    elif [ -f "$1_k2" ]; then
      1_gpname=$(awk -F'/' '{split($NF-1,",");print $a[1]}')
    fi
    # 如果 profile 名称存在
    if [ -n "$1_gpname" ]; then
      1_gpdir="/etc/dconf/$1_gpname.d"
      # 检查 profile 文件是否存在
      if grep -P -- "\system-db:$1_gpname\" /etc/dconf/profile/*"; then
        1_output="$1_output -n dconf 数据库 profile 文件 \$(grep -Gpl\"\system-db:$1_gpname\" /etc/dconf/profile/*\") 存在"
      else
        1_output="$1_output\n 未设置 dconf 数据库 profile"
      fi
      # 检查 dconf 数据库是否存在
      if [ -f "$1_gpname" ]; then
        1_output="$1_output -n dconf 数据库 \$1_gpname\" 存在"
      else
        1_output="$1_output2\n - dconf 数据库 \$1_gpname\" 不存在"
      fi
      # 检查 dconf 目录是否存在
      if [ -d "$1_gpdir" ]; then
        1_output="$1_output\n dconf 目录 \$1_dir 存在"
      else
        1_output="$1_output2\n dconf 目录 \$1_dir 不存在"
      fi
    fi
}

# 检查 automount 设置
if grep -P '^auttomount-behavior=false' "$1_kfile"; then
    1_output="$1_output\n - \"automount\" 已设置为 false"
else
    1_output="$1_output2\n 未设置 \"automount\""
fi

# 检查 automount-open 设置
if grep -P '^auttomount-open=false' "$1_k2"; then
    1_output="$1_output\n - \"automount-open\" 已设置为 false"
else
    1_output="$1_output2\n 未设置 \"automount-open\""
fi

else
    # 设置不存在。无需检查
    1_output="$1_output2\n - 未设置 \"automount\" 或 \"automount-open\""
fi

else
    1_output="$1_output\n 系统未安装 GNOME 桌面管理器 - 建议不适用"
fi

# 报告结果。如果 1_output 中没有失败，则通过
if [ -z "$1_output2" ]; then
    echo -e "\n- 审计结果：通过\n$1_output"
else
    echo -e "\n- 审计结果：失败\n- 审计原因:\n$1_output"
    [ -n "$1_output" ] && echo -e "\n 设置正确:\n$1_output"
fi
```
