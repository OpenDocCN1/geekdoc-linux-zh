// 设置 cookies
function init_cookies()
{
    global $session_id;
    if (!isset($session_id)) {
        // session_status 取值：
        // 0 - 未登录，正在浏览
        // 1 - 已登录
        // 2 - 完成购买。
        $query = "INSERT INTO sessions (session_status) VALUES('0')";
        $result = MYSQL_QUERY($query);
        $session_id=mysql_insert_id();
        setcookie("session_id",$session_id);
        // 注意 - 这在本质上是不安全的。例如，简单的
        // 在浏览器中增加 cookie ID 很容易。
        // 最好生成一个随机的或难以
        // 猜测的唯一 ID。
    }
}

// 注销用户
function logout_user()
{
    global $session_id;
    global $session_status;
    global $login_success;
    global $logout;
    if ($logout) {
        $query="UPDATE sessions SET user_id = '' WHERE session_id=$session_id";
        $result=MYSQL_QUERY($query);
        $query="UPDATE sessions SET session_status = '0' WHERE session_id=$session_id";
        $result=MYSQL_QUERY($query);
        $session_status=0;
        $login_success=0;
    }
}

{
