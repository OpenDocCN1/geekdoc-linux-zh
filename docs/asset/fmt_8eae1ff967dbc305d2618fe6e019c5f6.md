```bash
echo "$phone_number"|egrep '^([0-9]{3}\)? [0-9]{3}-[0-9]{4}$' >/dev/null 2>&1
if [ $? -eq 0 ]; then
  print "GOOD: $phone_number"
else
  print "NOT VALID: $phone_number"
fi
```
