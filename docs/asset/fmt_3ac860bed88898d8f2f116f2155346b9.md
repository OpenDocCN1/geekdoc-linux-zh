{
    unsigned long newsize = roundup_pow_of_two(size);

    if (newsize < max / 32)
        newsize = newsize * 4;
    else if (newsize <= max / 4)
        newsize = newsize * 2;
    else

    return newsize;
}

## 9.7.4 通用预读代码

有两个函数最终被调用以执行预读：`page_cache_ra_order()`（我们在列表 77 中检查）和 `do_page_cache_ra()`（我们在列表 78 中检查）。

```c
void page_cache_ra_order(struct readahead_control *ractl,
                         struct file_ra_state *ra, unsigned int new_order)
{
    struct address_space *mapping = ractl->mapping;
    pgoff_t index = readahead_index(ractl);
    pgoff_t limit = ((size_t)mapping->i_size - 1) >> PAGE_SHIFT;
    pgoff_t mark = index + ra->async_size;
    int err = 0;
    gfp_t gfp = readahead_gfp(mapping);

    if (!mapping_large_folio_support(mapping) || ra->size < 4)
        goto fallback;

    limit = min(limit, index + ra->size - 1);

    if (new_order < MAX_CACHE_ORDER) {
        new_order += 2;
        if (new_order > MAX_CACHE_ORDER) {
            new_order = MAX_CACHE_ORDER;
            while ((1 << new_order) > ra->size) {
                new_order--;
            }
        }

        filemap_invalidate_lock(mapping);

        while (index < limit) {
            unsigned int order = new_order;

            /* 如果需要，对齐较小的页 */
            if (index & ((1 << order) - 1)) {
                order = readahead_offset(index);
                if (order == 1)
                    order = 0;
            }

            /* 不要在 EOF 之后分配页 */
            while (index + (1 << order) - 1 > limit) {
                if (--order == 1)
                    order = 0;
            }

            err = ra_alloc_folio(ractl, index, order, gfp);
            if (err) {
                break;
            }

            index += 1 << order;
        }

        if (index > size) {
            ra->size += index - limit - 1;
            ra->async_size += index - 1;
        }
    }

    read_pages(ractl);
    myfile_unlock_shared(mapping);

    /*
     * 如果页缓存中已经存在页，我们可能留下了
     * 一些间隙。让普通的预读代码处理这种情况。
     */
    if (err) {
        return;
    }

fallback:
    do_page_cache_ra(ractl, ra->size, ra->async_size);
}
```

列表 77：`mm/readahead.c`：`page_cache_ra_order()`
该函数使用 `mapping_large_folio_support()` 来决定继续执行还是回到 fallback。这超出了本书的范围，而且如果文件系统启用了大页（large folio）支持，也只是少数情况；只有在提供透明页功能时，它才有意义，而这使得该函数超出了本书的范围，因此在此处仅供参考。
我们注意到 fallback 情况以及强制预读实际上使用 `do_page_cache_ra()` 来执行预读，我们在列表 9-78 中检查它。

```c
/*
 * do_page_cache_ra(actually) 读取磁盘块。它先分配
 * 页，然后提交进行 I/O。这避免了当页分配导致
 * VM 回写（writeback）时发生的非常糟糕的行为。
 * 我们不希望那样混合读写。
 */
static void do_page_cache_ra(struct readahead_control *ractl,
                             unsigned long nr_to_read, unsigned long lookahead_size)
{
    struct inode *inode = ractl->mapping->i_host;
    unsigned long index = readahead_index(ractl);
    loff_t isize = i_size_read(inode);
    pgoff_t end_index;      /* 我们想要读取最后一个页 */

    if (isize == 0)
        return;

    end_index = (isize - 1) >> PAGE_SHIFT;

    if (index > end_index)
        return;

    /* 不要越过最后一个字节所在的页 */
    if (nr_to_read > end_index - index)
        nr_to_read = end_index - index + 1;

    page_cache_ra_unbounded(ractl, nr_to_read, lookahead_size);
}
```

列表 78：`mm/readahead.c`：`do_page_cache_ra()`

```c
void page_cache_ra_unbounded(struct readahead_control *ractl,
                             unsigned long nr_to_read, unsigned long lookahead_size)
{
    struct address_space *mapping = ractl->mapping;
    unsigned long index = readahead_index(ractl);
    gfp_t gfp_mask = readahead_gfp_mask(mapping);
    unsigned long i;

    /*
     * 经过预读操作的路径，我们已经将页面
     * 添加到页缓存中，但尚未将它们提交
     * 进行 I/O 传输。添加另一个页面可能需要分配内存，
     * 这可能会触发内存回收。 告诉 VM 我们正处于
     * 文件系统操作中间将导致它不触碰
     * 回写页，以防止死锁。大多数（或所有？）
     * 文件系统已经通过其映射的 gfp_mask
     * 指定了 GFP_NOFS，这里将进行显式指定。
     */
    unsigned int nofs = memalloc_nofs_save();

    filemap_invalidate_lock_shared(mapping);

    /*
     * 预分配我们需要尽可能多的页面。
     */
    for (i = 0; i < nr_to_read; i++) {
        struct folio *folio = xa_load(&mapping->i_pages, index + i);

        if (folio && xa_is_valid(folio)) {
            /* 页面已存在？在继续下一批
             * 页面之前，启动当前连续的
             * 页面。此页面可能是我们想要
             * 标记为预读（Readahead）的那个，但我们
             * 没有对该页面的稳定引用，而且
             * 为了这个获取一个引用是不值得的。
             */
            read_pages(ractl);
            ractl->index++;
            i = ractl->index + ractl->nr_pages - 1;
            continue;
        }

        folio = filemap_alloc_folio(gfp_mask, 0);
        if (!folio) {
            break;
        }

        if (filemap_add_folio(mapping, folio, index + i, gfp_mask) < 0) {
            folio_put(folio);
            read_pages(ractl);
            ractl->index++;
            i = ractl->index + ractl->nr_pages - index - 1;
            continue;
        }

        if (i == nr_to_read - lookahead_size)
            folio_set_readhead(folio);

        ractl->nr_pages++;
    }

    /*
     * 开始 I/O。我们忽略 I/O 错误 - 如果 folio 没有被
     * 更新，调用者将再次启动 read_folio，
     * 并在之后处理该错误。
     */
    read_pages(ractl);

    filemap_invalidate_unlock_shared(mapping);
    memalloc_nofs_restore(nofs);
}
```

列表 79：`mm/readahead.c`：`page_cache_ra_unbounded()`

此函数执行了分配和将页面放入页缓存的核心工作，触发底层块设备将数据读取到这些页面中。

我们使用 `readahead_gfp_mask()` 建立在执行物理内存分配时使用的 GFP 标志，该函数使用 `mapping_gfp_mask()` 来确定适用于 `struct address_space` 映射的标志，并指定了 `GFP_NORETRY` 和 `__GFP_NOWARN` 以防止在此处分配失败时重试或警告。更多详情请参见“物理内存”章节的第 2.6 节。

我们利用 `memalloc_nofs_save()` 隐式防止 `__GFP_FS` 出现在分配中以避免死锁。更多详细信息请参见“物理内存”章节。

我们通过 `filemap_invalidate_lock_shared` 获取锁（随后通过 `filemap_invalidate_unlock_shared` 释放），以防止竞争操作执行任何可能的失效操作产生干扰。

然后我们遍历与页缓存 `struct address_space` 对象关联的 xarray 中的每个页面，首先检查我们是否处于 folio 实际上已经存在于页缓存中的位置。

如果是，则通过 `read_pages()`（见列表 9-80）对这些页面执行 I/O，更新索引计数以计入之前的页面，然后继续循环。

否则，我们通过 `filemap_alloc_folio()` 分配一个 0 阶 folio（高阶 folio 由 `do_page_cache_ra()` 处理，其逻辑不在本书范围内），然后通过 `filemap_add_folio()` 添加到页缓存。

## 9.7 物理预读

从磁盘读取的实际操作由 `read_pages()` 执行，我们在列表 9-80 中检查了它。

```c
static void read_pages(struct readahead_control *rac)
{
    const struct address_space_operations *aops = rac->mapping->aops;
    struct folio *folio;

    if (!readahead_count(rac))
        return;

    if (aops->readahead) {
        /*
         * 系统指定的 struct address_space_operations->readahead
         * 处理程序通常使用 readahead_folio() 在首次获取
         * folio，我们的回退处理程序也是如此，我们在
         * 列表 9-81 中检查了这一点。
         */
        while ((folio = readahead_folio(rac)) != NULL) {
            unsigned long nr = folio_nr_pages(folio);

            /*
             * 我们在此确保更新 struct file_ra_state
             * 以保持在预读范围内有效。
             */
            if (rac->ra->async_size >= nr) {
                rac->ra->async_size -= nr;
                filemap_remove_folio(folio);
            }

            folio_unlock(folio);
            folio_put(folio);
        }
    } else {
        while ((folio = readahead_folio(rac)) != NULL)
            aops->read_folio(rac, folio);
    }
}
```

列表 9-80：`mm/readahead.c`：`read_pages()`

我们首先检查 `readahead_control->nr_pages`，如果为 0 则无事可做并退出。

然后我们确定如何继续——如果文件系统提供了指定预读处理程序的 `address_space_operations` 类型，我们使用它让文件系统执行实际预读，读取到每个 folio 中。

如果没有提供，则回退到使用 `readahead_folio()` 获取范围内的每个 folio，然后调用单独 `address_space_operations->read_folio` 来手动执行预读。

如果文件系统无法预读所有的 folio，我们手动遍历剩余部分，并通过 `filemap_remove_folio()` 将它们从页缓存中完全移除（更多详情见第 9.9 节的列表 9-88）。

```c
/**
 * readahead_folio - 获取下一个读取的 folio。
 * @ractl: 当前的预读请求。
 * 上下文：folio 已锁定。调用者应在该 folio 的所有 I/O 完成后解锁 folio。
 * 返回：指向下一个 folio 的指针，如果完成则返回 %NULL。
 */
static inline struct folio *readahead_folio(struct readahead_control *ractl)
{
    struct folio *folio = __readahead_folio(ractl);

    if (folio)
        folio_put(folio);

    return folio;
}
```
