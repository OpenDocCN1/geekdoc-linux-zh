```python
def countdown(start):
    n = start
    def display():
        print('T-minus', n)
    def decrement():
        nonlocal n
        n -= 1  # 修改外层的 n
    while n > 0:
        display()
        decrement()
```

```python
import time

def after(seconds, func):
    time.sleep(seconds)
    func()

# 使用示例
def greeting():
    print('Hello World')

after(10, greeting)  # 10秒后打印 'Hello World'
```

```python
def main():
    name = 'Guido'
    def greeting():
        print('Hello', name)
    after(10, greeting) # 输出: 'Hello Guido'

main()
```
