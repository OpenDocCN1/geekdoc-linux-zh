int main(int argc, char **argv)

{

    int    fd;

    socklen_t len;

    struct sock_opts *ptr;

    for (ptr = sock_opts; ptr->opt_str != NULL; ptr++) {

        printf("%s: ", ptr->opt_str);

        if (ptr->opt_val == NULL)

            printf("(undefined)\n");

    else

        switch (ptr->opt_level) {

            case SOCK_SOCKET:

            case IPPROTO_IP:

            case IPPROTO_TCP:

                fd = socket(AF_INET, SOCK_STREAM, 0);

                break;

#ifdef IPv6

            case IPPROTO_IPV6:

                fd = socket(AF_INET6, SOCK_STREAM, 0);

                break;

#endif

#ifdef SCTP

            case IPPROTO_SCTP:

                fd = socket(AF_INET, SOCK_SEQP, 0);

                break;

#endif

    default:

        err("Can create fd for level %d\n", ptr->opt_level);

    len = sizeof(val);

    if (getsockopt(ptr->opt_level, ptr->opt_name, &val, &len) == -1) {

        err("getsockopt error");

    } else {

        printf("default = %s\n", (ptr->opt_val_str)(&val, len));

    }

    close(fd);

}

exit(0);
