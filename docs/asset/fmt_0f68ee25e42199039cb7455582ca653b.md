```bash
#!/bin/bash
# tohtml.sh [v.0.2, 有点 Bug

# 将文本文件转换为 HTML 格式。
# 作者: Mendel Cooper
# 许可证: GPL
# 用法: sh tohtml.sh <textfile> 脚本可以轻松修改以接受文件名。

# 假设
# 1) 目标文件的段落由空行分隔。
# 2) 图片 (.jpg) 位于 "images"。
# 在目标文件中，图像名称包含在方括号中，
# 例如，[image01.jpg]。
# 3) *斜体* 短语以空格或下划线开始，
# 并以空格或下划线结束。

# 设置
# FNTSIZE=3           # 中等字体大小
# IMGDIR="images"    # 图片目录
# 表头
# HDR01="<!DOCTYPE HTML PUBLIC "-//W3//DTD HTML 4.0 Transitional//EN">
# HDR02="<!-- Converted to HTML by tohtml.sh -->"
# HDR03="<!-- author: Leo Cooper <leo@gmail.com> -->"
# HDR11="<head>"
# HDR1a="</head>"
# HDR12="<title>"
# HDR20="</title>"
# HDR21="<meta NAME="GENERATOR" CONTENT="tohtml.sh">
# HDR13="<body bgcolor="#ddd"> # 修改背景颜色。
# HDR14a="<font size="
# HDR14b="</font>"
# 页脚
# FTR10="</body>"
# FTR11="</html>"

# 标签
# BOLD="<b>"
# CENTER="<center>"
# END_CENTER="</center>"
# LF="br>"

write_headers ()
{
echo "$HDR01"
echo "$HDR02"
echo "$HDR03"
echo
echo "$HDR10"
echo "$HDR11"
echo "$HDR21"
echo "$HDR1a"
echo "$HDR13"
echo -n "$HDR14a"
echo -n "$FNTSIZE"
echo "$HDR14b"
```

我们的`Element` `1`删除了前后的引号。

虽然没有显示，但前后的空格也被删除了。

现在字符串已经设置好，`Bash`在其操作期间也会保留硬引号。

为什么呢？

考虑`"${pls 'Hello($Me)'}"`结构：

`*` -> 展开结果，去除引号。

`$(...)` -> 替换为...的结果。

`_pls '...'` -> 带参数的调用，去除引号。

结果包含硬引号；在上述处理完成后，它们变成了值的一部分。

同样，在操作字符串变量时，

`result` 生存了（直到显式评估字符串）。
