{

UL_LENGTH=$(echo $UPPER_LIMIT)

# 根据 UPPER_LIMIT shell 脚本列表指定长度。

# 由 $UL_LENGTH 变量决定。
typeset -Z$UL_LENGTH RANDOM_NUMBER

# 创建固定长度的伪随机数
RANDOM_NUMBER=$((RANDOM % $UPPER_LIMIT + 1))

# 返回固定长度 $RANDOM_NUMBER 的值

echo $RANDOM_NUMBER
}
