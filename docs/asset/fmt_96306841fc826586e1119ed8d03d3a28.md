## 权限检查

```c
/* 此函数决定是否允许操作
 *（返回零）或不允许（返回非零），
 * 后指示为什么不允许。 */
操作可以是以下值之一：
* 0 执行（运行“文件” - 在我们的案例中没有意义）
* 2 写入（输入内核模块）
* 4 读取（从内核模块输出）
```

```c
/* 这是检查文件的真实函数
 * ls -l 返回的权限仅供
 * 参考，可以在这里覆盖。 */
```

```c
static int module_permission(struct inode *inode, int op)
{
    /* 我们允许所有人从我们的模块读取，
     * 只有 root (uid 0) 可以写入 */
    if (op == 4 || (op == 2 && current->uid == 0))
        return 0;

    /* 如果是其他情况，访问被拒绝 */
    return -EACCES;
}
```

## 注册 /proc 文件

/* 用于注册为 /proc 文件的结构，
* 包含所有相关函数的指针。 */

```c
/* 我们 proc 文件的文件操作。在这里
 * 我们放置在尝试对我们的文件进行操作时调用的
 * 函数指针。NULL 意味着我们不想处理某些操作。 */
static struct file_operations File_Ops_4_Proc_File = {
    NULL,        /* seek */
    module_output, /* 从文件读取 */
    module_input,  /* 写入文件 */
    NULL,        /* readdir */
    NULL,        /* select */
    NULL,        /* ioctl  */
    NULL,        /* mmap */
    module_open, /* 当 /proc 文件打开时调用 */
#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
    NULL,        /* flush */
#endif
    module_close, /* 关闭时调用 */
};
```

```c
/* 我们 proc 文件的 inode 操作。我们需要它，
 * 因为这样我们就有地方可以指定要使用的
 * 文件操作结构体，以及我们用于权限的
 * 函数。也可以指定在对 inode 进行其他
 * 操作时调用的函数（尽管我们没费心，直接放了 NULL）。 */
static struct inode_operations Node_Ops_4_Proc_File = {
    &File_Ops_4_Proc_File,
    NULL, /* create */
    NULL, /* lookup */
    NULL, /* link */
    NULL, /* unlink */
    NULL, /* symlink */
    NULL, /* mkdir */
    NULL, /* rmdir */
    NULL, /* knod */
    NULL, /* rename */
    NULL, /* readlink */
    NULL, /* follow_link */
    NULL, /* readpage */
    NULL, /* writepage */
    NULL, /* bmap */
    NULL, /* truncate */
    module_permission /* check for permissions */
};
```
