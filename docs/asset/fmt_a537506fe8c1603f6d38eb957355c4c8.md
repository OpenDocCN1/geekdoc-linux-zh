function showCard {
  # 这使用除法和取模来确定花色和点数，尽管
  # 在这个游戏中只有点数重要。尽管如此，演示是
  # 的，所以这有助于界面变得漂亮。
  card=$1
  if [ $card -lt 1 -o $card -gt 52 ]; then
    echo "错误的牌值: $card"
    exit 1
  fi

  # 除法和取模——看，所有的数学计算都浪费了！
  suit="$(( ($card - 1 ) / 13 + 1 ))"
  rank=$(( $card % 13))

  case $suit in
    1 ) suit="Hearts" ;;
    2 ) suit="Clubs" ;;
    3 ) suit="Spades" ;;
    4 ) suit="Diamonds" ;;
    *) echo "错误的花色值: $suit"
  esac

  case $rank in
    0 ) rank="King" ;;
    1 ) rank="Ace" ;;
    11 ) rank="Jack" ;;
    12 ) rank="Queen" ;;
  esac

  cardname="$rank of $suit"
}
