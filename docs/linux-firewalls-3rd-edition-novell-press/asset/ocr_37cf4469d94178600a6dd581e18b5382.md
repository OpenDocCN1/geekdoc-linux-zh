# APPENDIX B: Firewall Examples and Support Scripts

```
if [ "$CONNECTION_TRACKING" = "1" ]; then
$IPT -A local-tcp-client-request -p tcp \
-d $POP_SERVER --dport 110 \
-m state --state NEW \
-j ACCEPT
fi

$IPT -A local-tcp-client-request -p tcp \
-d $POP_SERVER --dport 110 \
-j ACCEPT

$IPT -A remote-tcp-server-response -p tcp ! --syn \
-s $POP_SERVER --sport 110 \
-j ACCEPT
```
