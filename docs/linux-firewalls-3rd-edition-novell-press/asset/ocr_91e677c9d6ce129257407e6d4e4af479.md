```

```

# Incoming ping

```
if [ "$CONNECTION_TRACKING" = "1" ]; then
$IPT -A EXT-icmp-in -p icmp \
-s $MY_ISP \
--icmp-type echo-request \
-m state --state NEW \
-j ACCEPT
fi
```

```
$IPT -A EXT-icmp-in -p icmp \
--icmp-type echo-request \
-s $MY_ISP -j ACCEPT
```

```
$IPT -A EXT-icmp-out -p icmp \
--icmp-type echo-reply \
-d $MY_ISP -j ACCEPT
```
