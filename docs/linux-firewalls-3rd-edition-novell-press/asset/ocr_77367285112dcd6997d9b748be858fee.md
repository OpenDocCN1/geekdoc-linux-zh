# APPENDIX B: Firewall Examples and Support Scripts

```
########################################################
# ICMP Control and Status Messages

# Log and drop initial ICMP fragments
SIPT -A INPUT --fragment -p icmp -j LOG \
    --log-prefix 'Fragmented incoming ICMP: '
SIPT -A INPUT --fragment -p icmp -j DROP

SIPT -A OUTPUT --fragment -p icmp -j LOG \
    --log-prefix 'Fragmented outgoing ICMP: '
SIPT -A OUTPUT --fragment -p icmp -j DROP

SIPT -A FORWARD --fragment -p icmp -j LOG \
    --log-prefix 'Fragmented forwarded ICMP: '
SIPT -A FORWARD --fragment -p icmp -j DROP

SIPT -A INPUT -p icmp \
    --icmp-type source-quench -d $DMZ_IPADDR -j ACCEPT

SIPT -A OUTPUT -p icmp \
    --icmp-type source-quench -j ACCEPT
SIPT -A FORWARD -p icmp \
    --icmp-type source-quench -j ACCEPT

SIPT -A INPUT -p icmp \
    --icmp-type parameter-problem -j ACCEPT

SIPT -A OUTPUT -p icmp \
    --icmp-type parameter-problem -j ACCEPT

SIPT -A FORWARD -p icmp \
    --icmp-type parameter-problem -j ACCEPT

SIPT -A INPUT -p icmp \
    --icmp-type destination-unreachable -j ACCEPT

SIPT -A OUTPUT -o SLAN_INTERFACE -p icmp \
    --icmp-type destination-unreachable -d SLAN_ADDRESSES -j ACCEPT

SIPT -A FORWARD -o SLAN_INTERFACE -p icmp \
    --icmp-type destination-unreachable -d SLAN_ADDRESSES -j ACCEPT

SIPT -A OUTPUT -p icmp \
    --icmp-type fragmentation-needed -j ACCEPT
```
