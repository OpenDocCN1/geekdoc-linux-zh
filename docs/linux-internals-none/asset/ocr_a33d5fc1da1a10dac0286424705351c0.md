The alarm and/or interrupt frequency are programmed into the RTC via ioctl(2) calls, as listed in include/linux/mc146818rtc.h (see the data structure below).

The following user program (supplied with the kernel documentation with RedHat's distribution) best illustrates how to use ioctls to control the real time clock:

```
```
/
* Real Time Clock Driver Test/Example Program
*
* Compile with:
* gcc -s -Wall -Wstrict-prototypes rcttest.c -o rcttest
*

* included herein by reference.

#include <stdio.h>
#include <linux/mc146818rtc.h>
#include <sys/ioctl.h>
#include <sys/time.h>
#include <sys/types.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>

void main(void) {

int i, fd, retval, irqcount = 0;
unsigned long tmp, data;
struct rtc_time rtc_tm;

fd = open ("/dev/rtc", O_RDONLY);

if (fd == -1) {
    perror("/dev/rtc");
    exit(errno);
}

fprintf(stderr, "\n\t\t\tRTC Driver Test Example.\n\n");

/* Turn on update interrupts (one per second) */
retval = ioctl(fd, RTC_UIE_ON, 0);
if (retval == -1) {
    perror("ioctl");
    exit(errno);
}

fprintf(stderr, "Counting 5 update (1/sec) interrupts from reading /dev/rtc:");
