#!/bin/sh
validtype() {
    if [ "$1" = "TYPEA" ] ||
       [ "$1" = "YPEB" ] ||
       [ "$1" = "TYPEC" ] ||
       [ "$1" = "TYPE" ] ||
       [ "$1" = "YPEE" ];
    then
        exit 0
    else
        exit 1
    fi
}
