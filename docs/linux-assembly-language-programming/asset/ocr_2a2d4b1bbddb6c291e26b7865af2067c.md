static inline char char_fill( char x)
{
/* Convert four-bit code into a C character */
char y;
if (x == 15) y = 'F';
else if (x == 14) y = ',';
else if (x == 13) y = 0; /* Termination flag!! */
else if (x == 12) y = '-';
else if (x == 11) y = 'B';
else if (x == 10) y = 'A'; /* Not used. */
else if (x < 10) y = x + '0';
else y = 0; /* Keep the compiler happy. */
return y;
}
