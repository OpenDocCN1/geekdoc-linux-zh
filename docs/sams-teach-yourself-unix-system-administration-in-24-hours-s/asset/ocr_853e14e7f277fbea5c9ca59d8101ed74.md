our %domain_cache;
sub domain {
    my $ip = shift;
    if ( !defined $domain_cache{$ip} ) {
        my @quads = split( /\./, $ip );
        my $domain = gethostbyaddr( pack('C4', @quads), AF_INET );
        if ( $domain ) {
            my @segments = split( /\./, $domain );
            $domain_cache{$ip} = join( '.', @segments[2..-1] );
        } else {
            $domain_cache{$ip} = $ip;
        }
    }
    return $domain_cache{$ip};
}
