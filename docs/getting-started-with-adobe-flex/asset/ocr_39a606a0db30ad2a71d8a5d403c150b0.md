```actionscript
private function displayGDP (data:Object,  field:String,  index:Number, 
percentValue:Number):String {
	var temp:String= (" " + percentValue).substr(0,6);
	return data.Country + " : " + '\n' + "Total GDP: " + data.GDP + 
" Billion $" + '\n' + temp + "%";
}
```
