The int_settimer() function, shown in Listing 6-11, sets the interval timer value. three types of interval timers: ITIMER_REAL, which counts down in real time; ITIMER_VIRTUAL, which counts down in while the process is running, and the ITIMER_PROF, which counts down both the virtual timer and the real itimer for kernel profiling.

# Listing 6-11: Viewing the int_settimer() function

int_settimer(int which, struct itimerval *value, struct itimerval *ovalue)
{
    register unsigned long i, j;
    int k;

    i = tvtojiffies(&value->it_interval);
    j = tvtojiffies(&value->it_value);
    if (ovalue && (k = __getitimer(which, ovalue)) < 0)
        return k;
    switch (which) {
    case ITIMER_REAL:
        del_timer(&current->real_timer);
        current->it_real_value = j;
        current->it_real_incr = i;
        if (!j)
            break;
        i = j + jiffies;
        /* check for overflow.. */
        if (i < j)
            i = ULONG_MAX;
        current->real_timer.expires = i;
        add_timer(&current->real_timer);
        break;
    case ITIMER_VIRTUAL:
        if (j)
            j++;
        current->it_virt_value = j;
        current->it_virt_incr = i;
        break;
    case ITIMER_PROF:
        if (j)
            j++;
        current->it_prof_value = j;
        current->it_prof_incr = i;
        break;
    default:
        return -EINVAL;
