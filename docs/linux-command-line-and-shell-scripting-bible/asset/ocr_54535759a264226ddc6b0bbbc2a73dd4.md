```
function menu {
  clear
  echo
  echo -e "\t\tSys Admin Menu\n"
  echo -e "\t1. Display disk space"
  echo -e "\t2. Display logged on users"
  echo -e "\t3. Display memory usage"
  echo -e "^[[1m0. Exit program\n\^[[44:33m"
  echo -en "\tEnter option: *"
}
```

```
echo "^[[44:33m"
while [ 1 ];
  menu
  case $option in
    0)
      break ::
    1)
      diskspace ::
    2)
      whoseon ::
    3)
      memusage ::
    *)
      clear
      echo -e "^[[5m\t\tSorry, wrong selection^[[0m^[[44:33m":
      echo -en "\n\n\tHit any key to continue"
      read -n 1 line
  esac
done
echo "^[[0m"
clear
```

The menu appear in yellow text on a blue background. Note that it's important to your foreground and background colors before using the clear command; otherwise, the clear command will paint the terminal emulation screen with the default colors, making the menu items not blend in with the rest of the background.
