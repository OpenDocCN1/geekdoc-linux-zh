# Display any errors

if test -s "$ERRORS"; then
    printf "&nbsp;<br />"
    printf "<table border=\"1\" summary=\"Error\">"
    printf "<tr bgcolor=\"White\"><td>"
    printf "<font color=\"Black\" size=\"3\">"
    printf "<p><b>Errors</b><br />"
    printf "</pre>"
    cat "$ERRORS"
    printf "</pre>"
    printf "&nbsp;<br />"
    printf "<p>If the problem persists, contact tech support.</p>"
    printf "</font></td></tr></table>"
    rm -f "$ERRORS"
fi
