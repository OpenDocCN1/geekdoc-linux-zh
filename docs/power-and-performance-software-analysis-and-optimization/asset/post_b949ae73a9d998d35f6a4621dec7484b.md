
## 14.2 Querying cache topology

### CPUID

As essentially any relevant processor information, the cache details on Intel Architecture are exposed via the CPUID instruction. In fact, multiple CPUID leaves report information about the cache.

The Basic CPUID Information leaf, EAX = 2, reports the type and size of each cache, TLB, and hardware prefetcher. Each of the registers is listed with a descriptor corresponding to an exact description of one of the resources. Multiple CPUID invocations may be necessary in order to retrieve all of the information. In order to encode such information in few bytes, each descriptor is merely a number, referencing a table in the Intel® Software Developer Manual (SDM) that describes the full configuration. By this, in order to properly interpret entries dynamically, a copy of the data in that table from the SDM must be included in the code.

The Deterministic Cache Parameters Leaf, EAX = 4, reports the properties of a specific cache. Each cache is identified by an index number, which is selected by the value of the ECX register upon invocation of CPUID. This index can be continuously incremented until a special NULL type is returned, indicating that all caches have been iterated. Unlike the Basic CPUID Information leaf, this leaf enters the information, such as the cache line size, number of ways, and number of sets, in bitfields returned in the registers. Because it doesn't require a pre-populated table, this approach yields itself better than the other leaf. Listing 14.1 provides an example of how to iterate the caches using this leaf, and how the report should be interpreted.

```c
#include <stdio.h>
#include <assert.h>
#ifdef _PIC_
static inline void cpuid(unsigned *restrict const a, unsigned *restrict const b, unsigned *restrict const c, unsigned *restrict const d) {
	asm __volatile (
	chg %%ebx, %%l\n"
	cpuid\n"
	chg %%1, %%ebx\n"
	"+a" (*a),
	"+b" (*b),
	"+c" (*c),
	"+d" (*d)
	);
#else /* Non PIC *\
static inline void cpuid(unsigned *restrict const a, unsigned *restrict const b).

25  [
26      :asm __volatile__ (

33  )

36  enum cache_type {\  /* From Table 3-17 of Volume 2 SDM */
37      TYPE_NULL = 0,
38      TYPE_DATA = 1,

40      TYPE_UNIF = 3,
41      TYPE_MAX = 4;

44  static const char *const type_str[TYPE_MAX] = {
45      [TYPE_NULL] = "NULL",
46      [TYPE_DATA] = "Data Cache",
47      [TYPE_INST] = "Instruction Cache",
48      [TYPE_UNIF] = "Unified Cache"
49  };

50  union ax {
52      unsigned e;
53      struct cache {\  /* From Table 3-17 */
54          unsigned type : 5;
55          unsigned lvl : 3;
56      } /* Skip Rest */
57      } b;
58  };

60  union bx {
61      unsigned e_sz;
62      struct cache_sz {\  /* From Table 3-17 */
63          unsigned linesz_ml : 12;
64          unsigned parts_ml : 10;
65          unsigned ways_ml : 10;
66      } b;
67  };

69  void iterate_caches(void)
unsigned cache_idx = 0;
static_assert(sizeof(union ax) == sizeof(unsigned),
    "bitfield A wrong");
static_assert(sizeof(union bx) == sizeof(unsigned),
    "bitfield B wrong");

while (1) {
    const char *units[] = { " B", "KB", "MB", "GB",
    NULL };

    union ax;
    union bx;
    unsigned cx, dx, cache_sz;
    const char *unit_str;

    ax.e = 0x04;
    cx = cache_idx;
    cpuid(&ax.e, &bx.e, &cx, &dx);

    if (ax.b.type == TYPE_NULL)
        break;

    cache_sz = (bx.b.ways_ml + 1) * (bx.b.parts_ml + 1 *
        (bx.b.lines_ml + 1) * (cx + 1);

    unit_str = &units[0];
    while (cache_sz >= 1024) {
        if (*unit_str + 1) == NULL)
            cache_sz /= 1024;
            unit_str++;
    }

    printf("%Index %d %3d %s Ld %s [Line Size: %dB]\n",
        cache_idx, cache_sz, unit_str, ax.b.lvl.
        type_str[ax.b.type], bx.b.lines_ml + 1);

    cache_idx++;
}
```
