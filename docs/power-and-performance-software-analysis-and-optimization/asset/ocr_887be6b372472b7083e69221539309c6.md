gettimeofday(2) system call is essentially identical to clock_gettime(2)
using CLOCK_REALTIME. The timezone parameter deprecated, and should be set to
NULL.

```
#include <stdio.h>
#include <math.h>
#include <sys/time.h>

static inline double time_diff_ms(const struct timeval *const s,\ const struct timeval *const e)
{
    double s_ms, e_ms;

    s_ms = (s)->tv_sec * 1000) + (s)->tv_use / 1000);
    e_ms = (e)->tv_sec * 1000) + (e)->tv_use / 1000);
    return e_ms - s_ms;
}

static void run_bench(double *avg, double *std_dev)
{
    size_t i;

    *avg = 0.0;
    *std_dev = 0.0;
    for (i = 0; i < NUM_RUNS; i++) {
        double measure, t;
        struct timeval s, e;

        gettimeofday(&s, NULL);
        /* do benchmarked code */
        gettimeofday(&e, NULL);

        measure = time_diff_ms(&s, &e);
        *avg += measure / (i + 1);
        *std_dev += (t * (measure - *avg));
    }
    *std_dev = sqrt(*std_dev) / (NUM_RUNS - 1));
}
```
