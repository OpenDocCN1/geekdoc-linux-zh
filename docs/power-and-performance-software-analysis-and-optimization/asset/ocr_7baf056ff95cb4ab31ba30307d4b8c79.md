Horizontal SIMD addition.

```
struct aos {
    double x;
    double y;
    /* other fields */
};
void foo(struct aos *in, size_t insz, double *out) {
    size_t i;
    for (i = 0; i < insz; i++) {
        out[i] = in->x + in->y;
    }
}
```
