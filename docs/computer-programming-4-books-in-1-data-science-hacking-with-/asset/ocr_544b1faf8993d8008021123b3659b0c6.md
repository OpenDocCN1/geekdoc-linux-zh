import plotly
import plotly.graph_objs as go

import pandas as pd

data = pd.read_csv('train.csv')
corpmat = data.corr()

trace = go.Heatmap(z=corpmat, x=corpmat.columns.tolist(), y=corpmat.columns.tolist())
plot_data = [trace]
plotly.offline.plot(plot_data, filename='basic-heatmap')
