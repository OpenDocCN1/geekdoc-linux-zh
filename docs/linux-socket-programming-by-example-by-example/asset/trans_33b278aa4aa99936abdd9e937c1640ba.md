
- 3. 辅助数据只能在接收数据时才能接收。因此，在第 26 行声明了一个小的数据缓冲区以接收部分数据。
- 4. 消息头 `msg` 和缓冲区 `buf` 被清零（第 31 行和第 32 行）。为了效率，您可以选择省略此步骤，但在调试时建议保留。
- 5. 由于不需要套接字地址（这里使用的是流式套接字），消息头成员 `msg_name` 和 `msg_namelen` 被置为空。如果您使用的是数据报套接字，您希望接收套接字地址，以便知道数据报来自何处。
- 6. 指向 I/O 向量的指针在第 43 行和第 44 行建立。此外，还确定它恰好有一个条目。
- 7. I/O 向量本身在第 50 行和第 51 行建立。第一个 I/O 向量条目 `iov[0]` 被设置为将数据接收进缓冲区 `d_buf`（第 50 行），最大尺寸由第 51 行指示。
- 8. `recvmsg(2)` 函数在第 62 到 64 行被调用。如果发生错误，该函数通过第 67 行的 return 语句退出。
- 9. 使用 `CMSG_FIRSTHDR()` 和 `CMSG_NXTHDR()` 宏（第 73 到 84 行）遍历辅助数据。如果发现级别为 `SOL_SOCKET` 且类型为 `SCM_RIGHTS` 的辅助数据对象，函数在第 82 行返回文件描述符。
- 10. 如果套接字服务器拒绝了请求或失败，数据缓冲区将包含服务器 errno 值的 `(int)` 值。该函数在第 93 行将此 errno 值存储到调用进程的 errno 变量中。如果接收的数据不是 `sizeof(int)`，则函数将 errno 值设置为 `ENOENT`（第 95 行）。除非联系到了错误的服务器，或者服务器存在 bug，否则这种情况绝不应发生。
- 11. 当错误已填入 errno 时，返回值为 -1（第 97 行）。

## `sock_srv` 服务器程序

套接字服务器是将以 root 身份运行的服务器程序。这使其具备在 80 端口创建和绑定套接字的能力。然而，这是唯一以该权限运行的组件，通过让服务器检查请求的凭据，您将完全控制谁获得什么。列表 17.8 显示了套接字服务器的源代码。

### **列表 17.8：** *sockserv.c*——套接字服务器

```c
1: /* * sockserv.c:
2:
3: * 这个简单的服务器将把套接字
4: * 提供给有效的接收者：
5: */
6: #include "common.h"
7: #include <pwd.h>
8:
9: /*
10:    * 检查用户的访问权限：
11:    *
12:    * 返回：
13:    *   ptr    如果授予，指向 (struct passwd *)。
14:    *   NULL   拒绝访问。
15:    */
16: static struct passwd *
17: check_access(
18:         int port,          /* 请求的端口 */
19:         struct ucred *pcred, /* 用户凭据 */
20:         char **uidlist) { /* 有效用户列表 */
21:         int x;
22:         struct passwd *pw; /* 用户 passwd 条目 */
23:
24:         /*
25:          * 在 /etc/passwd 文件中
26:          * 查找用户的 uid 号：
27:          */
28:         if ( ( pw = getpwnuid(pcred->uid) ) == 0 ) {
29: /*
30:          * 确保请求来自
31:          * 可接受的用户之一：
32:          */
33:         for ( x=0; uidlist[x]; ++x )
34:             if ( !strcmp(uidlist[x],pw->pw_name) )
35:                 break;
36:             if ( !uidlist[x] )
37:                 pw = 0;    /* 拒绝访问 */
38:         }
39:
40:         /*
41:          * 筛选端口号。对于这个演示，
42:          * 只允许 80 端口。
43:          */
44:         if ( port != 80 )
45:             pw = 0;         /* 拒绝访问 */
46:
47:    return pw;  /* 授予则返回 NULL 或指针 */
48: }
49:
50: /*
51:  * 访问已授予：将套接字
52:  * 发送给客户端。
53:  *
54:  * 参数：
55:  *    c      客户端套接字
56:  *    port   请求的端口
57:  *
58:  * 返回：
59:  *    0       成功
60:  *    -1      失败：检查 errno
61:  */
62: static int
63: grant_access(int c,int port) {
64:     int z;
65:     int fd = -1;             /* 新套接字 fd */
66:     int b = TRUE;            /* 布尔 TRUE */
67:     struct sockaddr_in addr; /* 工作地址  */
68:
69:     /*
70:      * 创建一个新的 TCP/IP 套接字：
71:      */
72:     fd = socket(PF_INET,SOCK_STREAM,0);
73:     if ( fd == -1 ) {
74:        perror("socket(2)");
75:        goto errxit;
76:     }
77:
78:     /*
79:      * 打开 SO_REUSEADDR：
80:      */
81:     z = setsockopt(fd,SOL_SOCKET,
82:                    SO_REUSEADDR,&b,sizeof b);
83:     if ( z == -1 )
84:        bail("setsockopt(2)");
85:
86:     /*
87:      * 创建要绑定的地址：
88:      */
89:     memset(&addr,0,sizeof addr);
90:     addr.sin_family = AF_INET;
91:     addr.sin_port = ntohs(port);
92:   addr.sin_addr.s_addr = ntohl(INADDR_ANY);   
93:
94:   /*
95:    * 绑定请求的地址：
96:    */
97:   z = bind(fd,
98:        (struct sockaddr *)&addr,
99:        sizeof addr);
100:  if ( z == -1 ) {
101:     fprintf(stderr,"s: binding port %d\n",    
102:            strerror(errno),port);
103:     goto errxit;
104:  }
105:
106:   /*
107:    * 将 fd 发送回
108:    * 请求的客户端：
109:    */
110:   z = send_fd(c,fd,NULL,0);
111:  if ( z == -1 ) {
112:     perror("send_fd()");
113:     goto errxit;
114:  }
115:
116:   close(fd);      /* 结束对 fd 的使用 */
117:   return 0;                  /* 成功*/
118:
119:  errxit:
120:   z = errno;               /* 保存 errno */
121:   if ( fd )
122:     close(fd);         /* 释放套接字 */
123:    errno = z;           /* 恢复 errno */
124:    return -1;
125:  }
126:
127:  /*
128:   * 处理已连接客户端的请求：
129:   */
130:  void
131:  proc_client(
132:     int c,             /* 客户端套接字 */
133:     char *uidlist      /* 有效用户列表  */
134:    ) {
135:    int z;
136:    int er;            /* 捕获的 errno 值 */
137:    int b = TRUE;      /* 布尔：真 */
138: struct ucred cred; /* 客户端凭据 */
139: short port; /* 请求的端口 */
140: struct passwd *pw; /* 用户 passwd 条目 */
141:
142: /*
143:  * 现在确保我们可以
144:  * 在此套接字上接收凭据：
145:  */
146: z = setsockopt(c,
147:               SOL_SOCKET,
148:               SO_PASSCRED,
149:               &b,
150:               sizeof b);
151: if ( z )
152:     bail("setsockopt(2)");
153:
154: /*
155:  * 接收带有
156:  * 用户凭据的请求：
157:  */
158: z = recv_cred(c, /* 套接字 */
159:               &cred, /* 返回的凭据 */
160:               &port, /* 返回的端口号 */
161:               sizeof port, /* 数据大小 */
162:               NULL, 0); /* 无套接字地址 */
163:
164: if ( z == -1 )
165:     perror("recv_cred()");
166:
167: /*
168:  * 现在检查访问权限。 如果 pw
169:  * 作为非 NULL 返回，则请求有效。
170:  */
171: pw = check_access(port,&cred,uidlist);
172:
173: if ( pw ) {
174:     if ( grant_access(c,port) ) {
175:         close(c);
176:         return; /* 请求成功 */
177:     }
178:     /* 失败 */
179:     er = errno; /* 捕获原因 */
180: } else {
181:     /*
182:     * UserId 未知，或不在
183:     * 特权列表中：
184:     */
185:     err = EACCES;    /* 权限被拒绝 */
186: }
187: 
188: /*
189:  * 如果请求失败
190:  * 或被拒绝，控制流到达此处：
191:  *
192:  * 这里我们只是发送错误
193:  * 代码，而不带文件
194:  * 描述符。这种缺少 fd
195:  * 将被客户端检测到。
196:  */
197: do {
198:     z = write(c,&err,sizeof er);
199: } while ( z == -1 && errno == EINTR );
200: 
201:     if ( z == -1 )
202:         perror("write(2)");
203: }
204: 
205: /*
206:  * 主程序：
207:  */
208: int
209: main(int argc,char **argv) {
210:     int z;
211:     int s;        /* 服务器 UDP 套接字 */
212:     int c;        /* 客户端套接字 */
213:     int alen;           /* 地址长度 */
214:     struct sockaddr_un asrv;  /* 服务器 @ */
215:     struct sockaddr_un acint; /* 客户端 @ */
216: 
217: /*
218:  * 确保指定了 userid：
219:  */
220:     if ( argc < 2 ) {
221:         fputs("必须至少有 1 个 userid.\n",
222:             stderr);
223:         exit(1);
224:     }
225: 
226: /*
227:  * 创建一个 Unix 套接字：
228:  */
229:     s = socket(PF_LOCAL,SOCK_STREAM,0);
```
