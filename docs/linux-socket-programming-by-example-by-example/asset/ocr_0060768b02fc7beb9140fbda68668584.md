# sockaddr_x25 and x25_address

```
```c
#include <linux/x25.h>

struct sockaddr_x25 {
    sa_family_t  sx25_family;  /* Must be AF_X25 */
    x25_address  sx25_addr;     /* X.121 Address */
};

typedef struct {
    char  x25_addr[16];
} x25_address;
```

# sockaddr_in6 and in6_addr

```
```c
struct sockaddr_in6 {
    sa_family_t  sin6_family;   /* port # */
    uint16_t     sin6_port;     /* flow info */
    uint32_t     sin6_flowinfo; /* IPv6 address */
    struct in6_addr sin6_addr;
};

struct in6_addr {
    union {
        uint8_t    u6_addr8[16];
        uint16_t   u6_addr16[8];
        uint32_t   u6_addr32[4];
    } in6_u;
};
```

# sockaddr_atalk and at_addr

```
```c
#include <netatalk/at.h>

struct sockaddr_atalk {
    sa_family_t  sat_family;
    u_char       sat_port;      /* port */
    struct at_addr sat_addr;    /* net/node */
};

struct at_addr {
    u_short      s_net;
    u_char       s_node;
};
```
