tcp_ack_snd_check() net/ipv4/tcp_input.c

```
* tcp_ack_snd_check(sk, offo_possible) checks for various cases where acknowledements can be sent. Also, it checks the type of acknowledgement (i.e. whether it should be quick or delayed)
static __inline void tcp_ack_snd_check(struct sock *sk) {
    struct tcp_opt *tp = &(sk->tp_info.of_tcp);
    if (!tp->ack_scheduled(tp)) {
        /* We nest a data segment already. */
        return;
    }
    /* More than one full frame received... */
    if ((tp->rcv_nxt - tp->rcv_wup) > tp->rcv_nd && right edge of window advances far enough.
        /* tcp_recvmsg(sk) will send ack otherwise). Or... */
        && _tcp_select_window(sk) >= tp->rcv_nd ||
        /* We ack each frame or... */
        tcp_in_quack_mode(tp) ||
        /* We have out of data. */
    {
        skb_peek(&tp->out_of_order_queue) != NULL
        tcp_snd_ack(sk); /* Then ack it now */
    }
else {
        tcp_snd_delayed_ack(sk); /* Else, send delayed ack. */
    }
```
