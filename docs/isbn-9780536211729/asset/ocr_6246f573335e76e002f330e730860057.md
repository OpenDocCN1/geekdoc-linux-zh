while true
do
  for i in eth0 eth1
    if ifconfig $i | grep PROMIS > /dev/null
    then
      (echo $ Promisc;fpromisc) \\
      2>&1 Mail -s PROMISCUOUS \\
      sysadmin sysadmin@homesys.com
    fi
    sleep 1800
done
