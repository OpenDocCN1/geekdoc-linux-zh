static void
hr_min_sec(char *buf, long sec) {
	if (sec >= 3600) {
		(void) sprintf(buf, "%ld:%21d:%26d",
		sec / 3600, sec % 3600 / 60, sec % 60);
	} else if (sec > 0) {
		(void) sprintf(buf, "%ld:%21d",
		sec / 60, sec % 60);
	} else {
		(void) sprintf(buf, "%ld", sec);
	}
}
