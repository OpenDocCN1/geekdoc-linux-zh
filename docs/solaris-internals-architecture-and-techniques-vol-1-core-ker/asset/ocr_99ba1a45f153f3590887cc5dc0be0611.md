```
static char    *command;
static char    procname[64];
```

```
main(int argc, char **argv)
{
    int ctlfd;
    long ctl[2];
    pid_t pid;
    struct siginfo info;
    int status;

    if ((strchr(argv[0], '/')) != NULL)
        command++;
    else
        command = argv[0];

    if (argc <= 1) {
        (void) fprintf(stderr, "usage:%s command [args ...]\\n", command);
        (void) fprintf(stderr, "      (time command using microstate accounting)\
");
        return (1);
    }

    switch (pid = fork()) {
    case -1:
        (void) fprintf(stderr, "%s: cannot fork\nn", command);
        return (2);
    case 0:
        /* newly created process */
        /* open the /proc ctl file and turn on microstate accounting */
        (void) sprintf(procname, "/proc/%d/ctl", getpid());
        ctlfd = open(procname, O_WRONLY);
```
