/*
 * Turn on microstate accounting, and print all field resource
 * usage and microstat accounting fields when process terminates.
 * 
 * Borrowed largely from ptime.c
 * (Thanks Roger Faulkner and Mike Shapiro)

 * Usage: msacct command
 */

#include <sys/types.h>
#include <sys/time.h>
#include <procs.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <string.h>
#include <errno.h>
#include <math.h>
#include <wait.h>
#include <signal.h>

static int     look(pid_t);
static void    hr_min_sec(char *, long);
static void    prtime(char *, timestruc_t *);
static int     perr(const char *);
static void    tsadd(timestruc_t *result, timestruc_t *a, timestruc_t *b);
static void    tssub(timestruc_t *result, timestruc_t *a, timestruc_t *b);
static char    *command;
static char    procname[64];

main(int argc, char **argv)
{
    int ctlfd;
    long ctl[2];
    pid_t pid;
    struct siginfo info;
    int status;

    if ((strchr(argv[0], '/')) != NULL)
        command++;
    else
        command = argv[0];

    if (argc <= 1) {
        (void) fprintf(stderr, "usage:%s command [args ...]\n", command);
        (void) fprintf(stderr, "      (time command using microstate accounting)\n");
        return (1);
    }

    switch (pid = fork()) {
    case -1:
        (void) fprintf(stderr, "%s: cannot fork\n", command);
        return (2);
    case 0:
        /* newly created process */
        /* open the /proc ctl file and turn on microstate accounting */
        (void) sprintf(procname, "/proc/%d/ctl", getpid());
        ctlfd = open(procname, O_RDWR);
        // ... (rest of logic follows)
    }
}
