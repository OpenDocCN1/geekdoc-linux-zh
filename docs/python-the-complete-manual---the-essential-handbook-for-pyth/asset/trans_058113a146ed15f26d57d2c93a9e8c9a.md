
```python
#!/usr/bin/env python
# 一个演示基础的 Python 程序
import sys

# 获取用户名
name = input('请输入你的名字： ')

# 打印问候
print(f'Hello, {name}!')

# 简单循环
for i in range(5):
    print(f'Iteration {i}')

# 函数定义
def greet(person):
    return f"Welcome, {person}!"

# 调用函数
print(greet("World"))

# 错误处理
try:
    number = int(input("输入一个数字： "))
    print(f"{number} 的平方是 {number**2}")
except ValueError:
    print("这不是一个数字!")

if __name__ == "__main__":
    print("程序结束。")
```
