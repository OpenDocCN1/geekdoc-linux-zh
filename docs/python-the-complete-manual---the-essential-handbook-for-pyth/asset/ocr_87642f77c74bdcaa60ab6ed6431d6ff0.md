```
import simpleguikit as simpleguikit
import random

w, h = 600, 400
tux = 20
pad_w = 0
pad_h = 80

def tux_spawn(right):
    global tux_pos, tux_vel
    tux_pos = [0, 0]
    tux_vel = [0, 0]
    tux_pos[0] = w/2
    tux_pos[1] = h
    if right:
        tux_vel[0] = random.randint(2, 4)
    else:
        tux_vel[0] = -random.randint(2, 4)
        tux_vel[1] = random.randint(1, 3)

def start():
```

```
def keydown(key):
    global paddle1_pos, paddle2_vel
    acc = 3
    if key == simplegui.KEY_MAP['w']:
        paddle1_vel[1] += acc
    elif key == simplegui.KEY_MAP['s']:
        paddle1_vel[1] -= acc
    elif key == simplegui.KEY_MAP['up']:
        paddle2_vel[1] += acc
    elif key == simplegui.KEY_MAP['down']:
        paddle2_vel[1] -= acc
```

```
frame = simplegui.create('Pong for Two', w, h)
frame.set_key_handler(keydown)
frame.set_key_handler(keyup)
frame.set_image_load('http://upload.wikipedia.org/wikipedia/commons/a/af/Tux.png')
```
