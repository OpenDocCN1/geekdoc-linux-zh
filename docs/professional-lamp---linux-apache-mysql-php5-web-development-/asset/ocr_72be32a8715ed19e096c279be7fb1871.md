# Variable Declaration

+   - PHP
+     `$variable = value;`
+   - ASP/VBScript
+     Dim variable
+   - ASP.NET/C#
+     type variable;
+   - JSP/Java
+     type variable;

# Constant Declaration

+   - PHP
+     define('NAME', value);
+   - ASP/VBScript
+     Const name = value
+   - ASP.NET/C#
+     const type name = value;
+   - JSP/Java
+     visibility final type = value;
