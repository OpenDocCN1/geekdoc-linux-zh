    kinfo_proc *procList = NULL;
    size_t *procCount = 0;

    m_processList.clear();

    if (!GetBSDProcessList(&procList, &procCount)) {
        if (procList) {
            for (int i = 0; i < procCount; i++) {
                Process proc;
                proc.SetPID(
                    procList[i].kp_proc.p_pid);
                proc.SetName(
                    procList[i].kp_proc.p_comm);
                m_processList.push_back(proc);
