# 如果用户提供 'all' 作为参数，则使用文件名和 `for..do` 语句来打印当前工作目录下的每一个 `.tex` 文件，一个接一个。

```bash
if [ "$1" = --all ]; then
  for LATEXFILE in *.tex; do
    echo "Printing $LATEXFILE"
    DVIFILE=$(echo $LATEXFILE | sed 's/.tex$/dvi/')

    latex "$LATEXFILE"
    dvips "$DVIFILE"
  done
  exit 0
fi
```

# 变量 `COUNTER` 将保存我们正在打印的序列中的当前数字。我们将从 1 开始，并在每次打印另一个文件时使用 `expr` 对其进行加 1。

```bash
COUNTER=1

while [ -f "$LATEXFILE$COUNTER.tex" ]; do
  echo "Printing $LATEXFILE$COUNTER.tex"
  latex $LATEXFILE$COUNTER.tex
  dvips $LATEXFILE$COUNTER.dvi
  COUNTER=$(expr $COUNTER + 1)

done
else

echo "似乎没有一个名为 $LATEXFILE 的文件。"
exit 1

fi
fi
```
