## 共享内存简介

我们现在调试图 12.10 中的程序，使用 Posix 内存信号量代替 Posix 信号量，并将此信号量存储在共享内存中。图 12.12 是新程序。

```c
#include <unipc.h>

struct shared {
    sem_t mutex;      /* 互斥锁：一个 Posix 内存信号量 */
    int count;         /* 计数器 */
};

int main(int argc, char **argv) {
    int fd, i, nloop;
    struct shared *ptr;

    if (argc < 2)
        err_quit("usage: <name> <loops>");
    nloop = atoi(argv[1]);

    /* 初始化为 0，映射到内存 */
    fd = open("/dev/shm", O_RDWR | O_CREAT, FILE_MODE);
    write(fd, &shared, sizeof(struct shared));
    ptr = mmap(NULL, sizeof(struct shared), PRO_READ | PRO_WRITE,
               MAP_SHARED, fd, 0);

    /* 初始化进程间共享的信号量 */
    sem_init(ptr->mutex, 1, 1);
    setvbuf(stdout, NULL); /* stdout 无缓冲 */
    if (Fork() == 0) {
        for (i = 0; i < nloop; i++) {
            sem_wait(ptr->mutex);
            printf("child: %d\n", ptr->count++);
            sem_post(ptr->mutex);
        }
        exit(0);
    }
    /* 父进程 */
    for (i = 0; i < nloop; i++) {
        sem_wait(ptr->mutex);
        printf("parent: %d\n", ptr->count++);
        sem_post(ptr->mutex);
    }
    exit(0);
}
```

图 12.12 计数器和信号量都在共享内存中。

## 4.4BSD 匿名内存映射
