sub wrap {
    # 创建行数组，每行
    # 指定字符数，并且仅在空白
    # 处折行
    # 获取限制行的字符数。
    $limit = pop(@_);
    # 调试
    # 输出 '进入 wrap 子程序'
    # 输出 '行长度限制为 $limit'
    # 读取文件，解析并放入数组。
    open(FILE, "<$file") or die "无法打开文件: $!";
    while (<FILE>) {
        $line = $_;
        # 调试
        # 输出 '行是:\n$line'
        # 如果超过限制，则折行。
        while (length($line) > $limit) {
            # 调试
            # 输出 '正在折行...';
            # 获取前 $limit + 1 个字符。
            $part = substr($line, 0, $limit+1);
            # 调试
            # 输出 '部分行为:\n$part'
            # 检查最后一个字符是否为
            # 空格。
            $last_char = substr($part,-1, 1);
            if (' ' eq $last_char) {
                # 如果是，打印剩余。
                # 调试
                # 输出 '最后一个字符是空格'
                substr($line, 0, $limit + 1) = '';
                substr($part,-1,1) = '';
                push(@newlines,'$part\n');
            } else {
                # 如果不是，在子行中找到最后一个空格
                # 并打印到那里。
                # 调试
                # 输出 '最后一个字符不是空格'
                # 移除超过 $limit 的字符
                substr($part,-1,1) = '';
                # 反转行以便于找到
                # 最后一个空格。
                $revpart = reverse($part);
                $index = index($revpart,'*');
                if ($index > 0) {
                    substr($line,0,$limit-$index) = '';
                    push(@newlines,
                        substr($part,0,$limit-$index) = "\n");
                } else {
                    # 行中没有空格，所以
                    # 打印到 $limit。
                    substr($line,0,$limit) = '';
                    push(@newlines,substr($part,0,$limit));
                }
            }
        }
    }
    push(@newlines,$line);
    close(FILE);
}
