// 使用新缓冲区再次调用 sysctl。如果我们

// 得到 ENOMEM 错误，丢弃缓冲区并

// 开始。

```c
if (err == 0) {
    err = sysctl((int *), sizeof(name) /
          sizeof(name)) - 1, result, &length,
          NULL, 0);
    if (err == -1) {
        err = errno;
    }
    if (err == 0) {
        done = true;
    } else if (err == ENOMEM) {
        assert(result != NULL);
        free(result);
        result = NULL;
        err = 0;
    }
}
```
