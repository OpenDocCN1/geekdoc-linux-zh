# rlogin

## 高级 Bash 脚本指南

```bash
for i in [list] missing...
  do ==> 所以是对命令行参数进行操作
  case "$i" in
    0*) ibase=2; # 二进制
    0x*[a-f]*[A-F]*) ibase=16; # 十六进制
    ibase=8; # 八进制
    [1-9]*) ibase=10; # 十进制
    *)
      echo "illegal number $i - ignored"
      continue;
  esac

  # 前缀，将十六进制数字转换为大写 (bc 需要此)
  number=$(echo "$i" | sed -e 's:0x:[0-9a-fA-F]//' | tr '[a-f]' '[A-F]')
  # 使用 \ 作为分隔符，而不是 ""
  # 将数字转换为十进制
  dec=$(echo "ibase=$ibase;$number" | bc) # 计算器工具
  case "$dec" in
    [0-9]*) # 数字正常
      continue;
    *) ) # 错误：忽略
  esac

  # 在同一行打印所有转换结果。
  # == 'here document' 命令列表输入到 'bc'
  bc <<EOF
    obase=16; "hex": $dec
    obase=10; "dec": $dec
    obase=8; "oct": $dec
    obase=2; "bin": $dec
EOF | sed -e 's/"//g'
done

while [ $# -gt 0 ];
  do
    case "$1" in
      -\) shift; break;
      -h) Usage; ==> 帮助信息。
         Usage;
         break;
      *) ) # 第一个数字
    esac
    ==> 对非法输入进行错误检查将会很有用。
  shift
done

if [ $# -gt 0 ];
  then
    PrintBases "$@" # 从标准 读取
  else
    while read line
      do
        PrintBases $line
      done
  fi
```

一种使用嵌入在 `[command substitution]` 块中的 `[here document]` 来调用 `bc` 的替代方法。当脚本需要向 `bc` 传递一组选项和命令时，这种方法特别适用。

```bash
#!/bin/bash
# 使用命令替换
# 结合 'here document' 来调用 'bc'

var1=$(bc << EOF
18.33 * 19.78
EOF)
echo $var1      # 362.56

# $(...) 记法同样有效。
v1=23.53
v2=17.881
v3=83.501
v4=171.63

var2=$(bc << EOF
scale = 4
a = ($v1 + $v2)
b = ($v3 + $v4)
a * b + 15.35
EOF)
echo $var2      # 593847.8452

var3=$(bc -1 << EOF
scale = 9
s ( 1.7 )
EOF)
# 返回 1.7 弧度的正弦值。
# "-1" 选项调用 'bc' 数学库。
echo $var3      # .991664810
```

# 示例 12–36. 安全删除文件

```bash
#!/bin/bash
# blotout.sh: 擦除文件的所有痕迹。
# 此脚本交替使用随机字节和零字节覆盖目标文件，
## 最后将其删除。
# 在此之后，即使是检查原始磁盘扇区
## 也无法发现原始文件数据。

PASSES=7
# 文件粉碎的次数。
BLOCKSIZE=1
# 与 /dev/urandom 进行 I/O 需要单位块大小，
# 否则你会得到奇怪的结果。
E_BADARGS=70
E_NOT_FOUND=71
E_CHANGED_MIND=72

if [ -z "$1" ]; then  # 未指定文件名。
then
    echo "Usage: basename $0 filename"
    exit $E_BADARGS
fi

file=1

if [ ! -e "$file" ]; 
then
    echo "File $file not found."
    exit $E_NOT_FOUND
fi

echo; echo -n "Are you absolutely sure you want to blot out $file? (y/n) "
read answer
case "$answer" in
[nN]) echo "Changed CHANGED_MIND, huh?"
    exit $E_CHANGED_MIND
    ;
*) echo "Blotting out $file.";
    break
esac

flength=$(ls -1 "$PASSES") | awk '{print $5}'  # 第 5 列是文件长度。

pass_count=1

echo

while [ "$pass_count" -le "$PASSES" ];
do
    echo "$pass_count"

    # 刷新缓冲区。
    sync

    dd if=/dev/urandom of=$file bs=$BLOCKSIZE count=$flength

    # 填充随机字节。

    sync

    # 再次刷新缓冲区。

    dd if=/dev/zero of=$file bs=$BLOCKSIZE count=$flength

    # 填充零。

    sync

    # 再次刷新缓冲区。

    let "$pass_count += 1"
done

rm -f $file  # 最后，删除乱码且粉碎的文件。

sync

# 最后刷新一次缓冲区。
```
