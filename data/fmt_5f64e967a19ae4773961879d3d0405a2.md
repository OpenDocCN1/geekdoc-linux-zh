import libardrone

def main():

    drone.speed = 0.5

    # 向左移动
    for i in range(5):
        drone.move_left()

    # 向前移动
    for i in range(5):

    # 让无人机转向相机

    # 再次转回来

        drone.turn_left()

    # 向右移动完成正方形
    for i in range(10):
        drone.move_right()

    # 360度全旋转

        drone.turn_left()

    # 返回起点

        drone.move_left()
        drone.move_backward()
