// path: kernel/exit.c
static inline int send_sig(long sig, struct task_struct *p, int priv)
{
    if (!p || sig < 1 || sig > 32) // 检查
        return -EINVAL;
    if (priv || ((current->uid == p->uid) || isset()))
        // 找到对应的位置
        p->signal |= (1 << (sig - 1));
        // 信号位图中的 SIGUSR1 信号。
        // processsig 进程的信号，然后将其设置为 1。
    else
        return -EPERM;
    return 0;
}
