
# 地址簿示例程序

```c
#include <stdio.h>
#include <stdlib.h>
#include <io.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>

/* num_lines 是屏幕显示的行数 */
#define num_lines  25

typedef struct
{
    char name[30];
    char address[30];
    char town[30];
    char county[30];
    char post[12];
    char telephone[15];
}
data;

data record;
int handle;

/* 函数原型 */

void ADD_REC(void);
void CLS(void);
void DISPDATA(void);
void FATAL(char *);
void GETDATA(void);
void MENU(void);
void OPENDATA(void);
int SEARCH(void);

void CLS()
{
int n;

for (n = 0; n < num_lines; n++)
puts(" ");
}

void FATAL(char *error)
{
printf("\nFATAL ERROR: %s", error);
exit(0);
}

void OPENDATA()
{
/* 检查数据文件是否存在，如果不存在则创建它 */
/* 否则，以读/写方式打开它并在文件末尾进行 */
handle = open("address.dat", O_RDWR | O_APPEND, S_IWRITE);

if (handle == -1)
{
handle = open("address.dat", O_RDWR | O_CREAT, S_IWRITE);
if (handle == -1)
FATAL("Unable to create data file");
}
}

void GETDATA()
{
/* 从操作员获取地址数据 */

CLS();

printf("Name ");
gets(record.name);
printf("\nAddress ");
gets(record.address);
printf("\nTown ");
gets(record.town);
printf("\nCounty ");
gets(record.county);
printf("\nPost Code ");
gets(record.post);
printf("\nTelephone ");
gets(record.telephone);
}

void DISPDATA()
{
/* 显示地址数据 */
char text[5];

CLS();

printf("Name %s",record.name);
printf("\nAddress %s",record.address);
printf("\nTown %s",record.town);
printf("\nCounty %s",record.county);
printf("\nPost Code %s",record.post);
printf("\nTelephone %s\n\n",record.telephone);

puts("Press RETURN to continue");
gets(text);
}

void ADD_REC()
{
/* 将新记录插入或附加到数据文件中 */
int result;

result = write(handle,&record,sizeof(data));

if (result == -1)
FATAL("Unable to write to data file");
}

int SEARCH()
{
char text[100];
int result;

printf("Enter data to search for ");
gets(text);
if (*text == 0)
return(-1);

/* 定位到文件开头 */
lseek(handle,0,SEEK_SET);

do
{
/* 将记录读入内存 */
result = read(handle,&record,sizeof(data));
if (result > 0)
{
/* 扫描记录以查找匹配的数据 */
if (strstr(record.name,text) != NULL)
return(1);
if (strstr(record.address,text) != NULL)
return(1);
if (strstr(record.town,text) != NULL)
return(1);
if (strstr(record.county,text) != NULL)
return(1);
if (strstr(record.post,text) != NULL)
return(1);
if (strstr(record.telephone,text) != NULL)
return(1);
}
}
while(result > 0);
return(0);
}

void MENU()
{
 int option;
 char text[10];

do
{
 CLS();
puts("\n\t\t\tSelect Option");
puts("\n\n\t\t\t1 Add new record");
puts("\n\n\t\t\t2 Search for data");
puts("\n\n\t\t\t3 Exit");
puts("\n\n\n\n");
gets(text);
option = atoi(text);

 switch(option)
{
 case 1 : GETDATA();
/* 定位到文件末尾以附加新记录 */
 lseek(handle,0,SEEK_END);
 ADD_REC();
 break;

 case 2 : if (SEARCH())
 DISPDATA();
 else
 {
 puts("NOT FOUND!");
 puts("Press RETURN to continue");
 gets(text);
 }
 break;

 case 3 : break;
 }
 while(option != 3);
}

void main()
{
 CLS();
 OPENDATA();
 MENU();
}
```

## 位字段

C 语言允许在结构体中包含小于 8 位的变量。这些变量被称为 *位字段*，可以是任何从 1 位开始声明的大小。声明位字段的一般形式如下：

```
类型 名称 : 位数;
```

例如，要声明一组状态标志，每个占据 1 位：

```c
typedef struct
{
    unsigned carry : 1;
    unsigned zero  : 1;
    unsigned over  : 1;
    unsigned parity : 1;
}
df;

df flags;
```

那么，变量 `flags` 在内存中仅占据 4 位，但它由四个变量组成，可以像其他结构体字段一样进行访问。

## 联合体

C 语言为高效利用可用内存提供的另一个设施是 *联合体* 结构，这是一组共享相同内存存储地址的变量。因此，在给定时间只能访问其中一个变量。联合体定义的一般形式如下：

```c
union 名称
{
    类型 变量名;
    类型 变量名;
    .
    .
    .
    类型 变量名;
} ;
```

## 文件 I/O

C 语言提供带缓冲的文件流用于文件访问。某些 C 平台（如 UNIX 和 DOS）也提供不带缓冲的文件句柄。

### 带缓冲流

带缓冲流通过类型为文件指针的变量进行访问。数据类型 `FILE` 定义在头文件 `stdio.h` 中。因此，要声明一个文件指针，应使用：

```c
#include <stdio.h>
FILE *ptr;
```

为了打开一个流，C 语言提供了函数 `fopen()`，该函数接受两个参数：要打开的文件名以及打开文件时使用的访问模式。访问模式可以是以下任一值：

| 模式 | 描述 |

| :--- | :--- |

| r | 以读模式打开。 |

| w | 以写模式创建，销毁任何现有文件。 |

| a | 以追加模式打开；如果文件不存在则创建新文件。 |

| r+ | 以读写模式打开现有文件。 |

| w+ | 以读写模式创建；销毁任何现有文件。 |

| a+ | 以追加模式打开；如果文件不存在则创建新文件。 |
