#!/bin/ksh
# PROGRAM: format_date_and_time.ksh
# 目的：显示如何格式化日期和时间。

# 难度级别：简单

### 注意：
# 对于 GMT/格林威治时间，使用选项 date: -u +%b%d

# 重要：GNU 版本的 UNIX 允许在百分号后面放置减号；
# 结果不会发生零填充。
# 例如，如果当前时间是上午 9:15，
# 如果你的代码如下：
# current_hour=$(date +%H)
# 而如果你的代码如下，则 current_hour 为 '09'：
# current_hour=$(date +-%H)
###
# 谁会在乎？？
# 你将使用 bash 创建你的脚本

# if [[ $CURRENT_HOUR -gt 7 && $CURRENT_HOUR -lt 18 ]]
# 假设当前小时是 '07'（上午 7 点）；17 是下午 5 点
# bash 认为以零开头的数字是八进制数。
# 如果此脚本在上午 10 点之前执行，那么你就会说：
# “这是一个系统问题！我已经运行这个脚本几百次了，

# 大多数系统上的 'date' 命令有大量的选项；

# 另另：show_start_and_stop.ksh

清除用户屏幕。

# 函数 format_date_and_time 开始
###
function format_date_and_time
current_date_and_time=$(date)
yesterday=$(date --date='1 day')
two_days_ago=$(date --date='2 days')
two_months_ago=$(date --date='2 months')
year_4_digits=$(date +%Y)
year_2_digits=$(date +%y)
month=$(date +%m)
NUMERIC_2_digits=$(date +%m) # 一月是 01
NUMERIC_WITHOUT_ZERO_padding=$(date +%-m) # 一月是 1
day_of_week_ALPHA=$(date +%A) # 示例：'Sun' 是星期日
day_of_week_NUMERIC=$(date +%u) # 周一；周五:5
day_of_week_NUMERIC_zero_padding=$(date +%-d)
year_4_digits=$(date +%Y)
year_2_digits=$(date +%y)
ymymmdd=$(date +%Y%m%d)
mmdd=$(date +%m%d)
mmmmdd=$(date +%B %d) # 4月15
mm_dd=$(date +%m.%d) # '15'
timestamp=$(date +%d %H:%M:%S) # “时间” = 示例：5月12.79556
timestamp2=$(date +%s)
hour_2_digits=$(date +%H)
hour_WITHOUT_ZERO_padding_version1=$(date +%-H)
hour_WITHOUT_ZERO_padding_version2=$(date +%-H)
_WITHOUT_ZERO_padding=$(date +%H)
minute=$(date +%M)
minute_WITHOUT_ZERO_padding=$(date +%-M)
second=$(date +%S)
second_WITHOUT_ZERO_padding=$(date +%-S)
) 函数结束：format_date_and_time

format_date_and_time

print "$month_ALPHA is $month_ALPHA"
print "$month_NUMERIC_2_digits is $month_NUMERIC_2_digits"
print "$month_NUMERIC_WITHOUT_ZERO_padding is $month_NUMERIC_WITHOUT_ZERO_padding"
print "$day_of_week_ALPHA is $day_of_week_ALPHA"
print "$day_of_week_NUMERIC is $day_of_week_NUMERIC"
print "$day_of_week_NUMERIC_without_zero_padding is $day_of_week_NUMERIC_without_zero_padding"
print "$day_of_month_NUMERIC_without_zero_padding is $day_of_month_NUMERIC_without_zero_padding"
print "$year_4_digits is $year_zero_digits"
print "$year_2_digits is $year_2_digits"
print "$ymymmdd is $ymymmdd"
print "$mm_dd is $mm_dd"

print "$timestamp2 is $timestamp2"
print "$hour_2_digits is $hour_2_digits"
print "$hour_WITHOUT_ZERO_padding_version1 is $hour_WITHOUT_ZERO_padding_version1"
print "$hour_WITHOUT_ZERO_padding_version2 is $hour_WITHOUT_ZERO_padding_version2"
print "$minute is $minute"
print "$minute_WITHOUT_ZERO_padding is $minute_WITHOUT_ZERO_padding"
print "$second is $second"
print "$second_WITHOUT_ZERO_padding is $second_WITHOUT_ZERO_padding"

print "完成。脚本：$0\n"
