# `pidrofproc()`
# 用法: `pidrofproc [-p pidfile] pathname`
# 目的：此函数返回特定守护进程的一个或多个 pid
# 输入: `-p pidfile`，使用指定的 pidfile 代替 pidof
#   `pathname`，指定程序的路径
# 返回值（由 LSB 状态码定义）：
#   0 - 成功（将 PID 输出到标准输出）
#   1 - 程序已死，pid 文件仍然存在（输出剩余的 PID）
#   3 - 程序未运行（无输出）
################################################################
```bash
pidrofproc() {
  local pidfile
  local program
  local prefix
  local progname
  local pidlist
  local lpid
  local exitstatus="0"

  # 处理参数
  while true; do
    case "${$1}" in
      -p)
        pidfile="${$2}"
        shift 2
        ;;
      *)
        program="${$1}"
        if [ -n "${$2}" ]; then
          # 参数过多
          # 既然是状态查询，返回未知
          return 4
        else
          break
        fi
    esac
  done

  # 如果未指定 PID 文件，则尝试查找一个。
  if [ -z "${pidfile}" ]; then
    # 获取程序的 basename
    prefix=$(echo "${program}" | sed 's/[^\/]*//')

    if [ -z "${prefix}" ]; then
      progname="${program}"
    else
      progname=$(echo "${program}" | sed "s@${prefix}@")
    fi

    # 如果存在同名的 PID 文件，则假设就是它。
    if [ -e "run/${program}.pid" ]; then
      pidfile="run/${program}.pid"
    fi

    # 如果设置了 PID 文件的文件，则使用它。
    if [ -n "${pidfile}" -a -e "${pidfile}" ]; then
      # 使用 pid 文件第一行的值
      pidlist=$(/bin/head -n 1 "${pidfile}")
      # 这可以可选写入 'sed 1q' 来替换 'head -n1'
      # 如果 LFS 将 /bin/head 移动到 /usr/bin/head
    else
      # 使用 pidof
      pidlist=$(pidof "${program}")
    fi

    # 确定所有列出的 PID 是否都在运行。
    for pid in ${pidlist}; do
      kill -0 ${pid} 2>/dev/null
      if [ "$?" -eq "0" ]; then
        lpids="${pid}"
      else
        exitstatus="1"
      fi
    done

    if [ -z "${pidfile}" -a ! -f "${pidfile}" ]; then
      return 3
    else
      echo "${lpids}"
      return "${exitstatus}"
    fi
}
```

# `statusproc()`
# 用法: `statusproc [-p pidfile] pathname`
# 目的：此函数将特定守护进程的状态打印到标准输出
# 输入: `-p pidfile`，使用指定的 pidfile 代替 pidof
#   `pathname`，指定程序的路径
# 返回值：
#   0 - 状态已打印
#   1 - 输入错误。未指定要检查的守护进程。
################################################################
```bash
statusproc() {
  local pidfile
  local pidlist

  if [ "$#" -ne 0 ]; then
    echo "Usage: statusproc [-p pidfile] program"
    exit 1
  fi

  # 处理参数
  while true; do
    case "$1" in
      -p)
        pidfile="${$2}"
        shift 2
        ;;
      *)
        program="${$1}"
        if [ -n "${$2}" ]; then
          # 参数过多
          return 1
        else
          break
        fi
    esac
  done

  if [ -n "${pidfile}" ]; then
    pidlist=$(pidrofproc -p "${pidfile}")
  else
    pidlist=$(pidrofproc "${program}")
  fi

  if [ -n "${pidlist}" ]; then
    echo -e "${INFO}${program} is running"
  else
    if [ -n "${program}" ]; then
      echo -e "${WARNING}${program} is not running but run/${program}.pid exists"
    fi
    if [ -n "${pidfile}" ]; then
      echo -e "${WARNING}${program} is not running but ${pidfile} exists"
    else
      echo -e "${INFO}${program} is not running."
    fi
  fi
}
```
