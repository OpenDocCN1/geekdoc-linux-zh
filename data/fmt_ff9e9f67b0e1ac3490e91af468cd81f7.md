int _do_munmap(struct mm_struct *mm, unsigned long start, size_t len,
      struct list_head *uf, bool downgrade)
{
    unsigned long end;
    struct vm_area_struct *vma, *prev, *last;

    if ((offset_in_page(start)) || start > TASK_SIZE || len > TASK_SIZE - start)
        return -EINVAL;

    len = PAGE_ALIGN(len);
    end = start + len;
    if (len == 0)
        return -EINVAL;

    /*
     * arch_unmap(might) 可能会解映射自身。它必须在代码运行之前
     * 被调用并完成任何红黑树操作，同时也为了操作红黑树。
     */
    arch_unmap(start, start, end);

    /* 找到第一个 start < vma->vm_end 的重叠 VMA */
    vma = find_vm_intersection(mm, start, end);
    if (!vma)
        return 0;
    prev = vma->vm_prev;

    /*
     * 如果我们需要分割 vma，现在就做以避免以后的麻烦。
     *
     * 注意：mmap() 的 move_vm vma (VM_ACCOUNT) 假设部分解映射
     * 的 vm_area_struct 将继续保持在使用状态：因此，较低的 split_vma
     * 将 tmp vma 放在上方，而较高的 split_vma 将 tmp vma 放在下方。
     */
    if (start > vma->vm_start) {
        int error;

        /*
         * 确保来自 munmap() 的 map_count 不超过其限制；
         * 但让 map_count 临时超过其限制，以帮助释放资源。
         */
        if (end < vma->vm_end && mm->map_count >= sysctl_max_map_count)
            return -ENOMEM;
        error = __split_vma(mm, vma, start, 0);
        if (error)
            return error;
        prev = vma;
    }
    /* 它会分割最后一个一个吗？ */
    last = find_vma(mm, end);
    if (last && end > last->vm_start) {
        int error = __split_vma(mm, last, end, 1);
        if (error)
            return error;
    }
    vma = vma_next(mm, prev);

    /* 从红黑树中分离 vmas */
    if (detach_vma_to_unmapped(mm, vma, prev, end))
        downgrade = false;

    if (downgrade)
        mmap_write_downgrade(mm);

    unmap_region(mm, vma, prev, start, end);

    /* 修复所有 VM 信息 */
    remove_vma_list(mm, vma);

    return downgrade ? 1 : 0;
}

## 5.1 VMA 合并与拆分

### 5.1.1 VMA 合并

由 `struct vm_area_struct` (VMAs) 表示的内存虚拟区域根据其在内存中的位置和属性（由 `can_vma_merge_before()` 和 `can_vma_merge_after()` 决定——两者很快就会失效）而分离。当通过 mmap 或 brk 系统调用映射/重新映射，或通过例如 `mprotect()` 更改属性时，我们可能合并它们。此合并操作由 `vma_merge()` 执行（带有对 `struct mm_struct->mmap_lock` 信号量头部的写锁），其主要参数为：

- 与进程关联的 `mm` 结构。
- `prev` - 紧邻要合并虚拟范围之前的 VMA（如果插入的 VMA 将是地址空间中的第一个 VMA，则为 NULL）。
- `addr` - 要插入地址空间的 VMA 地址或正在更改属性的范围的起始地址。
- `end` - 该范围的排他边界（即位于该范围上方但不属于该范围的虚拟地址）。
- `vm_flags` - 合并区域的标志。

合并可能发生的八种情况：

1. 新映射，合并两者 - `prev->vm_end` 等于 `addr`。
2. 新映射，合并之前的 - `prev->vm_end` 等于 `addr`。
3. 新映射，合并下一个 - `end` 等于 `next->vm_start`。
4. 调整映射，合并下一个 - `addr` 等于 `prev->vm_end`。
5. 调整映射，合并之前的 - `addr` 等于 `prev->vm_end`。
6. 调整映射，合并两者 - `addr` 等于 `prev->vm_end` 且 `end` 等于 `next->vm_start`。
7. 调整映射，合并之前的 - `addr` 等于 `prev->vm_end`。
8. 调整映射，合并下一个 - `end` 等于 `next->vm_start`。

决定 VMA 属性的一部分是 NUMA 内存策略（有关整体 NUMA 的更多信息，请参见 NUMA 章节）。在这种情况下，调用了 `mbind()` 系统调用，我们必须与其他 VMA 调整情况类似，检查是否与相邻 VMA 合并，否则我们必须对其进行拆分（情况 4 - 8）。

对于调整地址（即指定跨越 VMA 的地址范围）的函数，这里有一个重要的微妙之处：它要么等于 VMA 的起始（标记为 area），在这种情况下 `prev` 将是所跨越区域之前的 VMA；要么如果是偏移，`prev` 将指定被跨越的 VMA 并在情况 4 中合并。
