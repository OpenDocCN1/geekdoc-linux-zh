
## 7.3.6 ARM

ARM: Optimizing Keil (Thumb mode)

```assembly
call    printf

cmp     eax, 1

jne     LN2@main

mov     edx, DWORD PTR [rsp] ; 'You entered %d...'

lea     rcx, OFFSET:$SG27

call    printf

jmp     LN1@main

$LN2@main:

lea     rcx, OFFSET:$SG29 ; 'What you entered? Huh?'

call    printf

$LN1@main:

xor     eax, eax

add     rsp, 56

ret

main

ENDP
```

```assembly
var_8      = -8

PUSH      {R3,LR}

ADR       R0,aEnter      ; "Enter X:\n"

BL        printf

MOV       R1,SP

ADR       R0,aD           ; "%d"

BL        scanf

CMP       W0, #1

BEQ       loc1E

ADR       R0,aEntered  ; "What you entered?\n"

BL        printf

POP      {R3,PC}           ; XREF: main+12

loc1E

LDR       R1, [SP,#8]

ADR       R0,aEnded  ; "You entered...\n"

BL        printf
```

这里见到了`CMP`和`BEQ`指令。它们都是参数比较，并设置标志位。ARM系统的指令与x86相似。`BEQ`是条件跳转指令（Branch if Equal），在操作数相等，`ZF`寄存器的值就是1，那么`BEQ`就会跳转到它的偏移量。它与x86指令相同。其余的指令并不理解：程序有两个分支，这两个分支终于`loc1A`处，通过`MOV R0,#0`指令将返回值保存在`R0`寄存器，然后退出。

### ARM64

指令清单：Non-optimizing GCC

```assembly
LC0:
    .string "Enter X:"

LC1:
    .string "%d"

LC2:
    .string "You entered...\n"

LC3:
    .string "What you entered? Huh?"

f6:
    ; save FP and LR in stack frame;
    stp     x29, x30, [sp, -32]! ; set stack frame (FP=SP)
    add     x29, sp, #0
    ; load pointer to the "Enter X:" string:
    adrp    x0, LC0
    add     x0, x0, :lo:LC0
    puts
    ; calculate address of x in the local stack
    add     x1, x29, #28
    bl      scanf ; scanf returned result in W0.
    cmp     w0, #1
    bne     Branch If Not Equal ; if W0!=0, jump to L2 will occurred
    bne     L2 ; at moment W0=1, meaning no error
    ; load x1, [x29,28]
    ; load pointer to the "You entered...\n" string:
    adrp    x0, LC2
    add     x0, x0, :lo:LC2
    printf
```
