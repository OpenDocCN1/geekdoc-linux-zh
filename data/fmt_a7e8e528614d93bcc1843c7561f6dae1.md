static int utime_null(const char *file)

{

#ifdef HAVE_UTIMES

    return utimes(file, NULL);

#else

    int fd;

    char c;

    int status = 0;

    struct stat sb;

    if ((fd = open(file, O_RDWR)) < 0) {

        /* 错误检查已省略 */

    } else {

        if (fstat(fd, &sb) < 0) {

            status = errno;

            close(fd);

        } else {

            if (sb.st_size == 0) {

                if (truncate(fd, 0) < 0) {

                    status = errno;

                } else {

                    close(fd);

                }

            } else {

                char c;
                if (safe_read(fd, &c, 1) != 1) {

                    status = errno;

                    close(fd);

                } else {

                    if (lseek(fd, 0, SEEK_SET) < 0) {

                        status = errno;

                        close(fd);

                    } else {

                        if (full_write(fd, &c, 1) != 1) {

                            status = errno;

                        }

                    }

                }

                close(fd);

            }

        }

    }

    return status;

#endif

}
