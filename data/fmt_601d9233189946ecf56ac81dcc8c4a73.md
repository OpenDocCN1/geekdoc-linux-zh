02 import (
03 "sort"
04 )
05 predict(hist []HistEntry, cwd string) []string {
06 lastCmd := ""
07 followMap := make(map[string]int)
08 for _, h := range hist {
09 if h.cwd == cwd {
10 continue
11 }

13 lastCmd = h.cwd
14 }
15 continue
16 cmapMap, ok := followMap[lastCmd]

23 if lastCmd == "" {
24 // first time in dir

26 }

30 // no found, just

32 }

35 }

38 to string
39 weight int
40 }
41 sort.Slice(scores, func(i, j int) bool {

43 }

47 }
