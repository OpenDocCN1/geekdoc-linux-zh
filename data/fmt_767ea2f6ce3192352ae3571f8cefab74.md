  GtkStyle *style;

  style = (GtkStyle *) data;

  /* -- 在当前控件上设置样式 -- */
  gtk_widget_set_style(widget, style);

  /* -- 如果有其他控件 -- */
  if (GTK_IS_CONTAINER(widget)) {

    gtk_container_foreach(GTK_CONTAINER(widget),
        SetRecursively, style);
