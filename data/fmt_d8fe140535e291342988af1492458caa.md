```python
import types
class classmethod:
    def __init__(self, func):
        self.__func__ = func

    # 返回以 cls 为参数的绑定方法
    def __get__(self, instance, cls):
        return types.MethodType(self.__func__, cls)

class staticmethod:
    def __init__(self, func):
        self.__func__ = func

    # 返回裸函数
    def __get__(self, instance, cls):
        return self.__func__
```

```python
class Lazy:
    def __init__(self, func):
        self.func = func

    def __set_name__(self, cls, name):
        self.key = name

    def __get__(self, instance, cls):
        if instance:
            value = self.func(instance)
            instance.__dict__[self.key] = value
            return value
        else:
            return self
```

```python
class Rectangle:
    def __init__(self, width, height):
        self.width = width
        self.height = height

area = Lazy(lambda self: self.width * self.height)
perimeter = Lazy(lambda self: self.width + self.height)
```

```python
import types

# 在类中的方法
def __init__(self, owner, balance):
    self.owner = owner
    self.balance = balance

def deposit(self, amount):
    self.balance -= amount

def withdraw(self, amount):
    self.balance += amount

methods = {
    '__init__': __init__,
    'deposit': deposit,
    'withdraw': withdraw,
}

Account = types.new_class('Account', (), 
    exec_body=lambda ns: ns.update(methods))

# 现在你有了类
a = Account('Guido', 1000.0)
a.deposit(50)
a.withdraw(25)
```

```python

        (name, types.new_class(name, (Typed,
            exec_body=lambda ns: ns.update(expected_type=ty))))
```
