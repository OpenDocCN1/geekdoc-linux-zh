# 第22章

## 1. 一可能的简化方法是

```
# 初始化目标目录
DESTDIR="$2"

# 检查目标是否存在
if [ ! -d "$DESTDIR" ]; then
    # 如果不存在，则尝试获取绝对路径
    DESTDIR=$(cd "$2" && pwd)
fi

# 获取目标目录的基文件名
NEWNAME=$(basename "$DESTDIR")
# 获取父目录
PARENTDIR=$(dirname "$DESTDIR")

# 如果父目录不存在，则退出
if [ ! -d "$PARENTDIR" ]; then
    echo "Parent directory does not exist."
    exit 1
fi
```

- 4. 一可能的解决方案

```
################################################################
# Name: isSpaceAvailable
# Desc: 如果空间可用则返回真 (0)
# Args: $1 -> 要检查的目录
# $2 -> 要检查的空间量
# $3 -> 单位（可选）
# k 千字节
# m 兆字节
# g 吉兆字节
################################################################
isSpaceAvailable() {
    if [ $# -lt 2 ]; then
        echo "Insufficient Arguments"
        return 1
    fi

    if [ ! -d "$1" ]; then
        echo "$1 is not a directory."
        return 1
    fi

    SPACE_MIN="$2"

    case "$3" in
        k|m)
            SPACE_MIN=$(echo "$SPACE_MIN * 1024" | bc) ;;
        g)
            SPACE_MIN=$(echo "$SPACE_MIN * 1024" | bc) ;;
    esac

    if [ `getSpaceFree "$1") -gt "$SPACE_MIN" ]; then
        return 0
    fi
    return 1
}
```

- 5. 一种可能的解决方案

```
############################################
# Name: isUserRoot
# Desc: 如果 UID=0 则返回真 (0)
# Args: $1 -> 用户名（可选）
############################################
isUserRoot() {
    if [ "$(id -u)" -eq 0 ]; then
        return 0
    fi
    return 1
}
```
