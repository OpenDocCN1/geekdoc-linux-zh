```bash
/lib/sh/init-functions
case "$1" in
  start)
    # 在记录任何消息之前确保 /run可用
    if mountpoint /run >/dev/null; then
        mount /run || failed=1
    fi

    mkdir -p /run/lock
    chmod 1777 /run/lock

    log_info_msg "挂载虚拟文件系统: ${INFO}" "/run"

    if mountpoint /proc >/dev/null; then
        log_info_msg2 "${INFO}" "/proc"
        mount -o nosuid,nodeexec,/proc || failed=1
    fi

    if mountpoint /sys >/dev/null; then
        log_info_msg2 "${INFO}" "/sys"
        mount -o nosuid,nodeexec,/sys || failed=1
    fi

    if mountpoint /dev >/dev/null; then
        log_info_msg2 "${INFO}" "/dev"
        mount -o mode=0755,nosuid /dev || failed=1
    fi

    mkdir -p /dev/shm
    log_info_msg2 "${INFO}" "/dev/shm"
    mount -o nosuid,nodev /dev/shm || failed=1

    mkdir -p /sys/fs/cgroup
    log_info_msg2 "${INFO}" "/sys/fs/cgroup"
    mount -o nosuid,nodeexec,/sys/fs/cgroup || failed=1

    exit (${failed})
    ;;
esac
```

```bash
/lib/lsb/init-functions

case "${1}" in
start)
  # 如果没有模块文件或没有
  # 有效条目，则退出
  [ -r /etc/sysconfig/modules ] || exit 0
  grep -Ev '^#(${#})' /etc/sysconfig/modules || exit 0

  log_info_msg "Loading modules:"

  # 只有当用户确实提供了一些需要加载的模块时，才尝试加载。

while read module args; do

  # 忽略注释和空行。
  case "$module" in
    ""${#}") continue;
  esac

  # 尝试加载模块，传递任何提供的参数。
  modprobe $(module) ${args} >/dev/null

  # 如果成功则打印模块名，否则记录。
  if [ $? -eq 0 ]; then
    log_info_msg "${module}"
  else
    failedmod="${failedmod} ${module}"
  fi
done < /etc/sysconfig/modules

  # 在正确的行上打印关于成功加载模块的消息。
  log_success_msg$2

  # 打印失败消息，并列出任何可能加载失败的模块。
  if [ -n "${failedmod}" ]; then
    log_failure_msg "Failed to load modules:${failedmod}"
    exit 1
  fi

*)
  echo "Usage: ${0} (start)"
  exit 1
  fi
esac

exit 0
```
