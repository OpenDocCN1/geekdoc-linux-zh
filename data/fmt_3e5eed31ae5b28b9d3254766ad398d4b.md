void str_echo(int sockfd)
{
    ssize_t n;
    int i;
    char line[MAXLINE];
    for ( ; ; ) {
        if ( (n = read(sockfd , line , MAXLINE-1)) == 0 )
            return;
        for ( i = 0; i < n; i++ )
            if ( islower(line[i]))
                line[i] = toupper(line[i]);
        write(sockfd , line , n);
    }
}
