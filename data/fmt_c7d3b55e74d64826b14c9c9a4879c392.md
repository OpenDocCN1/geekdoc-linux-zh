
# 实用 TSR 程序（弹出式地址簿数据库）

```c
/**
 * 一个实用的 TSR 程序（弹出式地址簿数据库）
 * 在小内存模型下编译，关闭堆栈检查
 */

#include <dos.h>
#include <stdio.h>
#include <string.h>
#include <dir.h>

static union REGS rg;

/*
    保持驻留的程序大小
    需要实验以使其尽可能小
*/
unsigned sizeprogram = 28000/16;

/* 使用 Alt . 激活 */
unsigned scancode = 52;  /* . */
unsigned keymask = 8;    /* ALT */

char signature[] = "POPADDR";
char fpath[40];

/*
    函数原型
*/

void curr_cursor(int *x, int *y);
int resident(char *, void interrupt(*)( ));
void resinit(void);
void terminate(void);
void restart(void);
void wait(void);
void resident_psp(void);
void exec(void);

/*
 来自 DOS 的入口点
*/

main(int argc, char *argv[])
{
    void interrupt ifunc();
    int ivc;

    /*
    为简单起见，假设数据文件在 C: 驱动器的根目录
    */
    strcpy(fpath,"C:\\ADDRESS.DAT");

    if ((ivec = resident(signature,ifunc)) != 0)
    {
    /* TSR 已驻留 */
    if (argc > 1)
    {
        rg.x.ax = 0;
        if (strncmp(argv[1],"quit") == 0)
            rg.x.ax = 1;
        else if (strncmp(argv[1],"restart") == 0)
            rg.x.ax = 2;
        else if (strncmp(argv[1],"wait") == 0)
            rg.x.ax = 3;
    if (rg.x.ax)
    {
	int86(ivec,&rg,&rg);
	return;
    }

    printf("\n弹出式地址簿已驻留");

    }
    else
    {
	/* TSR 程序初始加载 */
    printf("弹出式地址簿已驻留。\n按 Alt . 激活...\n");
    resinit();
    }
}

void interrupt ifunc(bp,di,si,ds,es,dx,cx,bx,ax)
{
	if(ax == 1)
	terminate();
	else if(ax == 2)
	restart();
	else if(ax == 3)
	wait();
}

popup()
{
	int x,y;
	
	curr_cursor(&x,&y);
	
	/* 在此处调用 TSR C 程序 */
	exec();
	cursor(x,y);
}

static int diskflag = 0;
static int kbsval;

void dores(void);
void pidaddr(void);

void resinit()
{
    segread(&seg);
    myss = seg.ss;

    rg.h.ah = 0x34;
    intdos(&rg,&rg);
    dosseg = _ES;
    dosbusy = rg.x.bx;

    mydta = getdta();
    pidaddr();
    oldtimer = getvect(0x1c);
    oldt28 = getvect(0x28);
    oldkb = getvect(9);
    olddisk = getvect(0x13);
    setvect(0x1c,newtimer);

    setvect(9,newkb);
    setvect(0x28,new28);
    setvect(0x13,newdisk);

    stack = (sizeprogram - (seg.ds - seg.cs)) * 16 - 300;
    rg.x.ax = 0x3100;
    rg.x.dx = sizeprogram;
    intdos(&rg,&rg);
}

void interrupt newdisk(bp,di,si,ds,es,dx,cx,bx,ax,ip,cs,flgs)
{
    diskflag++;
    (*olddisk)();
    ax = _AX;
    newcrit();
    flgs = cflag;
    --diskflag;
}

void interrupt newcrit(bp,di,si,ds,es,dx,cx,bx,ax,ip,cs,flgs)
{
    ax = 0;
    cflag = flgs;
}

void interrupt newkb()
{
    if (inportb(0x60) == scancode)
    {
        kbval = peekb(0,0x417);
        if (!resoff && ((kbval & keymask) ^ keymask) == 0)
        {
            kbval = inportb(0x61);
            outportb(0x61,kbval | 0x80);
            outportb(0x61,kbval);
            disable();
            outportb(0x20,0x20);
            enable();
            if (!running)
                poflg = 1;
            return;
        }
    }
    (*oldkb)();
}

void interrupt newtimer()
{
    (*oldtimer)();
}
```
