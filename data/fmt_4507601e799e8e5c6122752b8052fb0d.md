```c
void sdr7am_init(void)
{
    ulong sdtr1;
    ulong rtr;
    int i;

    /*
     * 支持 100MHz 和 133MHz SDRAM
     */
    if (get_bus_freq(0) > 100000000) {
        /*
         * 133 MHz SDRAM
         */
        sdtr1 = 0x01704015;
        rtr = 0x07f00000;
    } else {
        /*
         * 默认: 100 MHz SDRAM
         */
        sdtr1 = 0x0086400d;
        rtr = 0x05f00000;
    }

    for (i = 0; i < N_MB0CF; i++) {
        /*
         * 禁用内存控制器。
         */
        mtdram0(mem_mcopt1, 0x00000000);

        /*
         * 设置 bank 0 的 MB0CF。
         */
        mtdram0(mem_mb0cf, mb0cf[i].reg);

        mtsdram0(mem_sdtr1, sdtr1);
        mtsdram0(mem_rtr, rtr);

        udel(200);

        /*
         * 设置内存控制器选项寄存器，MCEPT1。
         * 将 DC_EN 设置为 '1'并将 BRD_PRF 设置为 '01' 以进行 16 字节 PLB 突发
         * 读/预取
         */
        mtsdram0(mem_mctopt1, 0x80800000);

        if (get_ram_size(0, mb0cf[i].size) == mb0cf[i].size) {
            /*
             * OK，检测检测 -> 完成
             */
            return;
        }
    }
}
```

第一个操作读取 405GP 处理器上的 SDRAM 绑定，以确定内存时钟的设计值。在这种情况下，我们可以看到支持两个可能的值：100MHz 和 133MHz。根据此选择，选择了将要使用的常量。
