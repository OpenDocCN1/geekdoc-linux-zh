
### 1. 设备声明

```c
/* 设备名称，它将出现在 /proc */
#define DEVICE_NAME "char_dev"

/* 消息的最大长度 */
#define BUF_LEN 80

/* 设备现在打开了吗？用于防止并发访问 */
static int Device_Open = 0;

/* 设备提供的消息 */
static char Message[BUF_LEN];

/* 读取进程获取了多少消息？如果消息比
* 我们用于填充 device_read 的缓冲区大，则有用。 */
static char *Message_Ptr;
```

### 2. 打开设备函数

每当进程尝试打开设备文件时调用此函数：

```c
static int device_open(struct inode *inode, struct file *file)
{
#ifdef DEBUG
    printk("device_open(%p)\n", file);
#endif
    /* 我们不想同时与两个进程通信 */
    if (Device_Open)
        return -EBUSY;

    /* 如果这是一个进程，在这里必须更加小心，因为一个进程可能在另一个进程增加它之前检查了 Device_Open。然而，我们在内核中
    * 受到上下文切换的保护。
    * 这不是正确的做法，因为我们可能在 SMP
    * 环境下运行，但我们将在后面的章节中处理 SMP。
    */
    Device_Open++;

    /* 初始化消息 */
    Message_Ptr = Message;

    MOD_INC_COUNT;

    return SUCCESS;
}
```

### 3. 释放设备函数

当进程关闭设备文件时调用此函数。它没有返回值，因为不可能失败。无论发生什么，我们都应该能够关闭设备（在 2.0 中，2.2 设备文件可能无法关闭）：

```c
#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
static int device_release(struct inode *inode, struct file *file)
#else
static void device_release(struct inode *inode, struct file *file)
#endif
{
#ifdef DEBUG
    printk("device_release(%p,%p)\n", inode, file);
#endif
    /* 我们现在已经准备好处理下一个调用者了 */
    Device_Open--;

    MOD_DEC_USE_COUNT;

#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
    return 0;
#endif
}
```

### 4. 读取设备函数

每当已经打开设备文件的进程尝试从其读取数据时，该函数就会被调用：

```c
#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
static ssize_t device_read(
    struct file *file,
    char *buffer,       /* 用于填充数据的缓冲区 */
    size_t length,      /* 缓冲区的长度 */
    loff_t *offset      /* 指向文件的偏移量 */
)
#else
static int device_read(
    struct inode *inode,
    struct file *file,
    char *buffer,       /* 用于填充数据的缓冲区 */
    int length          /* 缓冲区的长度 (不能写入超出范围!) */
)
#endif
{
    /* 实际写入缓冲区的字节数 */
    int bytes_read = 0;

#ifdef DEBUG
    printk("device_read,%p,%p)d\n", file, buffer, length);
#endif

    /* 如果我们到达消息的末尾，返回 0 (表示文件结束) */
    if (Message_Ptr == 0)
        return 0;

    /* 实际上将数据放入缓冲区 */
    while (length && *Message_Ptr) {
        /* 因为缓冲区位于用户数据段中，
        * 而不是内核数据段，直接赋值将不起作用。
        * 相反，我们必须使用 put_user，将数据从内核数据段拷贝到用户数据段。 */
        put_user(*Message_Ptr++, buffer++);
        length--;
        bytes_read++;
    }

#ifdef DEBUG
    printk("Read %d bytes, %d left\n", bytes_read, length);
#endif

    /* 读取函数应该返回实际插入到缓冲区中的字节数 */
    return bytes_read;
}
```

### 5. 写入设备函数

当有人尝试向我们的设备文件写入时，该函数就会被调用：

```c
#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
static ssize_t device_write(struct file *file,
                            const char *buffer,
                            size_t length,
                            loff_t *offset)
#else
static int device_write(struct inode *inode,
                        struct file *file,
                        const char *buffer,
                        int length)
#endif
{
    int i;

#ifdef DEBUG
    printk("device_write%p,%s%d",
           file, buffer, length);
#endif

    for (i = 0; i < length && i < BUF_LEN; i++)
#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
        get_user(Message[i], buffer + i);
#else
        Message[i] = get_user(buffer + i);
#endif

    Message_Ptr = Message;

    /* 再次返回使用的输入字符数 */
    return i;
}
```

### 6. IOCTL 操作函数

每当进程尝试对我们的设备文件执行 `ioctl` 操作时，该函数就会被调用。我们得到了两个额外的参数（除了 `inode` 和 `file` 结构体，所有设备函数都会获取这些结构）：被调用的 `ioctl` 以及传递给 `ioctl` 函数的参数。如果 `ioctl` 是写或读/写（意味着输出被返回给调用进程），则 `ioctl` 调用将返回此函数的输出。

```c
int device_ioctl(
    struct inode *inode,
    struct file *file,
    unsigned int ioctl_num,      /* ioctl 的数量 */
    unsigned long ioctl_param    /* 传递它的参数 */
) {
    int i;
    char *temp;
#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
    char ch;
#endif

    /* 根据调用的 ioctl 进行分支 */
    switch (ioctl_num) {
        case IOCTL_SET_MSG:
            /* 接收指向消息（在用户空间）的指针
            * 并将其设置为设备的消息。 */
            /* 获取进程传递给 ioctl 的参数 */
            temp = (char *) ioctl_param;

            /* 获取消息的长度 */
#if LINUX_VERSION_CODE >= KERNEL_VERSION(2,2,0)
            get_user(ch, temp);
            for (i = 0; ch && i < BUF_LEN; i++, temp++)
                get_user(ch, temp);
#else
            for (i = 0; get_user(temp) && i < BUF_LEN; i++, temp++)
                ;
#endif
    }
}
```
