# 示例 8-1 最大公约数

```bash
#!/bin/bash

# gcd.sh: 最大公约数

# 使用欧几里得算法

# “最大公约数”(gcd)

# 是能整除两个数且无余数的最大整数。

# 欧几里得算法使用连续除法。

# 在每次循环中，

# 被除数 <--- 除数

# 除数 <--- 余数

# 直到 $remainder = 0

# 在最后一次中，gcd = 被除数。

# 关于欧几里得算法的极佳讨论，请参见

# Jim Loy 的网站，http://www.jimloy.com/number/euclids.htm。

---

# 参数检查

ARGS=2

E_BADARGS=85

if [ $# -ne "$ARGS" ];

then

    echo "Usage: \`basename $0\` first-number second-number"

    exit $E_BADARGS

fi

---

gcd() {

    dividend=$1    # 任意赋值。

    divisor=$2     # 两个数哪个更大并不重要。

                   # 为什么呢？

    remainder=1     # 如果在测试括号内使用未初始化的变量

                   # 会导致错误信息。

    until [ "$remainder" -eq 0 ];

    do

        # 必须预先初始化！

        let remainder=$dividend % $divisor

        dividend=$divisor

        divisor=$remainder # 现在对两个最小的数重复。

    done

    echo $dividend # 欧几里得算法

}

gcd $1 $2 # 最后的 $
```
