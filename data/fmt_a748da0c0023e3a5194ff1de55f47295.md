.equ END_OF_FILE, 0  #这是 read 的返回值

```
.equ NUMBER_ARGUMENTS, 2

section .bss

#Buffer - 这是我们从数据文件
## 加载数据并写入
## 输出文件的地方。出于各种原因，
## 这个值永远不应该
## 超过 16,000。
.equ BUFFER_SIZE, 500

.lcomm BUFFER_DATA, BUFFER_SIZE
```

```
section .text
```

```
#栈位置定义
.equ ST_SIZE_RESERVE, 8
.equ ST_FD_IN, -4
.equ ST_FD_OUT, -8
.equ ST_ARGC, 0  #参数个数
.equ ST_ARGV_0, 4  #程序名称
.equ ST_ARGV_1, 8  #输入文件名
.equ ST_ARGV_2, 12  #输出文件名
```

```
.glob _start

_start:

### 初始化程序###
#保存栈指针
movl %esp, %ebp
#在栈上为我们的文件描述符
#分配空间
subl $ST_SIZE_RESERVE, %esp
```

```
open_files:

open_fd_in:

### 输入文件###
## open 系统调用
movel $SYS_OPEN, %eax
#将文件名读入 %ebx
movl ST_ARGV_1(%ebp), %ebx
#只读标志
movel $O_RDONLY, %ecx
## 对于读取来说这并不重要
movel $0666, %edx
#调用 Linux
int $LINUX_SYSCALL
```

```
store_fd_in:
#保存给的文件描述符
movel %eax, ST_FD_IN(%ebp)
```

```
open_fd_out:
### 打开输出文件###
## 打开文件
movel $SYS_OPEN, %eax
#将文件名读入 %ebx
movel ST_ARGV_2(%ebp), %ebx
#向文件写入的标志
movel $O_CREAT_WRONLY_TRUNC, %ecx
#新文件的模式（如果创建了的话）
movel $0666, %edx
#调用 Linux
int $LINUX_SYSCALL

store_fd_out:
#在此存储文件描述符
mov %eax, ST_FD_OUT(%ebp)
```

```
### 开始主循环###
read_loop_begin:
```

```
### 从输入文件中读取一个数据块###
mov $SYS_READ, %eax
#获取输入文件描述符
mov ST_FD_IN(%ebp), %ebax
## 读取到该位置的位置
mov $BUFFER_DATA, %ecx
## 缓冲区的大小
mov $BUFFER_SIZE, %edx
## 读取的缓冲区大小返回在 %eax 中
int $LINUX_SYSCALL
```

```
### 如果到达末尾则退出###
#检查文件结束标记
cmpl $END_OF_FILE, %eax
## 如果找到或出错，跳转到末尾
le end_loop
```

```
continue_read_loop:
### 将数据块转换为大写###
push $BUFFER_DATA #缓冲区的位置
push %eax #缓冲区的大小
call convert_to_upper
popl %eax # 获取回大小
addl $4, %esp # 恢复 %esp
```

```
### 将数据块写入输出文件###
## 缓冲区的大小
mov %eax, %edx
mov $SYS_WRITE, %eax
# 使用的文件文件
mov ST_FD_OUT(%ebp), %ebx
## 缓冲区的位置
mov $BUFFER_DATA, %ecx
int $LINUX_SYSCALL
```

```
### 继续循环###
jmp read_loop_begin
```

```
end_loop:
### 关闭文件并检查###
## 注意 - 我们不需要对这些进行错误检查
## - 因为错误条件在这里
## 并不表示任何特殊含义
mov $SYS_CLOSE, %eax
mov ST_FD_OUT(%ebp), %ebx
int $LINUX_SYSCALL

mov $SYS_CLOSE, %eax
mov ST_FD_IN(%ebp), %ebx
int $LINUX_SYSCALL
```

```
### 退出###
mov $SYS_EXIT, %eax
mov $0, %ebx
int $LINUX_SYSCALL
```

## 目的：此函数实际上执行了
## 将数据块转换为大写的转换

## 输入：第一个参数是要转换
## 内存块的位置
## 第二个参数是缓冲区的长度

## 输出：此函数用大写版本
## 覆盖当前的缓冲区。

## 变量：
## `%eax` - 缓冲区起始位置
## `%ebx` - 缓冲区长度
## `%edi` - 当前缓冲区偏移量
## `%cl` - 当前正在检查的字节
## `(%ecx` 的第一部分)

#PURPOSE: This function actually does the
#conversion to upper case for a block

#INPUT: The first parameter is the location
# of the block of memory to convert
# The second parameter is the length of
# buffer

#OUTPUT: This function overwrites the current
#buffer with the upper-case version.

#VARIABLES:
# %eax - beginning of buffer
# %ebx - length of buffer
# %edi - current buffer offset
# %cl - current byte being examined
# (first part of %ecx)
