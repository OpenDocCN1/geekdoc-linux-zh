```cpp

    int r;
    int g;
    int b;
    double percent;
    char code;

    QFile qfile("qstext.txt");
    if(qfile.open(Q_ReadOnly)) {

        stream >> name;
        stream >> r >> g >> b;
        stream >> percent;
        stream << code;
```
