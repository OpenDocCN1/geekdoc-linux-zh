run() {
  if [ -z ${interactive} ]; then
    ${1} ${2}
    return $?
  fi
while true; do
  read -p "运行 ${1} ${2}？(是/否/继续) " -n 1
  echo
  case ${1} in
    c | C)
      interactive=""
      ${1} ${2}
      ret=$?
      break
      ;;
    n | N)
      return 0
      ;;
    y | Y)
      ${1} ${2}
      ret=$?
      break
      ;;
  esac
done
return $ret
}
