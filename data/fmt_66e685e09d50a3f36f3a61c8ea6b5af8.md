if [ "$1" != "" ]; then
  runlevel=$1
fi
if [ "$runlevel" = "" ]; then
  echo "Usage: $0 <runlevel>"
  exit 1
fi

previous=$PREVLEVEL
if [ "$previous" = "" ]; then
  export runlevel
fi

# 是否有该运行级别的 rc 目录
if [ -d /etc/$runlevel.d ]; then
  # 首先，处理该运行级别的 KILL 脚本
  if [ "$previous" != N ]; then
    for i in /etc/$runlevel.d/K*
    do
      [ -f "$i" ] && suffix=$etc/$runlevel.d/S[0-9][0-9]
      previous_start=$etc/$runlevel.d/S[0-9][0-9]
      # 如果上一个级别有启动脚本，则停止服务
      [ -f $previous_start ] && [ -f $stop ] &&
        continue
      startup $i
      error_value=$?
      if [ $error_value != 0 ]; then
        print_error_msg
      fi
    done
  fi
  # 现在运行该运行级别的 START 脚本
  for i in /etc/$runlevel.d/S*
  do
    [ -f "$i" ] && continue
    if [ "$previous" != N ]; then
      # 在上一个级别查找启动脚本和停止脚本
      suffix=$etc/$runlevel.d/S[0-9][0-9]
      stop=$etc/$runlevel.d/K[0-9][0-9]
      previous_start=$etc/$runlevel.d/S[0-9][0-9]
      # 如果上一个级别有启动且停止脚本，则不需要重启服务
      [ -f $previous_start ] && [ ! -f $stop ] &&
        continue
    fi
    case "$runlevel" in
    0|6)
        startup $i stop
        error_value=$?
        if [ $error_value != 0 ]; then
          print_error_msg
        fi
      *)
        startup $i start
        error_value=$?
        if [ $error_value != 0 ]; then
          print_error_msg
        fi
    esac
  done
fi
# End /etc/init.d/rc

## 创建 rcS 脚本

第二个主启动脚本是 `rcS` 脚本。创建一个新文件 `/etc/init.d/rcS`，包含以下内容：

```bash
#!/bin/sh
# Begin /etc/init.d/rcS
runlevel=S
prelevel=N
umask 022
export runlevel prelevel
trap "-" TST
for i in /rcS.d/$S??
do
    [ -f "$i" ] && continue
    $i start
done
# End /etc/init.d/rcS
```

## 创建 functions 脚本

创建一个新文件 `/etc/init.d/functions`，包含以下内容：

```bash
#!/bin/sh
# Begin /etc/init.d/functions
COL=70
SET_COL="echo -en \033${COL}G"
NORMAL="echo -en \033[0;39m"
SUCCESS="echo -en \033[1;32m"
FAILURE="echo -en \033[1;31m"
evaluate_retval()
{
  if [ $? = 0 ]; then
    print_status success
  else
    print_status failure
  fi
}
print_status()
{
  if [ $# = 0 ]; then
    echo "Usage: print_status {success|failure}"
    exit 1
  fi
  case "$1" in
    success)
      $SET_COL
      echo -n "["
      $SUCCESS
      echo -n "OK"
      $NORMAL
      echo -n "]"
      ;;
    failure)
      $SET_COL
      echo -n "["
      $FAILURE
      echo -n "FAILED"
      $NORMAL
      echo -n "]"
      ;;
  esac
}
loadproc()
{
  if [ $# = 0 ]; then
    echo "Usage: loadproc {program} [signal]"
    exit 1
  fi
  base='basename $1'
  pidlist=`pidof -o $$ -o PID -o %PPID -x $base`
  pid=""
  for apid in $pidlist
  do
    if [ -d /proc/$apid ]; then
      pid="$pid $apid"
    fi
  done
  if [ ! -n "$pid" ]; then
    evaluate_retval
  else
    print_status failure
  fi
}
killproc()
{
  if [ $# = 0 ]; then
    echo "Usage: killproc {program} [signal]"
    exit 1
  fi
  base='basename $1'
  if [ -n "$2" ]; then
    runlevel=$2
  else
    runlevel=1
  fi
  pidlist=`pidof -o $$ -o PID -o %PPID -x $base`
  pid=""
  for apid in $pidlist
  do
    if [ -d /proc/$apid ]; then
      pid="$pid $apid"
    fi
  done
  if [ -n "$pid" ]; then
    if [ "$runlevel" = 1 ]; then
      kill -TERM $pid
      if ps -h $pid >/dev/null 2>&1; then
        :
      fi
    else
      kill -KILL $pid
    fi
    if ps -h $pid >/dev/null 2>&1; then
      print_status failure
    else
      rm -f /var/run/$base.pid
      print_status success
    fi
  fi
}
reloadproc()
{
  if [ $# = 0 ]; then
    echo "Usage: reloadproc {program} [signal]"
    exit 1
  fi
  base='basename $1'
  if [ -n "$2" ]; then
    runlevel=$2
  else
    runlevel=1
  fi
  pidlist=`pidof -o $$ -o PID -o %PPID -x $base`
  pid=""
  for apid in $pidlist
  do
    if [ -d /proc/$apid ]; then
      pid="$pid $apid"
    fi
  done
  if [ -n "$pid" ]; then
    if [ "$runlevel" = 1 ]; then
      kill -TERM $pid
      if ps -h $pid >/dev/null 2>&1; then
        :
      fi
    else
      kill -KILL $pid
    fi
  fi
}
```
