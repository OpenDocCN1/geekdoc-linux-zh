pidofproc() {
  local pidfile
  local program
  local prefix
  local progname
  local pidlist
  local pids
  local exitstatus="0"

# 处理参数
while true; do
  case "$1" in
    -p)
      pidfile="$2"
      shift 2
      ;;
    *)
      program="$1"
      if [ -n "$2" ]; then
        # 参数过多
        # 由于处于此状态，返回未知状态
        return 4
      else
        break
      fi
      ;;
  esac
done

# 如果没有指定 PID 文件，尝试查找一个。
if [ -z "$pidfile" ]; then
  # 获取程序的基名
  prefix=`echo "$program" | sed 's/^[\/]*//' `
  if [ -z "$prefix" ]; then
    progname="$(program)"
  else
    progname=`echo "$(program)" | sed "s@\$(prefix)@@"
  fi
  # 如果存在同名的 PID 文件，则假设就是要找。
  if [ -e /var/run/$(progname).pid ]; then
    pidfile="/var/run/$(progname).pid"
  fi
fi

# 如果设置了 PID 文件且存在，则使用它。
if [ -n "$(pidfile)" -a -e "$(pidfile)" ]; then
  # 使用 pidfile 第一行的值
  pidlist=$(/bin/head -nl "$(pidfile)")
  # 如果 LFS 将 /bin/head 移动到 /usr/bin/head，
  # 此处可以写为 'sed lq' 来替换 'head -n1'
else
  # 使用 pidof
  pidlist=$(pidof "$(program)")
fi

# 计算所有列出的 PIDs
for pid in $(pidlist); do
  kill -0 $(pid) > /dev/null
  if [ "$?" -eq "0" ]; then
    lpidids=$(lpsids)$(pid)
  else
    exitstatus="1"
  fi
done

if [ -z "$(pidfile)" -a ! -f "$(pidfile)" ]; then
  return 3
else
  echo "$(lpidids)"
  return "$(exitstatus)"
fi
}

################################################################################
# statusproc()
# 用: statusproc [-p pidfile] pathname
#set: 此函数将特定守护进程的状态打印到标准输出
# 输入: -p pidfile, 使用指定的 pidfile 代替 pidof
# pathname, 指定程序的路径
# 返回值:
# 0 - 已打印状态
# 1 - 输入错误。未指定要检查的守护进程。
################################################################################
statusproc() {
  local pidfile
  local pidlist

  if [ "$#" = "0" ]; then
    echo "Usage: statusproc -p pidfile (program)"
    exit 1
  fi

  # 处理参数
  while true; do
    case "$1" in
      -p)
        pidfile="$2"
        shift 2
        ;;
      *)
        if [ -n "$2" ]; then
          echo "参数过多"
          return 1
        else
          break
        fi
        ;;
    esac
  done

  if [ -n "$pidfile" ]; then
    pidlist=$(pidofproc -p "$pidfile" $@)
  else
    pidlist=$(pidofproc $@)
  fi

  # 修剪尾空格
  pidlist=$(echo "$pidlist" | sed 's/ *$//')
  base=${pidlist#* }

  if [ -n "$pidlist" ]; then
    echo -e "${INFO}$base is running"
  else
    if [ -n "$base" ]; then
      if [ -e /var/run/${base}.pid ]; then
        echo -e "${WARNING}$base is not running"
      else
        echo -e "${INFO}$base is not running"
      fi
    fi
  fi
  return 0
}

################################################################################
# timespec()
#set: 一个用于格式化启动日志文件时间戳的实用函数。
#设置 STAMP 变量。
# 返回值: 未使用
################################################################################
timespec() {
  STAMP=$(echo `date +%Y-%m-%d %H:%M:%S` `hostname`)
}

################################################################################
# log_success_msg()
# 用: log_success_msg ["message"]
#set: 将成功状态打印到屏幕和
# 启动日志文件中。
# 输入: $@ -get
# 返回值: 未使用
################################################################################
log_success_msg() {
  echo -n "${BMPREFIX}$@}"
  echo -e "${CURS_ZERO}${SUCCESS_PREFIX}${SET_COL}${SUCCESS_SUFFIX}"

  # 从日志文件中移除不可打印字符
  logmessage=$(echo "${@}" | sed 's/\033[a-zA-Z]*//g')
  echo -e "${STAMP} ${logmessage} OK" >> ${BOOTLOG}

  return 0
}

log_success_msg2() {
  echo -n "${BMPREFIX}$@}"
  echo -e "${CURS_ZERO}${SUCCESS_PREFIX}${SET_COL}${SUCCESS_SUFFIX}"

  echo "OK" >> ${BOOTLOG}

  return 0
}
