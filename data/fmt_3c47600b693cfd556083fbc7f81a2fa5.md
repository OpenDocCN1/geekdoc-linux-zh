```bash
# Default-Stop: 0

```

```bash
# 注意：由于远程日志配置，syslogd 不运行级别 2 下启动。

/sbin/logd
evaluate_retval
;;

reload)
log_info_msg "正在重新加载日志守护进程..."
pid=`cat /var/run/logd.pid`
kill -H "$pid"
evaluate_retval
;;

restart)
$($0) stop
sleep 1
$($0) start
;;

status)
status /sbin/logd
;;

*)
echo "Usage: $0 {start|restart|status}"
exit 1
esac
```

```bash
### BEGIN INIT INFO

# Provides: $network

# Required-Start: $local_fs

# Should-Start: $syslog

# Required-Stop: $local_fs

# Should-Stop: $syslog

# Default-Start: 3

# Default-Stop: 0

# Short-Description: 启动并配置网络接口。

# Description: 启动并配置网络接口。

# X-LFS-Provided-By: LFS

### END INIT INFO

```

```bash

case "$1" in

start)

  # 启动所有网络接口

  for file in /etc/sysconfig/ifconfig*;

  do

    interface=${file*/}

    # 如果没有找到接口，则跳过

    if [ "${interface}" = "*" ]; then

      continue

    fi

    /sbin/ifup ${interface}

  done

  ;;

stop)

  # 反转列表

  net_files=""

  for file in /etc/sysconfig/ifconfig*;

  do

    net_files="${file} ${net_files}"

  done

  # 停止所有网络接口

  for file in $net_files;

  do

    interface=${file*/}

    # 如果没有找到接口，则跳过

    if [ "${interface}" = "*" ]; then

      continue

    fi

    /sbin/ifdown ${interface}

  done

  ;;

restart)

  $($0) stop

  sleep 1

  $($0) start

  ;;

*)

  echo "Usage: $0 {start|stop|restart}"

  exit 1

  ;;

esac

exit 0

# 结束 network
```
