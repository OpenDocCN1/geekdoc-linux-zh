
`OSTaskkInit()` 的代码如下。

```
MOV         [SP], AX;

MOV         AX, pdata;

MOV         [SP+2], AX;

MOV         AX, ptos;

MOV         [SP+4], AX;

MOV         AX, opt;

MOV         [SP+6], AX;

/* 恢复寄存器 */

POP         DI;

POP         CX;

POP         DX;

POP         BX;

POP         BP;

POP         SI;

POP         AX;

POP         ES;

POP         DS;

}
```

```c
void `OSTaskkInit`(void *`task`, void *`pdata`, void *`ptos`, uint16 `opt`)
{
    /* 保存任务栈帧 */
    PUSH        DS;
    PUSH        ES;
    PUSH        AX;
    PUSH        SI;
    PUSH        BP;
    PUSH        BX;
    PUSH        DX;
    PUSH        CX;
    PUSH        DI;

    /* 设置栈帧 */
    MOV         AX, `task`;
    MOV         [SP], AX;
    MOV         AX, `pdata`;
    MOV         [SP+2], AX;
    MOV         AX, `ptos`;
    MOV         [SP+4], AX;
    MOV         AX, `opt`;
    MOV         [SP+6], AX;

    /* 恢复寄存器 */
    POP         DI;
    POP         CX;
    POP         DX;
    POP         BX;
    POP         BP;
    POP         SI;
    POP         AX;
    POP         ES;
    POP         DS;
}
```
