int main(int argc, char **argv) {
  int g_rep;
  setup_gpio(0);
  IN_GPIO(7);
  OUT_GPIO(7);
  for (rep=0;rep<1; rep++) {
    GPIO_SET = 1<<7;
    sleep(1);
    GPIO_CLR = 1<<7;
    sleep(1);
  }
  return 0;
}
