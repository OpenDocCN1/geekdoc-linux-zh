```python
import pathlib
import re
def get_paths(topdir, pattern):
    for path in pathlib.Path(topdir).rglob(pattern):
        if path.exists():
            yield path
def get_files(paths):
    for path in paths:
        with path.open('rt', encoding='latin-1') as file:
            yield file
def get_lines(files):
    for file in files:
        yield from file
```

```python
def get_comments(lines):
    for line in lines:
        m = re.match('.*(#.*)$', line)
        if m:
            yield m.group(1)
```

```python
def print_matching(lines, substring):
    for line in lines:
        if substring in lines:
            print(substring)
```

```python
paths = get_paths('.', '*.py')
files = get_files(paths)
lines = get_lines(files)
comments = get_comments(lines)
print_matching(comments, 'spam')
```

```python
def flatten(items):
```
```python
    stack = [ iter(items) ]
    while stack:
        try:
            item = next(stack[-1])
            if isinstance(item, list):
                stack.append(iter(item))
            else:
                yield item
        except StopIteration:
            stack.pop()
```

```python
def receiver():
    print('Ready to receive')
    while True:
        n = yield
        print('Got', n)
```

```python
class Manager:
    def __init__(self, gen):
        self.gen = gen
    def __enter__(self):
        # 到 yield
        return self.gen.send(None)
    def __exit__(self, ty, val, tb):
        try:
            if ty:
                self.gen.throw(ty, val, tb)
        except StopIteration:
            return True
```
