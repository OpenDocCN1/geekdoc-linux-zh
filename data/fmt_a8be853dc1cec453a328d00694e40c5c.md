function AIX_classic_printing
{
  for Q in $(enq -AW | tail +13 | grep DOWN | awk '{print $1}')
  do
    enable $Q
    if (( $Q == 0 )) || echo "\n$Q print queue FAILED to enable.\n"
  done
}

function AIX_SYSV_printing
{
  LOOP=0      # 循环计数器 - 一次获取三行
  lpc all | egrep ':printing|queueing' | while read LINE
  do
    # 加载三行
    case $LINE in
      *) Q=$(echo $LINE | cut -f1)
      printing*) PSTATUS=$(echo $LINE | awk '{print $3}')
      queueing*) QSTATUS=$(echo $LINE | awk '{print $3}')
    esac

    # 增加循环计数器
    (( LOOP = LOOP + 1 ))
    if (( LOOP == 3 )) # 我们有三行数据
    then
      # 检查打印状态
      case $PSTATUS in
        disabled)
          lpc start $Q >/dev/null
          (( $? == 0 )) && echo "\n$Q printing-started\n"
          ;;
        enabled*) # 无操作
      esac
      # 检查队列状态
      case $QSTATUS in
        disabled)
          lpc enable $Q >/dev/null
          (( $? == 0 )) && echo "\n$Q queuing re-enabled\n"
          ;;
        enabled*) # 无操作
      esac
      LOOP=0 # 重置循环计数器
    fi
  done
}

function HP_UX_printing
{
  lpc status | grep 'Warning:' | while read LINE
  do
    if echo $LINE | grep 'is down'
    then
      enable $(echo $LINE | awk '{print $3}')
    fi
    if echo $LINE | grep 'is off'
    then
      enable $(echo $LINE | awk '{print $3}')
    fi
  done
}

function Linux_printing
{
  lpc status | tail +2 | while read stat1 stat2 stat3
  do
    # 首先检查打印状态
    case $stat2 in
      disabled)
        echo "$stat1 is disabled"
        lpc start $stat1
        (( $? == 0 )) && echo "$stat1 Restarted"
        ;;
      enabled*) # 无操作
    esac
    # 接下来检查队列状态
    case $stat3 in
      disabled)
        echo "$stat1 is queueing"
        lpc enable $stat1
        (( $? == 0 )) && echo "$stat1 queuing-enabled"
        ;;
      enabled*) # 无操作
    esac
  done
}
