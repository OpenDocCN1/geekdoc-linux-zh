# 方法 2C

```bash
print "\n##########"
print "方法 2C。"
print "其中我们允许逗号、加号、减号和小数。"
print "并且已经删除了小数点：\""
input+='123,456,989.0'
print "$input"
print "在移除逗号、加号、减号和"
print "小数点后："
# -d 意味着 '删除' .sed tr.sh
edited_input=$(echo "$input" | tr -d '"\,+,_')
# 查看修改后的输入 结果
print "$edited_input"
# 查看关于 'sed' 命令如何工作的说明（上文）。
n_0_thru="$(echo $edited_input | sed 's/[0-9]//g')"
if [ -z "$n_0_thru" ]; then
    print "检查：良好。你的输入是纯数字。"
else
    print "警告：你的输入中部分数据是非数字的。"
    print "这是输出中非数字的部分："
    print "$n_0_thru_"
fi
```
