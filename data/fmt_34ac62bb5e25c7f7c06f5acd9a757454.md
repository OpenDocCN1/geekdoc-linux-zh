
## D.8. `/etc/rc.d/init.d/checkfs`

```bash
#!/bin/sh
####################################################################
# 开始 checkfs
# Description : 文件系统检查
# Authors : Gerard Beekmans - gerard@linuxfromscratch.com
       A. Luecke - luecke@users.sourceforge.net
       DJ Lucas - dj@linuxfromscratch.org
# Update : Bruce Dubbs - bdubbs@linuxfromscratch.org
# Version : LFS 7.0
# 基于 LFS-3.1 及早期版本的 checkfs 脚本
# 自 man fsck
# 0 - 无错误
# 1 - 文件系统错误已修正
# 2 - 系统应该重启
# 4 - 文件系统错误仍未修正
# 8 - 运行错误
# 16 - 使用或语法错误
# 32 - 用户错误取消 Fck
# 128 - 共享库错误
####################################################################

### BEGIN INIT INFO
# Provides : checkfs
# Required-Start: udev swap $time
# Should-Start: 
# Required-Stop: 
# Should-Stop: 
# Default-Start: S
# Default-Stop: 
# Short-Description: 在挂载前检查本地文件系统。
# Description: 在挂载前检查本地文件系统。
# X-LFS-Provided-By: LFS
### END INIT INFO
```

```bash
. /lib/lsb/init-functions

case "${1}" in
start)
  if [ -f /fastboot ]; then
    msg="找到 fastboot，将按照要求跳过文件系统检查。\n"
    log_info_msg "${msg}"
    exit 0
  fi

  log_info_msg "正在以只读模式挂载文件系统..."
  mount -n -o remont,ro / /dev/null

  if [ ($?) -ne 0 ]; then
    log_failure_msg="\n无法检查根目录"
    msg="${msg}文件系统，因为它无法挂载"
    msg="${msg}以只读模式。\n\n"
    msg="${msg}在按下回车键后，此系统将"
    msg="${msg}停止并关机。\n\n"
    log_failure_msg="${msg}"
  fi

  log_info_msg "按回车继续..."
  wait_for_user
  /etc/rc.d/init.halt stop
  else
    log_success_msg2
  fi

  if [ -f /forceck ]; then
    msg="找到 forceck，将按照要求强制检查文件系统。"
    log_success_msg "${msg}"
    options="-f"
  else
    options=""
  fi

  log_info_msg "正在检查文件系统..."
  # 注意：-a 选项曾用于 -p；但在 fsck.minix 等系统上会失败
  if is_true VERBOSE_FSCK; then
    fsck ${options} -a -A -C -T
  else
    fsck ${options} -a -A -C -T >/dev/null
  fi

  error_value=$?

  if [ "${error_value}" = 0 ]; then
    log_success_msg2
  fi

  if [ "${error_value}" = 1 ]; then
    msg="\n警告：n发现文件系统错误"
    msg="${msg}并已完成修正。\n"
    msg="${msg}    您可能需要双检查"
    msg="${msg}一切是否已妥善修复。"
    log_warning_msg "${msg}"
  fi

  if [ "${error_value}" = 2 ] || [ "${error_value}" = 3 ]; then
    msg="\n警告：n文件系统错误"
    msg="${msg}已发现并已"
    msg="${msg}修正，但该值要求"
    msg="${msg}此系统重启。\n\n"
    msg="${msg}在您按下回车键后，"
    msg="${msg}此系统将重启\n\n"
    log_failure_msg="${msg}"
    log_info_msg "按回车继续..."
    wait_for_user
    boot -f
  fi

  if [ "${error_value}" -gt 3 -a "${error_value}" -lt 16 ]; then
    msg="失败：n遇到了文件系统错误"
    msg="${msg}无法自动"
    msg="${msg}修复。此系统"
    msg="${msg}无法继续启动，并将"
    msg="${msg}停止，直到这些"
    msg="${msg}错误由系统管理员手动"
    msg="${msg}修复。\n\n"
    msg="${msg}在您按下回车键后，此系统将"
    msg="${msg}停止并关机。\n\n"
    log_failure_msg="${msg}"
    log_info_msg "按回车继续..."
    wait_for_user
    /etc/rc.d/init.d halt
  fi

  if [ "${error_value}" -ge 16 ]; then
    msg="失败：n意外故障"
    msg="${msg}运行 fsck 退出代码为"
    msg="${msg}:${error_value}。\n\n"
    log_info_msg ${msg}
    exit ${error_value}
  fi

  exit 0
  ;;
  *)
    echo "Usage: $(0) (start)"
    exit 1
  ;;
esac
# 结束检查
```
