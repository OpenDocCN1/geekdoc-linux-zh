# 远程日志配置

```bash
/lib/lsb/init-functions

case "${1}" in
start)
    log_info_msg "Starting system log daemon..."
    parms=${SYSLOGD_PARMS:'-m 0'}
    start_daemon /sbin/syslogd $parms
    evaluate_retval

    log_info_msg "Starting kernel log daemon..."
    start_daemon /bin/klogd
    evaluate_retval
;;

stop()
    log_info_msg "Stopping kernel log daemon..."
    killproc /bin/klogd
    evaluate_retval

    log_info_msg "Stopping system log daemon..."
    killproc /bin/syslogd
    evaluate_retval
;;

reload()
    log_info_msg "Reloading system log daemon config file..."
    pid="$pidofproc syslog"
    kill -HUP "${pid}"
    evaluate_retval
;;

restart()
    $0 stop
    sleep 1
    $0 start
;;

status()
    statusproc /bin/syslogd
    statusproc klogd
;;

*)
    echo "Usage: ${0} {start|stop|reload|restart|status}"
    exit 1
;;
esac
```

```bash
#

######################################################################

### BEGIN INIT INFO

# Provides: $network

# Required-Start: $local_fs swap localnet

# Should-Start: $syslog

# Required-Stop: $local_fs swap localnet

# Should-Stop: $syslog

# Default-Start: 0

# Default-Stop: 0 1 2

# Short-Description: 启动并配置网络接口。

# Description: 启动并配置网络接口。

# X-LFS-Provided-By: LFS

### END INIT INFO

case "${1}" in

start)

  # 启动所有网络接口

  for file in /etc/sysconfig/if.config;

  do

    interface=$(file#*/if.config)

    # 如果 $file 是 * 则跳过（因为没有找到内容）

    if [ "${interface}" = "*" ]; then

      continue

    fi

    /sbin/ifup ${interface}

  done

stop)

  # 卸载任何网络挂载的文件系统

  umount --all --types nfs,cifs,nfs4

  # 反转列表

  net_files=""

  for file in /etc/sysconfig/if.config;

  do

    net_files="${net_files} $file"

  done

  # 停止所有网络接口

  for file in ${net_files};

  do

    interface=$(file#*/if.config)

    # 如果 $file 是 * 则跳过（因为没有找到内容）

    if [ "${interface}" = "*" ]; then

      continue

    fi

    /sbin/ifdown ${interface}

  done

;;

restart)
```
