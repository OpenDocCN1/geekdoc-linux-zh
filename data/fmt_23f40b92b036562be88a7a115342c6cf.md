
# usave.sh /tmp/pipe - save to remote tape

```bash
#------------------------------------------------

set -x

PROG=$(basename $0)

HOST=$(hostname)

TODAY=$(date +%H:%M:%S)

cleanup () {

    ec=$1

    error=$2

    case "$ec" in

        "$USAGE_EC")

            echo "Usage:$PROG DeviceName" >&2

            ;;

        "$NOTAP_EC")

            echo "error:$PROG: $DEVICE is not available on the system." >&2

            ;;

        "$LISTE_EC")

            echo "error:$PROG: not create tar list for $LOGNAME." >&2

            ;;

        "$NOTAR_EC")

            echo "error:$PROG: tar command failed." >&2

            ;;

        "$PIPEP_EC")

            echo "error:$PROG: mknod failed." >&2

            ;;

        "$NORSH_EC")

            echo "error:$PROG: Shell command failed." >&2

            ;;

        "$RHOST_EC")

            echo "error:$PROG: Remote Host." >&2

            ;;

        *)

            ;;

    esac

    rm -f ${LIST} ${PIPE} 2>/null

    [ -n "$error" ] && echo "$error"

    trap '' 0 25

    exit "$ec"

}

# Variables

USAGE_EC=1 # exit code for usage error

NOMNT_EC=2 # exit code wrong device name

NOTAP_EC=3 # exit code no available

LISTE_EC=4 # exit code backup error

NOTAR_EC=5 # exit code for wrong tar

TRAPP_EC=6 # exit code for trap

PIPEP_EC=7 # exit code for pipe

RHOST_EC=8 # exit code for bad ping

NORSH_EC=9 # exit code for bad rsh

DEVICE="$1" # device to tar into

LIST="/tmp/.tar.$LOGNAME.$$" #

REMOTEH="" # Remote host for backup

REMOTET="" # Remote tape for backup

tapedev=

PIPE="/tmp/pipe" # Pipe for remote backup

main() {

    tput clear

    echo "$PROG started from $LOGNAME@$HOST on $TERM at $TODAY."

    rm -f $LIST 2>/null

    # Trap on exit/interrupt/break to clean up

    trap "cleanup $TRAPP_EC \"Abnormal termination. $PROG\"" 0 1 25

    # Check command options

    [ "$#" -ne 1 ] && cleanup "$USAGE_EC" ""

    # Check device name

    [ 'expr "$DEVICE" : "[/"' -eq 0 ] && cleanup "$NOMNT_EC" "$PROG: Backup device or file name must start with a '/'."

    # Check tape device

    case "$DEVICE" in

        # Fix the block size if $DEVICE is a tape device

        /dev/rmt[0-9]*)

            echo "$PROG: Rewinding tape to begin.........(please wait)"

            tctl -f $DEVICE rewind 2>/null

            ;;

        *)

            ;;

    esac

    # Fix the block size if $DEVICE is a tape device

    /dev/rmt[0-9]*)

        echo "$PROG: Rewinding tape to begin.........(please wait)"

        tctl -f $DEVICE rewind 2>/null

        ;;

        *)

            ;;

    esac

    rm -f ${LIST} ${PIPE} 2>/null

    [ -n "$error" ] && echo "$error"

    trap '' 0 25

    exit "$ec"

}
```
