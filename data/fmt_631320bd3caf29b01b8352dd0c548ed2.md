### 1.13 主逻辑与选项解析

提取提取脚本使用的任何选项。
基于《The Bash Shell》(O'Reilly)。

```bash
while getopts ":save:cookie$helplist$run:inpu$wopt" opt
do

case $opt in

    save) save_func; break ;; # 保存我们的会话以备后用。

    cook) cookie_func; break ;; # 更改 cookie 文件。

    help) usage; break ;; # 获取帮助。

    list) list_func; break ;; # 允许使用 URL 列表。

    run) run_func; break ;; # 如果我们从 cron 脚本调用 wget 则有用。

    input) run_func; break ;; # 当你不知道文件名时。

    wopt) wopts; break ;; # 将选项直接传递给 wget

    \?) echo "不是有效选项。"

    echo "使用 -${wopt} 直接传递选项给 wget"

    echo "或使用 -${help} 获取帮助"; break; # 捕捉任何其他内容。

esac

done

shift $((OPTIND - 1)) # 对进行神奇魔法操作
```

## 2. 播客下载脚本示例

```bash
# --> ################################################################
# --> 什么是“播客 (podcasting)”？
# --> 它是在互联网上广播“电台节目”。
# --> 这些节目可以在 iPod 和其他音乐播放器上播放。
# --> 此脚本实现了这一切。
# --> 详阅上方脚本作者网站上的文档。
# --> ################################################################

# 使脚本兼容 crontab：

cd $0

# --> 切换到此脚本所在的目录。

# datadir 是你想要保存播客文件的目录：

datadir=$(date +%Y-%m-%d)

# --> 将创建一个以日期命名的目录，名为：YYYY-MM-DD

# 检查 datadir 是否存在，并在必要时创建：

if test -d $datadir

then

mkdir $datadir

# 删除任何临时文件：

rm -f temp.log

# 读取 bp.conf 文件并获取 podcast.log 文件中尚未
# 包含任何 url：

while read podcast

do

# --> 后面是主操作。

file=$(wget -q $podcast -o - | tr '\' '\\n' | tr '\' '\\' | \
sed -n 's/.*\(.*\*/\\1/p')

for url in $file

do

  echo $url >> temp.log

  if grep -q "$url" $datadir/podcast.log > /dev/null

  then

    wget -q -P $datadir "$url"

  fi

done < bp.conf

# 将动态创建的日志文件移动到永久日志：

cat podcast.log >> temp.log

sort temp.log | uniq -u > podcast.log

rm temp.log

# 创建 m3u 播放列表：

ls $datadir | grep -v m3u > $datadir/podcast.m3u

exit 0
```
