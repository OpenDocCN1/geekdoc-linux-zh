## 列表 13-4 float_subtract.ksh shell 脚本（续）

```ksh
case $S) SCALE=$OPTARG
    ;;
    ?) # 因为我们可能有负数，我们需要
       # 测试以连字符 (-) 开头的 ARGUMENT
       # 是否为数字，而不是无效的选项！
       for TST_ARG in "$@"
       do
           if [ $(echo $TST_ARG | cut -c1) = '-' ] \
           && [ $TST_ARG =~ '-s' -a $TST_ARG != '-S' ]
           then
               case $TST_ARG in
                   +(-[0-9])) : # 无操作，什么都不做
                       ;;
                   +(-[0-9]).-[0-9]))
               esac
           fi
       done
       ;;
esac
done
```

```ksh
do
    ((COUNT = COUNT + 1))
    TOKEN=$1
    case $TOKEN in
        -s|-S) shift 2
            ((COUNT = COUNT + 1))
            -s$(SCALE) shift
            ;;
            -S$(SCALE) shift
            ;;
            *) NUM_LIST="$NUM_LIST $TOKEN"
                ((COUNT < MAX_COUNT)) && shift
    esac
done
```

```bash
for X in $NUM_LIST
do
  # 如果数字带有 + 前缀，将其移除！
  if [[ $(echo $X | cut -c1) = '+' ]]
  then
    X=$(echo $X | cut -c2)
  fi
  SUBRACT="$SUBRACT MINUS $X"
  MINUS='-'
done
```
