    set_timer( ITIMER_REAL, 1000/speed, 1000/speed );

    mvaddstr(ROW, 0, MESSAGE);

            finished = update_from_input(&speed, &dir);
            input_ready = 0;
        }
        else
            pause();
    clear();
    endwin();
    return 0;
}
