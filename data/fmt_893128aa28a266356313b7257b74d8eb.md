mode(on, off)
	int on, off;
{
	struct sgttyb b;

	ptyflush();
	ioctl(pty, TIOCGGETP, &b);
	b.sg_flags |= on;
	b.sg_flags &= ~off;
	ioctl(pty, TIOCSETP, &b);
}

/*
 * 向 pty 另一侧的进程发送中断信号。
 * 如果处于原始模式，仅写入 NULL；
 * 否则，写入中断字符。
 */
interrupt()
{
	struct sgttyb b;
	struct tchars tchars;

	ptyflush();   /* 半心半意 */
	ioctl(pty, TIOCGGETP, &b);
	if (b.sg_flags & RAW) {
		*pfrontp++ = '\0';
		return;
	}
*pfrontp++ = ioctl(pty, TIOCGETC, &tchars) < 0 ?
    '\177' : tchars.t_intrc;
}

/*
 * 向 pty 另一侧的进程发送退出信号。
 * 如果处于原始模式，仅写入 NULL；
 * 否则，写入退出字符。
 */
sendbrk()
{
    struct sgttyb b;
    struct tchars tchars;

    ptyflush();  /* 半心半意 */
    ioctl(pty, TIOCGETP, &b);
    if (b.sg_flags & RAW) {
        *pfrontp++ = '\0';
        return;
    }
    *pfrontp++ = ioctl(pty, TIOCGETC, &tchars) < 0 ?
    '\034' : tchars.t_quitc;
}

/*
 * nextitem()
 *
 * 返回 TELNET 数据流中下一个“项”的地址。
 * 如果当前地址是用户数据字符，则返回下一个字符的地址；
 * 如果当前地址是 TELNET IAC（“我是一个命令”）
 * 字符，则返回紧随 TELNET 命令之后的字符的地址。
 */
char *
nextitem(current)
char    *current;
{
        if ((*current&0xff) != IAC) {
return current+1;
}
switch (* (current+1) & 0xff) {
case DO:
case DONT:
case WILL:
case WONT:
return current+3;
case SB:       /* 循环查找 SE */
{
register char *look = current+2;

for (;;) {
if ((*look++ & 0xff) == IAC) {
if ((*look++ & 0xff) == SE) {
return look;
}
}
}
}
default:
return current+2;
}
}

/*
* netclear()
*
* 我们即将执行 TELNET SYNCH 操作。清除
* 到网络的路径。
*
* 由于我们可能已经向网络发送了之前 TELNET 命令
* 的第一个字节，情况有些棘手。
* 因此，我们必须从网络缓冲区开始扫描，
* 直到我们到达想要的位置。
*
* 我们操作的一个副作用（为了保持简单）
* 是清除紧急数据指针。主要调用者
* 无论如何都应在调用我们之后设置紧急数据指针。
*/
netclear()
{
register char *thisitem, *next;
char *good;
#define  wewant(p) ((nfrontp > p) && ((*p&0xff) == IAC) && \
((*p+1)&0xff) != EC) && ((*p+1) &0xff) != EL))

thisitem = netobuf;

while ((next = nextitem(thisitem)) <= nbackp) {
thisitem = next;
} 

/* 现在，thisitem 位于边界之前/边界处。 */ 

good = netobuf; /* 正常字节存放的位置 */ 

while (nfrontp > thisitem) { 
if (ewant(thisitem)) { 
int length; 

next = thisitem; 
do { 
next = nextitem(next); 
} while (ewant(next) && (nfrontp > next)); 

length = next-thisitem; 
bcopy(thisitem, good, length); 
good += length; 
thisitem = next; 
} else { 
thisitem = nextitem(thisitem); 
} 
} 

nbackp = netobuf; 
nfrontp = good; /* 下一个要发送的字节 */ 
neturg = 0; 
}
