```c
/* 返回一个描述文件系统 PATH 类型的字符串。 */
const char *get_file_type(const char *path)
{
    struct stat stat;
    lstat(path, &stat);
    if (S_ISLNK(stat.st_mode))
        return "symbolic link";
    else if (S_ISDIR(stat.st_mode))
        return "directory";
    else if (S_ISCHR(stat.st_mode))
        return "character device";
    else if (S_ISBLK(stat.st_mode))
        return "block device";
    else if (S_ISFIFO(stat.st_mode)) 
        return "fifo";
    else if (S_ISSOCK(stat.st_mode))
        return "socket";
    else if (S_ISREG(stat.st_mode))
        return "regular file";
    else
        /* 异常。每个条目应为上述类型之一。 */
        assert(0);
}

int main (int argc, char* argv[])
{
    char* dir_path;
    DIR* dir;
    struct dirent* entry;
    char entry_path[PATH_MAX + 1];
    size_t path_len;

    if (argc >= 2)
        /* 如果命令行上指定了目录，则使用它。 */
        dir_path = argv[1];
    else
        /* 否则，使用当前目录。 */
        dir_path = ".";

    /* 将目录路径拷贝到 entry_path 中。 */
    strncpy(entry_path, dir_path, sizeof(entry_path));
    path_len = strlen(dir_path);

    /* 如果目录不以斜杠结尾，则追加一个。 */
    if (entry_path[path_len - 1] != '/')
    {
        entry_path[path_len] = '/';
        entry_path[path_len + 1] = '\0';
        ++path_len;
    }

    /* 开始对命令行指定的目录进行列表操作。 */
    dir = opendir(dir_path);
    /* 遍历所有目录条目。 */
    while ((entry = readdir(dir)) != NULL) {
        const char* type;
        /* 通过拼接条目构建目录条目的路径。 */
        strncpy(entry_path + path_len, entry->d_name,
               sizeof(entry_path) - path_len);
        /* 确定条目的类型。 */
        type = get_file_type(entry_path);
        /* 打印条目的类型和路径。 */
        printf("%-8s: %s\n", type, entry_path);
    }

    /* 全部完成。 */
    closedir (dir);
    return 0;
}
```
