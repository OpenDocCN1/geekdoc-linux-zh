### 示例：在客户端和服务器之间传递结构体

我们修改客户端和服务器传递二进制值而不是字符串。当客户端和服务器位于字节序不同的主机时，我们会发现这无法工作。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <netinet/in.h>
#include <arpa/inet.h>
struct args {
    long arg1;
    long arg2;
};
struct result {
    long sum;
};
void cli(char *hostname) {
    struct sockaddr_in servaddr;
    struct args a;
    struct result res;
    int sockfd;
    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    servaddr.sin_family = AF_INET;
    servaddr.sin_port = htons(8080);
    inet_pton(AF_INET, hostname, &servaddr.sin_addr);
    if (connect(sockfd, (struct sockaddr *)&servaddr, sizeof(servaddr)) < 0) {
        perror("connect error");
        exit(1);
    }
    a.arg1 = 11;
    a.arg2 = 22;
    write(sockfd, &a, sizeof(a));
    if (read(sockfd, &res, sizeof(res)) <= 0) {
        perror("read error");
        exit(1);
    }
    printf("Sum: %ld\n", res.sum);
    close(sockfd);
}
int main(int argc, char *argv[]) {
    if (argc != 2) {
        printf("Usage: %s <hostname>\n", argv[0]);
        exit(1);
    }
    cli(argv[1]);
    return 0;
}
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <netinet/in.h>
#include <arpa/inet.h>
struct args {
    long arg1;
    long arg2;
};
struct result {
    long sum;
};
void echo(int sockfd) {
    struct args a;
    struct result res;
    size_t n;
    for (;;) {
        n = read(sockfd, &a, sizeof(a));
        if (n <= 0) {
            /* connection closed by other end */
            break;
        }
        res.sum = a.arg1 + a.arg2;
        write(sockfd, &res, sizeof(res));
    }
}
int main() {
    int sockfd, client_fd;
    struct sockaddr_in servaddr, cli_addr;
    socklen_t len_len = sizeof(cli_addr);
    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    servaddr.sin_family = AF_INET;
    servaddr.sin_port = htons(8080);
    servaddr.sin_addr.s_addr = INADDR_ANY;
    if (bind(sockfd, (struct sockaddr *)&servaddr, sizeof(servaddr)) < 0) {
        perror("bind error");
        exit(1);
    }
    if (listen(sockfd, 10) < 0) {
        perror("listen error");
        exit(1);
    }
    while (1) {
        client_fd = accept(sockfd, (struct sockaddr *)&cli_addr, &len_len);
        if (client_fd < 0) {
            perror("accept error");
            continue;
        }
        echo(client_fd);
        close(client_fd);
    }
    return 0;
}
```
