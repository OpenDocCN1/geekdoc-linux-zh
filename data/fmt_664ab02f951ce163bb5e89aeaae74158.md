
# Linux From Scratch - 版本 12.1

# 输入：`-p pidfile`，使用指定的 `pidfile` `pathname`，指定程序的路径 `signal`，向 `pathname` 发送此信号

# 返回值（如由 LSB 退出码定义）：
# `0` 程序 (`pathname`) 已停止或处于停止状态，或运行中的程序收到信号并成功停止
# `1` - 通用或未指定的错误
# `2` - 无效或参数过多的参数
# `5` 程序未安装
# `7` 程序未运行且已提供信号

killproc() {
  local pidfile
  local program
  local prefix
  local programme
  local signal="-TERM"
  local fallback="-KILL"
  local nosig
  local pidlist
  local retry
  local pid
  local delay="30"
  local piddead
  local dtime

  # 处理参数
  while true; do
    case "${1}" in
      -p)
        pidfile="${2}"
        shift 2
        ;;
      *)
        if [ -n "${2}" ]; then
          signal="${2}"
          fallback=""
        else
          nosig=1

          # 如果有额外参数报错
          if [ -n "${3}" ]; then
            return 2
          else
            break
          fi
        fi
        ;;
    esac
  done

  # 检查程序有效
  if [ ! -e "${program}" ]; then return 5; fi

  # 检查信号有效
  check_signal="${signal}"
  if [ "$?" -ne "0" ]; then return 2; fi

  # Linux From Scratch - 版本 12.1

  # 获取 pid 列表
  if [ -z "${pidfile}" ]; then
    # 通过发现确定 pid
    pidlist=$(pidofproc "${1}")
    retval=${?}
  else
    # PID 文件包含了所需的 PIDs
    # 注意，根据 LSB 要求，路径必须提供给 pidofroc（此处此处应为 pidofproc），
    # 但当前实现或标准并未使用它。
    pidlist=$(pidofproc -p "${pidfile}" "${1}")
    retval=${?}
  fi

  # 仅返回值
  # 日志记录由初始化脚本（或发行版函数）负责！
  case "${retval}" in
    0)
      # 程序运行正常
      # 此处什么都不做，让 killproc 继续。
      ;;
    1)
      # 程序未运行，但存在无效的 pid 文件
      #  pid 文件。
      progname=${program##*/}

      if [ -e "run/${progname}.pid" ]; then
        pidfile=run/${progname}.pid
        rm -f "${pidfile}"
      fi

      # 如果没有传递信号，则视为成功。
      if [ -n "${nosig}" ]; then
        return 0
      else
        # 这部分逻辑在原始 OCR 中略有错，但遵循其结构：
        # 程序未运行且不存在 pidfile
        # 如果没有传递信号，则视为成功。
        if [ -n "${nosig}" ]; then
          return 0
        else
          return 7
        fi
      fi
      ;;
    *)
      # 其他由状态值返回的结果不被解释，
      # 并作为未指定的错误返回
      return 1
      ;;
  esac

  # 对退出信号和控制信号执行不同操作
  check_sig_type "${signal}" 

  if [ "$?" -eq "0" ]; then 
    # 信号用于终止程序
    # 考虑到空 pidlist（pid 文件仍存在且未提供
    # 信号
    if [ "${pidlist}" != "" ]; then
      # 杀死 pid 列表
      for pid in ${pidlist}; do
        kill -0 "${pid}" 2>/dev/null
        if [ "$?" -ne "0" ]; then
          # 进程已，继续下一个并假设一切正常
          continue
        else
          kill "${signal}" "${pid}" 2>/dev/null
          # 等待最多 "${delay}"/10 秒直到 "${pid}"
          # 以十分之一秒为单位进行终止
          while [ "${delay}" -ne "0" ]; do
            kill -0 "${pid}" 2>/dev/null || piddead="1"
            if [ "${piddead}" = "1" ]; then break; fi
            sleep 0.1
            delay=$(( ${delay} - 1 ))
          done

          # 如果设置了备选方案且仍在运行，
          # 则使用备选方案
          if [ -n "${fallback}" -a "${piddead}" != "1" ]; then
            kill "${fallback}" "${pid}" 2>/dev/null
            sleep 1
          fi
        fi
        # 再次检查，如果仍在运行
        kill -0 "${pid}" 2>/dev/null && return 1
      done
    fi

    # 检查并移除过期的 PID 文件。
    if [ -n "${pidlist}" ]; then
      # Find $program 的基名
      prefix=$(echo "${program}" | sed 's|/[^/]*$||')
      progame=$(echo "${program}" | sed "s|^\${prefix}/|@")

      if [ -e "/run/${progame}.pid" ]; then
        rm -f "/run/${progame}.pid" 2>/dev/null
      fi
    fi

    # 对于信号
    if [ "$?" -ne "0" ]; then 
      # 信号用于终止程序
    fi
  fi
}
