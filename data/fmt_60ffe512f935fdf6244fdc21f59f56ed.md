```c
/*
* GameX
* 获取屏幕的 -x- 值并转换为
* 游戏坐标 -x- 值。
*/
int GameX(int x) {
    int xPos;
    /* 相对于英雄 --- */
    xPos = (int) (x + nRelativeAdjustment);
    /* 确保它不在范围内 -- */
    if (xPos < 0) xPos += X_RANGE;
    if (xPos > X_RANGE) xPos -= X_RANGE;
    return (xPos);
}
```

```c
/*
* Move
* 让单位在 X 方向上移动 vx (x 速度)，
* 在 Y 方向上移动 vy。
*/
void Move(typUnit *unit) {
    /* --- 移动单位 --- */
    unit->y += unit->vy;
    unit->x += unit->vx;
    /* --- 将其保持在世界范围内 --- */
    if (unit->x < 0) unit->x += X_RANGE;
    if (unit->x > X_RANGE) unit->x -= X_RANGE;
```
